def dsl1(I):
    # Global periodicity comparison: choose color by vperiod vs hperiod of 8s; equality -> 7
    p8 = ofcolor(I, EIGHT)
    o8 = toobject(p8, I)
    v = vperiod(o8)
    h = hperiod(o8)
    eq = equality(v, h)
    basecol = branch(greater(v, h), TWO, THREE)
    col = branch(eq, SEVEN, basecol)
    S = switch(I, EIGHT, col)
    O = replace(S, ONE, ZERO)
    return O

def dsl3(I):

    lc = leastcolor(I)
    col = branch(equality(lc, ONE), THREE, TWO)

    O = replace(S, ONE, ZERO)

def dsl4(I):
