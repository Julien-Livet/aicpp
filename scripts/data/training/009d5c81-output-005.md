def dsl1(I):
    # Global replace: zero-out ones, then map all 8s to a value from bottom-half ones
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    no_ones = replace(I, ONE, ZERO)
    O = replace(no_ones, EIGHT, val)
    return O

def dsl2(I):
    # Object extraction: recolor 8-objects and paint on blank canvas
    f = rbind(ofcolor, ONE)
    val = chain(halve, size, f)(bottomhalf(I))
    base = canvas(ZERO, shape(I))
    allobjs = objects(I, T, F, T)
    eobjs = colorfilter(allobjs, EIGHT)
    recol = merge(apply(lbind(recolor, val), eobjs))
    O = paint(base, recol)
    return O

def dsl3(I):
    # Pixel mask fill: fill 8-indices with value on zero canvas
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    inds8 = ofcolor(I, EIGHT)
    base = canvas(ZERO, shape(I))
    O = fill(base, val, inds8)
    return O

def dsl4(I):
    # Geometric: normalize 8-object, recolor, and move back to original UL corner
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    inds8 = ofcolor(I, EIGHT)
    obj8 = toobject(inds8, I)
    ul = ulcorner(obj8)
    norm8 = normalize(obj8)
    rec = recolor(val, norm8)
    base = canvas(ZERO, shape(I))
    O = move(base, rec, ul)
    return O

def dsl5(I):
    # Partition-based: recolor foreground 8-objects and underpaint on blank canvas
    f = rbind(ofcolor, ONE)
    val = chain(halve, size, f)(bottomhalf(I))
    base = canvas(ZERO, shape(I))
    parts = fgpartition(I)
    eobjs = colorfilter(parts, EIGHT)
    recol = merge(apply(lbind(recolor, val), eobjs))
    O = underpaint(base, recol)
    return O
