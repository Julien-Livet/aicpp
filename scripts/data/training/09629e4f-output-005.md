    C33 = index(C, astuple(SEVEN, SEVEN))

    TR = canvas(C13, astuple(THREE, THREE))
    ML = canvas(C21, astuple(THREE, THREE))

    MR = canvas(C23, astuple(THREE, THREE))
    BL = canvas(C31, astuple(THREE, THREE))
