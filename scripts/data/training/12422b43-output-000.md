def dsl1(I):
    G = replace(I, FIVE, ZERO)
    obj = asobject(G)
    s1 = shift(obj, DOWN)
    s2 = shift(obj, multiply(DOWN, TWO))
    s3 = shift(obj, multiply(DOWN, THREE))
    S = insert(s2, insert(s1, initset(s3)))
    U = merge(S)
    O = underpaint(I, U)
    return O
