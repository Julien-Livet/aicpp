def dsl1(I):
    # Relational: 5s in straight line-of-sight from any 4 through contiguous 5s
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    allowed = combine(fou, fiv)
    segs = sfilter(prapply(connect, fou, fiv), fork(either, hline, vline))
    pure = sfilter(segs, matcher(compose(size, rbind(difference, allowed)), ZERO))
    mask = merge(pure)
    target = intersection(mask, fiv)
    O = fill(I, TWO, target)
    return O

def dsl2(I):
    # Object extraction: recolor line-shaped 5-objects touching any 4
    fou = ofcolor(I, FOUR)
    fiveobjs = colorfilter(objects(I, T, F, T), FIVE)
    touching = sfilter(fiveobjs, lbind(adjacent, fou))
    lines = sfilter(touching, fork(either, hline, vline))
    target = merge(mapply(toindices, lines))
    O = fill(I, TWO, target)
    return O

def dsl3(I):
    # Geometric: 5s strictly between pairs of 4s in same row/column without other colors
    fou = ofcolor(I, FOUR)
    allowed = combine(fou, ofcolor(I, FIVE))
    segs = sfilter(prapply(connect, fou, fou), fork(either, hline, vline))
    pure = sfilter(segs, matcher(compose(size, rbind(difference, allowed)), ZERO))
    mask = merge(pure)
    target = intersection(mask, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O

def dsl4(I):
    # Color filtering: 5s within two axial steps from any 4, restricted to 4-rows/cols
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    rows = merge(mapply(hfrontier, fou))
    cols = merge(mapply(vfrontier, fou))
    axis = combine(rows, cols)
    nbr1 = merge(mapply(neighbors, fou))
    step1 = intersection(nbr1, fiv)
    nbr2 = merge(mapply(neighbors, step1))
    step2 = intersection(nbr2, fiv)
    target = intersection(step2, axis)
    O = fill(I, TWO, target)
    return O

def dsl5(I):
    # Global geometric: 5s on horizontal/vertical connectors between centers of 4-objects
    fourobjs = colorfilter(objects(I, T, F, T), FOUR)
    centers = mapply(center, fourobjs)
    segs = sfilter(prapply(connect, centers, centers), fork(either, hline, vline))
    mask = merge(segs)
    target = intersection(mask, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O
