def dsl1(I):
    # Relational: color 5s that lie strictly between two 4s along a straight line (horizontal or vertical).
    p4 = ofcolor(I, FOUR)
    p5 = ofcolor(I, FIVE)
    paths = prapply(connect, p4, p4)
    allowed = combine(p4, p5)
    pure = compose(lbind(equality, ZERO), compose(size, rbind(difference, allowed)))
    linear = fork(either, hline, vline)
    cond = fork(both, pure, linear)
    good = mfilter(paths, cond)
    mask = intersection(merge(good), p5)
    O = fill(I, TWO, mask)
    return O

def dsl2(I):
    # Relational variant: color 5s on straight paths from a 4 to a zero that is adjacent to 5s (tight endpoints).
    p4 = ofcolor(I, FOUR)
    p5 = ofcolor(I, FIVE)
    z0 = ofcolor(I, ZERO)
    dirs = insert(LEFT, insert(RIGHT, insert(UP, initset(DOWN))))
    n5 = merge(apply(lbind(shift, p5), dirs))
    znear5 = intersection(z0, n5)
    allowed = merge(insert(p4, insert(p5, initset(znear5))))
    paths = prapply(connect, p4, znear5)
    pure = compose(lbind(equality, ZERO), compose(size, rbind(difference, allowed)))
    linear = fork(either, hline, vline)
    cond = fork(both, pure, linear)
    good = mfilter(paths, cond)
    mask = intersection(merge(good), p5)
    O = fill(I, TWO, mask)
    return O

def dsl3(I):
    # Geometric: two-step rays from each 4 into 5s, but only from the side where the run abuts a zero.
    p4 = ofcolor(I, FOUR)
    p5 = ofcolor(I, FIVE)
    z0 = ofcolor(I, ZERO)
    # Left
    L1 = intersection(shift(p4, LEFT), p5)
    L2 = intersection(shift(L1, LEFT), p5)
    EL1 = intersection(shift(L1, LEFT), z0)
    EL2 = intersection(shift(L2, LEFT), z0)
    CL = merge(insert(shift(EL1, RIGHT), insert(shift(EL2, RIGHT), initset(shift(shift(EL2, RIGHT), RIGHT)))))
    # Right
    R1 = intersection(shift(p4, RIGHT), p5)
    R2 = intersection(shift(R1, RIGHT), p5)
    ER1 = intersection(shift(R1, RIGHT), z0)
    ER2 = intersection(shift(R2, RIGHT), z0)
    CR = merge(insert(shift(ER1, LEFT), insert(shift(ER2, LEFT), initset(shift(shift(ER2, LEFT), LEFT)))))
    # Up
    U1 = intersection(shift(p4, UP), p5)
    U2 = intersection(shift(U1, UP), p5)
    EU1 = intersection(shift(U1, UP), z0)
    EU2 = intersection(shift(U2, UP), z0)
    CU = merge(insert(shift(EU1, DOWN), insert(shift(EU2, DOWN), initset(shift(shift(EU2, DOWN), DOWN)))))
    # Down
    D1 = intersection(shift(p4, DOWN), p5)
    D2 = intersection(shift(D1, DOWN), p5)
    ED1 = intersection(shift(D1, DOWN), z0)
    ED2 = intersection(shift(D2, DOWN), z0)
    CD = merge(insert(shift(ED1, UP), insert(shift(ED2, UP), initset(shift(shift(ED2, UP), UP)))))
    mask = merge(insert(CL, insert(CR, insert(CU, initset(CD)))))
    O = fill(I, TWO, mask)
    return O

def dsl4(I):
    # Color filtering + local structure: color 5s that are flanked by 4s (4-5-4) horizontally or vertically.
    p4 = ofcolor(I, FOUR)
    p5 = ofcolor(I, FIVE)
    midH = intersection(intersection(p5, shift(p4, LEFT)), shift(p4, RIGHT))
    midV = intersection(intersection(p5, shift(p4, UP)), shift(p4, DOWN))
    mask = merge(insert(midH, initset(midV)))
    O = fill(I, TWO, mask)
    return O

def dsl5(I):
    # Object extraction: color entire 5-components that touch at least one 4.
    p4 = ofcolor(I, FOUR)
    objs = objects(I, T, F, T)
    fiveobjs = colorfilter(objs, FIVE)
    touching = mfilter(fiveobjs, lbind(adjacent, p4))
    five_indices_sets = apply(toindices, touching)
    mask = merge(five_indices_sets)
    O = fill(I, TWO, mask)
    return O
