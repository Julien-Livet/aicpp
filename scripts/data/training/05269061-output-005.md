def dsl1(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    COL = lbind(index, I)
    SUP = rbind(shoot, UP_RIGHT)
    SDL = rbind(shoot, DOWN_LEFT)
    DIAG = fork(combine, SUP, SDL)
    OBJF = fork(recolor, COL, DIAG)
    SEED = merge(apply(OBJF, NZ))
    H = height(I)
    W = width(I)
    RO = interval(0, H, THREE)
    CO = interval(0, W, THREE)
    RV3 = apply(toivec, RO)
    CV3 = apply(tojvec, CO)
    OFFS = prapply(subtract, RV3, CV3)
    SH = apply(lbind(shift, SEED), OFFS)
    BIG = merge(SH)
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O

def dsl2(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    HEADS = difference(NZ, shift(NZ, UP_RIGHT))
    COL = lbind(index, I)
    SUP = rbind(shoot, UP_RIGHT)
    SDL = rbind(shoot, DOWN_LEFT)

    H = height(I)
    RO = interval(0, H, THREE)
    DV = apply(toivec, RO)
    NDV = apply(invert, DV)
    VOFFS = combine(DV, NDV)
    SH = apply(lbind(shift, SEED), VOFFS)

def dsl3(I):
    COLORS = remove(ZERO, palette(I))
    INDC = lbind(ofcolor, I)
    SUP = rbind(shoot, UP_RIGHT)
    SDL = rbind(shoot, DOWN_LEFT)
    LUP = compose(merge, compose(lbind(apply, SUP), INDC))
    LDL = compose(merge, compose(lbind(apply, SDL), INDC))
    DIAGC = fork(combine, LUP, LDL)
    FULL = merge(apply(DIAGC, COLORS))
    H = height(I)
    W = width(I)
    RO = interval(0, H, THREE)
    CO = interval(0, W, THREE)
    DV = apply(toivec, RO)
    NDV = apply(invert, DV)
    DH = apply(tojvec, CO)
    NDH = apply(invert, DH)
    VOFFS = combine(DV, NDV)
    HOFFS = combine(DH, NDH)
    OFFS = combine(VOFFS, HOFFS)
    SH = apply(lbind(shift, FULL), OFFS)
    BIG = merge(SH)
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O

def dsl4(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    COL = lbind(index, I)
    SUP = rbind(shoot, UP_RIGHT)
    OBJF = fork(recolor, COL, SUP)
    HALF = merge(apply(OBJF, NZ))
    S0 = initset(HALF)
    S1 = insert(hmirror(HALF), S0)
    S2 = insert(vmirror(HALF), S1)
    S3 = insert(vmirror(hmirror(HALF)), S2)
    BIG = merge(S3)
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O
