def dsl1(I):
    ones = ofcolor(I, ONE)
    ul = ulcorner(ones)
    lr = lrcorner(ones)
    span = subtract(lr, ul)
    n = size(ones)
    k = decrement(n)
    vec = divide(span, k)
    start = add(lr, vec)
    path = shoot(start, vec)
    twos = recolor(TWO, path)
    O = underpaint(I, twos)
    return O
