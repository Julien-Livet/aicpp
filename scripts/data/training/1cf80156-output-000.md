def dsl1(I):
    P = difference(asindices(I), ofcolor(I, ZERO))
    O = crop(I, ulcorner(P), shape(backdrop(P)))
    return O

def dsl2(I):
    P = mapply(toindices, fgpartition(I))
    O = subgrid(P, I)
    return O

def dsl3(I):
    C = remove(ZERO, palette(I))
    P = mapply(lbind(ofcolor, I), C)
    O = subgrid(P, I)
    return O

def dsl4(I):
    P = difference(asindices(I), ofcolor(I, ZERO))
    S = ulcorner(P)
    D = papply(add, subtract(lrcorner(P), S), UNITY)
    O = crop(I, S, D)
    return O

def dsl5(I):
    P = difference(asindices(I), ofcolor(I, ZERO))
    G = subgrid(P, I)
    N = normalize(P)
    OBJ = toobject(N, G)
    BASE = canvas(ZERO, shape(G))
    O = paint(BASE, OBJ)
    return O
