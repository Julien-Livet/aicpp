def dsl1(I):
    # O = overlay of all four rotations of the input (global symmetry)
    A = asobject(I)
    R1 = asobject(rot90(I))
    R2 = asobject(rot180(I))
    R3 = asobject(rot270(I))
    stamps = merge(insert(A, insert(R1, insert(R2, initset(R3)))))
    O = paint(canvas(ZERO, shape(I)), stamps)
    return O

def dsl2(I):
    # O = fill each object's bounding box with its color (object extraction)
    objs = objects(I, T, F, T)
    bb = compose(backdrop, toindices)
    stamped = apply(fork(recolor, color, bb), objs)
    O = paint(canvas(ZERO, shape(I)), merge(stamped))
    return O

def dsl3(I):
    # O = draw bbox outlines only for the most common color objects (color filtering)
    objs = objects(I, T, F, T)
    mc = mostcolor(I)
    sel = colorfilter(objs, mc)
    outline = compose(box, toindices)
    stamped = apply(fork(recolor, color, outline), sel)
    O = paint(canvas(ZERO, shape(I)), merge(stamped))
    return O

def dsl4(I):
    # O = connect each object's center to the grid center (geometric reasoning)
    objs = objects(I, T, F, T)
    gcen = center(asindices(I))
    ceno = compose(center, toindices)
    to_g = rbind(connect, gcen)
    ray = compose(to_g, ceno)
    stamped = apply(fork(recolor, color, ray), objs)
    O = paint(canvas(ZERO, shape(I)), merge(stamped))
    return O

def dsl5(I):
    # O = fill the ring (bbox minus object) for each object with its color (relational)
    objs = objects(I, T, F, T)
    ring = compose(delta, toindices)
    stamped = apply(fork(recolor, color, ring), objs)
    O = paint(canvas(ZERO, shape(I)), merge(stamped))
