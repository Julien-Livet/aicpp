def dsl1(I):
    # Global: recolor 8s by whole-grid midlines (quadrants of the full grid)
    P = ofcolor(I, EIGHT)
    hi = height(I); wi = width(I)
    mi = halve(hi); mj = halve(wi)
    TL = asindices(crop(I, ORIGIN, astuple(mi, mj)))
    TR = shift(asindices(crop(I, astuple(ZERO, mj), astuple(mi, subtract(wi, mj)))), astuple(ZERO, mj))
    BL = shift(asindices(crop(I, astuple(mi, ZERO), astuple(subtract(hi, mi), mj))), astuple(mi, ZERO))
    BR = shift(asindices(crop(I, astuple(mi, mj), astuple(subtract(hi, mi), subtract(wi, mj)))), astuple(mi, mj))
    base = replace(I, EIGHT, ZERO)
    O = fill(base, FOUR, intersection(P, TL))
    O = fill(O, TWO, intersection(P, TR))
    O = fill(O, THREE, intersection(P, BL))
    O = fill(O, ONE, intersection(P, BR))
    return O

def dsl2(I):
    # Objectwise: recolor each 8-object by COM sign vs 8-box center, merging axis cases
    P = ofcolor(I, EIGHT)
    l = leftmost(P); r = rightmost(P)
    u = uppermost(P); d = lowermost(P)
    gc = astuple(halve(add(u, d)), halve(add(l, r)))
    objs = colorfilter(objects(I, T, F, T), EIGHT)
    vec = compose(rbind(subtract, gc), centerofmass)
    sgn = compose(sign, vec)
    is_tl = compose(lbind(equality, NEG_UNITY), sgn)
    is_tr = compose(lbind(equality, UP_RIGHT), sgn)
    is_bl = compose(lbind(equality, DOWN_LEFT), sgn)
    is_br = compose(lbind(equality, UNITY), sgn)
    is_up = compose(lbind(equality, UP), sgn)
    is_dn = compose(lbind(equality, DOWN), sgn)
    is_lf = compose(lbind(equality, LEFT), sgn)
    is_rt = compose(lbind(equality, RIGHT), sgn)
    is_ct = compose(lbind(equality, ORIGIN), sgn)
    tl = combine(sfilter(objs, is_tl), sfilter(objs, is_ct))
    tr = combine(sfilter(objs, is_tr), sfilter(objs, is_up))
    bl = combine(sfilter(objs, is_bl), sfilter(objs, is_lf))
    br = combine(sfilter(objs, is_br), combine(sfilter(objs, is_rt), sfilter(objs, is_dn)))
    tl_r = merge(apply(lbind(recolor, FOUR), tl))
    tr_r = merge(apply(lbind(recolor, TWO), tr))
    bl_r = merge(apply(lbind(recolor, THREE), bl))
    br_r = merge(apply(lbind(recolor, ONE), br))
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(paint(paint(base, tl_r), tr_r), bl_r), br_r)
    return O

def dsl3(I):
    # Color filtering: default all 8s to 1 then overlay quadrant masks from 8-subgrid halves
    P = ofcolor(I, EIGHT)
    base = replace(I, EIGHT, ZERO)
    O1 = fill(base, ONE, P)
    S = subgrid(P, I)
    hi = height(S); wi = width(S)
    miL = halve(hi); mjL = halve(wi)
    UL = ulcorner(P)
    TL = crop(S, ORIGIN, astuple(miL, mjL))
    TR = crop(S, astuple(ZERO, mjL), astuple(miL, subtract(wi, mjL)))
    BL = crop(S, astuple(miL, ZERO), astuple(subtract(hi, miL), mjL))
    BR = crop(S, astuple(miL, mjL), astuple(subtract(hi, miL), subtract(wi, mjL)))
    TLm = shift(asindices(TL), UL)
    TRm = shift(asindices(TR), add(UL, astuple(ZERO, mjL)))
    BLm = shift(asindices(BL), add(UL, astuple(miL, ZERO)))
    BRm = shift(asindices(BR), add(UL, astuple(miL, mjL)))
    O = fill(O1, FOUR, intersection(P, TLm))
    O = fill(O, TWO, intersection(P, TRm))
    O = fill(O, THREE, intersection(P, BLm))
    # BR remains as default ONE
    return O

def dsl4(I):
    # Geometric: synthesize a colored 2x2 canvas on the 8-box and mask by 8 pixels
    P = ofcolor(I, EIGHT)
    S = subgrid(P, I)
    hi = height(S); wi = width(S)
    miL = halve(hi); mjL = halve(wi)
    TLg = canvas(FOUR, astuple(miL, mjL))
    TRg = canvas(TWO, astuple(miL, subtract(wi, mjL)))
    BLg = canvas(THREE, astuple(subtract(hi, miL), mjL))
    BRg = canvas(ONE, astuple(subtract(hi, miL), subtract(wi, mjL)))
    top = hconcat(TLg, TRg)
    bot = hconcat(BLg, BRg)
    Sc = vconcat(top, bot)
    UL = ulcorner(P)
    base = replace(I, EIGHT, ZERO)
    C4 = shift(ofcolor(Sc, FOUR), UL)
    C2 = shift(ofcolor(Sc, TWO), UL)
    C3 = shift(ofcolor(Sc, THREE), UL)
    C1 = shift(ofcolor(Sc, ONE), UL)
    O = fill(base, FOUR, intersection(P, C4))
    O = fill(O, TWO, intersection(P, C2))
    O = fill(O, THREE, intersection(P, C3))
    O = fill(O, ONE, intersection(P, C1))
    return O

def dsl5(I):
