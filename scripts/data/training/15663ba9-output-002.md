def dsl1(I):
    # Global directional-endpoints: mark horizontal endpoints with 4, vertical endpoints with 2
    c = leastcolor(I)
    S = ofcolor(I, c)
    hasL = intersection(S, shift(S, RIGHT))
    hasR = intersection(S, shift(S, LEFT))
    hasU = intersection(S, shift(S, DOWN))
    hasD = intersection(S, shift(S, UP))
    hendL = difference(hasR, hasL)
    hendR = difference(hasL, hasR)
    vendU = difference(hasD, hasU)
    vendD = difference(hasU, hasD)
    hend = combine(hendL, hendR)
    vend = combine(vendU, vendD)
    G = fill(I, FOUR, hend)
    O = fill(G, TWO, vend)
    return O

def dsl2(I):

    c = leastcolor(I)

    straightv = intersection(U, D)
    corners = intersection(combine(L, R), combine(U, D))

    endpoints = difference(S, deg2)
    G = fill(I, FOUR, corners)
    O = fill(G, TWO, endpoints)
    return O

def dsl3(I):

    c = leastcolor(I)

    U = intersection(S, shift(S, DOWN))

    rd = intersection(R, D)

    G2 = fill(G1, FOUR, endpoints)
