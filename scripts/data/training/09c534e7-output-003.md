def dsl1(I):
    H = hmirror(I)
    O1 = paint(I, asobject(H))
    V = vmirror(O1)
    O = paint(O1, asobject(V))
    return O

def dsl2(I):
    FG = fgpartition(I)
    Seeds = difference(FG, colorfilter(FG, ONE))
    ones = ofcolor(I, ONE)
    outlinef = compose(box, toindices)
    maskf = compose(lbind(intersection, ones), outlinef)
    recolorf = fork(recolor, color, maskf)
    Paints = mapply(recolorf, Seeds)
    O = paint(I, Paints)
    return O

def dsl3(I):
    mc = mostcolor(I)
    O = replace(I, ONE, mc)
    return O

def dsl4(I):
    R = rot90(I)
    O = paint(I, asobject(R))
    return O

def dsl5(I):

    maskf = compose(lbind(intersection, ones), bboxf)
