def dsl1(I):
    BG = ofcolor(I, ZERO)
    FG = difference(asindices(I), BG)
    U = toobject(FG, I)
    G = subgrid(U, I)
    H = height(I)
    W = width(I)
    HG = height(G)
    WG = width(G)
    Zb = canvas(ZERO, astuple(subtract(H, HG), WG))
    T = vconcat(G, Zb)
    Zr = canvas(ZERO, astuple(H, subtract(W, WG)))
    O = hconcat(T, Zr)
    return O

def dsl3(I):
    A = objects(I, T, F, T)
    FBB = compose(inbox, toindices)
    N = compose(normalize, FBB)
    FRC = fork(recolor, color, N)
    R = apply(FRC, A)
    U = merge(R)
    Z = canvas(ZERO, shape(I))
    O = paint(Z, U)
    return O
