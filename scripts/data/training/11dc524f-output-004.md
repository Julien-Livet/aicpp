def dsl1(I):
    dims = shape(I)
    bg = canvas(SEVEN, dims)
    s2 = ofcolor(I, TWO)
    s5 = ofcolor(I, FIVE)
    v = gravitate(s2, s5)
    s2p = shift(s2, multiply(v, TWO))
    mh = hmirror(s2p)
    mv = vmirror(s2p)
    c5 = centerofmass(s5)
    ch = centerofmass(mh)
    cv = centerofmass(mv)
    posh = shift(mh, subtract(c5, ch))
    posv = shift(mv, subtract(c5, cv))
    best = branch(adjacent(posh, s5), posh, posv)
    O = paint(paint(bg, recolor(TWO, s2p)), recolor(FIVE, best))
    return O

def dsl2(I):
    dims = shape(I)
    bg = canvas(SEVEN, dims)
    objs = objects(I, T, F, T)
    o2 = first(colorfilter(objs, TWO))
    o5 = first(colorfilter(objs, FIVE))
    s2 = toindices(o2)
    s5 = toindices(o5)
    v = gravitate(s2, s5)
    s2p = shift(s2, multiply(v, TWO))
    rel = position(s2p, s5)
    off = invert(rel)
    mh = hmirror(s2p)
    mv = vmirror(s2p)
    ph = shift(mh, off)
    pv = shift(mv, off)
    cset = insert(pv, initset(ph))
    best = argmin(cset, lbind(manhattan, s5))
    O = paint(paint(bg, recolor(TWO, s2p)), recolor(FIVE, best))
    return O
