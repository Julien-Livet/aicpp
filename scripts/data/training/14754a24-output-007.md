def dsl1(I):
    # Relational/structural: 5s that lie on a straight pure path to a 4 and also to a 0, plus 5s strictly between two 4s.
    p4 = ofcolor(I, FOUR)
    p5 = ofcolor(I, FIVE)
    p0 = ofcolor(I, ZERO)
    linear = fork(either, hline, vline)
    # 5 <-> 4
    paths54 = prapply(connect, p5, p4)
    allowed54 = merge(insert(p5, initset(p4)))
    pure54 = compose(lbind(equality, ZERO), compose(size, rbind(difference, allowed54)))
    good54 = mfilter(paths54, fork(both, pure54, linear))
    mask54 = intersection(merge(good54), p5)
    # 5 <-> 0
    paths50 = prapply(connect, p5, p0)
    allowed50 = merge(insert(p5, initset(p0)))
    pure50 = compose(lbind(equality, ZERO), compose(size, rbind(difference, allowed50)))
    good50 = mfilter(paths50, fork(both, pure50, linear))
    mask50 = intersection(merge(good50), p5)
    between40 = intersection(mask54, mask50)
    # 4 <-> 4
    paths44 = prapply(connect, p4, p4)
    allowed44 = merge(insert(p4, initset(p5)))
    pure44 = compose(lbind(equality, ZERO), compose(size, rbind(difference, allowed44)))
    good44 = mfilter(paths44, fork(both, pure44, linear))
    mask44 = intersection(merge(good44), p5)
    mask = combine(between40, mask44)
    O = fill(I, TWO, mask)
    return O

def dsl3(I):
    # Color-filtering (local): 5s immediately flanked by 4s or by a 0 and a 4, horizontally or vertically.
    p4 = ofcolor(I, FOUR)
    p5 = ofcolor(I, FIVE)
    p0 = ofcolor(I, ZERO)
    # 4-5-4
    midH44 = intersection(intersection(p5, shift(p4, LEFT)), shift(p4, RIGHT))
    midV44 = intersection(intersection(p5, shift(p4, UP)), shift(p4, DOWN))
    # 0-5-4 and 4-5-0
    midH04 = intersection(intersection(p5, shift(p0, LEFT)), shift(p4, RIGHT))
    midH40 = intersection(intersection(p5, shift(p4, LEFT)), shift(p0, RIGHT))
    midV04 = intersection(intersection(p5, shift(p0, UP)), shift(p4, DOWN))
    midV40 = intersection(intersection(p5, shift(p4, UP)), shift(p0, DOWN))
    mask1 = combine(midH44, midV44)
    mask2 = combine(combine(midH04, midH40), combine(midV04, midV40))
    mask = combine(mask1, mask2)
    O = fill(I, TWO, mask)
    return O
