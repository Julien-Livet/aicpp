def dsl1(I):
    # 4-to-5 pure axial connectors: contiguous 5-rays from each 4
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    segs = prapply(connect, fou, fiv)
    axial = sfilter(segs, fork(either, hline, vline))
    allowed = combine(fou, fiv)
    pure = sfilter(axial, matcher(compose(size, rbind(difference, allowed)), ZERO))
    mask = merge(pure)
    target = intersection(mask, fiv)
    O = fill(I, TWO, target)
    return O

def dsl2(I):
    # Line-shaped 5-objects that touch any 4
    fou = ofcolor(I, FOUR)
    fiveobjs = colorfilter(objects(I, T, F, T), FIVE)
    touching = sfilter(fiveobjs, lbind(adjacent, fou))
    lines = sfilter(touching, fork(either, hline, vline))
    obj = mapply(lbind(recolor, TWO), lines)
    O = paint(I, obj)
    return O

def dsl3(I):
    # Expand from 5-seeds adjacent to 4s along pure axial 5-runs
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    nbh = mapply(neighbors, fou)
    seeds = intersection(nbh, fiv)
    segs = prapply(connect, seeds, fiv)
    axial = sfilter(segs, fork(either, hline, vline))
    pure = sfilter(axial, matcher(compose(size, rbind(difference, fiv)), ZERO))
    mask = merge(pure)
    target = intersection(mask, fiv)
    O = fill(I, TWO, target)
    return O

def dsl4(I):
    # Intersections of all 4-rows with all 4-columns on 5s
    fou = ofcolor(I, FOUR)
    rows = mapply(hfrontier, fou)
    cols = mapply(vfrontier, fou)
    cross = intersection(rows, cols)
    target = intersection(cross, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O

def dsl5(I):
    # 4-to-0 axial connectors where path is only 4/5 until the terminal 0
    fou = ofcolor(I, FOUR)
    zer = ofcolor(I, ZERO)
    segs = prapply(connect, fou, zer)
    axial = sfilter(segs, fork(either, hline, vline))
    allowed = combine(fou, ofcolor(I, FIVE))
    tip = sfilter(axial, matcher(compose(size, rbind(difference, allowed)), ONE))
    mask = merge(tip)
    target = intersection(mask, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O
