def dsl1(I):
    # Crop one-pixel outer border
    O = crop(I, UNITY, subtract(shape(I), TWO_BY_TWO))
    return O

def dsl2(I):
    # Crop to the bounding box of the largest object
    objs = objects(I, T, F, T)
    big = argmax(objs, size)
    O = subgrid(backdrop(big), I)
    return O

def dsl3(I):
    # Crop to the bounding box of the most common nonzero color
    cols = remove(ZERO, palette(I))
    mc = argmax(cols, lbind(colorcount, I))
    obj = toobject(ofcolor(I, mc), I)
    O = subgrid(backdrop(obj), I)
    return O

def dsl4(I):
    # Interior defined by grid frontiers
    fr = frontiers(I)
    bd = merge(fr)
    inner = delta(bd)
    O = subgrid(inner, I)
    return O

def dsl5(I):
    # Keep cells equal to their horizontal mirror (symmetry core)
    O = cellwise(I, hmirror(I), ZERO)
    return O
