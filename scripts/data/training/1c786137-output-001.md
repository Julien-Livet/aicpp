def dsl2(I):
    OBJS = objects(I, T, F, T)

    O = switch(I, c, ZERO)

def dsl5(I):
    OBJS = objects(I, T, F, T)
    L = argmin(OBJS, leftmost)
    R = argmax(OBJS, rightmost)
    pL = center(L)
    pR = center(R)
    seg = connect(pL, pR)
    line = recolor(THREE, seg)
    Z = canvas(ZERO, shape(I))
    O = paint(Z, line)
