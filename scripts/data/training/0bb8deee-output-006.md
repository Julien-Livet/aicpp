def dsl1(I):
    P = remove(ZERO, palette(I))
    mc = argmax(P, lbind(colorcount, I))
    cross = ofcolor(I, mc)
    objs = objects(I, T, F, T)
    others = difference(objs, colorfilter(objs, mc))
    othersidx = toindices(merge(others))
    c = centerofmass(cross)
    sgn = compose(sign, rbind(subtract, c))
    TLidx = sfilter(othersidx, matcher(sgn, NEG_UNITY))
    TRidx = sfilter(othersidx, matcher(sgn, UP_RIGHT))
    BLidx = sfilter(othersidx, matcher(sgn, DOWN_LEFT))
    BRidx = sfilter(othersidx, matcher(sgn, UNITY))
    TLg = compress(subgrid(TLidx, I))
    TRg = compress(subgrid(TRidx, I))
    BLg = compress(subgrid(BLidx, I))
    BRg = compress(subgrid(BRidx, I))
    O = compress(vconcat(hconcat(TLg, TRg), hconcat(BLg, BRg)))
    return O
