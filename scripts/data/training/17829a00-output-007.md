def dsl1(I):
    # Half-wise vertical ray casting from all non-background cells to nearest border, recolor by border colors, exclude border rows, then redraw border rows
    BG = mostcolor(I)
    H = height(I)
    Half = halve(H)
    TT = interval(ZERO, Half, ONE)
    TB = interval(Half, H, ONE)
    FT = mapply(hfrontier, apply(toivec, TT))
    FB = mapply(hfrontier, apply(toivec, TB))
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    cT = color(argmin(BOR, uppermost))
    cB = color(argmax(BOR, lowermost))
    ALL = asindices(I)
    NB = difference(ALL, ofcolor(I, BG))
    Tcells = intersection(NB, FT)
    Bcells = intersection(NB, FB)
    RU = mapply(rbind(shoot, UP), Tcells)
    RD = mapply(rbind(shoot, DOWN), Bcells)
    R0 = hfrontier(astuple(ZERO, ZERO))
    Rb = hfrontier(astuple(subtract(H, ONE), ZERO))
    Tproj = difference(RU, R0)
    Bproj = difference(RD, Rb)
    C = canvas(BG, shape(I))
    O0 = fill(C, cT, Tproj)
    O1 = fill(O0, cB, Bproj)
    O2 = fill(O1, cT, R0)
    O = fill(O2, cB, Rb)
    return O
