def dsl1(I):
    # Global via frontiers: restrict tips to frontier cells; neighbor-inside gets 2
    c = leastcolor(I)
    S = ofcolor(I, c)
    Fobjs = frontiers(I)
    Fparts = apply(toindices, Fobjs)
    F = merge(Fparts)
    B = intersection(F, S)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    H = intersection(L, R)
    V = intersection(U, D)
    S1 = combine(combine(L, R), combine(U, D))
    LU = intersection(L, U)
    LD = intersection(L, D)
    RU = intersection(R, U)
    RD = intersection(R, D)
    B2 = combine(combine(LU, LD), combine(RU, RD))
    U2 = combine(H, V)
    U2p = combine(U2, B2)
    Eall = difference(S1, U2p)
    E = intersection(Eall, B)
    N1 = combine(shift(E, LEFT), shift(E, RIGHT))
    N2 = combine(shift(E, UP), shift(E, DOWN))
    N = intersection(combine(N1, N2), S)
    G = fill(I, FOUR, E)
    O = fill(G, TWO, N)
    return O
