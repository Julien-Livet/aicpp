def dsl1(I):
    S = compress(subgrid(ofcolor(I, TWO), I))
    P = ofcolor(S, TWO)
    C = centerofmass(P)
    A = astuple(THREE, THREE)
    ST = subtract(C, A)
    O = crop(S, ST, astuple(SEVEN, SEVEN))
    return O

def dsl2(I):
    P = toobject(ofcolor(I, TWO), I)
    N = normalize(P)
    C = center(N)
    OFS = subtract(astuple(THREE, THREE), C)
    R = recolor(TWO, N)
    S = shift(R, OFS)
    G = canvas(ZERO, astuple(SEVEN, SEVEN))
    O = paint(G, S)
    return O

def dsl3(I):
    G0 = canvas(ZERO, shape(I))
    R = recolor(TWO, toobject(ofcolor(I, TWO), I))
    S = paint(G0, R)
    T = trim(S)
    C = compress(T)
    H = height(C)
    W = width(C)
    FH = branch(greater(H, SEVEN), divide(H, SEVEN), ONE)
    FW = branch(greater(W, SEVEN), divide(W, SEVEN), ONE)
    F = branch(greater(FH, FW), FH, FW)
    D = downscale(C, F)
    O = crop(D, ORIGIN, astuple(SEVEN, SEVEN))
    return O

def dsl4(I):
    P = ofcolor(I, TWO)
    B = box(P)
    R = recolor(TWO, toobject(B, I))
    G0 = canvas(ZERO, shape(I))
    S = paint(G0, R)
    C = compress(S)
    O = crop(C, ORIGIN, astuple(SEVEN, SEVEN))
    return O

def dsl5(I):
    S = compress(subgrid(ofcolor(I, TWO), I))
    L = first(vsplit(S, TWO))
    R = last(vsplit(S, TWO))
    LT = first(hsplit(L, TWO))
    LB = last(hsplit(L, TWO))
    RT = first(hsplit(R, TWO))
    RB = last(hsplit(R, TWO))
    CLT = compress(LT)
    CLB = compress(LB)
    CRT = compress(RT)
    CRB = compress(RB)
    OLT = recolor(TWO, normalize(toobject(ofcolor(CLT, TWO), CLT)))
    OLB = recolor(TWO, normalize(toobject(ofcolor(CLB, TWO), CLB)))
    ORT = recolor(TWO, normalize(toobject(ofcolor(CRT, TWO), CRT)))
    ORB = recolor(TWO, normalize(toobject(ofcolor(CRB, TWO), CRB)))
    OX = tojvec(FOUR)
    OY = toivec(FOUR)
    OD = add(OX, OY)
    SLT = shift(OLT, ORIGIN)
    SRT = shift(ORT, OX)
    SLB = shift(OLB, OY)
    SRB = shift(ORB, OD)
    U1 = combine(SLT, SRT)
    U2 = combine(SLB, SRB)
    U = combine(U1, U2)
    G = canvas(ZERO, astuple(SEVEN, SEVEN))
    O = paint(G, U)
    return O
