def dsl1(I):
    # Color filtering + local expansion: build 5x5 masks around least-color centers, then lay plus and diagonals
    c = leastcolor(I)
    centers = ofcolor(I, c)
    nb = remove(ZERO, palette(I))
    arm = other(nb, c)
    n1 = mapply(neighbors, centers)
    d1 = mapply(ineighbors, centers)
    r1 = combine(combine(centers, n1), d1)
    n2 = mapply(neighbors, r1)
    d2 = mapply(ineighbors, r1)
    mask = combine(combine(r1, n2), d2)
    plus = combine(mapply(hfrontier, centers), mapply(vfrontier, centers))
    plus = intersection(plus, mask)
    dirs1 = insert(NEG_UNITY, initset(UNITY))
    dirs2 = insert(DOWN_LEFT, initset(UP_RIGHT))
    diag = combine(prapply(shoot, centers, dirs1), prapply(shoot, centers, dirs2))
    diag = intersection(diag, mask)
    O = canvas(ZERO, shape(I))
    O = fill(O, arm, plus)
    O = fill(O, c, diag)
    return O

def dsl2(I):
    # Object extraction: get center objects, expand masks, then paint recolored line objects
    c = leastcolor(I)
    objs = partition(I)
    cobjs = colorfilter(objs, c)
    centers = mapply(toindices, cobjs)
    nb = remove(ZERO, palette(I))
    arm = other(nb, c)
    n1 = mapply(neighbors, centers)
    d1 = mapply(ineighbors, centers)
    r1 = combine(combine(centers, n1), d1)
    n2 = mapply(neighbors, r1)
    d2 = mapply(ineighbors, r1)
    mask = combine(combine(r1, n2), d2)
    h = mapply(hfrontier, centers)
    v = mapply(vfrontier, centers)
    plus = intersection(combine(h, v), mask)
    dirs1 = insert(NEG_UNITY, initset(UNITY))
    dirs2 = insert(DOWN_LEFT, initset(UP_RIGHT))
    diag = intersection(combine(prapply(shoot, centers, dirs1), prapply(shoot, centers, dirs2)), mask)
    O = canvas(ZERO, shape(I))
    O = paint(O, recolor(arm, plus))
    O = paint(O, recolor(c, diag))
    return O

def dsl3(I):
    # Global-style mask growth via functional expansion; same pattern drawing with different construction
    c = leastcolor(I)
    centers = ofcolor(I, c)
    nb = remove(ZERO, palette(I))
    arm = other(nb, c)
    g = lbind(mapply, neighbors)
    h = lbind(mapply, ineighbors)
    r1 = combine(combine(centers, g(centers)), h(centers))
    r2 = combine(combine(r1, g(r1)), h(r1))
    plus = intersection(combine(mapply(hfrontier, centers), mapply(vfrontier, centers)), r2)
    dirsA = insert(NEG_UNITY, initset(UNITY))
    dirsB = insert(DOWN_LEFT, initset(UP_RIGHT))
    diag = intersection(combine(prapply(shoot, centers, dirsA), prapply(shoot, centers, dirsB)), r2)
    O = canvas(ZERO, shape(I))
    O = underfill(O, arm, plus)
    O = fill(O, c, diag)
    return O
