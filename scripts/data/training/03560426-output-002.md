def dsl1(I):
    C = compress(I)
    R = rot270(C)
    H = height(I)
    W = width(I)
    HR = height(R)
    WR = width(R)
    Db = astuple(subtract(H, HR), WR)
    Zb = canvas(ZERO, Db)
    T = vconcat(R, Zb)
    Dr = astuple(H, subtract(W, WR))
    Zr = canvas(ZERO, Dr)
    O = hconcat(T, Zr)
    return O

def dsl2(I):
    BG = ofcolor(I, ZERO)

    H = height(I)
    W = width(I)

def dsl3(I):
    P = palette(I)
    Cset = remove(ZERO, P)
    F = lbind(colorcount, I)
    C = valmax(Cset, F)
    Ind = ofcolor(I, C)
    Obj = toobject(Ind, I)
    N = normalize(Obj)
    Z = canvas(ZERO, shape(I))
    O = paint(Z, N)
    return O

def dsl5(I):
    A = objects(I, T, F, T)
    N = apply(normalize, A)
    L = apply(leftmost, A)
    V = apply(tojvec, L)
    S = mpapply(shift, N, V)
    U = merge(S)
