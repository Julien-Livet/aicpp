def dsl1(I):
    T = trim(I)
    R = rot270(T)
    hI = height(I)
    wI = width(I)
    hR = height(R)
    wR = width(R)
    rightpad = canvas(ZERO, astuple(hR, subtract(wI, wR)))
    toprow = hconcat(R, rightpad)
    bottompad = canvas(ZERO, astuple(subtract(hI, hR), wI))
    O = vconcat(toprow, bottompad)
    return O

def dsl2(I):
    objs = fgpartition(I)
    norms = apply(normalize, objs)
    packed = merge(norms)
    base = canvas(ZERO, shape(I))
    O = paint(base, packed)
    return O

def dsl3(I):
    colors = remove(ZERO, palette(I))
    comp = lbind(colorcount, I)
    c = valmax(colors, comp)
    mask = ofcolor(I, c)
    obj = recolor(c, mask)
    base = canvas(ZERO, shape(I))
    onlyc = paint(base, obj)
    C = compress(onlyc)
    hI = height(I)
    wI = width(I)
    hC = height(C)
    wC = width(C)
    rightpad = canvas(ZERO, astuple(hC, subtract(wI, wC)))
    toprow = hconcat(C, rightpad)
    bottompad = canvas(ZERO, astuple(subtract(hI, hC), wI))
    O = vconcat(toprow, bottompad)
    return O

def dsl4(I):

def dsl5(I):
    allidx = asindices(I)
    zeros = ofcolor(I, ZERO)
    nonzeros = difference(allidx, zeros)
    colors = remove(ZERO, palette(I))
    comp = lbind(colorcount, I)
    c = valmax(colors, comp)
    mask_c = ofcolor(I, c)
    anchor = toobject(mask_c, I)
    others_mask = difference(nonzeros, mask_c)
    others = toobject(others_mask, I)
    direction = gravitate(others, anchor)
    moved_others = shift(others, direction)
    base = canvas(ZERO, shape(I))
    combined = combine(anchor, moved_others)
    O = paint(base, combined)
    return O
