def dsl1(I):
    dims = shape(I)
    blank = canvas(ZERO, dims)
    objs = objects(I, T, F, T)
    isline = fork(either, hline, vline)
    isborder = lbind(bordering, I)
    keepcond = fork(both, isline, isborder)
    keep = sfilter(objs, keepcond)
    finalobj = merge(keep)
    O = paint(blank, finalobj)
    return O

def dsl4(I):
    dims = shape(I)
    blank = canvas(ZERO, dims)
    objs = objects(I, T, F, T)
    cols = remove(ZERO, palette(I))
    percolor = apply(lbind(colorfilter, objs), cols)
    anchors = apply(rbind(argmax, size), percolor)
    rings = apply(compose(outbox, toindices), anchors)
    recol_rings = mapply(rbind(recolor, compose(color, identity)), rings)
    base = merge(anchors)
    union = combine(base, recol_rings)
    O = paint(blank, union)
    return O

def dsl5(I):
    dims = shape(I)
    blank = canvas(ZERO, dims)
    objs = objects(I, T, F, T)
    cols = remove(ZERO, palette(I))
    percolor = apply(lbind(colorfilter, objs), cols)
    anchors = apply(rbind(argmax, size), percolor)
    # pair color-objects with their anchor
    pairs = pair(percolor, anchors)
    # for each (color-objects, anchor): shift all other objects of the same color adjacent to the anchor
    def per_pair(cobjs_anchor):
        cobjs = first(cobjs_anchor)
        anch = last(cobjs_anchor)
        extras = remove(anch, cobjs)
        to_adj = mapply(fork(shift, identity, rbind(gravitate, anch)), extras)
        return combine(anch, to_adj)
    parts = apply(per_pair, pairs)
    finalobj = merge(parts)
    O = paint(blank, finalobj)
    return O
