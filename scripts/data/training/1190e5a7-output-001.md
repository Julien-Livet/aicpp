def dsl1(I):
    SC = leastcolor(I)
    BG = mostcolor(I)
    H = height(I)
    W = width(I)
    RH = astuple(ONE, W)
    CH = astuple(H, ONE)
    Rg = canvas(SC, RH)
    Cg = canvas(SC, CH)
    RO = asobject(Rg)
    CO = asobject(Cg)
    RC = size(occurrences(I, RO))
    CC = size(occurrences(I, CO))
    HD = add(RC, ONE)
    WD = add(CC, ONE)
    D = astuple(HD, WD)
    O = canvas(BG, D)
    return O

def dsl5(I):
    SC = leastcolor(I)
    BG = mostcolor(I)
    H = height(I)
    W = width(I)
    S = ofcolor(I, SC)
    s = colorcount(I, SC)
    L0 = first(asindices(I))
    HR = hfrontier(L0)
    C = size(intersection(S, HR))
    CH = multiply(C, H)
    N = subtract(s, CH)
    Dn = subtract(W, C)
    R = divide(N, Dn)
    HD = add(R, ONE)
    WD = add(C, ONE)
    D = astuple(HD, WD)
    O = canvas(BG, D)
    return O
