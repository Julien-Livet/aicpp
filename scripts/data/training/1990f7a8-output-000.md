def dsl1(I):
    # Global cropping + quadrant splitting
    m = ofcolor(I, TWO)
    bb = subgrid(m, I)
    tb = hsplit(bb, 2)
    top = first(tb)
    bottom = last(tb)
    tltr = vsplit(top, 2)
    blbr = vsplit(bottom, 2)
    tl = first(tltr)
    tr = last(tltr)
    bl = first(blbr)
    br = last(blbr)
    gtl = subgrid(ofcolor(tl, TWO), tl)
    gtr = subgrid(ofcolor(tr, TWO), tr)
    gbl = subgrid(ofcolor(bl, TWO), bl)
    gbr = subgrid(ofcolor(br, TWO), br)
