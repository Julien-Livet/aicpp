def dsl1(I):
    # Global: mirror bounding subgrid horizontally and place back
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    fg = difference(allidx, ofcolor(J, FOUR))
    S = subgrid(fg, J)
    sidx = asindices(S)
    sfg = difference(sidx, ofcolor(S, FOUR))
    Gs = paint(canvas(FOUR, shape(S)), toobject(sfg, S))
    Hs = hmirror(Gs)
    src = toobject(difference(asindices(Hs), ofcolor(Hs, FOUR)), Hs)
    u = ulcorner(fg)
    O = move(canvas(FOUR, shape(J)), src, u)
    return O

def dsl2(I):
    # Object extraction: mirror the leftmost object horizontally
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    s = argmin(objs, leftmost)
    base = canvas(FOUR, shape(J))
    G = paint(base, s)
    M = hmirror(G)
    O = underpaint(G, asobject(M))
    return O

def dsl3(I):
    # Color filtering: mirror the most-common non-bg color horizontally
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    cols = apply(color, objs)
    cstar = mostcommon(totuple(cols))
    foc = colorfilter(objs, cstar)
    merged = merge(foc)
    src = toobject(merged, J)
    G = paint(canvas(FOUR, shape(J)), src)
    H = hmirror(G)
    O = underpaint(G, asobject(H))
    return O

def dsl4(I):
    # Geometric: enforce fourfold rotational symmetry
    J = replace(I, FIVE, FOUR)
    fg = difference(asindices(J), ofcolor(J, FOUR))
    src = toobject(fg, J)
    G = paint(canvas(FOUR, shape(J)), src)
    R90 = rot90(G)
    R180 = rot180(G)
    R270 = rot270(G)
    O1 = underpaint(G, asobject(R90))
    O2 = underpaint(O1, asobject(R180))
    O = underpaint(O2, asobject(R270))
    return O

def dsl5(I):
    # Relational: gravitate foreground to the vertical frontier through its own center of mass
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    fg = difference(allidx, ofcolor(J, FOUR))
    src = toobject(fg, J)
    c = centerofmass(fg)
    vline = vfrontier(c)
    v = gravitate(fg, vline)
    O = move(canvas(FOUR, shape(J)), src, v)
    return O
