def dsl1(I):
    u = merge(objects(I, T, F, T))
    ui = toindices(u)
    bd = backdrop(ui)
    inside = difference(bd, ui)
    col = color(u)
    stepw = hperiod(u)
    steph = vperiod(u)
    w = width(bd)
    h = height(bd)
    ks = interval(0, w, stepw)
    iset = interval(0, h, steph)
    hvecs = apply(tojvec, ks)
    ivecs = apply(toivec, iset)
    ul = ulcorner(inside)
    ur = urcorner(inside)
    ll = llcorner(inside)
    lr = lrcorner(inside)
    startsUR = combine(apply(lbind(add, ll), hvecs), apply(lbind(add, ul), ivecs))
    raysUR = merge(apply(rbind(shoot, UP_RIGHT), startsUR))
    startsDL = combine(apply(lbind(add, ur), apply(invert, hvecs)), apply(lbind(add, lr), apply(invert, ivecs)))
    raysDL = merge(apply(rbind(shoot, DOWN_LEFT), startsDL))
    mask = intersection(combine(raysUR, raysDL), inside)
    O = underpaint(I, recolor(col, mask))
    return O

def dsl2(I):
    u = merge(objects(I, T, F, T))
    ui = toindices(u)
    bd = backdrop(ui)
    inside = difference(bd, ui)
    col = color(u)
    g = paint(canvas(ZERO, shape(I)), recolor(col, inside))
    cells = objects(g, T, F, T)
    bdf = compose(backdrop, toindices)
    ulf = compose(ulcorner, bdf)
    lrf = compose(lrcorner, bdf)
    urf = compose(urcorner, bdf)
    llf = compose(llcorner, bdf)
    line1 = fork(connect, ulf, lrf)
    line2 = fork(connect, urf, llf)
    a = merge(apply(line1, cells))
    b = merge(apply(line2, cells))
    mask = intersection(combine(a, b), inside)
    O = underpaint(I, recolor(col, mask))
    return O
