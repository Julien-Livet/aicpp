def dsl1(I):
    val = leastcolor(I)
    m = ofcolor(I, val)
    rlines = merge(apply(hfrontier, m))
    clines = merge(apply(vfrontier, m))
    cross = combine(rlines, clines)
    target = intersection(cross, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl5(I):

    parts = fgpartition(I)
