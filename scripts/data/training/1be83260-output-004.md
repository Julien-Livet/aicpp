def dsl1(I):
    # Fill internal zeros with the most common color inside the cropped foreground
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    O = fill(G, mostcolor(G), ofcolor(G, ZERO))
    return O

def dsl2(I):
    # Rebuild on a canvas of the most common color, then repaint the foreground
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    base = canvas(mostcolor(G), shape(G))
    O = paint(base, asobject(G))
    return O

def dsl3(I):
    # Direct replacement of zeros by the most common color in the cropped region
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    O = replace(G, ZERO, mostcolor(G))
    return O

def dsl4(I):
    # Geometric split: fill zeros in each vertical half with the most common color, then recombine
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    parts = vsplit(G, 2)
    L = first(parts)
    R = last(parts)
    L2 = replace(L, ZERO, mostcolor(G))
    R2 = replace(R, ZERO, mostcolor(G))
    O = hconcat(L2, R2)
    return O

def dsl5(I):
    # Relational fill: use the backdrop of the foreground to underfill background with the modal color
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    bg = backdrop(asobject(G))
    O = underfill(G, mostcolor(G), bg)
    return O
