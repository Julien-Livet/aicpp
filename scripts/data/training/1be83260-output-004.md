def dsl1(I):
    # Global bandwise fill: crop, compress, then fill each half with its dominant color while preserving details
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    C = compress(A)
    TOP = tophalf(C)
    BOT = bottomhalf(C)
    DT = mostcolor(TOP)
    DB = mostcolor(BOT)
    FT = underfill(TOP, DT, asindices(TOP))
    FB = underfill(BOT, DB, asindices(BOT))
    O = vconcat(FT, FB)
    return O

def dsl2(I):
    # Relational expansion: crop, keep non-dominant objects, expand them to their backdrops, paint on dominant canvas
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    DOM = mostcolor(A)
    BASE = canvas(DOM, shape(A))
    OBJS = objects(A, T, F, T)
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    TOT = totuple(OTH)
    COLS = totuple(apply(color, TOT))
    PATS = totuple(apply(compose(backdrop, toindices), TOT))
    COLORED = mpapply(recolor, COLS, PATS)
    EXP = merge(COLORED)
    O = paint(BASE, EXP)
    return O

def dsl3(I):
    # Color filtering per half: crop, then for each half fill zeros with that half's dominant color
    NZ = difference(asindices(I), ofcolor(I, ZERO))

    TOP = tophalf(A)
    BOT = bottomhalf(A)
    FT = underfill(TOP, mostcolor(TOP), asindices(TOP))
    FB = underfill(BOT, mostcolor(BOT), asindices(BOT))
    O = vconcat(FT, FB)
    return O

def dsl4(I):
    # Geometric composition: crop, compress, fill each half with its dominant color, mirror the bottom half
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    C = compress(A)
    TOP = tophalf(C)
    BOT = bottomhalf(C)
    FT = underfill(TOP, mostcolor(TOP), asindices(TOP))
    FB = underfill(BOT, mostcolor(BOT), asindices(BOT))
    MB = hmirror(FB)
    O = vconcat(FT, MB)
    return O
