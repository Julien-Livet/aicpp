def dsl1(I):
    # Global: 5s at intersections of any 4-row and any 4-column
    fours = ofcolor(I, FOUR)
    rows = merge(mapply(hfrontier, fours))
    cols = merge(mapply(vfrontier, fours))
    mask = intersection(rows, cols)
    target = intersection(mask, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O

def dsl2(I):
    # Object extraction: recolor 5-objects adjacent to any 4
    fou = ofcolor(I, FOUR)
    fiveobjs = colorfilter(objects(I, T, F, T), FIVE)
    touching = sfilter(fiveobjs, lbind(adjacent, fou))
    target = merge(mapply(toindices, touching))
    O = fill(I, TWO, target)
    return O

def dsl3(I):
    # Relational: 5s lying strictly between pairs of 4s in same row/col
    fours = ofcolor(I, FOUR)
    segs = sfilter(prapply(connect, fours, fours), fork(either, hline, vline))
    allowed = combine(fours, ofcolor(I, FIVE))
    pure = sfilter(segs, matcher(compose(size, rbind(difference, allowed)), ZERO))
    mask = merge(pure)
    target = intersection(mask, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O

def dsl4(I):
    # Geometric: 5s on rectangular outlines of 4-objects
    fourobjs = colorfilter(objects(I, T, F, T), FOUR)
    outlines = merge(mapply(box, fourobjs))
    target = intersection(outlines, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O

def dsl5(I):
    # Color filtering: 5s in any column that contains a 4
    cols = merge(mapply(vfrontier, ofcolor(I, FOUR)))
    target = intersection(cols, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O
