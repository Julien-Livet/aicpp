def dsl2(I):
    p4 = ofcolor(I, FOUR)
    p5 = ofcolor(I, FIVE)
    allowed = combine(p4, p5)
    paths = prapply(connect, p4, p5)
    pure = compose(lbind(equality, ZERO), compose(size, rbind(difference, allowed)))
    linear = fork(either, hline, vline)
    cond = fork(both, pure, linear)
    good = mfilter(paths, cond)
    mask = intersection(merge(good), p5)
    O = fill(I, TWO, mask)
    return O

def dsl3(I):
