def dsl1(I):
    # Global diagonal mask: 5s on counterdiagonal -> 2, other 5s -> 8
    A = asindices(I)
    UR = urcorner(A)
    LL = llcorner(A)
    AD = connect(UR, LL)
    S5 = ofcolor(I, FIVE)
    S2 = intersection(S5, AD)
    S8 = difference(S5, S2)
    O0 = fill(I, SEVEN, S5)
    O1 = fill(O0, EIGHT, S8)
    O = fill(O1, TWO, S2)
    return O

def dsl2(I):
    # Relational via vertical symmetry: 5s with a vertical mirror partner -> 8, others -> 2
    S5 = ofcolor(I, FIVE)
    Sm = vmirror(S5)
    Paired = intersection(S5, Sm)
    Singles = difference(S5, Paired)
    O0 = fill(I, SEVEN, S5)
    O1 = fill(O0, EIGHT, Paired)
    O = fill(O1, TWO, Singles)
    return O

def dsl3(I):
    # Color filtering: recolor all 5s to 8
    O = replace(I, FIVE, EIGHT)
    return O

def dsl4(I):
    # Object extraction: largest diagonal-connected 5-object -> 2, all other 5-objects -> 8
    OB = objects(I, T, T, T)
    OB5 = colorfilter(OB, FIVE)
    BIG = argmax(OB5, size)
    REST = remove(BIG, OB5)
    Sbig = toindices(BIG)
    Srest = merge(apply(toindices, REST))
    Sall = combine(Sbig, Srest)
    O0 = fill(I, SEVEN, Sall)
    O1 = fill(O0, EIGHT, Srest)
    O = fill(O1, TWO, Sbig)
    return O
