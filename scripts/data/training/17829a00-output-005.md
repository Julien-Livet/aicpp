def dsl1(I):
    # Object-wise gravitation to top/bottom, keep full moved shapes except the border rows, then restore border rows
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    TO = colorfilter(OB, cT)
    BO = colorfilter(OB, cB)
    R0 = hfrontier(astuple(ZERO, ZERO))
    H = height(I)
    Hm1 = subtract(H, ONE)
    Rb = hfrontier(astuple(Hm1, ZERO))
    ST = fork(shift, identity, rbind(gravitate, R0))
    SB = fork(shift, identity, rbind(gravitate, Rb))
    Tmov = apply(ST, TO)
    Bmov = apply(SB, BO)
    Itop = toindices(merge(Tmov))
    Ibot = toindices(merge(Bmov))
    Tproj = difference(Itop, R0)
    Bproj = difference(Ibot, Rb)
    BG = mostcolor(I)
    C = canvas(BG, shape(I))
    O0 = fill(C, cT, Tproj)
    O1 = fill(O0, cB, Bproj)
    O = paint(paint(O1, TOP), BOT)
    return O

def dsl3(I):
    # Rotational symmetry: solve top-only on I and on rot180(I), then rotate back and merge
    BG = mostcolor(I)
    # Top-half on I
    OB1 = objects(I, T, F, T)
    BOR1 = sfilter(OB1, rbind(bordering, I))
    TOP1 = argmin(BOR1, uppermost)
    c1 = color(TOP1)
    S1 = ofcolor(I, c1)
    O1obj = toobject(S1, I)
    R01 = hfrontier(astuple(ZERO, ZERO))
    ST1 = fork(shift, identity, rbind(gravitate, R01))
    M1 = merge(apply(ST1, initset(O1obj)))
    P1 = difference(toindices(M1), R01)
    Ct = canvas(BG, shape(I))
    Gt0 = fill(Ct, c1, P1)
    Gt = paint(Gt0, TOP1)
    # Top-only on rot180(I), then rotate back
    V = rot180(I)
    OB2 = objects(V, T, F, T)
    BOR2 = sfilter(OB2, rbind(bordering, V))
    TOP2 = argmin(BOR2, uppermost)
    c2 = color(TOP2)
    S2 = ofcolor(V, c2)
    O2obj = toobject(S2, V)
    R02 = hfrontier(astuple(ZERO, ZERO))
    ST2 = fork(shift, identity, rbind(gravitate, R02))
    M2 = merge(apply(ST2, initset(O2obj)))
    P2 = difference(toindices(M2), R02)
    Cv = canvas(mostcolor(V), shape(V))
    Gv0 = fill(Cv, c2, P2)
    Gv = paint(Gv0, TOP2)
    Gb = rot180(Gv)
    # Merge layers
    O0 = canvas(BG, shape(I))
    O1 = underpaint(O0, asobject(Gt))
    O = underpaint(O1, asobject(Gb))
    return O

def dsl5(I):
