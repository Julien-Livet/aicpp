def dsl1(I):
    # Global: trim border and fill background with the modal color of the trimmed grid
    G = trim(I)
    O = replace(G, ZERO, mostcolor(G))
    return O

def dsl2(I):
    # Object extraction: paint each non-modal object's backdrop on a modal-color canvas
    fgobjs = objects(I, T, F, T)
    fg = merge(fgobjs)
    G = subgrid(fg, I)
    modal = mostcolor(fg)
    base = canvas(modal, shape(G))
    modalobjs = colorfilter(fgobjs, modal)
    others = difference(fgobjs, modalobjs)
    B = totuple(apply(backdrop, others))
    C = totuple(apply(color, others))
    colored = mpapply(recolor, C, B)
    M = merge(colored)
    O = paint(base, M)
    return O

def dsl3(I):
    # Color filtering: emphasize least-common color objects via their inbox, over a modal-filled crop
    fgobjs = objects(I, T, F, T)
    fg = merge(fgobjs)
    G = subgrid(fg, I)
    modal = mostcolor(fg)
    base = replace(G, ZERO, modal)
    rare = leastcolor(fg)
    rareobjs = colorfilter(fgobjs, rare)
    B = totuple(apply(inbox, rareobjs))
    C = totuple(apply(color, rareobjs))
    colored = mpapply(recolor, C, B)
    M = merge(colored)
    O = underpaint(base, M)
    return O

def dsl4(I):
    # Geometric: enforce vertical agreement with its mirror, disagreements go to modal color
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    modal = mostcolor(fg)
    B = replace(G, ZERO, modal)
    V = vmirror(B)
    O = cellwise(B, V, modal)
    return O

def dsl5(I):
    # Relational/structural: recolor nonzero left/right halves with least/most colors respectively
    fgobjs = objects(I, T, F, T)
    fg = merge(fgobjs)
    G = subgrid(fg, I)
    Z = ofcolor(G, ZERO)
    NZ = difference(asindices(G), Z)
    wL = width(lefthalf(G))
    Lbox = asindices(lefthalf(G))
    Rbox = shift(asindices(righthalf(G)), tojvec(wL))
    Lmask = intersection(NZ, Lbox)
    Rmask = intersection(NZ, Rbox)
    A = leastcolor(fg)
    Bc = mostcolor(fg)
    LO = recolor(A, Lmask)
    RO = recolor(Bc, Rmask)
    base = replace(G, ZERO, Bc)
    O = paint(base, merge(insert(RO, initset(LO))))
    return O
