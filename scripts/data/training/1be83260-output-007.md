def dsl1(I):
    # Global: crop to nonzero bbox and replace all background with the dominant nonzero color
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    OBJS = objects(A, T, F, T)
    UNION = merge(OBJS)
    DOMNZ = mostcolor(UNION)
    O = replace(A, ZERO, DOMNZ)
    return O

def dsl2(I):
    # Object extraction: keep all non-dominant-nonzero objects on a dominant-nonzero canvas
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    OBJS = objects(A, T, F, T)
    UNION = merge(OBJS)
    DOMNZ = mostcolor(UNION)
    BASE = canvas(DOMNZ, shape(A))
    PURE = colorfilter(OBJS, DOMNZ)
    OTH = difference(OBJS, PURE)
    EXP = merge(OTH)
    O = paint(BASE, EXP)
    return O

def dsl3(I):
    # Color filtering: fill all background within the crop with the dominant nonzero color
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    OBJS = objects(A, T, F, T)
    UNION = merge(OBJS)
    DOMNZ = mostcolor(UNION)
    O = underfill(A, DOMNZ, asindices(A))
    return O

def dsl4(I):
    # Geometric: split crop into halves and fill each half's background by its own dominant nonzero color, then recombine
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    L = lefthalf(A)
    R = righthalf(A)
    LO = objects(L, T, F, T)
    RO = objects(R, T, F, T)
    LDOM = mostcolor(merge(LO))
    RDOM = mostcolor(merge(RO))
    LF = replace(L, ZERO, LDOM)
    RF = replace(R, ZERO, RDOM)
    O = hconcat(LF, RF)
    return O

def dsl5(I):
    # Relational: fill background with dominant nonzero color, then expand each non-dominant object to its bounding box recolored by its own color
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    OBJS = objects(A, T, F, T)
    UNION = merge(OBJS)
    DOMNZ = mostcolor(UNION)
    BASE = replace(A, ZERO, DOMNZ)
    PURE = colorfilter(OBJS, DOMNZ)
    OTH = difference(OBJS, PURE)
    TOT = totuple(OTH)
    COLS = apply(color, TOT)
    PATS = apply(compose(backdrop, toindices), TOT)
    COLORED = mpapply(recolor, totuple(COLS), totuple(PATS))
    EXP = merge(COLORED)
    O = paint(BASE, EXP)
    return O
