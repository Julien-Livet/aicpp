def dsl1(I):
    # Relational replication via shoots + frontiers with border-based orientation
    OS = objects(I, T, F, T)
    P = mapply(toindices, OS)
    H = height(I)
    W = width(I)
    top = uppermost(P)
    bot = lowermost(P)
    left = leftmost(P)
    right = rightmost(P)
    Hm1 = subtract(H, ONE)
    Wm1 = subtract(W, ONE)
    vert = both(equality(top, ZERO), equality(bot, Hm1))
    dv = subtract(bot, top)
    dh = subtract(right, left)
    o1 = first(OS)
    o2 = last(OS)
    # horizontal stripes for each seed
    l1 = first(toindices(o1))
    s1p = toivec(dv)
    s1n = toivec(invert(dv))
    L1 = combine(combine(shoot(l1, s1p), shoot(l1, s1n)), initset(l1))
    R1 = mapply(hfrontier, L1)
    H1 = recolor(color(o1), R1)
    l2 = first(toindices(o2))
    s2p = toivec(dv)
    s2n = toivec(invert(dv))
    L2 = combine(combine(shoot(l2, s2p), shoot(l2, s2n)), initset(l2))
    R2 = mapply(hfrontier, L2)
    H2 = recolor(color(o2), R2)
    HOBJ = combine(H1, H2)
    # vertical stripes for each seed
    c1p = tojvec(dh)
    c1n = tojvec(invert(dh))
    C1A = combine(combine(shoot(l1, c1p), shoot(l1, c1n)), initset(l1))
    V1 = mapply(vfrontier, C1A)
    VOBJ1 = recolor(color(o1), V1)
    C2A = combine(combine(shoot(l2, c1p), shoot(l2, c1n)), initset(l2))
    V2 = mapply(vfrontier, C2A)
    VOBJ2 = recolor(color(o2), V2)
    VOBJ = combine(VOBJ1, VOBJ2)
    OOBJ = branch(vert, VOBJ, HOBJ)
    O = paint(canvas(ZERO, shape(I)), OOBJ)
    return O
