def dsl1(I):
    # Global rows/cols defined by the bounding box of least-color region
    val = leastcolor(I)
    m = ofcolor(I, val)
    rect = backdrop(m)
    rows = merge(apply(hfrontier, rect))
    cols = merge(apply(vfrontier, rect))
    lines = combine(rows, cols)
    target = intersection(lines, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl2(I):
    # Per-component center-of-mass crosses of least-color components
    val = leastcolor(I)
    parts = colorfilter(fgpartition(I), val)
    coms = apply(centerofmass, parts)
    rows = merge(apply(hfrontier, coms))
    cols = merge(apply(vfrontier, coms))
    lines = combine(rows, cols)
    target = intersection(lines, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl4(I):
    # Object extraction by shape matching with a least-color exemplar
    val = leastcolor(I)
    parts = colorfilter(fgpartition(I), val)
    exemplar = argmax(parts, size)
    dims = shape(toindices(exemplar))
    onesobjs = colorfilter(fgpartition(I), ONE)
    getdims = compose(shape, toindices)
    eqdims = lbind(equality, dims)
    matches = sfilter(onesobjs, compose(eqdims, getdims))
    target = merge(apply(toindices, matches))
    O = fill(I, val, target)
    return O

def dsl5(I):
    # Relational alignment: rows/cols through least-color pixels clipped to its box
    val = leastcolor(I)
    m = ofcolor(I, val)
    rect = backdrop(m)
    h = merge(apply(hfrontier, m))
    v = merge(apply(vfrontier, m))
    lines = combine(h, v)
    clipped = intersection(lines, rect)
    target = intersection(clipped, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O
