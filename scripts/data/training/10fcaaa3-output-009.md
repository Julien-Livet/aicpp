def dsl1(I):
    T0 = hconcat(I, I)
    TT = vconcat(T0, T0)
    h = height(I)
    w = width(I)
    A = asindices(TT)
    rloc = toivec(h)
    cloc = tojvec(w)
    LH = intersection(hfrontier(rloc), A)
    LV = intersection(vfrontier(cloc), A)
    L = combine(LH, LV)
    Z = ofcolor(TT, ZERO)
    M = intersection(L, Z)
    O = fill(TT, EIGHT, M)
    return O
