def dsl1(I):
    # Global geometric: 2 on the S5 bbox counterdiagonal, others 8
    S5 = ofcolor(I, FIVE)
    D = connect(llcorner(S5), urcorner(S5))
    S2 = intersection(S5, D)
    S8 = difference(S5, S2)
    O = fill(fill(fill(I, SEVEN, S5), TWO, S2), EIGHT, S8)
    return O

def dsl2(I):
    # Object extraction: longest diagonal-connected 5-object -> 2, others -> 8
    S5 = ofcolor(I, FIVE)
    O5 = colorfilter(objects(I, T, T, T), FIVE)
    Long = argmax(O5, size)
    S2 = toindices(Long)
    S8 = difference(S5, S2)
    O = fill(fill(fill(I, SEVEN, S5), TWO, S2), EIGHT, S8)
    return O

def dsl3(I):
    # Color filtering: bottommost 5 per column -> 2, others -> 8
    S5 = ofcolor(I, FIVE)
    RaysD = prapply(shoot, S5, initset(DOWN))
    HasBelow = intersection(RaysD, S5)
    Bottom = difference(S5, HasBelow)
    Rest = HasBelow
    O = fill(fill(fill(I, SEVEN, S5), TWO, Bottom), EIGHT, Rest)
    return O

def dsl5(I):
    # Relational symmetry: vmirror-paired 5s -> 8, unpaired -> 2
    S5 = ofcolor(I, FIVE)
    Paired = intersection(S5, vmirror(S5))
