def dsl1(I):
    # Global: remove frontier lines/columns (e.g., full-8 separators)
    O = compress(I)

def dsl2(I):
    # Object extraction: crop to the largest non-8 object
    objs_all = fgpartition(I)
    bg = colorfilter(objs_all, EIGHT)
    fg = difference(objs_all, bg)
    big = argmax(fg, size)
    O = subgrid(big, I)
    return O

def dsl3(I):
    # Color filtering: promote filler 1s to separators 8
    O = replace(I, ONE, EIGHT)
    return O

def dsl4(I):
    # Geometric: rotate the grid 90 degrees clockwise
    O = rot90(I)
    return O

def dsl5(I):
    # Relational/structural: draw colored boxes around each non-8 object on an 8 canvas
    objs = objects(I, T, F, T)
    boxes = apply(fork(recolor, color, box), objs)
    allboxes = merge(boxes)
    base = canvas(EIGHT, shape(I))
