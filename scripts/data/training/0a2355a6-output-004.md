def dsl1(I):
    # Global: pixelwise recolor using quadrant masks of the 8-foreground bounding box
    P = ofcolor(I, EIGHT)
    S = subgrid(P, I)
    UL = ulcorner(P)
    TLm = shift(intersection(asindices(lefthalf(S)), asindices(tophalf(S))), UL)
    TRm = shift(intersection(asindices(righthalf(S)), asindices(tophalf(S))), UL)
    BLm = shift(intersection(asindices(lefthalf(S)), asindices(bottomhalf(S))), UL)
    BRm = shift(intersection(asindices(righthalf(S)), asindices(bottomhalf(S))), UL)
    pts = P
    base = replace(I, EIGHT, ZERO)
    tl = intersection(pts, TLm)
    tr = intersection(pts, TRm)
    bl = intersection(pts, BLm)
    br = intersection(pts, BRm)
    O = paint(paint(paint(paint(base, recolor(FOUR, tl)), recolor(TWO, tr)), recolor(THREE, bl)), recolor(ONE, br))
    return O

def dsl2(I):
    # Object extraction: classify 8-objects by overlap with foreground-box quadrant masks
    P = ofcolor(I, EIGHT)
    S = subgrid(P, I)
    UL = ulcorner(P)
    TLm = shift(intersection(asindices(lefthalf(S)), asindices(tophalf(S))), UL)
    TRm = shift(intersection(asindices(righthalf(S)), asindices(tophalf(S))), UL)
    BLm = shift(intersection(asindices(lefthalf(S)), asindices(bottomhalf(S))), UL)
    BRm = shift(intersection(asindices(righthalf(S)), asindices(bottomhalf(S))), UL)
    objs = colorfilter(objects(I, T, F, T), EIGHT)
    has = lambda M: chain(positive, size, compose(lbind(intersection, M), toindices))
    tl = sfilter(objs, has(TLm))
    tr = sfilter(objs, has(TRm))
    bl = sfilter(objs, has(BLm))
    br = sfilter(objs, has(BRm))
    tl_r = merge(apply(lbind(recolor, FOUR), tl))
    tr_r = merge(apply(lbind(recolor, TWO), tr))
    bl_r = merge(apply(lbind(recolor, THREE), bl))
    br_r = merge(apply(lbind(recolor, ONE), br))
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(paint(paint(base, tl_r), tr_r), bl_r), br_r)
    return O

def dsl3(I):
    # Global split/recompose inside the 8-foreground box, then embed back
    P = ofcolor(I, EIGHT)
    S = subgrid(P, I)
    TH = tophalf(S)
    BH = bottomhalf(S)
    TL = lefthalf(TH)
    TR = righthalf(TH)
    BL = lefthalf(BH)
    BR = righthalf(BH)
    TLc = replace(TL, EIGHT, FOUR)
    TRc = replace(TR, EIGHT, TWO)
    BLc = replace(BL, EIGHT, THREE)
    BRc = replace(BR, EIGHT, ONE)
    top = hconcat(TLc, TRc)
    bot = hconcat(BLc, BRc)
    Sc = vconcat(top, bot)
    UL = ulcorner(P)
    base = replace(I, EIGHT, ZERO)
    O = paint(base, recolor(FOUR, shift(ofcolor(Sc, FOUR), UL)))
    O = paint(O, recolor(TWO, shift(ofcolor(Sc, TWO), UL)))
    O = paint(O, recolor(THREE, shift(ofcolor(Sc, THREE), UL)))
    O = paint(O, recolor(ONE, shift(ofcolor(Sc, ONE), UL)))
    return O
