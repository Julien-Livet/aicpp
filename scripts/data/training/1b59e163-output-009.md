def dsl1(I):
    # Global: shift cropped foreground so its right edge aligns to grid's right edge
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    fg = difference(allidx, ofcolor(J, FOUR))
    S = subgrid(fg, J)
    sidx = asindices(S)
    sfg = difference(sidx, ofcolor(S, FOUR))
    src = toobject(sfg, S)
    v = subtract(urcorner(allidx), lrcorner(sfg))
    O = move(canvas(FOUR, shape(J)), src, v)
    return O

def dsl2(I):
    # Object extraction: mirror the uppermost object horizontally and union
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    s = argmin(objs, uppermost)
    G = paint(canvas(FOUR, shape(J)), s)
    M = hmirror(G)
    O = underpaint(G, asobject(M))
    return O

def dsl3(I):
    # Color filtering: mirror the least-common non-bg color horizontally and union
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    cols = apply(color, objs)
    c = leastcommon(totuple(cols))
    foc = colorfilter(objs, c)
    merged = merge(foc)
    src = toobject(merged, J)
    G = paint(canvas(FOUR, shape(J)), src)
    H = hmirror(G)
    O = underpaint(G, asobject(H))
    return O

def dsl4(I):
    # Geometric: enforce bilateral symmetry inside the bounding subgrid (union original + mirrored)
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    fg = difference(allidx, ofcolor(J, FOUR))
    S = subgrid(fg, J)
    sidx = asindices(S)
    sfg = difference(sidx, ofcolor(S, FOUR))
    src = toobject(sfg, S)
    Sm = hmirror(S)
    midx = asindices(Sm)
    mfg = difference(midx, ofcolor(Sm, FOUR))
    mir = toobject(mfg, Sm)
    base = canvas(FOUR, shape(J))
    u = ulcorner(fg)
    P1 = move(base, src, u)
    P2 = move(base, mir, u)
    O = underpaint(P1, asobject(P2))
    return O

def dsl5(I):
    # Relational: translate foreground so its center of mass equals grid center
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    fg = difference(allidx, ofcolor(J, FOUR))
    src = toobject(fg, J)
    v = subtract(center(allidx), centerofmass(fg))
    O = move(canvas(FOUR, shape(J)), src, v)
    return O
