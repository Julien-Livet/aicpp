def dsl1(I):
    # Global: trim outer border, then compress horizontally and vertically via rotation
    A = trim(I)
    B = compress(A)
    C = rot90(B)
    D = compress(C)
    O = rot270(D)
    return O

def dsl2(I):
    # Object extraction: crop to the union of all non-1 and non-8 objects
    OBJS = objects(I, T, F, T)
    NOTONE = compose(flip, matcher(color, ONE))
    NOTEIGHT = compose(flip, matcher(color, EIGHT))
    PRED = fork(both, NOTONE, NOTEIGHT)
    GOOD = sfilter(OBJS, PRED)
    UNION = merge(GOOD)
    O = subgrid(UNION, I)
    return O

def dsl3(I):
    # Color filtering: keep only non-1 and non-8 colors on an 8 canvas
    OBJS = objects(I, T, F, T)
    NOTONE = compose(flip, matcher(color, ONE))
    NOTEIGHT = compose(flip, matcher(color, EIGHT))
    PRED = fork(both, NOTONE, NOTEIGHT)
    GOOD = sfilter(OBJS, PRED)
    FG = merge(GOOD)
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, FG)
    return O

def dsl4(I):
    # Geometric: keep cells invariant under diagonal mirror, else 8
    A = dmirror(I)
    O = cellwise(I, A, EIGHT)
    return O

def dsl5(I):
    # Relational: paint the center of mass of each object with its own color on an 8 canvas
    OBJS = objects(I, T, F, T)
    DOTS = apply(fork(connect, center, center), OBJS)
    COLORED = apply(fork(recolor, color, identity), DOTS)
    ALL = merge(COLORED)
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, ALL)
    return O
