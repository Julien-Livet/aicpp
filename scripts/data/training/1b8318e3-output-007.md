def dsl1(I):
    # Global: keep only cells lying on rows or columns through centers of 5-objects
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    centers = apply(centerofmass, fives)
    rows = merge(apply(hfrontier, centers))
    cols = merge(apply(vfrontier, centers))
    mask = combine(rows, cols)
    O = paint(canvas(ZERO, shape(I)), toobject(mask, I))
    return O

def dsl2(I):
    # Object extraction: keep 5s and objects sharing any row or column with them
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    others = difference(objs, fives)
    vmatch = lbind(vmatching, fiveobj)
    hmatch = lbind(hmatching, fiveobj)
    selv = sfilter(others, vmatch)
    selh = sfilter(others, hmatch)
    kept = merge(combine(selv, selh))
    O = paint(canvas(ZERO, shape(I)), combine(fiveobj, kept))
    return O

def dsl3(I):
    # Color filtering: keep all objects whose colors occur adjacent to any 5
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    others = difference(objs, fives)
    adjf = lbind(adjacent, fiveobj)
    adjobjs = sfilter(others, adjf)
    keepcolors = palette(merge(adjobjs))
    isallowed = compose(rbind(contained, keepcolors), color)
    keepobjs = sfilter(objs, isallowed)
    O = paint(canvas(ZERO, shape(I)), merge(keepobjs))
    return O

def dsl4(I):
    # Geometric: move each non-5 toward the union of columns containing any 5
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    fidxs = toindices(fiveobj)
    vcols = merge(apply(vfrontier, fidxs))
    others = difference(objs, fives)
    dirf = rbind(gravitate, vcols)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    keep = combine(fiveobj, mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl5(I):
    # Relational: move each non-5 toward the union of per-5 bounding-box outlines
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    crnf = compose(box, toindices)
    perboxes = merge(apply(crnf, fives))
    others = difference(objs, fives)
    dirf = rbind(gravitate, perboxes)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    keep = combine(fiveobj, mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O
