def dsl1(I):
    # Global: recolor 8s by orientation of their bounding subgrid; erase 1s
    p8 = ofcolor(I, EIGHT)
    o8 = toobject(p8, I)
    g8 = subgrid(o8, I)
    col = branch(portrait(g8), TWO, THREE)
    S = switch(I, EIGHT, col)
    O = replace(S, ONE, ZERO)
    return O

def dsl2(I):
    # Object extraction: largest 8-object -> 3, other 8s -> 2; 1s erased
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    objs = objects(I, T, F, T)
    o8s = colorfilter(objs, EIGHT)
    big = argmax(o8s, size)
    bigi = toindices(big)
    rest = difference(p8, bigi)
    A = fill(base, THREE, bigi)
    O = fill(A, TWO, rest)
    return O

def dsl3(I):
    # Color filtering: parity of count of 1s selects recolor; then erase 1s
    p1 = ofcolor(I, ONE)
    n1 = size(p1)
    col = branch(even(n1), THREE, TWO)
    S = switch(I, EIGHT, col)
    O = replace(S, ONE, ZERO)
    return O

def dsl4(I):
    # Geometric: if 1-object is both H- and V-symmetric, use 2 else 3; erase 1s
    p1 = ofcolor(I, ONE)
    o1 = toobject(p1, I)
    g1 = subgrid(o1, I)
    symh = equality(g1, hmirror(g1))
    symv = equality(g1, vmirror(g1))
    sym = both(symh, symv)
    col = branch(sym, TWO, THREE)
    S = switch(I, EIGHT, col)
    O = replace(S, ONE, ZERO)
    return O

def dsl5(I):
    # Relational: if any 8 is diagonally adjacent to a 1 -> color 3 else 2; erase 1s
    p8 = ofcolor(I, EIGHT)
    p1 = ofcolor(I, ONE)
    dmask = mapply(ineighbors, p1)
    adj8 = intersection(p8, dmask)
    has = positive(size(adj8))
    col = branch(has, THREE, TWO)
    S = switch(I, EIGHT, col)
    O = replace(S, ONE, ZERO)
    return O
