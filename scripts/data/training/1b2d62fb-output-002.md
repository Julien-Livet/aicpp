def dsl1(I):
    axis = ofcolor(I, ONE)
    Z = ofcolor(I, ZERO)
    L1 = shift(axis, LEFT)
    L2 = shift(axis, multiply(LEFT, TWO))
    L3 = shift(axis, multiply(LEFT, THREE))
    leftarea = combine(L1, combine(L2, L3))
    R1 = shift(axis, RIGHT)
    R2 = shift(axis, multiply(RIGHT, TWO))
    R3 = shift(axis, multiply(RIGHT, THREE))
    rightarea = combine(R1, combine(R2, R3))
    LZ = intersection(Z, leftarea)
    RZ = intersection(Z, rightarea)
    SR = shift(RZ, multiply(LEFT, FOUR))
    P = intersection(LZ, SR)
    base = canvas(ZERO, shape(I))
    painted = fill(base, EIGHT, P)
    dims = astuple(height(I), THREE)
    O = crop(painted, ulcorner(leftarea), dims)
    return O
