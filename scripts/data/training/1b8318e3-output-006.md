def dsl1(I):
    # Geometric: move each non-5 toward the outline of the global bbox of all 5s
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    bbox = box(toindices(fiveobj))
    others = difference(objs, fives)
    dirf = rbind(gravitate, bbox)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    keep = combine(fiveobj, mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl2(I):
    # Structural: move each non-5 toward the union of per-5 bbox outlines
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    crnf = compose(box, toindices)
    perboxes = merge(mapply(crnf, fives))
    fiveobj = merge(fives)
    others = difference(objs, fives)
    dirf = rbind(gravitate, perboxes)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    keep = combine(fiveobj, mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl3(I):
    # Global: keep only cells on rows or columns passing through centers of 5-clusters
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    centers = mapply(centerofmass, fives)
    rows = merge(apply(hfrontier, centers))
    cols = merge(apply(vfrontier, centers))
    mask = combine(rows, cols)
    O = paint(canvas(ZERO, shape(I)), toobject(mask, I))
    return O

def dsl4(I):
    # Color filtering: keep all 5s and their 4-neighborhood ring
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    fidxs = toindices(fiveobj)
    neigh = merge(mapply(neighbors, fidxs))
    ring = difference(neigh, fidxs)
    mask = combine(fidxs, ring)
    O = paint(canvas(ZERO, shape(I)), toobject(mask, I))
    return O

def dsl5(I):
    # Object extraction: keep 5s and any objects adjacent to them
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    others = difference(objs, fives)
    adjfun = lbind(adjacent, fiveobj)
    sel = sfilter(others, adjfun)
    keep = combine(fiveobj, merge(sel))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O
