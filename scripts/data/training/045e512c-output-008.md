def dsl1(I):
    # O = enforce bilateral symmetry by overlaying horizontal and vertical mirrors (global transformation)
    A = asobject(I)
    H = asobject(hmirror(I))
    V = asobject(vmirror(I))
    HV = asobject(vmirror(hmirror(I)))
    stamps = merge(insert(A, insert(H, insert(V, initset(HV)))))
    O = paint(canvas(ZERO, shape(I)), stamps)
    return O

def dsl2(I):
    # O = draw the bbox outlines for all objects (object extraction)
    objs = objects(I, T, F, T)
    outline = compose(box, toindices)
    stamped = apply(fork(recolor, color, outline), objs)
    O = paint(canvas(ZERO, shape(I)), merge(stamped))
    return O

def dsl3(I):
    # O = fill the bounding boxes of least-common-color objects (color filtering)
    objs = objects(I, T, F, T)
    lc = leastcolor(I)
    sel = colorfilter(objs, lc)
    bb = compose(backdrop, toindices)
    stamped = apply(fork(recolor, color, bb), sel)
    O = paint(canvas(ZERO, shape(I)), merge(stamped))
    return O

def dsl4(I):
    # O = from each object's center, shoot a ray toward nearest border (geometric reasoning)
    objs = objects(I, T, F, T)
    border = box(asindices(I))
    ceno = compose(center, toindices)
    dirf = compose(rbind(gravitate, border), toindices)
    ray = fork(shoot, ceno, dirf)
    stamped = apply(fork(recolor, color, ray), objs)
    O = paint(canvas(ZERO, shape(I)), merge(stamped))
    return O

def dsl5(I):
    # O = mark the corners of each object's bounding box with the object's color (relational/structural)
