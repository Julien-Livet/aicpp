def dsl1(I):
    # Global: remove 5s and union with a 90-degree rotation
    J = replace(I, FIVE, FOUR)
    base = canvas(FOUR, shape(J))
    R = rot90(J)
    O1 = underpaint(base, asobject(J))
    O = underpaint(O1, asobject(R))
    return O

def dsl2(I):
    # Object extraction: select most common object size and mirror that subset vertically
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    sizes = apply(size, objs)
    sstar = mostcommon(totuple(sizes))
    sel = sizefilter(objs, sstar)
    merged = merge(sel)
    base = canvas(FOUR, shape(J))
    G = paint(base, merged)
    M = vmirror(G)
    O = underpaint(G, asobject(M))
    return O

def dsl3(I):
    # Color filtering: move least-common color to the central vertical frontier
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    cols = apply(color, objs)
    cstar = leastcommon(totuple(cols))
    foc = colorfilter(objs, cstar)
    merged = merge(foc)
    src = toobject(merged, J)
    allidx = asindices(J)
    dest = vfrontier(center(allidx))
    v = gravitate(merged, dest)
    O = move(canvas(FOUR, shape(J)), src, v)
    return O

def dsl4(I):
    # Geometric: overlay foreground with its diagonal-then-vertical mirror
    J = replace(I, FIVE, FOUR)

def dsl5(I):

    fg = difference(allidx, ofcolor(J, FOUR))
