def dsl1(I):
    # Relational: connect same-colored aligned objects with full rectangles (global, single pass)
    s = mostcolor(I)
    dims = shape(I)
    objs = objects(I, T, F, T)
    fg = sfilter(objs, compose(flip, matcher(color, s)))
    pairs = product(fg, fg)
    same = fork(equality, compose(color, first), compose(color, last))
    hA = fork(hmatching, first, last)
    vA = fork(vmatching, first, last)
    aligned = fork(either, hA, vA)
    good = sfilter(pairs, fork(both, same, aligned))
    idx = compose(backdrop, fork(combine, first, last))
    col = compose(color, first)
    can = compose(rbind(canvas, dims), col)
    mk = fork(toobject, idx, can)
    P = merge(apply(mk, good))
    O = underpaint(I, P)
    return O

def dsl2(I):
    # Object extraction by color groups: per-color aligned pair rectangles
    s = mostcolor(I)
    dims = shape(I)
    objs = objects(I, T, F, T)
    fg = sfilter(objs, compose(flip, matcher(color, s)))
    cols = apply(color, fg)
    groups = apply(lbind(colorfilter, fg), cols)
    pairsF = fork(product, identity, identity)
    same = fork(equality, compose(color, first), compose(color, last))
    hA = fork(hmatching, first, last)
    vA = fork(vmatching, first, last)
    aligned = fork(either, hA, vA)
    cond = fork(both, same, aligned)
    filtPairs = compose(rbind(sfilter, cond), pairsF)
    idx = compose(backdrop, fork(combine, first, last))
    col = compose(color, first)
    can = compose(rbind(canvas, dims), col)
    mk = fork(toobject, idx, can)
    perGroup = compose(merge, compose(lbind(apply, mk), filtPairs))
    P = merge(apply(perGroup, groups))
    O = underpaint(I, P)
    return O

def dsl4(I):
    # Orientation split: vertical via full rectangles, horizontal via interior of boxes
    s = mostcolor(I)
    dims = shape(I)
    objs = objects(I, T, F, T)
    fg = sfilter(objs, compose(flip, matcher(color, s)))
    pairs = product(fg, fg)
    same = fork(equality, compose(color, first), compose(color, last))
    vcond = fork(both, same, fork(vmatching, first, last))
    hcond = fork(both, same, fork(hmatching, first, last))
    vIdx = compose(backdrop, fork(combine, first, last))
    hIdx = compose(delta, compose(box, fork(combine, first, last)))
    col = compose(color, first)
    can = compose(rbind(canvas, shape(I)), col)
    vObj = fork(toobject, vIdx, can)
    hObj = fork(toobject, hIdx, can)
    PV = merge(apply(vObj, sfilter(pairs, vcond)))
    PH = merge(apply(hObj, sfilter(pairs, hcond)))
    P = combine(PV, PH)
    O = underpaint(I, P)
    return O
