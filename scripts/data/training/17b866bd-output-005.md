def dsl1(I):
    objs = objects(I, T, F, F)
    eights = colorfilter(objs, EIGHT)
    panes = sizefilter(eights, 16)
    non8 = difference(objs, eights)
    zeros = colorfilter(non8, ZERO)
    seeds = difference(non8, zeros)
    ulp = compose(ulcorner, identity)
    pt = fork(add, ulp, rbind(identity, LEFT))
    colat = compose(rbind(index, I), pt)
    nz = compose(flip, fork(equality, colat, rbind(identity, ZERO)))
    ne = compose(flip, fork(equality, colat, rbind(identity, EIGHT)))
    chosen = sfilter(panes, fork(both, nz, ne))
    allidx = compose(toindices, identity)
    crn = compose(corners, identity)
    diamond = fork(difference, allidx, crn)
    recol = apply(fork(recolor, colat, diamond), chosen)
    recolmerged = merge(recol)
    seeds_idx = mapply(toindices, seeds)
    base = fill(I, ZERO, seeds_idx)
    O = paint(base, recolmerged)
    return O
