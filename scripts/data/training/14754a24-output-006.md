def dsl1(I):
    # Geometric connectors: 5s on pure horizontal/vertical lines between 4s
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    allowed = combine(fou, fiv)
    segs = sfilter(prapply(connect, fou, fou), fork(either, hline, vline))
    pure = sfilter(segs, matcher(compose(size, rbind(difference, allowed)), ZERO))
    mask = merge(pure)
    target = intersection(mask, fiv)
    O = fill(I, TWO, target)
    return O

def dsl2(I):
    # Object extraction: recolor line-shaped 5-objects that touch any 4
    fou = ofcolor(I, FOUR)
    fiveobjs = colorfilter(objects(I, T, F, T), FIVE)
    touching = sfilter(fiveobjs, lbind(adjacent, fou))
    lines = sfilter(touching, fork(either, hline, vline))
    recol = mapply(lbind(recolor, TWO), lines)
    obj = merge(recol)
    O = paint(I, obj)
    return O

def dsl3(I):
    # Rays from 4s: all 5s collinear with any 4 along four axial directions
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    raysR = merge(mapply(lbind(shoot, RIGHT), fou))
    raysL = merge(mapply(lbind(shoot, LEFT), fou))
    raysU = merge(mapply(lbind(shoot, UP), fou))
    raysD = merge(mapply(lbind(shoot, DOWN), fou))
    axis = merge(combine(combine(raysR, raysL), combine(raysU, raysD)))
    target = intersection(axis, fiv)
    O = fill(I, TWO, target)
    return O

def dsl4(I):
    # Global row/column mask through all 4s: recolor 5s on those axes
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    rows = merge(mapply(hfrontier, fou))
    cols = merge(mapply(vfrontier, fou))
    axis = merge(combine(rows, cols))
    target = intersection(axis, fiv)
    O = fill(I, TWO, target)
    return O

def dsl5(I):
    # Bounding box of all 4s: recolor 5s inside the 4s' bounding rectangle (excluding 4s)
    fou = ofcolor(I, FOUR)
    fouobj = toobject(fou, I)
    boxmask = delta(fouobj)
    target = intersection(boxmask, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O
