def dsl1(I):
    # geometric: 2 on the counterdiagonal of the 5-bbox; others -> 8
    S5 = ofcolor(I, FIVE)
    Axis = connect(urcorner(S5), llcorner(S5))
    S2 = intersection(S5, Axis)
    S8 = difference(S5, S2)
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O

def dsl2(I):
    # object extraction: pick the 5-object with max overlap with the counterdiagonal axis
    S5 = ofcolor(I, FIVE)
    O5 = colorfilter(objects(I, T, T, T), FIVE)
    Axis = combine(shoot(urcorner(S5), DOWN_LEFT), shoot(llcorner(S5), UP_RIGHT))
    score = chain(size, lbind(intersection, Axis), toindices)
    Best = argmax(O5, score)
    S2 = toindices(Best)
    S8 = difference(S5, S2)
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O

def dsl3(I):
    # symmetry: cmirror of the main 5-bbox axis yields the counterdiagonal axis
    S5 = ofcolor(I, FIVE)
    Main = combine(shoot(ulcorner(S5), DOWN_RIGHT), shoot(lrcorner(S5), UP_LEFT))
    Axis = cmirror(Main)
    S2 = intersection(S5, Axis)
    S8 = difference(S5, S2)
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O

def dsl4(I):
    # global rays within bbox: 2 where 5 intersects the UR↘LL↖ rays clipped to the 5-bbox
    S5 = ofcolor(I, FIVE)
    Rays = combine(shoot(urcorner(S5), DOWN_LEFT), shoot(llcorner(S5), UP_RIGHT))
    AxisBBox = intersection(Rays, backdrop(S5))
    S2 = intersection(S5, AxisBBox)
    S8 = difference(S5, S2)

    return O

def dsl5(I):
    # relational centerline: 2 where 5 lies on the counterdiagonal through the bbox center
    S5 = ofcolor(I, FIVE)
    C = center(S5)
    Axis = combine(shoot(C, UP_RIGHT), shoot(C, DOWN_LEFT))
