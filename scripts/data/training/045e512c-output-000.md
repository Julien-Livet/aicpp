def dsl1(I):
    comp = compress(I)
    hf = divide(width(I), width(comp))
    vf = divide(height(I), height(comp))
    O = vupscale(hupscale(comp, hf), vf)
    return O

def dsl2(I):
    nz = difference(asindices(I), ofcolor(I, ZERO))
    sub = subgrid(nz, I)
    obj = toobject(asindices(sub), sub)
    O = paint(canvas(ZERO, shape(I)), obj)
    return O

def dsl3(I):
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    inds = ofcolor(I, c)
    obj = recolor(c, inds)
    O = paint(canvas(ZERO, shape(I)), obj)
    return O

def dsl4(I):
    nz = difference(asindices(I), ofcolor(I, ZERO))
    obj = toobject(nz, I)
    u1 = combine(obj, hmirror(obj))
    u2 = combine(u1, vmirror(obj))
    u3 = combine(u2, dmirror(obj))
    u4 = combine(u3, cmirror(obj))
    O = paint(canvas(ZERO, shape(I)), u4)
    return O

def dsl5(I):
    nz = difference(asindices(I), ofcolor(I, ZERO))
    obj = toobject(nz, I)
    base = normalize(obj)
    hp = hperiod(obj)
    vp = vperiod(obj)
    w = width(I)
    h = height(I)
    offsH = interval(0, w, hp)
    vecsH = apply(tojvec, offsH)
    tileH = mapply(lbind(shift, base), vecsH)
    offsV = interval(0, h, vp)
    vecsV = apply(toivec, offsV)
    tileHV = mapply(lbind(shift, tileH), vecsV)
    O = paint(canvas(ZERO, shape(I)), tileHV)
    return O
