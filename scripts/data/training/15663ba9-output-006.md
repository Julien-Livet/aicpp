def dsl1(I):
    # Global: mark endpoints with 4 and neighbors-of-endpoints plus corners with 2
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    H = intersection(L, R)
    V = intersection(U, D)
    corners = difference(intersection(combine(L, R), combine(U, D)), combine(H, V))
    hends = difference(combine(L, R), H)
    vends = difference(combine(U, D), V)
    E = combine(hends, vends)
    Le = difference(L, R)
    Re = difference(R, L)
    Ue = difference(U, D)
    De = difference(D, U)
    N1 = combine(shift(Le, LEFT), shift(Re, RIGHT))
    N2 = combine(shift(Ue, UP), shift(De, DOWN))
    N = combine(N1, N2)
    Nonly = difference(combine(N, corners), E)
    G = fill(I, FOUR, E)
    O = fill(G, TWO, Nonly)
    return O
