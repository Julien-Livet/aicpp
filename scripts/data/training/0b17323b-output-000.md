def dsl1(I):
    ones = ofcolor(I, ONE)
    obj1 = toobject(ones, I)
    d = hperiod(obj1)
    step = add(toivec(d), tojvec(d))
    start = add(lrcorner(ones), step)
    path = shoot(start, step)
    ins = recolor(TWO, path)
    O = paint(I, ins)
    return O
