def dsl1(I):
    # Global: frontier pixels of leastcolor get 4; their neighbors inside S get 2
    c = leastcolor(I)
    S = ofcolor(I, c)
    frobjs = frontiers(I)
    frparts = apply(toindices, frobjs)
    Fr = merge(frparts)
    E = intersection(S, Fr)
    N1 = combine(shift(E, LEFT), shift(E, RIGHT))
    N2 = combine(shift(E, UP), shift(E, DOWN))
    N = intersection(combine(N1, N2), S)
    G = fill(I, FOUR, E)
    O = fill(G, TWO, N)
    return O

def dsl2(I):
    # Object extraction: per-component box-outline intersection as anchors; neighbors inside get 2
    c = leastcolor(I)
    comps = colorfilter(objects(I, T, F, T), c)

    S = merge(parts)
    frames = apply(box, parts)
    Frm = merge(frames)

    N1 = combine(shift(E, LEFT), shift(E, RIGHT))
    N2 = combine(shift(E, UP), shift(E, DOWN))
    N = intersection(combine(N1, N2), S)
    G = fill(I, FOUR, E)

def dsl3(I):
    # Color filtering: endpoints of leastcolor get 4; their neighbors inside get 2; restore base
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    A = combine(combine(L, R), combine(U, D))
    bends = combine(combine(intersection(L, U), intersection(L, D)), combine(intersection(R, U), intersection(R, D)))
    straights = combine(intersection(L, R), intersection(U, D))
    multi = combine(bends, straights)
    E = difference(A, multi)
    N1 = combine(shift(E, LEFT), shift(E, RIGHT))
    N2 = combine(shift(E, UP), shift(E, DOWN))
    N = intersection(combine(N1, N2), S)
    G0 = replace(I, c, ZERO)
    G1 = fill(G0, FOUR, E)
    G2 = fill(G1, TWO, N)
    O = underfill(G2, c, S)
    return O

def dsl4(I):
    # Geometric: convex corners of leastcolor as anchors (4); their immediate neighbors inside get 2
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    Eleft = difference(S, L)
    Eright = difference(S, R)
    Eup = difference(S, U)
    Edown = difference(S, D)
    C1 = intersection(Eleft, Eup)
    C2 = intersection(Eleft, Edown)
    C3 = intersection(Eright, Eup)
    C4 = intersection(Eright, Edown)
    E = combine(combine(C1, C2), combine(C3, C4))
    N1 = combine(shift(E, LEFT), shift(E, RIGHT))
    N2 = combine(shift(E, UP), shift(E, DOWN))
    N = intersection(combine(N1, N2), S)
    G = fill(I, FOUR, E)
    O = fill(G, TWO, N)
    return O

def dsl5(I):
    # Relational: junctions (deg>=2) of leastcolor get 2; endpoints (deg==1) get 4
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    straights = combine(intersection(L, R), intersection(U, D))
    bends = combine(combine(intersection(L, U), intersection(L, D)), combine(intersection(R, U), intersection(R, D)))
    J = combine(straights, bends)
    A = combine(combine(L, R), combine(U, D))
    E = difference(A, J)
    G = fill(I, TWO, J)
    O = fill(G, FOUR, E)
    return O
