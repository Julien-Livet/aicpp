def dsl1(I):
    # Global: remove separators by compressing across both axes via rotation
    A = rot90(I)
    B = compress(A)
    C = rot270(B)
    O = compress(C)
    return O

def dsl2(I):
    # Object extraction: crop to the union of all non-background objects
    OBJS = objects(I, T, F, T)
    MER = merge(OBJS)
    O = subgrid(MER, I)
    return O

def dsl3(I):
    # Color filtering: suppress color 1 by turning it into 8
    O = replace(I, ONE, EIGHT)
    return O

def dsl4(I):
    # Geometric: keep cells invariant under 180° rotation, else 8
    R = rot180(I)
    O = cellwise(I, R, EIGHT)
    return O

def dsl5(I):
    # Relational: draw each object's bounding box outline in its own color on an 8 canvas
    OBJS = objects(I, T, F, T)
    BX = apply(fork(recolor, color, box), OBJS)
    ALL = merge(BX)
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, ALL)
    return O
