def dsl1(I):
    # Global via compression: decide direction in compressed space, apply in original
    Ic = compress(I)
    P5c = ofcolor(Ic, FIVE)
    Bc = inbox(P5c)
    Gc = replace(Ic, SIX, NINE)
    P9outc = difference(ofcolor(Gc, NINE), Bc)
    Dc = sign(gravitate(box(Bc), P9outc))
    P5 = ofcolor(I, FIVE)
    B = inbox(P5)
    J = intersection(ofcolor(I, NINE), B)
    G = replace(I, SIX, NINE)
    M = shift(J, Dc)
    O = paint(fill(G, FIVE, J), recolor(NINE, M))
    return O

def dsl2(I):
    # Object extraction + move: move inner 9 by cardinal pull from the 5-block toward new 9s (from 6->9)
    P5 = ofcolor(I, FIVE)
    B = inbox(P5)
    J = intersection(ofcolor(I, NINE), B)
    G = replace(I, SIX, NINE)
    P9new = difference(ofcolor(G, NINE), ofcolor(I, NINE))
    D = sign(gravitate(B, P9new))
    MG = move(G, toobject(J, I), D)
    O = fill(MG, FIVE, J)
    return O

def dsl3(I):
    # Color filtering: isolate newly created 9s and pull the inner 9 one step along box-to-9s direction
    P5 = ofcolor(I, FIVE)
    B = inbox(P5)
    J = intersection(ofcolor(I, NINE), B)
    G = replace(I, SIX, NINE)
    P9new = difference(ofcolor(G, NINE), ofcolor(I, NINE))
    D = sign(gravitate(box(B), P9new))
    G2 = fill(G, FIVE, J)
    O = fill(G2, NINE, shift(J, D))
    return O

def dsl4(I):
