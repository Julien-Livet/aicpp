def dsl1(I):
    # Global color remapping: remove 1s and recolor all 8s to 7
    A = replace(I, ONE, ZERO)
    O = replace(A, EIGHT, SEVEN)
    return O

def dsl2(I):
    # Object extraction with bounding: recolor 8s outside the 1-box to 2
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    p1 = ofcolor(I, ONE)
    box1 = backdrop(p1)
    outside = difference(p8, box1)
    O = fill(base, TWO, outside)
    return O

def dsl3(I):
    # Pure color filtering: swap 8->2 and remove 1s
    S = switch(I, EIGHT, TWO)
    O = replace(S, ONE, ZERO)
    return O

def dsl4(I):
    # Geometric reasoning: mirror the 8-mask horizontally and paint it as 3
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    obj8 = toobject(p8, I)
    mir = hmirror(obj8)
    pm = toindices(mir)
    O = fill(base, THREE, pm)
    return O

def dsl5(I):
    # Relational rule: choose 2 vs 3 for 8s based on adjacency to 1s
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    p1 = ofcolor(I, ONE)
    col = branch(adjacent(p8, p1), TWO, THREE)
    O = fill(base, col, p8)
