def dsl1(I):
    # Global: fill the overall foreground bounding box with its dominant non-zero color
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    fg = merge(objs)
    dom = mostcolor(fg)
    O = paint(canvas(ZERO, shape(I)), recolor(dom, inbox(fg)))
    return O

def dsl2(I):
    # Object extraction: rectangularize only objects that touch the border (after removing 5)
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    keep = sfilter(objs, lbind(bordering, I2))
    parts = apply(fork(recolor, color, inbox), keep)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl3(I):
    # Color filtering: for each remaining color, draw the outline of its overall bounding box
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    cols = remove(ZERO, palette(I2))
    grp = compose(merge, lbind(colorfilter, objs))
    getbox = compose(box, compose(toindices, grp))
    parts = apply(fork(recolor, identity, getbox), cols)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl4(I):
    # Geometric: draw a diagonal path between opposite corners of each object (after removing 5)
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    diag = fork(connect, ulcorner, lrcorner)
    parts = apply(fork(recolor, color, diag), objs)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl5(I):
    # Relational: keep objects that share a row with any other object; rectangularize them
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    others_all = rbind(remove, objs)
    union_others = compose(merge, others_all)
    aligned = fork(hmatching, identity, union_others)
    keep = sfilter(objs, aligned)
    parts = apply(fork(recolor, color, inbox), keep)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O
