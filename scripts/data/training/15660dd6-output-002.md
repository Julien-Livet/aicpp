def dsl1(I):
    # Global: remove outer and internal full-8 separators by alternating trim/compress within a rotation frame
    O = rot270(compress(rot90(compress(trim(I)))))
    return O

def dsl2(I):
    # Object extraction: crop to the largest non-background object
    OBJS = fgpartition(I)
    BIG = argmax(OBJS, size)
    O = subgrid(BIG, I)
    return O

def dsl3(I):
    # Color filtering: remove the maximum palette color by turning it into 8
    P = palette(I)
    MX = maximum(P)
    O = replace(I, MX, EIGHT)
    return O

def dsl4(I):
    # Geometric: rotate, mirror vertically, and rotate back to re-align structures
    O = rot90(vmirror(rot270(I)))
    return O

def dsl5(I):
    # Relational/structural: draw each object's outline (box) in its own color on an 8 canvas
    OBJS = objects(I, T, F, T)
    OUTS = apply(fork(recolor, color, box), OBJS)
    ALL = merge(OUTS)
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, ALL)
    return O
