def dsl1(I):
    # Global diagonal band: 5s near the UR→LL diagonal -> 2, others -> 8
    S5 = ofcolor(I, FIVE)
    D = connect(urcorner(S5), llcorner(S5))
    B = combine(D, combine(shift(D, LEFT), shift(D, RIGHT)))
    S2 = intersection(S5, B)
    S8 = difference(S5, S2)
    O0 = fill(I, SEVEN, S5)
    O1 = fill(O0, TWO, S2)
    O = fill(O1, EIGHT, S8)
    return O

def dsl2(I):
    # Row-relational via shooting: leftmost 5s per row -> 2, others -> 8
    S5 = ofcolor(I, FIVE)
    RaysR = prapply(shoot, S5, initset(RIGHT))
    HasLeft = intersection(RaysR, S5)
    Leftmost = difference(S5, HasLeft)
    Rest = HasLeft
    O0 = fill(I, SEVEN, S5)
    O1 = fill(O0, TWO, Leftmost)
    O = fill(O1, EIGHT, Rest)
    return O

def dsl3(I):
    # Orientation-exclusive adjacency: (-45°) neighbors -> 2, (+45°) neighbors -> 8, others -> 2
    S5 = ofcolor(I, FIVE)
    Nneg = combine(shift(S5, DOWN_LEFT), shift(S5, UP_RIGHT))
    Npos = combine(shift(S5, DOWN_RIGHT), shift(S5, UP_LEFT))
    OriNeg = intersection(S5, Nneg)
    OriPos = intersection(S5, Npos)
    OnlyNeg = difference(OriNeg, OriPos)
    OnlyPos = difference(OriPos, OriNeg)
    Ambig = difference(S5, combine(OnlyNeg, OnlyPos))
    O0 = fill(I, SEVEN, S5)
    O1 = fill(O0, TWO, OnlyNeg)
    O2 = fill(O1, EIGHT, OnlyPos)
    O = fill(O2, TWO, Ambig)
    return O

def dsl4(I):
