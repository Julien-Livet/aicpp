def dsl1(I):
    # Global: crop, compress, fill background by modal color, then reinsert internal frontiers
    fgobjs = objects(I, T, F, T)
    fg = merge(fgobjs)
    G = subgrid(fg, I)
    C = compress(G)
    B = replace(C, ZERO, mostcolor(fg))
    E = merge(frontiers(C))
    O = paint(B, E)
    return O

def dsl2(I):
    # Object extraction: fill each object's backdrop with its own color over a modal-color canvas
    fgobjs = objects(I, T, F, T)
    fg = merge(fgobjs)
    G = subgrid(fg, I)
    base = canvas(mostcolor(fg), shape(G))
    modal = mostcolor(fg)
    modalobjs = colorfilter(fgobjs, modal)
    othersobjs = difference(fgobjs, modalobjs)
    B = apply(backdrop, othersobjs)
    C = apply(color, othersobjs)
    colored = mpapply(recolor, C, B)
    M = merge(colored)
    O = paint(base, M)
    return O

def dsl3(I):
    # Color filtering: fill crop background with modal color, then fill each non-modal inbox with its own color
    fgobjs = objects(I, T, F, T)
    fg = merge(fgobjs)
    G = subgrid(fg, I)
    modal = mostcolor(fg)
    base = replace(G, ZERO, modal)
    modalobjs = colorfilter(fgobjs, modal)
    othersobjs = difference(fgobjs, modalobjs)
    B = apply(inbox, othersobjs)
    C = apply(color, othersobjs)
    colored = mpapply(recolor, C, B)
    M = merge(colored)
    O = underpaint(base, M)
    return O

def dsl4(I):
    # Geometric: enforce horizontal symmetry by keeping only cells matching their mirror; others to modal color
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    B = replace(G, ZERO, mostcolor(fg))
    H = hmirror(B)
    O = cellwise(B, H, mostcolor(fg))
    return O

def dsl5(I):
    # Relational/structural: per-half delta mask underfill + preserve non-modal objects
    fgobjs = objects(I, T, F, T)
    fg = merge(fgobjs)
    G = subgrid(fg, I)
    L = lefthalf(G)
    R = righthalf(G)
    fL = merge(objects(L, T, F, T))
    fR = merge(objects(R, T, F, T))
    mL = normalize(delta(fL))
    mR = shift(normalize(delta(fR)), tojvec(width(L)))
    S = insert(mR, initset(mL))
    mask = merge(S)
    bg = underfill(G, mostcolor(fg), mask)
    modal = mostcolor(fg)
    othersobjs = difference(fgobjs, colorfilter(fgobjs, modal))
    O = underpaint(bg, normalize(merge(othersobjs)))
    return O
