def dsl1(I):
    # Global: crop nonzero region, compress, then replace zeros with dominant color of the crop
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    C = compress(A)
    O = replace(C, ZERO, mostcolor(A))
    return O

def dsl2(I):
    # Object extraction: keep non-dominant objects on a dominant-color canvas
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    DOM = mostcolor(A)
    BASE = canvas(DOM, shape(A))
    OBJS = objects(A, T, F, T)
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    EXP = merge(OTH)
    O = paint(BASE, EXP)
    return O

def dsl3(I):
    # Color filtering: fill all background within the crop with the dominant color
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    DOM = mostcolor(A)
    O = underfill(A, DOM, asindices(A))
    return O

def dsl4(I):
    # Geometric: fill background with dominant color, mirror left half to form right half
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    DOM = mostcolor(A)
    F = replace(A, ZERO, DOM)
    L = lefthalf(F)
    R = vmirror(L)
    O = hconcat(L, R)
    return O

def dsl5(I):
    # Relational: fill background with dominant color, expand each non-dominant object to its bounding box recolored by its own color
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    DOM = mostcolor(A)
    BASE = replace(A, ZERO, DOM)
    OBJS = objects(A, T, F, T)
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    TOT = totuple(OTH)
    COLS = totuple(apply(color, TOT))
    PATS = totuple(apply(compose(backdrop, toindices), TOT))
    COLORED = mpapply(recolor, COLS, PATS)
    EXP = merge(COLORED)
    O = paint(BASE, EXP)
    return O
