def dsl1(I):
    # Global overlay: corners -> 2, straight-segment endcaps -> 4
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    straighth = intersection(L, R)
    straightv = intersection(U, D)
    corners_any = intersection(combine(L, R), combine(U, D))
    corners = difference(corners_any, combine(straighth, straightv))
    hends = difference(combine(L, R), straighth)
    vends = difference(combine(U, D), straightv)
    endcaps = combine(hends, vends)
    G = fill(I, FOUR, endcaps)
    O = fill(G, TWO, corners)
    return O

def dsl2(I):

    c = leastcolor(I)

    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    H = intersection(L, R)
    V = intersection(U, D)
    corners = difference(intersection(combine(L, R), combine(U, D)), combine(H, V))
    hends = difference(combine(L, R), H)
    vends = difference(combine(U, D), V)
    endcaps = combine(hends, vends)
    O = fill(fill(I, FOUR, endcaps), TWO, corners)
    return O
