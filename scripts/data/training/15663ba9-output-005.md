def dsl1(I):
    objs = objects(I, T, F, T)
    tinds = toindices
    shL = rbind(shift, LEFT)
    shR = rbind(shift, RIGHT)
    hasR_fn = fork(intersection, tinds, compose(shL, tinds))  # has left neighbor
    hasL_fn = fork(intersection, tinds, compose(shR, tinds))  # has right neighbor
    noR_fn = fork(difference, tinds, hasR_fn)
    noL_fn = fork(difference, tinds, hasL_fn)
    rightEP_fn = fork(intersection, noR_fn, hasL_fn)
    leftEP_fn = fork(intersection, noL_fn, hasR_fn)
    RE_sets = apply(rightEP_fn, objs)
    LE_sets = apply(leftEP_fn, objs)
    RE = merge(RE_sets)
    LE = merge(LE_sets)
    endpoints = merge(insert(LE, initset(RE)))
    bg = ofcolor(I, ZERO)
    up_bg = shift(bg, UP)
    dn_bg = shift(bg, DOWN)
    up_end = intersection(endpoints, up_bg)
    dn_end = intersection(endpoints, dn_bg)
    bothUD = intersection(up_end, dn_end)
    o4 = recolor(FOUR, endpoints)
    o2 = recolor(TWO, bothUD)
    O = paint(I, o4)
    O = paint(O, o2)
    return O

def dsl4(I):
    objs = objects(I, T, F, T)
    tinds = toindices
    shL = rbind(shift, LEFT)
    shR = rbind(shift, RIGHT)
    hasR_fn = fork(intersection, tinds, compose(shL, tinds))  # has left neighbor
    hasL_fn = fork(intersection, tinds, compose(shR, tinds))  # has right neighbor
    union_fn = fork(merge, fork(insert, hasL_fn, compose(initset, hasR_fn)), compose(initset, initset))  # helper not directly used
    # Build endpoints via XOR = (hasL ∪ hasR) \ (hasL ∩ hasR)
    HL_sets = apply(hasL_fn, objs)
    HR_sets = apply(hasR_fn, objs)
    HL = merge(HL_sets)
    HR = merge(HR_sets)
    unionLR = merge(insert(HL, initset(HR)))
    bothLR = intersection(HL, HR)
    endpoints = difference(unionLR, bothLR)
    bg = ofcolor(I, ZERO)
    up_end = intersection(endpoints, shift(bg, UP))
    dn_end = intersection(endpoints, shift(bg, DOWN))
    bothUD = intersection(up_end, dn_end)
    O = paint(I, recolor(FOUR, endpoints))
    O = paint(O, recolor(TWO, bothUD))
    return O

def dsl5(I):
    bg = ofcolor(I, ZERO)
    U = asindices(I)
    nz = difference(U, bg)
    # sides by global bg adjacency
    leftSide = intersection(nz, shift(bg, RIGHT))
    rightSide = intersection(nz, shift(bg, LEFT))
    # require inward same-color neighbor to avoid singletons
    hasR_any = intersection(nz, shift(nz, RIGHT))
    hasL_any = intersection(nz, shift(nz, LEFT))
    leftEP = intersection(leftSide, hasR_any)
    rightEP = intersection(rightSide, hasL_any)
    endpoints = merge(insert(leftEP, initset(rightEP)))
    bothUD = intersection(intersection(endpoints, shift(bg, UP)), shift(bg, DOWN))
    O = paint(I, recolor(FOUR, endpoints))
    O = paint(O, recolor(TWO, bothUD))
    return O
