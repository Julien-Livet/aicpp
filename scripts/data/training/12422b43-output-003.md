def dsl1(I):
    # Global: replicate the top m-row motif (with 5->0) to fill the bottom, splice under the unchanged head
    W = width(I)
    H = height(I)
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    top_h = height(subgrid(NZ, I))
    M5 = ofcolor(I, FIVE)
    m = height(M5)
    tile = crop(I, ORIGIN, astuple(m, W))
    tile0 = replace(tile, FIVE, ZERO)
    obj0 = asobject(tile0)
    bottom_h = subtract(H, top_h)
    k = divide(bottom_h, m)
    times = interval(ZERO, k, ONE)
    iv = apply(toivec, times)
    step = astuple(m, ZERO)
    steps = apply(lbind(multiply, step), iv)
    offs = apply(lbind(add, astuple(top_h, ZERO)), steps)
    repobj = merge(mapply(lbind(shift, obj0), offs))
    head = crop(I, ORIGIN, astuple(top_h, W))
    tail = paint(canvas(ZERO, astuple(bottom_h, W)), repobj)
    O = vconcat(head, tail)
    return O

def dsl4(I):
    # Geometric: build a bottom-aligned periodic background from (5->0) over full height, then repaint original FG
    W = width(I)
    H = height(I)
    M5 = ofcolor(I, FIVE)
    m = height(M5)
    tile = crop(I, ORIGIN, astuple(m, W))
    tile0 = replace(tile, FIVE, ZERO)
    obj0 = asobject(tile0)
    kfull = divide(H, m)
    start = astuple(subtract(H, multiply(m, kfull)), ZERO)
    times = interval(ZERO, kfull, ONE)
    iv = apply(toivec, times)
    step = astuple(m, ZERO)
    steps = apply(lbind(multiply, step), iv)
    offs = apply(lbind(add, start), steps)
    baseobj = merge(mapply(lbind(shift, obj0), offs))
    base = paint(canvas(ZERO, astuple(H, W)), baseobj)
    FG = merge(objects(I, T, F, T))
    O = paint(base, FG)
    return O
