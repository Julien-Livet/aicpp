def dsl1(I):
    FG = fgpartition(I)
    Seeds = difference(FG, colorfilter(FG, ONE))
    ones = ofcolor(I, ONE)
    base = toindices
    hm = compose(hmirror, base)
    vm = compose(vmirror, base)
    dm = compose(dmirror, base)
    cm = compose(cmirror, base)
    u1 = fork(combine, base, hm)
    u2 = fork(combine, vm, dm)
    u3 = fork(combine, cm, u2)
    mirrors = fork(combine, u1, u3)
    maskf = compose(lbind(intersection, ones), mirrors)
    recolorf = fork(recolor, color, maskf)
    Paints = mapply(recolorf, Seeds)
    O = paint(I, Paints)
    return O
