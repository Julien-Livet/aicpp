def dsl1(I):
    # Relational: per-object vertical shift to be adjacent to top/bottom border rows (no horizontal shift)
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    TO = colorfilter(OB, cT)
    BO = colorfilter(OB, cB)
    Tbor = sfilter(TO, rbind(bordering, I))
    Bbor = sfilter(BO, rbind(bordering, I))
    Tnon = difference(TO, Tbor)
    Bnon = difference(BO, Bbor)
    H = height(I)
    Hm2 = subtract(H, TWO)
    drT = fork(subtract, lbind(identity, ONE), uppermost)
    vecT = compose(toivec, drT)
    ST = fork(shift, identity, vecT)
    drB = fork(subtract, lbind(identity, Hm2), lowermost)
    vecB = compose(toivec, drB)
    SB = fork(shift, identity, vecB)
    Tmov = apply(ST, Tnon)
    Bmov = apply(SB, Bnon)
    G = cover(cover(I, merge(Tnon)), merge(Bnon))
    O = paint(paint(G, merge(Tmov)), merge(Bmov))
    return O

def dsl2(I):
    # Global: project interior top/bottom colors vertically to the inward border rows using shoot
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    TO = colorfilter(OB, cT)
    BO = colorfilter(OB, cB)
    Tbor = sfilter(TO, rbind(bordering, I))
    Bbor = sfilter(BO, rbind(bordering, I))
    pT = ofcolor(I, cT)
    pB = ofcolor(I, cB)
    pBorT = toindices(merge(Tbor))
    pBorB = toindices(merge(Bbor))
    pNonT = difference(pT, pBorT)
    pNonB = difference(pB, pBorB)
    R1 = hfrontier(astuple(ONE, ZERO))
    H = height(I)
    Rb = hfrontier(astuple(subtract(H, TWO), ZERO))
    linesT = mapply(rbind(shoot, UP), pNonT)
    linesB = mapply(rbind(shoot, DOWN), pNonB)
    entryT = intersection(linesT, R1)
    entryB = intersection(linesB, Rb)
    BG = mostcolor(I)
    G0 = fill(I, BG, pNonT)
    G = fill(G0, BG, pNonB)
    O1 = fill(G, cT, entryT)
    O = fill(O1, cB, entryB)
    return O

def dsl4(I):
    # Geometric: replace each non-border object by the top/bottom edge of its inbox, shifted to the inward border rows
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    TO = colorfilter(OB, cT)
    BO = colorfilter(OB, cB)
    Tbor = sfilter(TO, rbind(bordering, I))
    Bbor = sfilter(BO, rbind(bordering, I))
    Tnon = difference(TO, Tbor)
    Bnon = difference(BO, Bbor)
    UL = compose(ulcorner, inbox)
    RL = compose(hfrontier, UL)
    TI = compose(inbox, identity)
    TopEdge = fork(intersection, RL, TI)
    BLL = compose(llcorner, inbox)
    BRL = compose(hfrontier, BLL)
    BotEdge = fork(intersection, BRL, TI)
    Tedge = apply(fork(recolor, color, TopEdge), Tnon)
    Bedge = apply(fork(recolor, color, BotEdge), Bnon)
    H = height(I)
    Hm2 = subtract(H, TWO)
    drT = fork(subtract, lbind(identity, ONE), uppermost)
    vecT = compose(toivec, drT)
    drB = fork(subtract, lbind(identity, Hm2), lowermost)
    vecB = compose(toivec, drB)
    ST = fork(shift, identity, vecT)
    SB = fork(shift, identity, vecB)
    Tmov = apply(ST, Tedge)
    Bmov = apply(SB, Bedge)
    G = cover(cover(I, merge(Tnon)), merge(Bnon))
    O = paint(paint(G, merge(Tmov)), merge(Bmov))
    return O
