def dsl1(I):
    # Global (via fgpartition): determine color from how 1s occupy bbox corners; recolor 8; remove 1
    o = first(colorfilter(fgpartition(I), ONE))
    P = toindices(o)
    crn = corners(backdrop(P))
    k = size(intersection(P, crn))
    col = branch(equality(k, FOUR), THREE, branch(equality(k, TWO), SEVEN, TWO))
    O = replace(switch(I, EIGHT, col), ONE, ZERO)
    return O
