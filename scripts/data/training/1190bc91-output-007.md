def dsl1(I):
    # Global symmetry/rotation overlay
    A = asobject(I)
    B = asobject(rot90(I))
    C = asobject(rot180(I))
    D = asobject(rot270(I))
    AB = combine(A, B)
    CD = combine(C, D)
    OBJ = combine(AB, CD)
    O = underpaint(I, OBJ)
    return O

def dsl2(I):
    # Object-based propagation along all diagonals
    FG = fgpartition(I)
    DIR0 = initset(UNITY)
    DIR1 = insert(NEG_UNITY, DIR0)
    DIR2 = insert(UP_RIGHT, DIR1)
    DIR = insert(DOWN_LEFT, DIR2)
    R0 = lbind(prapply, shoot)
    R1 = rbind(R0, DIR)
    PFN = compose(merge, R1)
    RC = fork(recolor, color, compose(PFN, toindices))
    PARTS = apply(RC, FG)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O

def dsl3(I):
    # Color filtering: propagate the most frequent nonzero color along all diagonals
    FG = fgpartition(I)
    ALL = merge(FG)
    C = mostcolor(ALL)
    S = ofcolor(I, C)
    DIR0 = initset(UNITY)
    DIR1 = insert(NEG_UNITY, DIR0)
    DIR2 = insert(UP_RIGHT, DIR1)
    DIR = insert(DOWN_LEFT, DIR2)
    R0 = lbind(prapply, shoot)
    R1 = rbind(R0, DIR)
    LINES = compose(merge, R1)(S)
    COLORED = recolor(C, LINES)
    O = underpaint(I, COLORED)
    return O

def dsl4(I):
    # Geometric alignment: draw horizontal and vertical frontiers through each object's center
    FG = fgpartition(I)
    HR = compose(hfrontier, center)
    VR = compose(vfrontier, center)
    AXES = fork(combine, HR, VR)
    RC = fork(recolor, color, AXES)
    PARTS = apply(RC, FG)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O

def dsl5(I):
    # Relational: from each object, shoot towards the nearest grid border (and opposite) using gravitate
    FG = fgpartition(I)
    BORDER = box(asindices(I))
    GFN = rbind(gravitate, BORDER)
    DFN = compose(sign, GFN, toindices)
    SD = compose(initset, DFN)
    DIRS = fork(insert, compose(invert, DFN), SD)
    R0 = lbind(prapply, shoot)
    RAW = fork(R0, toindices, DIRS)
    PFN = compose(merge, RAW)
    RC = fork(recolor, color, PFN)
    PARTS = apply(RC, FG)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O
