def dsl1(I):
    objs = objects(I, T, F, T)
    mc = mostcolor(I)
    crossobjs = colorfilter(objs, mc)
    cross = merge(crossobjs)
    others = difference(objs, crossobjs)
    c = centerofmass(cross)
    tocent = compose(rbind(subtract, c), centerofmass)
    sgn = compose(sign, tocent)
    isTL = matcher(sgn, NEG_UNITY)
    isTR = matcher(sgn, UP_RIGHT)
    isBL = matcher(sgn, DOWN_LEFT)
    isBR = matcher(sgn, UNITY)
    base = canvas(ZERO, shape(I))
    TL = compress(paint(base, merge(sfilter(others, isTL))))
    TR = compress(paint(base, merge(sfilter(others, isTR))))
    BL = compress(paint(base, merge(sfilter(others, isBL))))
    BR = compress(paint(base, merge(sfilter(others, isBR))))
    top = hconcat(TL, TR)
    bottom = hconcat(BL, BR)
    O = compress(vconcat(top, bottom))
