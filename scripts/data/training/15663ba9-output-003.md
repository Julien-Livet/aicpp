def dsl1(I):
    # Global neighbor-parity endpoints: exactly-one horizontal neighbor -> 4; exactly-one vertical neighbor -> 2
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    hend = difference(combine(L, R), intersection(L, R))
    vend = difference(combine(U, D), intersection(U, D))
    G = fill(I, FOUR, hend)
    O = fill(G, TWO, vend)
    return O

def dsl2(I):
    # Object extraction: draw each object's bounding box outline in 4
    c = leastcolor(I)
    objs = objects(I, T, F, T)
    outlines = mapply(box, objs)
    O = fill(I, FOUR, outlines)
    return O

def dsl3(I):
    # Color filtering: exact L-corners -> 4; true endpoints -> 2; restore original color elsewhere
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    straighth = intersection(L, R)
    straightv = intersection(U, D)
    corners_any = intersection(combine(L, R), combine(U, D))
    corners_exact = difference(corners_any, combine(straighth, straightv))
    deg2plus = combine(corners_any, combine(straighth, straightv))
    endpoints = difference(S, deg2plus)
    G0 = replace(I, c, ZERO)
    G1 = fill(G0, FOUR, corners_exact)
    G2 = fill(G1, TWO, endpoints)
    O = underfill(G2, c, S)
    return O

def dsl4(I):
    # Geometric reasoning: mark the four corners of each object's bounding box with 2
    c = leastcolor(I)
    objs = objects(I, T, F, T)
    corner_pts = mapply(corners, objs)
    O = fill(I, TWO, corner_pts)
    return O
