def dsl1(I):
    idx = asindices(I)
    at = compose(rbind(index, I), identity)
    shR = compose(rbind(add, RIGHT), identity)
    shD = compose(rbind(add, DOWN), identity)
    shL = compose(rbind(add, LEFT), identity)
    shU = compose(rbind(add, UP), identity)
    valR = compose(at, shR)
    valD = compose(at, shD)
    valL = compose(at, shL)
    valU = compose(at, shU)
    eq8 = rbind(equality, EIGHT)
    eq0 = rbind(equality, ZERO)
    c_here = compose(eq8, at)
    c_r = compose(eq8, valR)
    c_d = compose(eq8, valD)
    c_l0 = compose(eq0, valL)
    c_u0 = compose(eq0, valU)
    c1 = fork(both, c_here, c_r)
    c2 = fork(both, c_d, c_l0)
    c3 = fork(both, c1, c2)
    cond = fork(both, c3, c_u0)
    uls = sfilter(idx, cond)
    seedpos = compose(rbind(add, NEG_UNITY), identity)
    seedcol = compose(rbind(index, I), seedpos)
    nz = compose(flip, compose(eq0, seedcol))
    sel = sfilter(uls, nz)
    dims = double(TWO_BY_TWO)
    off = decrement(dims)
    lrcalc = compose(rbind(add, off), identity)
    cset = fork(combine, compose(initset, identity), compose(initset, lrcalc))
    rect = compose(backdrop, cset)
    diamond = fork(difference, rect, compose(corners, rect))
    recol = apply(fork(recolor, seedcol, diamond), sel)
    obj = merge(recol)
    spatches = apply(compose(initset, seedpos), sel)
    spatch = merge(spatches)
    base = fill(I, ZERO, spatch)
    O = paint(base, obj)
    return O
