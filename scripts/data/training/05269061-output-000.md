def dsl2(I):
    O0 = I
    O1 = underpaint(O0, asobject(rot90(I)))

def dsl3(I):
    OBJS = fgpartition(I)
    A = first(OBJS)
    OBJS2 = remove(A, OBJS)
    B = first(OBJS2)
    OBJS3 = remove(B, OBJS2)
    C = first(OBJS3)
    IA = toindices(A)
    HA = merge(apply(hfrontier, IA))
    VA = merge(apply(vfrontier, IA))
    MA = combine(HA, VA)
    SA = recolor(color(A), MA)
    IB = toindices(B)
    HB = merge(apply(hfrontier, IB))
    VB = merge(apply(vfrontier, IB))
    MB = combine(HB, VB)
    SB = recolor(color(B), MB)
    IC = toindices(C)
    HC = merge(apply(hfrontier, IC))
    VC = merge(apply(vfrontier, IC))
    MC = combine(HC, VC)
    SC = recolor(color(C), MC)
    SET = insert(SA, initset(SB))
    SET2 = insert(SC, SET)
    BIG = merge(SET2)
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O

def dsl5(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    TILE = subgrid(NZ, I)
    H = height(I)
    W = width(I)
    TH = height(TILE)
    TW = width(TILE)
    RO = interval(0, H, TH)
    CO = interval(0, W, TW)
    RV = apply(toivec, RO)
    CV = apply(tojvec, CO)
    OFFS = prapply(add, RV, CV)
    BASE = asobject(TILE)
    SHIFTED = mapply(lbind(shift, BASE), OFFS)
    O = paint(canvas(ZERO, shape(I)), SHIFTED)
    return O
