def dsl1(I):
    E = ofcolor(I, EIGHT)
    S = ofcolor(I, SEVEN)
    OI = difference(asindices(I), combine(E, S))
    OO = toobject(OI, I)
    RE = vmirror(OO)
    B = cover(I, OO)
    O = underpaint(B, RE)

def dsl5(I):
    OBJS = objects(I, T, F, T)
    OE = colorfilter(OBJS, EIGHT)
    OS = colorfilter(OBJS, SEVEN)
    FG = difference(OBJS, combine(OE, OS))
    IDX = apply(toindices, FG)
    E = ofcolor(I, EIGHT)
    DIRS = apply(rbind(gravitate, E), IDX)
    MOV = mapply(shift, FG, DIRS)
    BASE = cover(I, merge(IDX))
    O = underpaint(BASE, merge(MOV))
