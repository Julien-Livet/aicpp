def dsl1(I):
    G = trim(I)
    C = compress(G)
    R = rot90(C)
    H = height(I)
    W = width(I)
    HR = height(R)
    WR = width(R)
    Zb = canvas(ZERO, astuple(subtract(H, HR), WR))
    V = vconcat(R, Zb)
    Zr = canvas(ZERO, astuple(H, subtract(W, WR)))
    O = hconcat(V, Zr)
    return O
