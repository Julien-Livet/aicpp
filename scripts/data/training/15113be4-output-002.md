def dsl2(I):
    val = leastcolor(I)
    m = ofcolor(I, val)
    parts = fgpartition(I)
    onesobjs = colorfilter(parts, ONE)
    aligned = sfilter(onesobjs, lbind(hmatching, m))
    target = merge(apply(toindices, aligned))
    O = fill(I, val, target)
    return O

def dsl3(I):
    val = leastcolor(I)
    idx = ofcolor(I, val)
    rightshots = merge(apply(rbind(shoot, RIGHT), idx))
    leftshots = merge(apply(rbind(shoot, LEFT), idx))
    lines = combine(rightshots, leftshots)
    target = intersection(lines, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl5(I):
    val = leastcolor(I)
