def dsl1(I):
    # Object-based morphological expansion (8-neighborhood)
    objs = objects(I, T, F, T)
    n4 = compose(lbind(mapply, neighbors), toindices)
    n8 = compose(lbind(mapply, ineighbors), toindices)
    expand = fork(combine, fork(combine, toindices, n4), n8)
    ex = apply(fork(recolor, color, expand), objs)
    allcells = merge(ex)
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl2(I):
    # Geometric replication of object frames in a 3x3 grid using object size as step
    objs = objects(I, T, F, T)
    frame = fork(recolor, color, compose(box, toindices))
    wv = compose(tojvec, width)
    hv = compose(toivec, height)
    nwv = compose(invert, wv)
    nhv = compose(invert, hv)
    base = apply(frame, objs)
    r = apply(fork(shift, frame, wv), objs)
    l = apply(fork(shift, frame, nwv), objs)
    d = apply(fork(shift, frame, hv), objs)
    u = apply(fork(shift, frame, nhv), objs)
    rd = apply(fork(shift, frame, fork(add, wv, hv)), objs)
    ru = apply(fork(shift, frame, fork(add, wv, nhv)), objs)
    ld = apply(fork(shift, frame, fork(add, nwv, hv)), objs)
    lu = apply(fork(shift, frame, fork(add, nwv, nhv)), objs)
    allsets = combine(combine(combine(combine(base, r), combine(l, d)),
                              combine(u, combine(rd, ru))),
                      combine(ld, lu))
    allcells = merge(allsets)
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl3(I):
    # Color filtering: fill each object's bounding rectangle with its color
    objs = objects(I, T, F, T)
    rect = compose(backdrop, toindices)
    filled = apply(fork(recolor, color, rect), objs)
    allcells = merge(filled)
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl4(I):
    # Global transformation: extract and paint colored frontiers (edges)
    fr = frontiers(I)
    cfr = apply(rbind(toobject, I), fr)
    allcells = merge(cfr)
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl5(I):
    # Relational reasoning: connect object center to its four corners
    objs = objects(I, T, F, T)
    cen = compose(centerofmass, toindices)
    ul = compose(ulcorner, toindices)
    ur = compose(urcorner, toindices)
    ll = compose(llcorner, toindices)
    lr = compose(lrcorner, toindices)
    lines = fork(combine,
                 fork(combine, fork(connect, cen, ul), fork(connect, cen, ur)),
                 fork(combine, fork(connect, cen, ll), fork(connect, cen, lr)))
    drawn = apply(fork(recolor, color, lines), objs)
    allcells = merge(drawn)
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O
