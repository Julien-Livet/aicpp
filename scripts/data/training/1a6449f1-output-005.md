def dsl1(I):
    # Interior of the bounding box of all nonzero cells
    nz = remove(ZERO, palette(I))
    patches = apply(lbind(ofcolor, I), nz)
    allnz = merge(patches)
    inner = delta(allnz)
    O = subgrid(inner, I)
    return O

def dsl2(I):
    # Interior of the largest border-touching object
    objs = objects(I, T, F, T)
    borderobjs = sfilter(objs, rbind(bordering, I))
    big = argmax(borderobjs, compose(size, delta))
    inner = delta(big)
    O = subgrid(inner, I)
    return O

def dsl3(I):
    # Interior of the color whose union has the largest interior
    nz = remove(ZERO, palette(I))
    f1 = lbind(ofcolor, I)
    f2 = lbind(toobject, I)
    g = compose(f2, f1)
    h = compose(delta, g)
    comp = compose(size, h)
    mc = valmax(nz, comp)
    obj = toobject(ofcolor(I, mc), I)
    inner = delta(obj)
    O = subgrid(inner, I)
    return O

def dsl4(I):
    # Interior defined by the union of all horizontal and vertical line objects
    objs = objects(I, T, F, T)
    hls = sfilter(objs, hline)
    vls = sfilter(objs, vline)
    lines = merge(combine(hls, vls))
    inner = delta(lines)
    O = subgrid(inner, I)
    return O

def dsl5(I):
    # Interior from the dominant border color's border-touching union
    objs = objects(I, T, F, T)
    borderobjs = sfilter(objs, rbind(bordering, I))
    cols = remove(ZERO, palette(I))
    cf = lbind(colorfilter, objs)
    ib = lbind(intersection, borderobjs)
    mrg = compose(merge, ib)
    dsz = chain(size, delta, mrg)
    comp = compose(dsz, cf)
    bc = valmax(cols, comp)
    sel = intersection(borderobjs, colorfilter(objs, bc))
    bd = merge(sel)
    inner = delta(bd)
    O = subgrid(inner, I)
    return O
