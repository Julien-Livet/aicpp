def dsl1(I):
    ones = ofcolor(I, ONE)
    ul = ulcorner(ones)
    lr = lrcorner(ones)
    k = decrement(size(ones))
    step = divide(subtract(lr, ul), k)
    start = add(lr, step)
    path = intersection(shoot(start, step), asindices(I))
    O = underpaint(I, recolor(TWO, path))
    return O

def dsl2(I):
    ones = ofcolor(I, ONE)
    ul = ulcorner(ones)
    lr = lrcorner(ones)
    k = decrement(size(ones))
    seg = connect(ul, lr)
    gaps = decrement(size(seg))
    per = divide(gaps, k)
    dir = sign(subtract(lr, ul))
    step = multiply(dir, per)
    start = add(lr, step)
    path = intersection(shoot(start, step), asindices(I))
    O = underpaint(I, recolor(TWO, path))
    return O

def dsl3(I):
    allidx = asindices(I)
    ones = ofcolor(I, ONE)
    ul = ulcorner(ones)
    lr = lrcorner(ones)
    k = decrement(size(ones))
    step = divide(subtract(lr, ul), k)
    start = add(lr, step)
    rect = backdrop(combine(initset(start), initset(lrcorner(allidx))))
    ray = intersection(shoot(ul, step), rect)
    O = underpaint(I, recolor(TWO, ray))
    return O

def dsl4(I):

    rlr = lrcorner(rones)
