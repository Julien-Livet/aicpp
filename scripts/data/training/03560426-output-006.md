def dsl1(I):
    R = rot90(I)
    C = compress(R)
    U = rot270(C)
    H = height(I)
    W = width(I)
    HU = height(U)
    WU = width(U)
    Zb = canvas(ZERO, astuple(subtract(H, HU), WU))
    T = vconcat(U, Zb)
    Zr = canvas(ZERO, astuple(H, subtract(W, WU)))
    O = hconcat(T, Zr)
    return O

def dsl2(I):
    A = fgpartition(I)
    FN = compose(toindices, normalize)
    FRC = fork(recolor, color, FN)

    U = merge(R)
    Z = canvas(ZERO, shape(I))
    O = paint(Z, U)
    return O
