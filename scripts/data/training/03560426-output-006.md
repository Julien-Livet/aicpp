def dsl1(I):
    T0 = trim(I)
    P = rot90(T0)
    hI = height(I)
    wI = width(I)
    hP = height(P)
    wP = width(P)
    rightpad = canvas(ZERO, astuple(hP, subtract(wI, wP)))
    toprow = hconcat(P, rightpad)
    bottompad = canvas(ZERO, astuple(subtract(hI, hP), wI))
    O = vconcat(toprow, bottompad)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    ls = apply(leftmost, objs)
    minl = valmin(ls, identity)
    dx = compose(rbind(subtract, minl), leftmost)
    vvec = compose(toivec, dx)
    hvec = compose(tojvec, dx)
    off = fork(add, vvec, hvec)
    sh = fork(shift, normalize, off)
    g = fork(recolor, color, sh)
    parts = apply(g, objs)
    packed = merge(parts)
    base = canvas(ZERO, shape(I))
    O = paint(base, packed)
    return O
