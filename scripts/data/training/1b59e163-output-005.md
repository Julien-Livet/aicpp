def dsl1(I):
    # Global: remove 5s, overlay with vertical mirror of entire grid
    J = replace(I, FIVE, FOUR)
    base = canvas(FOUR, shape(J))
    O1 = underpaint(base, asobject(J))
    M = vmirror(J)
    O = underpaint(O1, asobject(M))
    return O

def dsl2(I):
    # Object extraction: take largest object and overlay its vertical-mirrored counterpart
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    big = argmax(objs, size)
    base = canvas(FOUR, shape(J))
    g = paint(base, big)
    mobjs = objects(vmirror(g), T, F, T)
    bigm = argmax(mobjs, size)
    O1 = underpaint(base, big)
    O = underpaint(O1, bigm)
    return O

def dsl3(I):
    # Color filtering: keep most common foreground color and add its vertical mirror
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    cols = apply(color, objs)
    cstar = mostcommon(totuple(cols))
    foc = colorfilter(objs, cstar)
    merged = merge(foc)
    base = canvas(FOUR, shape(J))
    G = paint(base, merged)
    M = vmirror(G)
    O = underpaint(G, asobject(M))
    return O

def dsl4(I):
    # Geometric: mirror only the foreground across the main diagonal
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    fg = difference(allidx, ofcolor(J, FOUR))
    base = canvas(FOUR, shape(J))
    G = paint(base, toobject(fg, J))
    D = dmirror(G)
    O = underpaint(base, asobject(D))
    return O

def dsl5(I):
    # Relational: translate foreground so its center-of-mass aligns with grid center
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    fg = difference(allidx, ofcolor(J, FOUR))
    src = toobject(fg, J)
    v = subtract(center(allidx), centerofmass(fg))
    O = move(canvas(FOUR, shape(J)), src, v)
    return O
