def dsl1(I):
    # Global transformation: remove frontiers then rotate
    O = rot90(compress(I))
    return O

def dsl2(I):
    # Object extraction: crop around all horizontal-line objects
    objs = objects(I, T, F, T)
    hobjs = mfilter(objs, hline)
    O = subgrid(merge(hobjs), I)
    return O

def dsl3(I):
    # Color filtering: crop around the most frequent non-zero color
    nz = remove(ZERO, palette(I))
    comp = compose(size, lbind(ofcolor, I))
    mc = valmax(nz, comp)
    O = subgrid(ofcolor(I, mc), I)
    return O

def dsl4(I):
    # Geometric reasoning: trim to foreground and mirror horizontally
    O = hmirror(trim(I))
    return O

def dsl5(I):
    # Relational reasoning: remove border-touching objects, then trim
    objs = objects(I, T, F, T)
    bord = mfilter(objs, rbind(bordering, I))
    inner = cover(I, merge(bord))
    O = trim(inner)
    return O
