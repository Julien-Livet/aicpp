def dsl1(I):
    # Global: crop the FIVE band, remove FIVE/zeros, and tile that band downward by its height
    A = ofcolor(I, FIVE)
    top = uppermost(A)
    bot = lowermost(A)
    k = add(subtract(bot, top), ONE)
    W = width(I)
    B = crop(I, astuple(top, ZERO), astuple(k, W))
    NZ = difference(asindices(B), ofcolor(B, ZERO))
    Sel = difference(NZ, ofcolor(B, FIVE))
    P = toobject(Sel, B)
    Base = shift(P, toivec(top))
    vec = multiply(DOWN, k)
    n = divide(height(I), k)
    steps = interval(ONE, add(n, ONE), ONE)
    Offs = apply(lbind(multiply, vec), steps)
    U = mapply(lbind(shift, Base), Offs)
    O = underpaint(I, U)
    return O
