def dsl1(I):
    # Global: remove only horizontal separator rows (8-lines) via rotated compress
    A = rot90(I)
    B = compress(A)
    O = rot270(B)
    return O

def dsl2(I):
    # Object extraction: crop to all non-1 foreground by removing color-1 indices before subgrid
    OBJS = objects(I, T, F, T)
    MER = merge(OBJS)
    IDX = toindices(MER)
    C1 = ofcolor(I, ONE)
    FG = difference(IDX, C1)
    OBJ = toobject(FG, I)
    O = subgrid(OBJ, I)
    return O

def dsl3(I):
    # Color filtering: suppress color 1, then globally compress separators
    A = replace(I, ONE, EIGHT)
    O = compress(A)
    return O

def dsl4(I):
    # Geometric: keep cells invariant under vertical mirror, else 8
    R = vmirror(I)
    O = cellwise(I, R, EIGHT)
    return O

def dsl5(I):
    # Relational: fill each object's bounding box (inbox) with its color on an 8 canvas
    OBJS = objects(I, T, F, T)
    INB = apply(fork(recolor, color, inbox), OBJS)
    ALL = merge(INB)
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, ALL)
    return O
