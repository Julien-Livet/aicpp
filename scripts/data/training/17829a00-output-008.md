def dsl1(I):
    # Global column selection per half: fill entire columns within each half if that column contains any non-background cell in that half
    BG = mostcolor(I)
    H = height(I)
    W = width(I)
    Half = halve(H)
    TT = interval(ZERO, Half, ONE)
    TB = interval(Half, H, ONE)
    FT = mapply(hfrontier, apply(toivec, TT))
    FB = mapply(hfrontier, apply(toivec, TB))
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    cT = color(argmin(BOR, uppermost))
    cB = color(argmax(BOR, lowermost))
    R0 = hfrontier(astuple(ZERO, ZERO))
    Rb = hfrontier(astuple(subtract(H, ONE), ZERO))
    ALL = asindices(I)
    NB = difference(ALL, ofcolor(I, BG))
    TNB = intersection(NB, FT)
    BNB = intersection(NB, FB)
    CJ = interval(ZERO, W, ONE)
    VC = mapply(vfrontier, apply(tojvec, CJ))
    condTc = compose(positive, compose(size, compose(rbind(intersection, TNB), identity)))
    condBc = compose(positive, compose(size, compose(rbind(intersection, BNB), identity)))
    Tcols = sfilter(VC, condTc)
    Bcols = sfilter(VC, condBc)
    Tmask = difference(intersection(merge(Tcols), FT), R0)
    Bmask = difference(intersection(merge(Bcols), FB), Rb)
    C = canvas(BG, shape(I))
    O0 = fill(C, cT, Tmask)
    O1 = fill(O0, cB, Bmask)
    O2 = fill(O1, cT, R0)
    O = fill(O2, cB, Rb)
    return O

def dsl4(I):
    # Geometric relational: connect centers of interior objects to centers of the nearest border object within each half; paint those connections with border colors; redraw borders
    BG = mostcolor(I)
    H = height(I)
    Half = halve(H)
    TT = interval(ZERO, Half, ONE)
    TB = interval(Half, H, ONE)
    FT = mapply(hfrontier, apply(toivec, TT))
    FB = mapply(hfrontier, apply(toivec, TB))
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TopB = argmin(BOR, uppermost)
    BotB = argmax(BOR, lowermost)
    cT = color(TopB)
    cB = color(BotB)
    R0 = hfrontier(astuple(ZERO, ZERO))
    Rb = hfrontier(astuple(subtract(H, ONE), ZERO))
    INOB = sfilter(OB, compose(flip, rbind(bordering, I)))
    condT = compose(positive, compose(size, compose(rbind(intersection, FT), toindices)))
    condB = compose(positive, compose(size, compose(rbind(intersection, FB), toindices)))
    TO = sfilter(INOB, condT)
    BO = sfilter(INOB, condB)
    CMt = centerofmass(TopB)
    CMb = centerofmass(BotB)
    Tpts = apply(centerofmass, TO)
    Bpts = apply(centerofmass, BO)
    Tlines = apply(rbind(connect, CMt), Tpts)
    Blines = apply(rbind(connect, CMb), Bpts)
    Tmask = difference(merge(Tlines), R0)
    Bmask = difference(merge(Blines), Rb)
    C = canvas(BG, shape(I))
    O0 = fill(C, cT, Tmask)
    O1 = fill(O0, cB, Bmask)
    O2 = fill(O1, cT, R0)
    O = fill(O2, cB, Rb)
    return O
