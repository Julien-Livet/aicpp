def dsl1(I):
    C = compress(I)
    R = rot90(C)
    H = height(I)
    W = width(I)
    HR = height(R)
    WR = width(R)
    Db = astuple(subtract(H, HR), WR)
    Zb = canvas(ZERO, Db)
    T = vconcat(R, Zb)
    Dr = astuple(H, subtract(W, WR))
    Zr = canvas(ZERO, Dr)
    O = hconcat(T, Zr)
    return O

def dsl3(I):
    A = objects(I, T, F, T)
    N = apply(normalize, A)
    C = apply(color, A)
    Vi = apply(toivec, C)
    Vj = apply(tojvec, C)
    V = mpapply(add, Vi, Vj)
    NI = apply(toindices, N)
    SI = mpapply(shift, NI, V)
    RO = mpapply(recolor, C, SI)
    U = merge(RO)
    Z = canvas(ZERO, shape(I))
    O = paint(Z, U)
    return O
