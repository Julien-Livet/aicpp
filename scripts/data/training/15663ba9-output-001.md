def dsl1(I):
    # Relational/structural: classify endpoints (deg=1) and corners/junctions (has both horiz and vert neighbors)
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    H = combine(L, R)
    V = combine(U, D)
    corners = intersection(H, V)
    straighth = intersection(L, R)
    straightv = intersection(U, D)
    deg2 = combine(corners, combine(straighth, straightv))
    endpoints = difference(S, deg2)
    G = fill(I, TWO, corners)
    O = fill(G, FOUR, endpoints)
    return O

def dsl2(I):
    # Object-extraction flavored: same local rule, structured via directional neighbor sets
    c = leastcolor(I)
    S = ofcolor(I, c)
    hasL = intersection(S, shift(S, RIGHT))
    hasR = intersection(S, shift(S, LEFT))
    hasU = intersection(S, shift(S, DOWN))
    hasD = intersection(S, shift(S, UP))
    horiz2 = intersection(hasL, hasR)
    vert2 = intersection(hasU, hasD)
    hv_any = combine(combine(hasL, hasR), combine(hasU, hasD))
    hv_cross = intersection(combine(hasL, hasR), combine(hasU, hasD))
    deg2plus = combine(hv_cross, combine(horiz2, vert2))
    ends = difference(S, deg2plus)
    G = fill(I, TWO, hv_cross)
    O = fill(G, FOUR, ends)
    return O

def dsl3(I):
    # Color filtering and layering: remove line color, place specials, then underfill straights with original color
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    corners = intersection(combine(L, R), combine(U, D))
    ends = difference(S, combine(corners, combine(intersection(L, R), intersection(U, D))))
    G0 = replace(I, c, ZERO)
    G1 = fill(G0, TWO, corners)
    G2 = fill(G1, FOUR, ends)
    O = underfill(G2, c, S)
    return O

def dsl4(I):

    c = leastcolor(I)

    objs = objects(I, T, F, T)

    O = fill(boxed, TWO, corners)

def dsl5(I):
    # Alternative structural formulation: endpoints = S minus any pairwise-neigh unions; corners by orthogonal pairs
    c = leastcolor(I)
    S = ofcolor(I, c)

    U = intersection(S, shift(S, DOWN))

    ld = intersection(L, D)
    ru = intersection(R, U)
    rd = intersection(R, D)
    hv_corners = combine(combine(lu, ld), combine(ru, rd))
    deg2plus = combine(hv_corners, combine(lr, ud))
    ends = difference(S, deg2plus)
    G = fill(I, TWO, hv_corners)
    O = fill(G, FOUR, ends)
