def dsl1(I):
    # Global halfwise column projection to nearest interior row; repaint borders
    BG = mostcolor(I)
    H = height(I)
    W = width(I)
    Half = halve(H)
    TT = interval(ZERO, Half, ONE)
    TB = interval(Half, H, ONE)
    FTset = mapply(hfrontier, apply(toivec, TT))
    FBset = mapply(hfrontier, apply(toivec, TB))
    FT = merge(FTset)
    FB = merge(FBset)
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TopB = argmin(BOR, uppermost)
    BotB = argmax(BOR, lowermost)
    cT = color(TopB)
    cB = color(BotB)
    R0 = hfrontier(astuple(ZERO, ZERO))
    Rb = hfrontier(astuple(subtract(H, ONE), ZERO))
    R1 = shift(R0, DOWN)
    Rm1 = shift(Rb, UP)
    NB = difference(asindices(I), ofcolor(I, BG))
    TNB = intersection(NB, FT)
    BNB = intersection(NB, FB)
    CJ = interval(ZERO, W, ONE)
    VC = mapply(vfrontier, apply(tojvec, CJ))
    condTc = compose(positive, compose(size, compose(rbind(intersection, TNB), identity)))
    condBc = compose(positive, compose(size, compose(rbind(intersection, BNB), identity)))
    Tcols = sfilter(VC, condTc)
    Bcols = sfilter(VC, condBc)
    Tmask = intersection(merge(Tcols), R1)
    Bmask = intersection(merge(Bcols), Rm1)
    C = canvas(BG, shape(I))
    O0 = fill(C, cT, Tmask)
    O1 = fill(O0, cB, Bmask)
    O2 = fill(O1, cT, R0)
    O = fill(O2, cB, Rb)
    return O

def dsl2(I):
    # Object extraction and sliding: gravitate interior objects in each half to their border; recolor; redraw borders
    BG = mostcolor(I)
    H = height(I)
    Half = halve(H)
    TT = interval(ZERO, Half, ONE)
    TB = interval(Half, H, ONE)
    FT = merge(mapply(hfrontier, apply(toivec, TT)))
    FB = merge(mapply(hfrontier, apply(toivec, TB)))
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TopB = argmin(BOR, uppermost)
    BotB = argmax(BOR, lowermost)
    cT = color(TopB)
    cB = color(BotB)
    R0 = hfrontier(astuple(ZERO, ZERO))
    Rb = hfrontier(astuple(subtract(H, ONE), ZERO))
    INOB = sfilter(OB, compose(flip, rbind(bordering, I)))
    condT = compose(positive, compose(size, compose(rbind(intersection, FT), toindices)))
    condB = compose(positive, compose(size, compose(rbind(intersection, FB), toindices)))
    TO = sfilter(INOB, condT)
    BO = sfilter(INOB, condB)
    TOI = apply(toindices, TO)
    BOI = apply(toindices, BO)
    Tdir = apply(rbind(gravitate, TopB), TOI)
    Bdir = apply(rbind(gravitate, BotB), BOI)
    Tshifts = mpapply(shift, TOI, Tdir)
    Bshifts = mpapply(shift, BOI, Bdir)
    Tmask = difference(merge(Tshifts), R0)
    Bmask = difference(merge(Bshifts), Rb)
    C = canvas(BG, shape(I))
    O0 = fill(C, cT, Tmask)
    O1 = fill(O0, cB, Bmask)
    O2 = fill(O1, cT, R0)
    O = fill(O2, cB, Rb)
    return O
