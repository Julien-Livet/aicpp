def dsl1(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    H = hmirror(OBJ)
    V = vmirror(OBJ)
    C = cmirror(OBJ)
    S = combine(combine(OBJ, H), combine(V, C))
    O = paint(canvas(ZERO, shape(I)), S)
    return O

def dsl2(I):
    FG = fgpartition(I)
    BIG = argmax(FG, size)
    C = color(BIG)
    STARTS = toindices(BIG)
    DIR = insert(UP_RIGHT, initset(DOWN_LEFT))
    R0 = lbind(prapply, shoot)
    R1 = rbind(R0, DIR)
    PFN = compose(merge, R1)
    LINES = PFN(STARTS)
    COLORED = recolor(C, LINES)
    O = underpaint(I, COLORED)
    return O

def dsl4(I):
    FG = fgpartition(I)
    RC = fork(recolor, color, box)
    PARTS = apply(RC, FG)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O
