    A = ofcolor(I, FIVE)
    top = uppermost(A)
    bot = lowermost(A)
    k = add(subtract(bot, top), ONE)
    W = width(I)
    B = crop(I, ORIGIN, astuple(k, W))
    Ab = ofcolor(B, FIVE)
    RowMask = merge(mapply(hfrontier, Ab))
    NZ = difference(asindices(B), ofcolor(B, ZERO))

    P0 = toobject(Sel, B)
    vec = multiply(DOWN, k)
    n = divide(height(I), k)

    U = merge(mapply(lbind(shift, P0), Offs))
    O = underpaint(I, U)
