def dsl1(I):
    a0 = asobject(I)
    a1 = asobject(rot90(I))
    a2 = asobject(rot180(I))
    a3 = asobject(rot270(I))
    allcells = merge(combine(combine(a0, a1), combine(a2, a3)))
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    frame = fork(recolor, color, compose(box, toindices))
    wv = compose(tojvec, width)
    hv = compose(toivec, height)
    nwv = compose(invert, wv)
    nhv = compose(invert, hv)
    base = apply(frame, objs)
    r = apply(fork(shift, frame, wv), objs)
    l = apply(fork(shift, frame, nwv), objs)
    d = apply(fork(shift, frame, hv), objs)
    u = apply(fork(shift, frame, nhv), objs)
    rd = apply(fork(shift, frame, fork(add, wv, hv)), objs)
    ru = apply(fork(shift, frame, fork(add, wv, nhv)), objs)
    ld = apply(fork(shift, frame, fork(add, nwv, hv)), objs)
    lu = apply(fork(shift, frame, fork(add, nwv, nhv)), objs)
    allcells = merge(combine(combine(combine(base, r), combine(l, d)), combine(combine(u, rd), combine(ru, combine(ld, lu)))))
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl3(I):
    c = leastcolor(I)
    mask = ofcolor(I, c)
    rect = backdrop(mask)
    colrect = recolor(c, rect)
    O = paint(canvas(ZERO, shape(I)), colrect)
    return O

def dsl4(I):
    a0 = asobject(I)
    a1 = asobject(dmirror(I))
    a2 = asobject(cmirror(I))
    allcells = merge(combine(a0, combine(a1, a2)))
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl5(I):
    objs = objects(I, T, F, T)
    nbors = compose(lbind(mapply, neighbors), toindices)
    expand = fork(combine, toindices, nbors)
    ex = apply(fork(recolor, color, expand), objs)
    allcells = merge(ex)
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O
