def dsl1(I):
    # Global: crop nonzeros, compress, rotate-compress-rotate to densify both axes
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    C = compress(A)
    RC = rot270(compress(rot90(C)))
    return RC

def dsl2(I):
    # Object extraction: expand each non-dominant object to its full bounding box, recolored by its own color
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    OBJS = objects(A, T, F, T)
    DOM = mostcolor(merge(OBJS))
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    TOT = totuple(OTH)
    F = fork(recolor, color, compose(backdrop, toindices))
    BBOBJ = apply(F, TOT)
    O = paint(canvas(ZERO, shape(A)), merge(BBOBJ))
    return O

def dsl3(I):
    # Color filtering: remove the dominant color and keep only non-dominant objects on a zero canvas
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    OBJS = objects(A, T, F, T)
    DOM = mostcolor(merge(OBJS))
    OTH = difference(OBJS, colorfilter(OBJS, DOM))
    O = paint(canvas(ZERO, shape(A)), merge(OTH))
    return O

def dsl4(I):
    # Geometric: split crop into left/right halves and fill each half's background by its own dominant color, then recombine
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    L = lefthalf(A)
    R = righthalf(A)
    LOBJS = objects(L, T, F, T)
    ROBJS = objects(R, T, F, T)
    LDOM = mostcolor(merge(LOBJS))
    RDOM = mostcolor(merge(ROBJS))
    LF = replace(L, ZERO, LDOM)
    RF = replace(R, ZERO, RDOM)
    O = hconcat(LF, RF)
    return O

def dsl5(I):
    # Relational: replace each non-dominant object by the border of its bounding box recolored by its own color
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    OBJS = objects(A, T, F, T)
    DOM = mostcolor(merge(OBJS))
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    TOT = totuple(OTH)
    G = fork(recolor, color, compose(outbox, toindices))
    BORD = apply(G, TOT)
    O = paint(canvas(ZERO, shape(A)), merge(BORD))
    return O
