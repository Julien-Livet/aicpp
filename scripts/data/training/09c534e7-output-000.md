def dsl1(I):
    # Relational expansion: grow non-1 colors into adjacent 1-cells by one step
    FG = fgpartition(I)
    C1 = colorfilter(FG, ONE)
    Seeds = difference(FG, C1)
    idxf = toindices
    nbf = lbind(mapply, dneighbors)
    nb_of_obj = compose(nbf, idxf)
    ringf = fork(difference, nb_of_obj, idxf)
    ones = ofcolor(I, ONE)
    adj1f = compose(lbind(intersection, ones), ringf)
    recolorf = fork(recolor, color, adj1f)
    Grown = mapply(recolorf, Seeds)
    O = paint(I, Grown)
    return O

def dsl2(I):
    # Global transformation: remove frontiers (compress)
    O = compress(I)
    return O

def dsl3(I):
    # Color filtering: remove all 1s
    O = replace(I, ONE, ZERO)
    return O

def dsl4(I):
    # Geometric reasoning: mirror the left half to enforce horizontal symmetry
    L = lefthalf(I)
    O = hconcat(L, hmirror(L))
    return O

def dsl5(I):
    # Object extraction: crop to the bounding box of all non-1 foreground pixels
    FG = fgpartition(I)
    C1 = colorfilter(FG, ONE)
    Not1 = difference(FG, C1)
    P = merge(apply(toindices, Not1))
    O = subgrid(P, I)
    return O
