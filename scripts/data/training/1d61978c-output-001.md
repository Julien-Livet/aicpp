def dsl1(I):
    # Color filtering with override: map all 5->8, then override the (-1) diagonal class to 2
    base = replace(I, FIVE, EIGHT)
    S = ofcolor(I, FIVE)
    minus_neighbors = combine(shift(S, UP_RIGHT), shift(S, DOWN_LEFT))
    minus = intersection(S, minus_neighbors)
    O = paint(base, recolor(TWO, minus))
    return O

def dsl2(I):
    # Relational (pixel-level): classify 5s by diagonal neighbors, reversed mapping
    S = ofcolor(I, FIVE)
    plus_neighbors = combine(shift(S, UNITY), shift(S, NEG_UNITY))
    minus_neighbors = combine(shift(S, UP_RIGHT), shift(S, DOWN_LEFT))
    plus = intersection(S, plus_neighbors)
    minus = intersection(S, minus_neighbors)
    colored = combine(recolor(TWO, plus), recolor(EIGHT, minus))
    O = paint(I, colored)
    return O

def dsl3(I):
    # Geometric per-object: subset-of UL-LR diagonal vs its complement, then fill safely
    objs = colorfilter(objects(I, T, T, T), FIVE)
    on_plus = fork(equality, toindices, fork(intersection, toindices, fork(connect, ulcorner, lrcorner)))
    asc = mfilter(objs, on_plus)
    asc_inds = merge(mapply(toindices, asc))
    S = ofcolor(I, FIVE)
    desc_inds = difference(S, asc_inds)
    Oa = branch(greater(size(asc_inds), ZERO), fill(I, EIGHT, asc_inds), I)
    Ob = branch(greater(size(desc_inds), ZERO), fill(Oa, TWO, desc_inds), Oa)
    return Ob

def dsl4(I):
    # Global symmetry: classify 5s by invariance under counterdiagonal mirroring
    S = ofcolor(I, FIVE)
    Sm = cmirror(S)
    sym = intersection(S, Sm)
    asym = difference(S, sym)
    Oa = branch(greater(size(sym), ZERO), fill(I, EIGHT, sym), I)
    Ob = branch(greater(size(asym), ZERO), fill(Oa, TWO, asym), Oa)
    return Ob

def dsl5(I):
    # Adaptive relational: choose mapping by which diagonal neighbor class is larger
    S = ofcolor(I, FIVE)
    plus = intersection(S, combine(shift(S, UNITY), shift(S, NEG_UNITY)))
    minus = intersection(S, combine(shift(S, UP_RIGHT), shift(S, DOWN_LEFT)))
    cond = greater(size(plus), size(minus))
    A = paint(I, combine(recolor(EIGHT, plus), recolor(TWO, minus)))
    B = paint(I, combine(recolor(TWO, plus), recolor(EIGHT, minus)))
    O = branch(cond, A, B)
    return O
