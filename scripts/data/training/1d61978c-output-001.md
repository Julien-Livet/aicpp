def dsl1(I):
    # Global: 5s on the counterdiagonal of the 5s' bounding box -> 2, others -> 8
    S5 = ofcolor(I, FIVE)
    UR = urcorner(S5)
    LL = llcorner(S5)
    AD = connect(UR, LL)
    S2 = intersection(S5, AD)
    S8 = difference(S5, S2)
    O0 = fill(I, SEVEN, S5)
    O1 = fill(O0, EIGHT, S8)
    O = fill(O1, TWO, S2)
    return O

def dsl2(I):
    # Object extraction: 5-objects touching border -> 2, other 5-objects -> 8
    OB = objects(I, T, T, T)
    OB5 = colorfilter(OB, FIVE)
    BORD = sfilter(OB5, rbind(bordering, I))
    Sbord = merge(apply(toindices, BORD))
    S5 = ofcolor(I, FIVE)
    Srest = difference(S5, Sbord)
    O0 = fill(I, SEVEN, S5)
    O1 = fill(O0, TWO, Sbord)
    O = fill(O1, EIGHT, Srest)
    return O

def dsl3(I):
    # Local adjacency: 5s with a DOWN_LEFT/UP_RIGHT neighbor -> 2, others -> 8
    S5 = ofcolor(I, FIVE)
    N1 = shift(S5, DOWN_LEFT)
    N2 = shift(S5, UP_RIGHT)
    N = combine(N1, N2)
    S2 = intersection(S5, N)
    S8 = difference(S5, S2)
    O0 = fill(I, SEVEN, S5)
    O1 = fill(O0, TWO, S2)
    O = fill(O1, EIGHT, S8)
    return O

def dsl4(I):
    # Geometric symmetry: 5s with a horizontal-mirror partner -> 8, others -> 2

def dsl5(I):
