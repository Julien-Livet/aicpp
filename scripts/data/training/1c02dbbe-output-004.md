def dsl1(I):
    # Relational pairing: horizontal/vertical strips per seed inside 5-region; pair same-color to fill their union bbox
    dims = shape(I)
    base = canvas(ZERO, dims)
    fidx = ofcolor(I, FIVE)
    fbase = fill(base, FIVE, fidx)
    seeds = difference(objects(I, T, F, T), colorfilter(objects(I, T, F, T), FIVE))
    hin5 = compose(rbind(intersection, fidx), compose(hfrontier, center))
    vin5 = compose(rbind(intersection, fidx), compose(vfrontier, center))
    hseg = fork(recolor, color, hin5)
    vseg = fork(recolor, color, vin5)
    Hraw = apply(hseg, seeds)
    Vraw = apply(vseg, seeds)
    nonempty = compose(positive, compose(size, toindices))
    H = sfilter(Hraw, nonempty)
    V = sfilter(Vraw, nonempty)
    pairs = product(H, V)
    samec = fork(equality, compose(color, first), compose(color, last))
    pp = sfilter(pairs, samec)
    tf = compose(toindices, first)
    tl = compose(toindices, last)
    bboxuv = compose(backdrop, fork(combine, tf, tl))
    cfirst = compose(color, first)
    rectpair = fork(recolor, cfirst, bboxuv)
    rectobj = merge(apply(rectpair, pp))
    O = paint(fbase, rectobj)
    return O
