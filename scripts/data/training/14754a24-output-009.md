def dsl1(I):
    # 5s at intersections of all 4-rows and all 4-columns
    fou = ofcolor(I, FOUR)
    rows = mapply(hfrontier, fou)
    cols = mapply(vfrontier, fou)
    cross = intersection(merge(rows), merge(cols))
    fiv = ofcolor(I, FIVE)
    target = intersection(cross, fiv)
    O = fill(I, TWO, target)
    return O

def dsl2(I):
    # 5s strictly between two 4s along pure axial segments (only 4/5 on the path)
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    segs = prapply(connect, fou, fou)
    axial = sfilter(segs, fork(either, hline, vline))
    allowed = combine(fou, fiv)
    pure = sfilter(axial, matcher(compose(size, rbind(difference, allowed)), ZERO))
    mask = intersection(merge(pure), fiv)
    O = fill(I, TWO, mask)
    return O

def dsl3(I):
    # Propagate from 5-seeds adjacent to 4 along pure axial 5-runs
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    nbh = mapply(neighbors, fou)
    seeds = intersection(merge(nbh), fiv)
    segs = prapply(connect, seeds, fiv)
    axial = sfilter(segs, fork(either, hline, vline))
    pure = sfilter(axial, matcher(compose(size, rbind(difference, fiv)), ZERO))
    mask = merge(pure)
    O = fill(I, TWO, mask)
    return O

def dsl4(I):
    # Portions of 5-objects (after removing 4s) that touch a 4 and lie on the 4-row/4-column cross
    fou = ofcolor(I, FOUR)
    obj4 = toobject(fou, I)
    cleared = cover(I, obj4)
    fiveobjs = colorfilter(objects(cleared, T, F, T), FIVE)
    touching = sfilter(fiveobjs, lbind(adjacent, fou))
    touchmask = merge(mapply(toindices, touching))
    rows = mapply(hfrontier, fou)
    cols = mapply(vfrontier, fou)
    cross = intersection(merge(rows), merge(cols))
    target = intersection(touchmask, cross)
    O = fill(I, TWO, target)
    return O

def dsl5(I):
    # 5-runs grown from 4-adjacent 5s, restricted to columns containing a 4
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    nbh = mapply(neighbors, fou)
    seeds = intersection(merge(nbh), fiv)
    segs = prapply(connect, seeds, fiv)
    axial = sfilter(segs, fork(either, hline, vline))
    pure = sfilter(axial, matcher(compose(size, rbind(difference, fiv)), ZERO))
    runmask = merge(pure)
    colmask = merge(mapply(vfrontier, fou))
    target = intersection(runmask, colmask)
    O = fill(I, TWO, target)
    return O
