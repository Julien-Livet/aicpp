def dsl1(I):
    # Global outline: draw the outline of the 8-mask as 3, remove 1s
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    obj8 = toobject(p8, I)
    outline = box(obj8)
    O = fill(base, THREE, outline)
    return O

def dsl2(I):
    # Object extraction via 1-bbox: color 8s inside the 1-box as 3, outside as 2
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    p1 = ofcolor(I, ONE)
    box1 = backdrop(p1)
    inside = intersection(p8, box1)
    outside = difference(p8, box1)
    A = fill(base, THREE, inside)
    O = fill(A, TWO, outside)
    return O

def dsl3(I):
    # Pure color filtering: swap 8->3 and erase 1s
    S = switch(I, EIGHT, THREE)
    O = replace(S, ONE, ZERO)
    return O

def dsl4(I):
    # Geometric symmetry: 8s that are symmetric under vertical mirror -> 3, others -> 2
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    obj8 = toobject(p8, I)
    mir = vmirror(obj8)
    pm = toindices(mir)
    sym = intersection(p8, pm)
    asym = difference(p8, sym)
    A = fill(base, THREE, sym)
    O = fill(A, TWO, asym)
    return O

def dsl5(I):
    # Relational rule: choose 3 vs 2 for all 8s based on vertical row-overlap with 1s
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    p1 = ofcolor(I, ONE)
    col = branch(vmatching(toobject(p8, I), toobject(p1, I)), THREE, TWO)
    O = fill(base, col, p8)
    return O
