    p5 = ofcolor(I, FIVE)
    b = backdrop(p5)
    inner = delta(p5)
    O0 = underfill(I, TWO, inner)
    rows5 = intersection(b, mapply(hfrontier, p5))
    cols5 = intersection(b, mapply(vfrontier, p5))
    grows = difference(b, rows5)
    gcols = difference(b, cols5)
    ring1h = intersection(outbox(b), mapply(hfrontier, grows))
    ring1v = intersection(outbox(b), mapply(vfrontier, gcols))
    outside = difference(asindices(I), b)
    ring2h = intersection(outside, mapply(hfrontier, ring1h))
    ring2v = intersection(outside, mapply(vfrontier, ring1v))
    ring = combine(combine(ring1h, ring1v), combine(ring2h, ring2v))

    return O
