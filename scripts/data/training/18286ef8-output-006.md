def dsl1(I):
    # Global: compute direction on compressed grid from UR-corner of 5-box to external 9s
    Ic = compress(I)
    P5c = ofcolor(Ic, FIVE)
    Bc = inbox(P5c)
    Gc = replace(Ic, SIX, NINE)
    P9oc = difference(ofcolor(Gc, NINE), Bc)
    D = sign(subtract(centerofmass(P9oc), urcorner(Bc)))
    P5 = ofcolor(I, FIVE)
    B = inbox(P5)
    J = intersection(ofcolor(I, NINE), B)
    G = replace(I, SIX, NINE)
    O = paint(fill(G, FIVE, J), recolor(NINE, shift(J, D)))
    return O

def dsl2(I):

    P5 = ofcolor(I, FIVE)
    B = inbox(P5)

    G = replace(I, SIX, NINE)

    return O

def dsl3(I):
    # Color filtering: derive axiswise signs from extremal coords (top of 5-box, right edge)
    P5 = ofcolor(I, FIVE)
    B = inbox(P5)
    J = intersection(ofcolor(I, NINE), B)
    G = replace(I, SIX, NINE)
    P9o = difference(ofcolor(G, NINE), B)
    Di = sign(subtract(uppermost(P9o), uppermost(B)))
    Dj = sign(subtract(leftmost(P9o), rightmost(B)))
    D = astuple(Di, Dj)
    G2 = fill(G, FIVE, J)
    O = fill(G2, NINE, shift(J, D))
    return O
