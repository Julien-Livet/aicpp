def dsl1(I):
    # Global symmetry: mirror each object across horizontal and vertical axes and overlay
    objs = objects(I, T, F, T)
    bidx = toindices
    idf = bidx
    hm = compose(hmirror, bidx)
    vm = compose(vmirror, bidx)
    hv = compose(hmirror, compose(vmirror, bidx))
    funcs = insert(idf, insert(hm, insert(vm, initset(hv))))
    per = compose(merge, lbind(rapply, funcs))
    stamps = apply(fork(recolor, color, per), objs)
    O = paint(canvas(ZERO, shape(I)), merge(stamps))
    return O

def dsl2(I):

def dsl3(I):
    # Color filtering: tile only the least-common-color objects on a 3x3 lattice with one-cell gaps
    objs = objects(I, T, F, T)
    lc = leastcolor(I)
    foc = colorfilter(objs, lc)
    bidx = toindices
    w = compose(width, bidx)
    h = compose(height, bidx)
    sw = compose(tojvec, compose(increment, w))
    sv = compose(toivec, compose(increment, h))
    nsw = compose(invert, sw)
    nsv = compose(invert, sv)
    base = bidx
    rs = fork(shift, bidx, sw)
    ls = fork(shift, bidx, nsw)
    ds = fork(shift, bidx, sv)
    us = fork(shift, bidx, nsv)
    rd = fork(shift, bidx, fork(add, sw, sv))
    ru = fork(shift, bidx, fork(add, sw, nsv))
    ld = fork(shift, bidx, fork(add, nsw, sv))
    lu = fork(shift, bidx, fork(add, nsw, nsv))
    funcs = insert(base, insert(rs, insert(ls, insert(ds, insert(us, insert(rd, insert(ru, insert(ld, initset(lu)))))))))
    per = compose(merge, lbind(rapply, funcs))
    stamps = apply(fork(recolor, color, per), foc)
    O = paint(canvas(ZERO, shape(I)), merge(stamps))
    return O

def dsl4(I):

def dsl5(I):
    # Relational/structural: draw the rectangle border connecting each pair of bbox corners
    objs = objects(I, T, F, T)
    ul = compose(ulcorner, toindices)

    ll = compose(llcorner, toindices)

    horiz = fork(combine, top, bot)
