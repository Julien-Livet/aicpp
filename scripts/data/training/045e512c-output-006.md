def dsl1(I):
    # Global tiling of the cropped nonzero subgrid across the full grid
    nz = difference(asindices(I), ofcolor(I, ZERO))
    sub = subgrid(nz, I)
    base = asobject(sub)
    ws = width(sub)
    hs = height(sub)
    wI = width(I)
    hI = height(I)
    offsH = interval(0, wI, ws)
    vecsH = apply(tojvec, offsH)
    tileH = mapply(lbind(shift, base), vecsH)
    offsV = interval(0, hI, hs)
    vecsV = apply(toivec, offsV)
    tileHV = mapply(lbind(shift, tileH), vecsV)
    O = paint(canvas(ZERO, shape(I)), tileHV)
    return O

def dsl2(I):
    # Object-period tiling: tile all normalized objects at the minimal object bbox period
    objs = objects(I, T, F, T)
    bbs = apply(rbind(subgrid, I), objs)
    ws = apply(width, bbs)
    hs = apply(height, bbs)
    pH = minimum(ws)
    pV = minimum(hs)
    norms = apply(normalize, objs)
    wI = width(I)
    hI = height(I)
    offsH = interval(0, wI, pH)
    vecsH = apply(tojvec, offsH)
    copiesH = prapply(shift, norms, vecsH)
    tileH = merge(copiesH)
    offsV = interval(0, hI, pV)
    vecsV = apply(toivec, offsV)
    tileHV = mapply(lbind(shift, tileH), vecsV)
    O = paint(canvas(ZERO, shape(I)), tileHV)
    return O

def dsl3(I):
    # Color filtering: draw all rows and columns through the least frequent nonzero color
    pal = remove(ZERO, palette(I))
    c = argmin(pal, lbind(colorcount, I))
    starts = ofcolor(I, c)
    rows = mapply(hfrontier, starts)
    cols = mapply(vfrontier, starts)
    lines = combine(rows, cols)
    obj = recolor(c, lines)
    O = paint(canvas(ZERO, shape(I)), obj)
    return O

def dsl4(I):
    # Geometric symmetry within cropped foreground: combine vertical and horizontal mirrors
    nz = difference(asindices(I), ofcolor(I, ZERO))
    sub = subgrid(nz, I)
    base = asobject(sub)
    sym = combine(base, vmirror(base))
    sym2 = combine(sym, hmirror(sym))
    O = paint(canvas(ZERO, shape(I)), sym2)
    return O

def dsl5(I):
    # Relational: draw the bounding box outline of each object in its own color
    objs = objects(I, T, F, T)
    recol_box = fork(recolor, color, box)
    outs = apply(recol_box, objs)
    allouts = merge(outs)
    O = paint(canvas(ZERO, shape(I)), allouts)
    return O
