def dsl1(I):
    # Global: 5s on the grid counter-diagonal -> 2, others -> 8
    S5 = ofcolor(I, FIVE)
    G = asindices(I)
    Axis = connect(llcorner(G), urcorner(G))
    S2 = intersection(S5, Axis)
    S8 = difference(S5, S2)
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O

def dsl2(I):
    # Object extraction: largest diag-connected 5-component -> 2, others -> 8
    O5 = colorfilter(objects(I, T, T, T), FIVE)
    Best = argmax(O5, size)
    S2 = toindices(Best)
    S8 = difference(ofcolor(I, FIVE), S2)
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O

def dsl3(I):
    # Color filtering (local): 5s that are endpoints (no both UR and DL neighbors) -> 2, interiors -> 8
    S5 = ofcolor(I, FIVE)
    HasUR = intersection(S5, shift(S5, DOWN_LEFT))
    HasDL = intersection(S5, shift(S5, UP_RIGHT))
    Interior = intersection(HasUR, HasDL)
    S2 = difference(S5, Interior)
    S8 = Interior
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O
