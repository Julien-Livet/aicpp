def dsl1(I):
    # Color filtering: fill the gaps inside each object's bounding box with its color
    objs = objects(I, T, F, T)
    inner = compose(delta, toindices)
    filled = apply(fork(recolor, color, inner), objs)
    allcells = merge(filled)
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl2(I):
    # Geometric reasoning: replicate each object on a 3x3 lattice using the largest object's size as step
    objs = objects(I, T, F, T)
    big = argmax(objs, compose(size, toindices))
    bidx = toindices(big)
    stepw = tojvec(width(bidx))
    stepv = toivec(height(bidx))
    nstepw = invert(stepw)
    nstepv = invert(stepv)
    rdv = add(stepw, stepv)
    ruv = add(stepw, nstepv)
    ldv = add(nstepw, stepv)
    luv = add(nstepw, nstepv)
    base = toindices
    rs = compose(rbind(shift, stepw), toindices)
    ls = compose(rbind(shift, nstepw), toindices)
    ds = compose(rbind(shift, stepv), toindices)
    us = compose(rbind(shift, nstepv), toindices)
    rd = compose(rbind(shift, rdv), toindices)
    ru = compose(rbind(shift, ruv), toindices)
    ld = compose(rbind(shift, ldv), toindices)
    lu = compose(rbind(shift, luv), toindices)
    funcs = insert(base, insert(rs, insert(ls, insert(ds, insert(us, insert(rd, insert(ru, insert(ld, initset(lu)))))))))
    per = compose(merge, lbind(rapply, funcs))
    stamps = apply(fork(recolor, color, per), objs)
    allcells = merge(stamps)
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl3(I):
    # Global transformation: radiate lines from each object's center across the grid
    objs = objects(I, T, F, T)
    cen = compose(center, toindices)
    up = rbind(shoot, UP)
    down = rbind(shoot, DOWN)
    leftf = rbind(shoot, LEFT)
    rightf = rbind(shoot, RIGHT)
    dirfuncs = insert(up, insert(down, insert(leftf, initset(rightf))))
    rays = compose(merge, lbind(rapply, dirfuncs))
    beams = apply(fork(recolor, color, compose(rays, cen)), objs)
    allcells = merge(beams)
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O
