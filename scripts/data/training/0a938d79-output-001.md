def dsl1(I):
    # Global stripes via shoots + frontiers with minimal-distance orientation (safe steps)
    OS = objects(I, T, F, T)
    P = mapply(toindices, OS)
    H = height(I)
    W = width(I)
    top = uppermost(P)
    bot = lowermost(P)
    left = leftmost(P)
    right = rightmost(P)
    dv = subtract(bot, top)
    dh = subtract(right, left)
    stepR = branch(equality(dv, ZERO), ONE, dv)
    stepC = branch(equality(dh, ZERO), ONE, dh)
    o1 = first(OS)
    o2 = last(OS)
    l1 = first(toindices(o1))
    l2 = first(toindices(o2))
    # horizontal pattern from both seeds
    L1h = combine(combine(shoot(l1, toivec(stepR)), shoot(l1, toivec(invert(stepR)))), initset(l1))
    R1 = mapply(hfrontier, L1h)
    H1 = recolor(color(o1), R1)
    L2h = combine(combine(shoot(l2, toivec(stepR)), shoot(l2, toivec(invert(stepR)))), initset(l2))
    R2 = mapply(hfrontier, L2h)
    H2 = recolor(color(o2), R2)
    HOBJ = combine(H1, H2)
    # vertical pattern from both seeds
    L1v = combine(combine(shoot(l1, tojvec(stepC)), shoot(l1, tojvec(invert(stepC)))), initset(l1))
    C1 = mapply(vfrontier, L1v)
    V1 = recolor(color(o1), C1)
    L2v = combine(combine(shoot(l2, tojvec(stepC)), shoot(l2, tojvec(invert(stepC)))), initset(l2))
    C2 = mapply(vfrontier, L2v)
    V2 = recolor(color(o2), C2)
    VOBJ = combine(V1, V2)
    # choose orientation with smaller nonzero distance
    hsel = either(equality(dh, ZERO), flip(either(equality(dv, ZERO), greater(dv, dh))))
    OOBJ = branch(hsel, HOBJ, VOBJ)
    O = paint(canvas(ZERO, shape(I)), OOBJ)
    return O

def dsl3(I):

    S2 = ofcolor(I, C2)
    l1 = first(S1)
    l2 = first(S2)
    P = combine(S1, S2)
    top = uppermost(P)
    bot = lowermost(P)
    left = leftmost(P)
    right = rightmost(P)
    H = height(I)
    W = width(I)
    dv = subtract(bot, top)
    dh = subtract(right, left)
    stepR = branch(equality(dv, ZERO), ONE, dv)
    stepC = branch(equality(dh, ZERO), ONE, dh)

    L1h = combine(combine(shoot(l1, toivec(stepR)), shoot(l1, toivec(invert(stepR)))), initset(l1))
    HMask1 = mapply(hfrontier, L1h)
    HObj1 = recolor(C1, HMask1)
    L2h = combine(combine(shoot(l2, toivec(stepR)), shoot(l2, toivec(invert(stepR)))), initset(l2))
    HMask2 = mapply(hfrontier, L2h)
    HObj2 = recolor(C2, HMask2)

    VObj1 = recolor(C1, VMask1)

    VMask2 = mapply(vfrontier, L2v)
    VObj2 = recolor(C2, VMask2)
    HB = canvas(ZERO, shape(I))
    VB = canvas(ZERO, shape(I))
    Hfilled = underpaint(underpaint(HB, HObj1), HObj2)
    Vfilled = underpaint(underpaint(VB, VObj1), VObj2)

    O = branch(hsel, Hfilled, Vfilled)
