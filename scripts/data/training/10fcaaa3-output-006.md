def dsl2(I):

    offR = tojvec(w)
    offD = toivec(h)

    G1 = move(base, MO, ORIGIN)
    G2 = move(G1, MO, offR)
    G3 = move(G2, MO, offD)

    C12 = crop(G, offR, astuple(h, w))
    C21 = crop(G, offD, astuple(h, w))

    B22 = shift(box(asindices(C22)), offRD)
    L1 = combine(B11, B12)
    L2 = combine(B21, B22)
    L = combine(L1, L2)
    O = underfill(G, EIGHT, L)
    return O
