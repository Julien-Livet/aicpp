def dsl1(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    O = replace(A, ZERO, ONE)
    return O

def dsl2(I):
    P2 = ofcolor(I, TWO)
    A = subgrid(P2, I)
    Z = ofcolor(A, ZERO)
    O = fill(A, ONE, Z)
    return O

def dsl3(I):
    P2 = ofcolor(I, TWO)
    A = subgrid(P2, I)
    BASE = canvas(ONE, shape(A))
    OBJS = objects(A, T, F, T)
    C2 = colorfilter(OBJS, TWO)
    OTH = difference(OBJS, C2)
    CELLS = merge(OTH)
    TMP = paint(BASE, CELLS)
    C2P = recolor(TWO, ofcolor(A, TWO))
    O = paint(TMP, C2P)
    return O

def dsl4(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    O = vmirror(A)
    return O

def dsl5(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    BASE = canvas(ONE, shape(A))
    FR = frontiers(A)
    INDS = mapply(toindices, FR)
    COL = mostcolor(A)
    GRIDOBJ = recolor(COL, INDS)
    OBJS = objects(A, T, F, T)
    PURE = colorfilter(OBJS, COL)
    OTH = difference(OBJS, PURE)
    CELLS = merge(OTH)
    TMP = paint(BASE, GRIDOBJ)
    O = paint(TMP, CELLS)
    return O
