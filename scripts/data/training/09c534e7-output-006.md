def dsl1(I):
    FG = fgpartition(I)
    Seeds = difference(FG, colorfilter(FG, ONE))
    ones = ofcolor(I, ONE)
    base = toindices
    hf = compose(hmirror, base)
    vf = compose(vmirror, base)
    hvf = chain(hmirror, vmirror, toindices)
    u1 = fork(combine, base, hf)
    u2 = fork(combine, vf, hvf)
    mirror4 = fork(combine, u1, u2)
    maskf = compose(lbind(intersection, ones), mirror4)
    recolorf = fork(recolor, color, maskf)
    Paints = mapply(recolorf, Seeds)
    O = paint(I, Paints)
    return O
