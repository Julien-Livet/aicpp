def dsl1(I):
    bg = ofcolor(I, ZERO)
    U = asindices(I)
    nz = difference(U, bg)
    hasR = intersection(nz, shift(nz, LEFT))
    hasL = intersection(nz, shift(nz, RIGHT))
    rightSide = intersection(nz, shift(bg, LEFT))
    leftSide = intersection(nz, shift(bg, RIGHT))
    rightEP = intersection(rightSide, hasL)
    leftEP = intersection(leftSide, hasR)
    vt = intersection(rightEP, shift(bg, DOWN))
    vb = intersection(rightEP, shift(bg, UP))
    tipsR = merge(insert(vt, initset(vb)))
    o4 = recolor(FOUR, merge(insert(leftEP, initset(rightEP))))
    o2 = recolor(TWO, tipsR)
    O = paint(I, o4)
    O = paint(O, o2)
    return O
