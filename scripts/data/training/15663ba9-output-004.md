def dsl1(I):
    # Global overlay via cover/paint: recolor exact L-corners with 4 and endpoints with 2
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    cLU = difference(intersection(L, U), combine(R, D))
    cRU = difference(intersection(R, U), combine(L, D))
    cLD = difference(intersection(L, D), combine(R, U))
    cRD = difference(intersection(R, D), combine(L, U))
    corners = combine(combine(cLU, cRU), combine(cLD, cRD))
    eL = difference(L, combine(combine(R, U), D))
    eR = difference(R, combine(combine(L, U), D))
    eU = difference(U, combine(combine(L, R), D))
    eD = difference(D, combine(combine(L, R), U))
    endpoints = combine(combine(eL, eR), combine(eU, eD))
    base = cover(I, S)
    O = paint(paint(base, recolor(FOUR, corners)), recolor(TWO, endpoints))
    return O

def dsl2(I):
    # Object extraction: rebuild leastcolor components, then recolor corners/endpoints
    c = leastcolor(I)
    comps = colorfilter(objects(I, T, F, T), c)
    parts = mapply(toindices, comps)
    S = merge(parts)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    straighth = intersection(L, R)
    straightv = intersection(U, D)
    corners_any = intersection(combine(L, R), combine(U, D))
    corners = difference(corners_any, combine(straighth, straightv))
    deg2plus = combine(corners, combine(straighth, straightv))
    endpoints = difference(S, deg2plus)
    G = fill(I, FOUR, corners)
    O = fill(G, TWO, endpoints)
    return O
