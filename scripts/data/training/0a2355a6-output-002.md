def dsl1(I):
    # Pixelwise quadrant recoloring via geometric masks (no object extraction)
    p = ofcolor(I, EIGHT)
    TL = asindices(lefthalf(tophalf(I)))
    TR = asindices(righthalf(tophalf(I)))
    BL = asindices(lefthalf(bottomhalf(I)))
    BR = asindices(righthalf(bottomhalf(I)))
    tl = intersection(p, TL)
    tr = intersection(p, TR)
    bl = intersection(p, BL)
    br = intersection(p, BR)
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(paint(paint(base, recolor(FOUR, tl)), recolor(TWO, tr)), recolor(THREE, bl)), recolor(ONE, br))
    return O

def dsl2(I):
    # Objectwise quadrant recoloring using halves of the full grid (object extraction)
    objs = colorfilter(objects(I, T, F, T), EIGHT)
    topspace = asindices(tophalf(I))
    leftspace = asindices(lefthalf(I))
    cond_top = compose(lbind(hmatching, topspace), toindices)
    cond_left = compose(lbind(vmatching, leftspace), toindices)
    tops = sfilter(objs, cond_top)
    lefts = sfilter(objs, cond_left)
    tl = intersection(tops, lefts)
    tr = difference(tops, tl)
    bl = difference(lefts, tl)
    bots = difference(objs, tops)
    br = difference(bots, bl)
    tl_r = merge(apply(lbind(recolor, FOUR), tl))
    tr_r = merge(apply(lbind(recolor, TWO), tr))
    bl_r = merge(apply(lbind(recolor, THREE), bl))
    br_r = merge(apply(lbind(recolor, ONE), br))
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(paint(paint(base, tl_r), tr_r), bl_r), br_r)
    return O

def dsl3(I):
    # Global reconstruction by recoloring each quadrant subgrid then reassembling
    TH = tophalf(I)
    BH = bottomhalf(I)
    TL = lefthalf(TH)
    TR = righthalf(TH)
    BL = lefthalf(BH)
    BR = righthalf(BH)
    TLc = replace(TL, EIGHT, FOUR)
    TRc = replace(TR, EIGHT, TWO)
    BLc = replace(BL, EIGHT, THREE)
    BRc = replace(BR, EIGHT, ONE)
    top = hconcat(TLc, TRc)
    bot = hconcat(BLc, BRc)
    O = vconcat(top, bot)
    return O

def dsl4(I):
    # Geometric masking via splits: use hsplit/vsplit to form quadrants and recolor
    TH = first(hsplit(I, 2))
    BH = last(hsplit(I, 2))
    TL = first(vsplit(TH, 2))
    TR = last(vsplit(TH, 2))
    BL = first(vsplit(BH, 2))
    BR = last(vsplit(BH, 2))
    TLc = replace(TL, EIGHT, FOUR)
    TRc = replace(TR, EIGHT, TWO)
    BLc = replace(BL, EIGHT, THREE)
    BRc = replace(BR, EIGHT, ONE)
    top = hconcat(TLc, TRc)
    bot = hconcat(BLc, BRc)
    O = vconcat(top, bot)
    return O

def dsl5(I):
    # Relational recoloring: classify 8-pixels by quadrant masks derived from grid halves
    TLm = asindices(lefthalf(tophalf(I)))
    TRm = asindices(righthalf(tophalf(I)))
    BLm = asindices(lefthalf(bottomhalf(I)))
    BRm = asindices(righthalf(bottomhalf(I)))
    pts = ofcolor(I, EIGHT)
    tl = intersection(pts, TLm)
    tr = intersection(pts, TRm)
    bl = intersection(pts, BLm)
    br = intersection(pts, BRm)
    base = replace(I, EIGHT, ZERO)
    colored = paint(paint(paint(paint(base, recolor(FOUR, tl)), recolor(TWO, tr)), recolor(THREE, bl)), recolor(ONE, br))
    return colored
