def dsl1(I):
    # Global counterdiagonal band from grid corners: near band -> 2, others -> 8
    S5 = ofcolor(I, FIVE)
    A = asindices(I)
    D = connect(urcorner(A), llcorner(A))
    Band = combine(D, combine(shift(D, LEFT), shift(D, RIGHT)))
    S2 = intersection(S5, Band)
    S8 = difference(S5, S2)
    O = fill(fill(fill(I, SEVEN, S5), TWO, S2), EIGHT, S8)
    return O

def dsl2(I):
    # Object-anchored rays through bbox center: on (-45°) ray -> 2, others -> 8
    S5 = ofcolor(I, FIVE)
    Bx = backdrop(S5)
    C = center(Bx)
    R = combine(shoot(C, DOWN_LEFT), shoot(C, UP_RIGHT))
    S2 = intersection(S5, R)
    S8 = difference(S5, S2)
    O = fill(fill(fill(I, SEVEN, S5), TWO, S2), EIGHT, S8)
    return O

def dsl3(I):
    # Color-first: replace 5->8, then override near cmirror (±1 col) as 2
    S5 = ofcolor(I, FIVE)
    M = cmirror(S5)
    Near = intersection(S5, combine(shift(M, LEFT), shift(M, RIGHT)))
    J = replace(I, FIVE, EIGHT)
    O = fill(J, TWO, Near)
    return O

def dsl4(I):
    # Geometric pairing via cmirror: paired -> 8, singles -> 2
    S5 = ofcolor(I, FIVE)
    Paired = intersection(S5, cmirror(S5))
    Singles = difference(S5, Paired)
    O = fill(fill(fill(I, SEVEN, S5), EIGHT, Paired), TWO, Singles)
    return O

def dsl5(I):
    # Relational: topmost 5 per column -> 2, others -> 8
    S5 = ofcolor(I, FIVE)
    RaysU = prapply(shoot, S5, initset(UP))
    HasAbove = intersection(RaysU, S5)
    Topmost = difference(S5, HasAbove)
    Rest = HasAbove
    O = fill(fill(fill(I, SEVEN, S5), TWO, Topmost), EIGHT, Rest)
    return O
