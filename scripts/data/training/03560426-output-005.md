def dsl1(I):
    G = compress(I)
    H = height(I)
    W = width(I)
    HR = height(G)
    WR = width(G)
    Zb = canvas(ZERO, astuple(subtract(H, HR), WR))
    T = vconcat(G, Zb)
    Zr = canvas(ZERO, astuple(H, subtract(W, WR)))
    O = hconcat(T, Zr)
    return O

def dsl2(I):
    A = objects(I, T, F, T)
    FN = compose(toindices, normalize)
    FRC = fork(recolor, color, FN)
    R = apply(FRC, A)
    U = merge(R)
    Z = canvas(ZERO, shape(I))
    O = paint(Z, U)
    return O
