def dsl1(I):
    # Global: keep only the bounding box around all 5s, zero everything outside
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    dest = mapply(toindices, fives)
    bb = backdrop(dest)
    allidx = asindices(I)
    outside = difference(allidx, bb)
    O = fill(I, ZERO, outside)
    return O

def dsl2(I):
    # Object extraction: crop to the bounding box of all 5s and re-embed at top-left
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    dest = mapply(toindices, fives)
    bb = backdrop(dest)
    sub = subgrid(bb, I)
    base = canvas(ZERO, shape(I))
    O = paint(base, asobject(sub))
    return O

def dsl3(I):
    # Color filtering: keep only 5s, remove all other colors
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fivecells = merge(fives)
    base = canvas(ZERO, shape(I))
    O = paint(base, fivecells)
    return O

def dsl4(I):
    # Geometric (horizontal alignment): move non-5 objects that share a row with any 5s horizontally until adjacent to a 5-cluster
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    others = difference(objs, fives)
    dest = mapply(toindices, fives)
    share_row = rbind(hmatching, dest)
    horiz = sfilter(others, share_row)
    rest = difference(others, horiz)
    dirf = rbind(gravitate, dest)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    movedcells = mapply(moved, horiz)
    restcells = merge(rest)
    fivecells = merge(fives)
    allcells = combine(fivecells, combine(movedcells, restcells))
    base = canvas(ZERO, shape(I))
    O = paint(base, allcells)
    return O

def dsl5(I):
    # Relational: gravitate every non-5 object towards the union of all 5s until adjacency, then repaint with 5s
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    others = difference(objs, fives)
    dest = mapply(toindices, fives)
    dirf = rbind(gravitate, dest)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    movedcells = mapply(moved, others)
    fivecells = merge(fives)
    allcells = combine(movedcells, fivecells)
    base = canvas(ZERO, shape(I))
    O = paint(base, allcells)
    return O
