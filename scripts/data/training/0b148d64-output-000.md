def dsl1(I):
    lc = leastcolor(I)
    P = ofcolor(I, lc)
    OBJ = toobject(P, I)
    N = normalize(OBJ)
    D = shape(subgrid(P, I))
    O = paint(canvas(ZERO, D), N)
    return O

def dsl2(I):
    lc = leastcolor(I)
    OBJS = fgpartition(I)
    CF = colorfilter(OBJS, lc)
    M = merge(CF)
    N = normalize(M)
    D = shape(subgrid(toindices(M), I))
    O = paint(canvas(ZERO, D), N)
    return O

def dsl3(I):
    lc = leastcolor(I)
    P = ofcolor(I, lc)
    UL = ulcorner(P)
    OFF = invert(UL)
    S0 = shift(P, OFF)
    D = shape(subgrid(P, I))
    O = fill(canvas(ZERO, D), lc, S0)
    return O

def dsl4(I):
    lc = leastcolor(I)
    P = ofcolor(I, lc)
    G = fill(canvas(ZERO, shape(I)), lc, P)
    O = subgrid(P, G)
    return O

def dsl5(I):
    lc = leastcolor(I)
    P = ofcolor(I, lc)
    BB = backdrop(P)
    G = fill(canvas(ZERO, shape(I)), lc, P)
    O = subgrid(BB, G)
    return O
