def dsl1(I):
    # Global scale-space: build macro-grid (factor 3), connect same-colored macro-objects both horizontally and vertically, upscale, underpaint
    G = downscale(I, THREE)
    dimsG = shape(G)
    sG = mostcolor(G)
    objsG = objects(G, T, F, T)
    notS = compose(flip, matcher(color, sG))
    notZ = compose(flip, matcher(color, ZERO))
    cond = fork(both, notS, notZ)
    fg = sfilter(objsG, cond)
    pairs = product(fg, fg)
    same = fork(equality, compose(color, first), compose(color, last))
    hA = fork(hmatching, first, last)
    vA = fork(vmatching, first, last)
    goodH = sfilter(pairs, fork(both, same, hA))
    goodV = sfilter(pairs, fork(both, same, vA))
    good = combine(goodH, goodV)
    seg = fork(connect, compose(centerofmass, first), compose(centerofmass, last))
    can = compose(rbind(canvas, dimsG), compose(color, first))
    mk = fork(toobject, seg, can)
    P = merge(apply(mk, good))
    G2 = underpaint(G, P)
    U = upscale(G2, THREE)
    O = underpaint(I, asobject(U))
    return O

def dsl3(I):
    # Color filtering + hybrid propagation: horizontal connections within rows + vertical projection of each object's two columns; underpaint
    s = mostcolor(I)
    dims = shape(I)
    objs = objects(I, T, F, T)
    notS = compose(flip, matcher(color, s))
    notZ = compose(flip, matcher(color, ZERO))
    cond = fork(both, notS, notZ)
    fg = sfilter(objs, cond)
    # Horizontal connections (same color, shared rows)
    pairs = product(fg, fg)
    same = fork(equality, compose(color, first), compose(color, last))
    hA = fork(hmatching, first, last)
    goodH = sfilter(pairs, fork(both, same, hA))
    segH = fork(connect, compose(centerofmass, first), compose(centerofmass, last))
    canH = compose(rbind(canvas, dims), compose(color, first))
    mkH = fork(toobject, segH, canH)
    PH = merge(apply(mkH, goodH))
    # Vertical projection of each object's two columns
    su = rbind(shoot, UP)
    sd = rbind(shoot, DOWN)
    topL = compose(su, ulcorner)
    botL = compose(sd, llcorner)
    topR = compose(su, urcorner)
    botR = compose(sd, lrcorner)
    vL = fork(combine, topL, botL)
    vR = fork(combine, topR, botR)
    vseg = fork(combine, vL, vR)
    canV = compose(rbind(canvas, dims), color)
    mkV = fork(toobject, vseg, canV)
    PV = merge(apply(mkV, fg))
    P = combine(PH, PV)
    O = underpaint(I, P)
    return O
