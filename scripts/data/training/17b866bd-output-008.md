def dsl1(I):
    # Object-extraction: turn each non-8, non-0 seed into a 4x4 diamond anchored at its incremented ul-corner
    objs = objects(I, T, F, F)
    non0 = difference(objs, colorfilter(objs, ZERO))
    seeds = difference(non0, colorfilter(non0, EIGHT))
    ul = compose(increment, ulcorner)
    dims = double(TWO_BY_TWO)
    off = decrement(dims)
    lr = compose(rbind(add, off), ul)
    cset = fork(combine, compose(initset, ul), compose(initset, lr))
    rect = compose(backdrop, cset)
    diamond = fork(difference, rect, compose(corners, rect))
    col = compose(color, identity)
    recol = apply(fork(recolor, col, diamond), seeds)
    obj = merge(recol)
    spatch = mapply(toindices, seeds)
    base = fill(I, ZERO, spatch)
    O = paint(base, obj)
    return O

def dsl2(I):

    idx = ofcolor(I, EIGHT)
    at = compose(rbind(index, I), identity)
    shR = compose(rbind(add, RIGHT), identity)
    shD = compose(rbind(add, DOWN), identity)
    shL = compose(rbind(add, LEFT), identity)
    shU = compose(rbind(add, UP), identity)
    valR = compose(at, shR)
    valD = compose(at, shD)
    valL = compose(at, shL)
    valU = compose(at, shU)
    eq8 = rbind(equality, EIGHT)
    eq0 = rbind(equality, ZERO)
    here8 = compose(eq8, at)
    r8 = compose(eq8, valR)
    d8 = compose(eq8, valD)
    l0 = compose(eq0, valL)
    u0 = compose(eq0, valU)
    c1 = fork(both, here8, r8)
    c2 = fork(both, d8, l0)
    c3 = fork(both, c1, c2)
    cond = fork(both, c3, u0)
