def dsl1(I):
    top = tophalf(I)
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    inds8_top = ofcolor(top, EIGHT)
    obj8_top = toobject(inds8_top, top)
    top_base = canvas(ZERO, shape(top))
    top_colored = paint(top_base, recolor(val, obj8_top))
    O = vconcat(top_colored, canvas(ZERO, shape(bh)))
    return O

def dsl3(I):
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    ones_obj = toobject(ofcolor(I, ONE), I)
    tmp = cover(I, ones_obj)
    tmp2 = replace(tmp, EIGHT, val)
    top = tophalf(tmp2)
    botz = canvas(ZERO, shape(bh))
    O = vconcat(top, botz)
    return O

def dsl4(I):
    bh = bottomhalf(I)
    inds1 = ofcolor(bh, ONE)
    obj1 = toobject(inds1, bh)
    ul = ulcorner(obj1)
    lr = lrcorner(obj1)
    p1 = initset(ul)
    p2 = initset(lr)
    val = halve(manhattan(p1, p2))
    inds8 = ofcolor(I, EIGHT)
    obj8 = toobject(inds8, I)
    base = canvas(ZERO, shape(I))
    O = paint(base, recolor(val, obj8))
    return O

def dsl5(I):
    top = tophalf(I)
    bh = bottomhalf(I)
    val = halve(colorcount(bh, ONE))
    inds8_top = ofcolor(top, EIGHT)
    top_base = canvas(ZERO, shape(top))
