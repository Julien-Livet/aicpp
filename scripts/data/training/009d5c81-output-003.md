def dsl1(I):
    # Global relation: draw the line between centers of 8-object and 1-object as 3; recolor all 8s to 2; erase 1s
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    p1 = ofcolor(I, ONE)
    obj8 = toobject(p8, I)
    obj1 = toobject(p1, I)
    c8 = center(obj8)
    c1 = center(obj1)
    path = connect(c8, c1)
    A = fill(base, TWO, p8)
    O = fill(A, THREE, path)
    return O

def dsl2(I):
    # Object extraction via local adjacency: 8s diagonally adjacent to any 1 -> 3, other 8s -> 2
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    p1 = ofcolor(I, ONE)
    dmask = mapply(ineighbors, p1)
    near = intersection(p8, dmask)
    far = difference(p8, near)
    A = fill(base, THREE, near)
    O = fill(A, TWO, far)
    return O

def dsl3(I):
    # Color filtering with palette-based choice: if 3 already present use 2, else use 3; erase 1s
    col = branch(contained(THREE, palette(I)), TWO, THREE)
    S = switch(I, EIGHT, col)
    O = replace(S, ONE, ZERO)
    return O

def dsl4(I):
    # Geometric symmetry (two axes): 8s symmetric under both H and V mirrors -> 3, others -> 2
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    obj8 = toobject(p8, I)
    hv = toindices(hmirror(obj8))
    vv = toindices(vmirror(obj8))
    symh = intersection(p8, hv)
    symhv = intersection(symh, vv)
    asym = difference(p8, symhv)
    A = fill(base, THREE, symhv)
    O = fill(A, TWO, asym)
    return O

def dsl5(I):
    # Relational/structural: 8s sharing any row or column with 1s -> 3, others -> 2
    dims = shape(I)
    base = canvas(ZERO, dims)
    p8 = ofcolor(I, EIGHT)
    p1 = ofcolor(I, ONE)
    rows = mapply(hfrontier, p1)
    cols = mapply(vfrontier, p1)
    align = combine(rows, cols)
    inmask = intersection(p8, align)
    outmask = difference(p8, inmask)
    A = fill(base, THREE, inmask)
    O = fill(A, TWO, outmask)
    return O
