    # Global: crop the FIVE-band, collect its non-zero, non-FIVE cells on those rows, align, then tile downward by the band's height
    A = ofcolor(I, FIVE)
    top = uppermost(A)
    bot = lowermost(A)
    k = add(subtract(bot, top), ONE)
    W = width(I)
    B = crop(I, astuple(top, ZERO), astuple(k, W))
    Ab = ofcolor(B, FIVE)
    L = merge(mapply(rbind(shoot, LEFT), Ab))
    R = merge(mapply(rbind(shoot, RIGHT), Ab))
    RowMask = combine(combine(L, R), Ab)
    NZ = difference(asindices(B), ofcolor(B, ZERO))
    Sel = intersection(RowMask, difference(NZ, Ab))

    Base = shift(P0, toivec(top))
    vec = multiply(DOWN, k)
    n = divide(height(I), k)

    Offs = apply(lbind(multiply, vec), steps)
    U = merge(mapply(lbind(shift, Base), Offs))
