def dsl1(I):
    # Global compression over the cropped foreground with background unified to modal foreground color
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    modal = mostcolor(fg)
    B = replace(G, ZERO, modal)
    O = compress(B)
    return O

def dsl2(I):
    # Object extraction: paint each object's outline on a modal-filled crop
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    objs = objects(G, T, F, T)
    modal = mostcolor(fg)
    base = replace(G, ZERO, modal)
    B = totuple(apply(box, objs))
    C = totuple(apply(color, objs))
    M = merge(mpapply(recolor, C, B))
    O = paint(base, M)
    return O

def dsl3(I):
    # Color filtering: emphasize least-common color objects via their backdrops on a modal base
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    objs = objects(G, T, F, T)
    modal = mostcolor(fg)
    base = replace(G, ZERO, modal)
    rare = leastcolor(fg)
    rareobjs = colorfilter(objs, rare)
    B = totuple(apply(backdrop, rareobjs))
    C = totuple(apply(color, rareobjs))
    M = merge(mpapply(recolor, C, B))
    O = paint(base, M)
    return O

def dsl4(I):
    # Geometric consensus: enforce agreement with horizontal and then vertical mirror, fallback to modal
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    modal = mostcolor(fg)
    B = replace(G, ZERO, modal)
    H = hmirror(B)
    A = cellwise(B, H, modal)
    V = vmirror(A)
    O = cellwise(A, V, modal)
    return O

def dsl5(I):
    # Relational: recolor the outline of the merged foreground with the least-common foreground color
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    modal = mostcolor(fg)
    base = replace(G, ZERO, modal)
    outline = box(fg)
    O = fill(base, leastcolor(fg), outline)
    return O
