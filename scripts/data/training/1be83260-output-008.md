def dsl1(I):
    # Global: crop, compress frontiers, then fill remaining background with dominant nonzero color
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    C = compress(A)
    OBJS = objects(C, T, F, T)
    DOM = mostcolor(merge(OBJS))
    O = replace(C, ZERO, DOM)
    return O

def dsl2(I):
    # Object extraction: keep only dominant-color shapes and all non-dominant objects on a zero canvas
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    OBJS = objects(A, T, F, T)
    DOM = mostcolor(merge(OBJS))
    DOBS = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, DOBS)
    BASE0 = canvas(ZERO, shape(A))
    BASE = paint(BASE0, merge(DOBS))
    O = paint(BASE, merge(OTH))
    return O

def dsl3(I):
    # Color filtering: retain only the least frequent nonzero color objects within the crop
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    OBJS = objects(A, T, F, T)
    LC = leastcolor(merge(OBJS))
    KEEP = colorfilter(OBJS, LC)
    O = paint(canvas(ZERO, shape(A)), merge(KEEP))
    return O

def dsl4(I):
    # Geometric: split crop into top/bottom halves, fill each half's background by its own dominant nonzero color, then recombine
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    TOP = tophalf(A)
    BOT = bottomhalf(A)
    TOBJS = objects(TOP, T, F, T)
    BOBJS = objects(BOT, T, F, T)
    TDOM = mostcolor(merge(TOBJS))
    BDOM = mostcolor(merge(BOBJS))
    TF = replace(TOP, ZERO, TDOM)
    BF = replace(BOT, ZERO, BDOM)
    O = vconcat(TF, BF)
    return O

def dsl5(I):
    # Relational: fill background with dominant color, then expand each non-dominant object to its bounding box recolored by its own color
    NZI = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZI, I)
    OBJS = objects(A, T, F, T)
    DOM = mostcolor(merge(OBJS))
    BASE = replace(A, ZERO, DOM)
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    TOT = totuple(OTH)
    PATS = apply(backdrop, apply(toindices, TOT))
    COLS = apply(color, TOT)
    MAKE = rbind(canvas, shape(A))
    CANS = apply(MAKE, COLS)
    COBJ = mpapply(toobject, totuple(PATS), totuple(CANS))
    EXP = merge(COBJ)
    O = paint(BASE, EXP)
    return O
