def dsl1(I):
    zeros8 = replace(I, ZERO, EIGHT)
    z1 = replace(zeros8, ONE, ZERO)
    z2 = replace(z1, NINE, ZERO)
    mirrored = vmirror(z2)
    both = cellwise(z2, mirrored, ZERO)
    axis = ofcolor(I, ONE)
    L1 = shift(axis, LEFT)
    L2 = shift(axis, multiply(LEFT, TWO))
    L3 = shift(axis, multiply(LEFT, THREE))
    leftarea = combine(L1, combine(L2, L3))
    O = crop(both, ulcorner(leftarea), astuple(height(I), THREE))
    return O

def dsl2(I):
    axis = ofcolor(I, ONE)
    Z = ofcolor(I, ZERO)
    L1 = shift(axis, LEFT)
    L2 = shift(axis, multiply(LEFT, TWO))
    L3 = shift(axis, multiply(LEFT, THREE))
    R1 = shift(axis, RIGHT)
    R2 = shift(axis, multiply(RIGHT, TWO))
    R3 = shift(axis, multiply(RIGHT, THREE))
    lz1 = intersection(Z, L1)
    lz2 = intersection(Z, L2)
    lz3 = intersection(Z, L3)
    rz1 = intersection(Z, R1)
    rz2 = intersection(Z, R2)
    rz3 = intersection(Z, R3)
    b1 = intersection(lz1, shift(rz1, multiply(LEFT, TWO)))
    b2 = intersection(lz2, shift(rz2, multiply(LEFT, FOUR)))
    b3 = intersection(lz3, shift(rz3, multiply(LEFT, SIX)))
    both = combine(b1, combine(b2, b3))
    base = canvas(ZERO, shape(I))
    painted = fill(base, EIGHT, both)
    leftarea = combine(L1, combine(L2, L3))
    O = crop(painted, ulcorner(leftarea), astuple(height(I), THREE))
    return O

def dsl3(I):
    axis = ofcolor(I, ONE)
    L1 = shift(axis, LEFT)
    L2 = shift(axis, multiply(LEFT, TWO))
    L3 = shift(axis, multiply(LEFT, THREE))
    R1 = shift(axis, RIGHT)
    R2 = shift(axis, multiply(RIGHT, TWO))
    R3 = shift(axis, multiply(RIGHT, THREE))
    leftarea = combine(L1, combine(L2, L3))
    rightarea = combine(R1, combine(R2, R3))
    dims = astuple(height(I), THREE)
    L = crop(I, ulcorner(leftarea), dims)
    R = crop(I, ulcorner(rightarea), dims)
    Ls = switch(L, ZERO, EIGHT)
    Lr1 = replace(Ls, ONE, ZERO)
    Lz = replace(Lr1, NINE, ZERO)
    Rs = switch(R, ZERO, EIGHT)
    Rr1 = replace(Rs, ONE, ZERO)
    Rz = replace(Rr1, NINE, ZERO)
    Rrev = rot180(hmirror(rot180(Rz)))
    O = cellwise(Lz, Rrev, ZERO)
    return O

def dsl4(I):
    axis = ofcolor(I, ONE)
    L1 = shift(axis, LEFT)
    L2 = shift(axis, multiply(LEFT, TWO))

    R1 = shift(axis, RIGHT)

    leftarea = combine(L1, combine(L2, L3))
    rightarea = combine(R1, combine(R2, R3))
    dims = astuple(height(I), THREE)
    L = crop(I, ulcorner(leftarea), dims)
    R = crop(I, ulcorner(rightarea), dims)
    Ls = replace(replace(replace(L, ZERO, EIGHT), ONE, ZERO), NINE, ZERO)
    Rs = replace(replace(replace(R, ZERO, EIGHT), ONE, ZERO), NINE, ZERO)
    Rrev = rot180(hmirror(rot180(Rs)))
    Lpatch = ofcolor(Ls, EIGHT)
    Rpatch = ofcolor(Rrev, EIGHT)
    overlap = intersection(Lpatch, Rpatch)
    O = fill(canvas(ZERO, dims), EIGHT, overlap)
    return O
