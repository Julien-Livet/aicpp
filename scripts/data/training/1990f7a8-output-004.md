def dsl1(I):
    R = recolor(TWO, toobject(ofcolor(I, TWO), I))
    G0 = canvas(ZERO, shape(I))
    S = paint(G0, R)
    T = trim(S)
    C = compress(T)
    H = height(C)
    W = width(C)
    FH = branch(greater(H, SEVEN), divide(H, SEVEN), ONE)
    FW = branch(greater(W, SEVEN), divide(W, SEVEN), ONE)
    F = branch(greater(FH, FW), FH, FW)
    D = downscale(C, F)
    E = compress(D)
    O = crop(E, ORIGIN, astuple(SEVEN, SEVEN))
    return O

def dsl2(I):
    P = toobject(ofcolor(I, TWO), I)
    N = normalize(P)
    C = center(N)
    OFS = subtract(astuple(THREE, THREE), C)
    S = shift(recolor(TWO, N), OFS)
    G = canvas(ZERO, astuple(SEVEN, SEVEN))
    O = paint(G, S)
    return O

def dsl3(I):
    P = toobject(ofcolor(I, TWO), I)
    N = normalize(P)
    A = recolor(TWO, N)
    B = recolor(TWO, hmirror(N))
    C = recolor(TWO, vmirror(N))
    D = recolor(TWO, dmirror(N))
    OX = tojvec(FOUR)
    OY = toivec(FOUR)
    OD = add(OX, OY)
    SA = shift(A, ORIGIN)
    SB = shift(B, OX)
    SC = shift(C, OY)
    SD = shift(D, OD)
    U1 = combine(SA, SB)
    U2 = combine(SC, SD)
    U = combine(U1, U2)
    G = canvas(ZERO, astuple(SEVEN, SEVEN))
    O = paint(G, U)
    return O

def dsl4(I):
    P = ofcolor(I, TWO)
    B = box(P)
    R = recolor(TWO, toobject(B, I))
    G0 = canvas(ZERO, shape(I))
    S = paint(G0, R)
    C = compress(S)
    M = centerofmass(ofcolor(C, TWO))
    ST = subtract(M, astuple(THREE, THREE))
    O = crop(C, ST, astuple(SEVEN, SEVEN))
    return O

def dsl5(I):
    S = compress(subgrid(ofcolor(I, TWO), I))
    L = first(vsplit(S, TWO))
    R = last(vsplit(S, TWO))
    LT = first(hsplit(L, TWO))
    LB = last(hsplit(L, TWO))
    RT = first(hsplit(R, TWO))
    RB = last(hsplit(R, TWO))
    CLT = compress(LT)
    CLB = compress(LB)
    CRT = compress(RT)
    CRB = compress(RB)
    OLT = recolor(TWO, normalize(toobject(ofcolor(CLT, TWO), CLT)))
    OLB = recolor(TWO, normalize(toobject(ofcolor(CLB, TWO), CLB)))
    ORT = recolor(TWO, normalize(toobject(ofcolor(CRT, TWO), CRT)))
    ORB = recolor(TWO, normalize(toobject(ofcolor(CRB, TWO), CRB)))
    C_LT = center(OLT)
    C_LB = center(OLB)
    C_RT = center(ORT)
    C_RB = center(ORB)
    TGT = astuple(ONE, ONE)
    OX = tojvec(FOUR)
    OY = toivec(FOUR)
    OD = add(OX, OY)
    SLT = shift(OLT, subtract(TGT, C_LT))
    SRT = shift(ORT, add(OX, subtract(TGT, C_RT)))
    SLB = shift(OLB, add(OY, subtract(TGT, C_LB)))
    SRB = shift(ORB, add(OD, subtract(TGT, C_RB)))
    U1 = combine(SLT, SRT)
    U2 = combine(SLB, SRB)
    U = combine(U1, U2)
    G = canvas(ZERO, astuple(SEVEN, SEVEN))
    O = paint(G, U)
    return O
