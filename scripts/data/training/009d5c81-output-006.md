def dsl1(I):
    # Single 1-object: center-filled -> 2, center-empty -> 7; multiple 1-objects -> 3
    objs1 = colorfilter(objects(I, T, F, T), ONE)
    n1 = size(objs1)
    o1 = first(objs1)
    c = center(o1)
    i1 = toindices(o1)
    hasc = contained(c, i1)
    col1 = branch(hasc, TWO, SEVEN)
    col = branch(greater(n1, ONE), THREE, branch(equality(n1, ONE), col1, TWO))
    S = switch(I, EIGHT, col)
    O = replace(S, ONE, ZERO)
    return O

def dsl2(I):

    objs1 = colorfilter(objects(I, T, F, T), ONE)

    c1 = colorcount(I, ONE)
