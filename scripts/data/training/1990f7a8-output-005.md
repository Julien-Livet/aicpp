def dsl1(I):
    # Global: compress + downscale to 7x7, then enforce a zero cross through the center
    S = compress(subgrid(ofcolor(I, TWO), I))
    H = height(S)
    W = width(S)
    FH = branch(greater(H, SEVEN), divide(H, SEVEN), ONE)
    FW = branch(greater(W, SEVEN), divide(W, SEVEN), ONE)
    FCT = branch(greater(FH, FW), FH, FW)
    D = downscale(S, FCT)
    E = compress(D)
    OOBJ = recolor(TWO, toobject(ofcolor(E, TWO), E))
    G7 = canvas(ZERO, astuple(SEVEN, SEVEN))
    O0 = paint(G7, OOBJ)
    R0 = astuple(THREE, ZERO)
    R6 = astuple(THREE, SIX)
    C0 = astuple(ZERO, THREE)
    C6 = astuple(SIX, THREE)
    ROW = connect(R0, R6)
    COL = connect(C0, C6)
    CROSS = combine(ROW, COL)
    O = fill(O0, ZERO, CROSS)
    return O

def dsl2(I):
    # Object extraction per quadrant with local rescale to 3x3 and placement in 7x7 tiles
    S = compress(subgrid(ofcolor(I, TWO), I))
    T = first(hsplit(S, TWO))
    B = last(hsplit(S, TWO))
    TLg = first(vsplit(T, TWO))
    TRg = last(vsplit(T, TWO))
    BLg = first(vsplit(B, TWO))
    BRg = last(vsplit(B, TWO))

    CTL = compress(TLg)
    CTR = compress(TRg)
    CBL = compress(BLg)
    CBR = compress(BRg)

    HTL = height(CTL); WTL = width(CTL)
    HTR = height(CTR); WTR = width(CTR)
    HBL = height(CBL); WBL = width(CBL)
    HBR = height(CBR); WBR = width(CBR)

    FHTL = branch(greater(HTL, THREE), divide(HTL, THREE), ONE)
    FWTL = branch(greater(WTL, THREE), divide(WTL, THREE), ONE)
    FHTR = branch(greater(HTR, THREE), divide(HTR, THREE), ONE)
    FWTR = branch(greater(WTR, THREE), divide(WTR, THREE), ONE)
    FHBL = branch(greater(HBL, THREE), divide(HBL, THREE), ONE)
    FWBL = branch(greater(WBL, THREE), divide(WBL, THREE), ONE)
    FHBR = branch(greater(HBR, THREE), divide(HBR, THREE), ONE)
    FWBR = branch(greater(WBR, THREE), divide(WBR, THREE), ONE)

    FTL = branch(greater(FHTL, FWTL), FHTL, FWTL)
    FTR = branch(greater(FHTR, FWTR), FHTR, FWTR)
    FBL = branch(greater(FHBL, FWBL), FHBL, FWBL)
    FBR = branch(greater(FHBR, FWBR), FHBR, FWBR)

    DTL = downscale(CTL, FTL)
    DTR = downscale(CTR, FTR)
    DBL = downscale(CBL, FBL)
    DBR = downscale(CBR, FBR)

    ETL = compress(DTL)
    ETR = compress(DTR)
    EBL = compress(DBL)
    EBR = compress(DBR)

    PTL = crop(ETL, ORIGIN, astuple(THREE, THREE))
    PTR = crop(ETR, ORIGIN, astuple(THREE, THREE))
    PBL = crop(EBL, ORIGIN, astuple(THREE, THREE))
    PBR = crop(EBR, ORIGIN, astuple(THREE, THREE))

    OTL = recolor(TWO, toobject(ofcolor(PTL, TWO), PTL))
    OTR = recolor(TWO, toobject(ofcolor(PTR, TWO), PTR))
    OBL = recolor(TWO, toobject(ofcolor(PBL, TWO), PBL))
    OBR = recolor(TWO, toobject(ofcolor(PBR, TWO), PBR))

    CT = center(OTL)
    CR = center(OTR)
    CB = center(OBL)
    CC = center(OBR)

    TGT = astuple(ONE, ONE)
    OFTL = subtract(TGT, CT)
    OFTR = subtract(TGT, CR)
    OFBL = subtract(TGT, CB)
    OFBR = subtract(TGT, CC)

    OX = tojvec(FOUR)
    OY = toivec(FOUR)
    OD = add(OX, OY)

    STL = shift(OTL, OFTL)
    STR = shift(OTR, add(OX, OFTR))
    SBL = shift(OBL, add(OY, OFBL))
    SBR = shift(OBR, add(OD, OFBR))

    U1 = combine(STL, STR)
    U2 = combine(SBL, SBR)
    U = combine(U1, U2)

    G7 = canvas(ZERO, astuple(SEVEN, SEVEN))
    O = paint(G7, U)
    return O

def dsl4(I):
    # Geometric: tile one normalized shape into the four corners without mirroring
    P = toobject(ofcolor(I, TWO), I)
    N = normalize(P)
    A = recolor(TWO, N)
    OX = tojvec(FOUR)
    OY = toivec(FOUR)
    OD = add(OX, OY)
    SA = shift(A, ORIGIN)
    SB = shift(A, OX)
    SC = shift(A, OY)
    SD = shift(A, OD)
    U1 = combine(SA, SB)
    U2 = combine(SC, SD)
    U = combine(U1, U2)
    G = canvas(ZERO, astuple(SEVEN, SEVEN))
    O = paint(G, U)

def dsl5(I):
    # Relational: split into quadrants, normalize each shape and center within its 3x3 tile
    S = compress(subgrid(ofcolor(I, TWO), I))
    L = first(vsplit(S, TWO))
    R = last(vsplit(S, TWO))
    LT = first(hsplit(L, TWO))
    LB = last(hsplit(L, TWO))
    RT = first(hsplit(R, TWO))
    RB = last(hsplit(R, TWO))

    OLT = recolor(TWO, normalize(toobject(ofcolor(compress(LT), TWO), compress(LT))))
    ORT = recolor(TWO, normalize(toobject(ofcolor(compress(RT), TWO), compress(RT))))
    OLB = recolor(TWO, normalize(toobject(ofcolor(compress(LB), TWO), compress(LB))))
    ORB = recolor(TWO, normalize(toobject(ofcolor(compress(RB), TWO), compress(RB))))

    C_LT = center(OLT)
    C_RT = center(ORT)
    C_LB = center(OLB)
    C_RB = center(ORB)

    TGT = astuple(ONE, ONE)
    SLT = shift(OLT, subtract(TGT, C_LT))
    SRT = shift(ORT, subtract(TGT, C_RT))
    SLB = shift(OLB, subtract(TGT, C_LB))
    SRB = shift(ORB, subtract(TGT, C_RB))

    OX = tojvec(FOUR)
    OY = toivec(FOUR)
    OD = add(OX, OY)

    STL = shift(SLT, ORIGIN)
    STR = shift(SRT, OX)
    SBL = shift(SLB, OY)
    SBR = shift(SRB, OD)

    U1 = combine(STL, STR)
    U2 = combine(SBL, SBR)
    U = combine(U1, U2)
    G = canvas(ZERO, astuple(SEVEN, SEVEN))
    O = paint(G, U)
    return O
