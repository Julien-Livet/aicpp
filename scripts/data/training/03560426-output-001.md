def dsl1(I):
    A = bottomhalf(I)
    B = rot270(A)
    H = height(I)
    W = width(I)
    WB = width(B)
    R = subtract(W, WB)
    D = astuple(H, R)
    Z = canvas(ZERO, D)
    O = hconcat(B, Z)
    return O

def dsl2(I):
    BG = ofcolor(I, ZERO)
    ALL = asindices(I)
    FG = difference(ALL, BG)
    U = toobject(FG, I)
    N = normalize(U)
    Z = canvas(ZERO, shape(I))
    O = paint(Z, N)
    return O

def dsl3(I):
    P = palette(I)
    Cset = remove(ZERO, P)
    F = lbind(colorcount, I)
    C = valmin(Cset, F)
    Ind = ofcolor(I, C)
    Obj = toobject(Ind, I)
    N = normalize(Obj)
    Z = canvas(ZERO, shape(I))
    O = paint(Z, N)
    return O

def dsl4(I):

    HB = height(B)
    R = subtract(H, HB)
    D = astuple(R, W)

def dsl5(I):
    A = objects(I, T, F, T)
    B = order(A, leftmost)
    P0 = initset(ORIGIN)
    Fdir = rbind(gravitate, P0)
    D = apply(Fdir, B)
    S = mpapply(shift, B, D)
    U = merge(S)
    Z = canvas(ZERO, shape(I))
    O = paint(Z, U)
