def dsl1(I):
    C = compress(I)
    B = mostcolor(I)
    O = replace(C, leastcolor(C), B)
    return O

def dsl2(I):
    OB = objects(I, T, F, T)
    SC = leastcolor(I)
    SO = colorfilter(OB, SC)
    HL = mfilter(SO, hline)
    VL = mfilter(SO, vline)
    BH = argmax(HL, size)
    BV = argmax(VL, size)
    HC = size(occurrences(I, BH))
    VC = size(occurrences(I, BV))
    HD = add(HC, ONE)
    WD = add(VC, ONE)
    D = astuple(HD, WD)
    BG = mostcolor(I)
    O = canvas(BG, D)
    return O

def dsl3(I):
    FR = frontiers(I)

    VFR = mfilter(FR, vline)

def dsl4(I):
    SC = leastcolor(I)
    BG = mostcolor(I)
    S = switch(I, SC, BG)
    C = compress(S)
    O = replace(C, SC, BG)
    return O
