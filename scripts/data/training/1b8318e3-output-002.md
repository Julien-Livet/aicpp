def dsl1(I):
    # Global (mirror within five-bbox)
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    dest = mapply(toindices, fives)
    bb = backdrop(dest)
    sg = subgrid(bb, I)
    mir = hmirror(sg)
    obj = asobject(mir)
    base = canvas(ZERO, shape(I))
    O = move(base, obj, ulcorner(bb))
    return O

def dsl2(I):
    # Object extraction (row-sharing with fives)
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    dest = mapply(toindices, fives)
    others = difference(objs, fives)
    rowmatch = rbind(hmatching, dest)
    sel = sfilter(others, rowmatch)
    keep = combine(merge(fives), merge(sel))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl3(I):
    # Color filtering (adjacent to five-bbox outline)
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    dest = mapply(toindices, fives)
    bb = backdrop(dest)
    outline = box(bb)
    others = difference(objs, fives)
    near = sfilter(others, rbind(adjacent, outline))
    keep = combine(merge(fives), merge(near))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl4(I):
    # Geometric (gravitate all non-5 to five-clusters)
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    dest = mapply(toindices, fives)
    others = difference(objs, fives)
    dirf = rbind(gravitate, dest)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    keep = combine(merge(fives), mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O
