def dsl1(I):
    tl = tophalf(lefthalf(I))
    top = hconcat(tl, tl)
    O = vconcat(top, top)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    notborder = compose(flip, rbind(bordering, I))
    internal = sfilter(objs, notborder)
    cells = merge(internal)
    O = paint(canvas(ZERO, shape(I)), cells)
    return O

def dsl3(I):
    objs = objects(I, T, F, T)
    cols = apply(color, objs)

    sel = colorfilter(objs, mc)
    cells = merge(sel)
    O = paint(canvas(ZERO, shape(I)), cells)

def dsl4(I):
    inter = cellwise(I, hmirror(I), ZERO)
    O = cellwise(inter, vmirror(I), ZERO)
    return O

def dsl5(I):
    objs = objects(I, T, F, T)
    centers = apply(center, objs)
    whole = asobject(I)
    gc = center(whole)
    raysets = apply(lbind(connect, gc), centers)
