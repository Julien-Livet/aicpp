def dsl1(I):
    # Global: trim borders and compress internal frontiers
    O = compress(trim(I))
    return O

def dsl2(I):
    # Object extraction: crop nonzero area, take largest object, then compress
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    SUB = subgrid(NZ, I)
    OBJS = objects(SUB, T, F, T)
    BIG = argmax(OBJS, size)
    A = subgrid(BIG, SUB)
    O = compress(A)
    return O

def dsl3(I):
    # Color filtering: keep all non-dominant objects on the dominant (within crop) canvas
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    DOM = mostcolor(A)
    BASE = canvas(DOM, shape(A))
    OBJS = objects(A, T, F, T)
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    EXP = merge(OTH)
    O = paint(BASE, EXP)
    return O

def dsl4(I):
    # Geometric: crop nonzero content, compress, then mirror horizontally
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    C = compress(A)
    O = hmirror(C)
    return O

def dsl5(I):
    # Relational/structural: expand each non-dominant object to its backdrop and paint on dominant canvas
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    DOM = mostcolor(A)
    BASE = canvas(DOM, shape(A))
    OBJS = objects(A, T, F, T)
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    TOT = order(OTH, uppermost)
    COLS = totuple(apply(color, TOT))
    PATS = totuple(apply(compose(backdrop, toindices), TOT))
    COLORED = mpapply(recolor, COLS, PATS)
    EXP = merge(COLORED)
    O = paint(BASE, EXP)
    return O
