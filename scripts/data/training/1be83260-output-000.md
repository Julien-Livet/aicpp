def dsl1(I):
    A = compress(I)
    O = replace(A, ZERO, ONE)
    return O

def dsl2(I):
    OBJS = objects(I, T, F, T)
    BIG = argmax(OBJS, size)
    A = subgrid(BIG, I)
    O = replace(A, ZERO, ONE)
    return O

def dsl3(I):
    P2 = ofcolor(I, TWO)
    A = subgrid(P2, I)
    O = replace(A, ZERO, ONE)
    return O

def dsl4(I):
    A = compress(I)
    B = hmirror(A)
    O = trim(B)
    return O

def dsl5(I):
    BASE = canvas(ONE, shape(I))
    FR = frontiers(I)
    INDS = mapply(toindices, FR)
    COL = mostcolor(I)
    COLOBJ = recolor(COL, INDS)
    O = paint(BASE, COLOBJ)
    return O
