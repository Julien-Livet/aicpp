def dsl1(I):
    # Global compression of the non-5 content, placed at top-left of original canvas
    I2 = replace(I, FIVE, ZERO)
    C = compress(I2)
    O = paint(canvas(ZERO, shape(I)), asobject(C))
    return O

def dsl2(I):
    # Keep only edge-touching objects and rectangularize them
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    keep = sfilter(objs, lbind(bordering, I2))
    rects = apply(fork(recolor, color, inbox), keep)
    O = paint(canvas(ZERO, shape(I)), merge(rects))
    return O

def dsl3(I):
    # Rectangularize the group of the most frequent nonzero color
    I2 = replace(I, FIVE, ZERO)
    cols = remove(ZERO, palette(I2))
    maj = valmax(cols, lbind(colorcount, I2))
    grp = compose(merge, lbind(colorfilter, objects(I2, T, F, T)))
    rect = recolor(maj, inbox(grp(maj)))
    O = paint(canvas(ZERO, shape(I)), rect)
    return O

def dsl4(I):
    # Draw the bounding box outline for every object (geometric boxing)
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    outlines = apply(fork(recolor, color, box), objs)
    O = paint(canvas(ZERO, shape(I)), merge(outlines))
    return O

def dsl5(I):
    # Keep objects that share any column with others; rectangularize them
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    others = rbind(remove, objs)
    union_others = compose(merge, others)
    aligned = sfilter(objs, fork(vmatching, identity, union_others))
    rects = apply(fork(recolor, color, inbox), aligned)
    O = paint(canvas(ZERO, shape(I)), merge(rects))
    return O
