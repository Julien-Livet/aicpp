def dsl1(I):
    # O = overlay all four rotations of the grid (global transformation)
    A0 = asobject(I)
    A1 = asobject(rot90(I))
    A2 = asobject(rot180(I))
    A3 = asobject(rot270(I))
    stamps = merge(insert(A0, insert(A1, insert(A2, initset(A3)))))
    O = paint(canvas(ZERO, shape(I)), stamps)
    return O

def dsl2(I):
    # O = fill each object's bounding box with the object's color (object extraction)
    objs = objects(I, T, F, T)
    bb = compose(backdrop, toindices)
    stamped = apply(fork(recolor, color, bb), objs)
    O = paint(canvas(ZERO, shape(I)), merge(stamped))
    return O

def dsl3(I):
    # O = for the least-common color, draw full rows and columns through its pixels (color filtering)
    lc = leastcolor(I)
    pts = ofcolor(I, lc)
    hs = apply(hfrontier, pts)
    vs = apply(vfrontier, pts)
    rays = merge(combine(hs, vs))
    O = paint(canvas(ZERO, shape(I)), recolor(lc, rays))
    return O

def dsl4(I):
    # O = from each object's center, shoot rays in both nearest-border direction and its opposite (geometric reasoning)
    objs = objects(I, T, F, T)
    border = box(asindices(I))
    ceno = compose(center, toindices)
    dirn = compose(rbind(gravitate, border), toindices)
    odir = compose(invert, dirn)
    rayf = fork(shoot, ceno, dirn)
    rayb = fork(shoot, ceno, odir)
    fwd = apply(fork(recolor, color, rayf), objs)
    bwd = apply(fork(recolor, color, rayb), objs)
    O = paint(canvas(ZERO, shape(I)), merge(combine(fwd, bwd)))
    return O

def dsl5(I):
    # O = draw a colored cross (row and column) through each object's center (relational/structural)
    objs = objects(I, T, F, T)
    cen = compose(center, toindices)
    hcen = compose(hfrontier, cen)
    vcen = compose(vfrontier, cen)
    hs = apply(fork(recolor, color, hcen), objs)
    vs = apply(fork(recolor, color, vcen), objs)
    O = paint(canvas(ZERO, shape(I)), merge(combine(hs, vs)))
    return O
