def dsl1(I):
    # Global: overlay diagonal mirror
    A = asobject(I)
    B = asobject(dmirror(I))
    OBJ = combine(A, B)
    O = underpaint(I, OBJ)
    return O

def dsl2(I):
    # Object extraction: draw boxes around each foreground color set
    FG = fgpartition(I)
    BX = compose(box, toindices)
    RC = fork(recolor, color, BX)
    PARTS = apply(RC, FG)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O

def dsl3(I):
    # Color filtering: propagate least frequent nonzero color along cardinal directions
    FG = fgpartition(I)
    ALL = merge(FG)
    C = leastcolor(ALL)
    S = ofcolor(I, C)
    DIR0 = initset(UP)
    DIR1 = insert(DOWN, DIR0)
    DIR2 = insert(LEFT, DIR1)
    DIR = insert(RIGHT, DIR2)
    R0 = lbind(prapply, shoot)
    FN = rbind(R0, DIR)
    LINES = compose(merge, FN)(S)
    COLORED = recolor(C, LINES)
    O = underpaint(I, COLORED)
    return O

def dsl4(I):
    # Geometric: symmetrize each object by union with its horizontal mirror
    FG = fgpartition(I)
    HM = compose(toindices, hmirror)
    UNION = fork(combine, toindices, HM)
    RC = fork(recolor, color, UNION)
    PARTS = apply(RC, FG)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O

def dsl5(I):
    # Relational: from each object, shoot towards nearest border and opposite
    FG = fgpartition(I)
    BORDER = box(asindices(I))
    GFN = rbind(gravitate, BORDER)
    DFN = compose(GFN, toindices)
    SD = compose(initset, DFN)
    NEGDIR = compose(invert, DFN)
    DIRS = fork(insert, NEGDIR, SD)
    R0 = lbind(prapply, shoot)
    RAW = fork(R0, toindices, DIRS)
    PFN = compose(merge, RAW)
    RC = fork(recolor, color, PFN)
    PARTS = apply(RC, FG)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O
