def dsl1(I):
    # Relational: straight segments from 4 to 5 containing only {4,5}
    fours = ofcolor(I, FOUR)
    fives = ofcolor(I, FIVE)
    segs = prapply(connect, fours, fives)
    straight = sfilter(segs, fork(either, hline, vline))
    allowed = combine(fours, fives)
    pure = sfilter(straight, matcher(compose(size, rbind(difference, allowed)), ZERO))
    mask = merge(pure)
    target = intersection(mask, fives)
    O = fill(I, TWO, target)
    return O

def dsl2(I):
    # Global edge-based: frontier cells that touch any 4, restricted to 5s
    frs = frontiers(I)
    fours = ofcolor(I, FOUR)
    touching = sfilter(frs, lbind(adjacent, fours))
    edges = merge(mapply(toindices, touching))
    target = intersection(edges, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O

def dsl3(I):
    # Object extraction: outbox ring around each 4-object, restricted to 5s
    fourobjs = colorfilter(objects(I, T, F, T), FOUR)
    rings = merge(mapply(outbox, fourobjs))
    target = intersection(rings, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O

def dsl4(I):
    # Local adjacency: 5s directly adjacent (4-neighborhood) to any 4
    fours = ofcolor(I, FOUR)
    dirs = insert(LEFT, insert(RIGHT, insert(UP, initset(DOWN))))
    nbrs = merge(mapply(lbind(shift, fours), dirs))
    target = intersection(nbrs, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O

def dsl5(I):
    # Global lines: rows/cols through each 4; restrict to 5s along those lines
    fours = ofcolor(I, FOUR)
    dirs = insert(LEFT, insert(RIGHT, insert(UP, initset(DOWN))))
    rays = prapply(shoot, fours, dirs)
