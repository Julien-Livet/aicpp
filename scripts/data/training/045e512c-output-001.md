def dsl1(I):
    objall = merge(partition(I))
    base = hmirror(I)
    step1 = underpaint(base, objall)
    base2 = vmirror(step1)
    O = underpaint(base2, objall)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    frameobj = fork(recolor, color, compose(box, toindices))
    wvec = compose(tojvec, width)
    hvec = compose(toivec, height)
    addvec = fork(add, hvec, wvec)
    fr = apply(frameobj, objs)
    sr = apply(fork(shift, frameobj, wvec), objs)
    sd = apply(fork(shift, frameobj, hvec), objs)
    sdiag = apply(fork(shift, frameobj, addvec), objs)
    allcells = merge(combine(combine(fr, sr), combine(sd, sdiag)))
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl3(I):
    objs = objects(I, T, F, T)
    small1 = sizefilter(objs, ONE)
    small3 = sizefilter(objs, THREE)
    smalls = combine(small1, small3)
    others = difference(objs, smalls)
    nbors = compose(lbind(mapply, neighbors), toindices)
    expand = fork(combine, toindices, nbors)
    brect = compose(backdrop, expand)
    fillbox = fork(recolor, color, brect)
    filled = apply(fillbox, smalls)
    union_objs = combine(others, filled)
    allcells = merge(union_objs)
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl4(I):
    objs = objects(I, T, F, T)
    dx4 = tojvec(FOUR)
    dx8 = double(dx4)
    dxm4 = invert(dx4)
    dxm8 = invert(dx8)
    sh4 = apply(rbind(shift, dx4), objs)
    sh8 = apply(rbind(shift, dx8), objs)
    sh_4 = apply(rbind(shift, dxm4), objs)
    sh_8 = apply(rbind(shift, dxm8), objs)
    tiled = combine(combine(sh_8, sh_4), combine(objs, combine(sh4, sh8)))
    allcells = merge(tiled)
    O = paint(canvas(ZERO, shape(I)), allcells)
    return O

def dsl5(I):
    objs = objects(I, T, F, T)
    diag1 = fork(connect, ulcorner, lrcorner)
    diag2 = fork(connect, urcorner, llcorner)
    diags = fork(combine, diag1, diag2)
    lines = apply(fork(recolor, color, diags), objs)
    allcells = merge(lines)
