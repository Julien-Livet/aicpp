def dsl1(I):
    FG = fgpartition(I)
    DIR = insert(UNITY, initset(NEG_UNITY))
    R0 = lbind(prapply, shoot)
    R1 = rbind(R0, DIR)
    PFN = compose(merge, R1)
    RC = fork(recolor, color, compose(PFN, toindices))
    PARTS = apply(RC, FG)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O

def dsl3(I):
    FG = fgpartition(I)
    ALL = merge(FG)
    C = mostcolor(ALL)
    S = ofcolor(I, C)
    DIR0 = initset(LEFT)
    DIR1 = insert(RIGHT, DIR0)
    DIR2 = insert(DOWN, DIR1)
    DIR = insert(UP, DIR2)
    R0 = lbind(prapply, shoot)
    R1 = rbind(R0, DIR)
    PFN = compose(merge, R1)
    LINES = PFN(S)
    COLORED = recolor(C, LINES)
    O = underpaint(I, COLORED)
    return O

def dsl4(I):
