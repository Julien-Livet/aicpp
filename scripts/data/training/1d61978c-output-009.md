def dsl1(I):
    # relational selection: choose between the two bbox diagonals by max overlap with 5, recolor chosen as 2 and the rest as 8
    S5 = ofcolor(I, FIVE)
    A1 = connect(urcorner(S5), llcorner(S5))
    A2 = connect(ulcorner(S5), lrcorner(S5))
    Axes = insert(A2, initset(A1))
    score = chain(size, lbind(intersection, S5), identity)
    Best = argmax(Axes, score)
    S2 = intersection(S5, Best)
    S8 = difference(S5, S2)
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O

def dsl2(I):
    # object extraction: select all 5-objects that touch the dominant bbox diagonal, paint them 2, others 8
    S5 = ofcolor(I, FIVE)
    A1 = connect(urcorner(S5), llcorner(S5))
    A2 = connect(ulcorner(S5), lrcorner(S5))
    Axes = insert(A2, initset(A1))
    score = chain(size, lbind(intersection, S5), identity)
    Best = argmax(Axes, score)
    O5 = colorfilter(objects(I, T, T, T), FIVE)
    cond = chain(positive, size, lbind(intersection, Best), toindices)
    Touching = sfilter(O5, cond)
    S2 = mapply(toindices, Touching)
    S8 = difference(S5, S2)
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O

def dsl3(I):
    # geometric symmetry: mirror the main bbox diagonal to get the counterdiagonal, clip to bbox, recolor 2 vs 8
    S5 = ofcolor(I, FIVE)
    Main = connect(ulcorner(S5), lrcorner(S5))
    Axis = intersection(cmirror(Main), backdrop(S5))
    S2 = intersection(S5, Axis)
    S8 = difference(S5, S2)
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O

def dsl4(I):
    # global rays within bbox: build UR↘ and LL↖ rays and clip to the 5-bbox as the axis for 2; others -> 8
    S5 = ofcolor(I, FIVE)
    Rays = combine(shoot(urcorner(S5), DOWN_LEFT), shoot(llcorner(S5), UP_RIGHT))
    Axis = intersection(Rays, backdrop(S5))
    S2 = intersection(S5, Axis)
    S8 = difference(S5, S2)
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O

def dsl5(I):
    # relational centerline: generate both diagonals through bbox center, pick the one with more 5 overlap for 2; others -> 8
    S5 = ofcolor(I, FIVE)
    C = center(S5)
    D1 = combine(shoot(C, UP_RIGHT), shoot(C, DOWN_LEFT))
    A1 = intersection(D1, backdrop(S5))
    A2 = intersection(cmirror(D1), backdrop(S5))
    Axes = insert(A2, initset(A1))
    score = chain(size, lbind(intersection, S5), identity)

    S8 = difference(S5, S2)
