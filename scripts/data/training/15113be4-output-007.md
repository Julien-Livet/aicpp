def dsl1(I):
    # Global: fill ones inside the bounding box of the least-color region
    val = leastcolor(I)
    rect = backdrop(ofcolor(I, val))
    target = intersection(rect, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl2(I):
    # Object extraction: fill one-objects that lie in the delta (bbox minus patch) of least-color
    val = leastcolor(I)
    ring = delta(ofcolor(I, val))
    ones = colorfilter(fgpartition(I), ONE)
    idx = toindices
    intr = compose(lbind(intersection, ring), idx)
    pred = chain(positive, size, intr)
    hits = sfilter(ones, pred)
    target = merge(apply(idx, hits))
    O = fill(I, val, target)
    return O

def dsl3(I):
    # Color filtering: fill ones on the outline (box) of the least-color region
    val = leastcolor(I)
    border = box(ofcolor(I, val))
    target = intersection(border, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl4(I):
    # Geometric: connect the center of least-color bbox to its corners and recolor intersecting ones
    val = leastcolor(I)
    rect = backdrop(ofcolor(I, val))
    c = center(rect)
    paths = merge(apply(lbind(connect, c), corners(rect)))
    target = intersection(paths, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl5(I):
    # Relational: for one-objects sharing a column with least-color, connect to its center of mass
    val = leastcolor(I)
    vp = ofcolor(I, val)
    cv = centerofmass(vp)
    ones = colorfilter(fgpartition(I), ONE)
    idx = toindices
    colmatch = compose(lbind(vmatching, vp), idx)
    hits = sfilter(ones, colmatch)
    comobj = compose(centerofmass, idx)
    pathfunc = compose(lbind(connect, cv), comobj)
    paths = merge(apply(pathfunc, hits))
    target = intersection(paths, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O
