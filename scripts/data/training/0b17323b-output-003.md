def dsl1(I):
    ones = ofcolor(I, ONE)
    obj = toobject(ones, I)
    d = hperiod(obj)
    allidx = asindices(I)
    diag = connect(ulcorner(allidx), lrcorner(allidx))
    ray = shoot(ulcorner(ones), multiply(UNITY, d))
    ext = intersection(ray, diag)
    twos = recolor(TWO, difference(ext, inbox(ones)))
    O = underpaint(I, twos)
    return O

def dsl2(I):
    ones = ofcolor(I, ONE)
    ul = ulcorner(ones)
    lr = lrcorner(ones)
    n = size(ones)
    k = decrement(n)
    vec = divide(subtract(lr, ul), k)
    start = add(lr, vec)
    path = shoot(start, vec)
    O = underpaint(I, recolor(TWO, path))
    return O
