def dsl1(I):
    # Global: compress after removing background
    MC = mostcolor(I)
    J = replace(I, ZERO, MC)
    O = compress(J)
    return O

def dsl2(I):
    # Object extraction: crop largest foreground object and fill holes
    OBJS = fgpartition(I)
    BIG = argmax(OBJS, size)
    A = subgrid(BIG, I)
    Z = ofcolor(A, ZERO)
    O = fill(A, ONE, Z)
    return O

def dsl3(I):
    # Color filtering: keep non-dominant colors on a dominant-color canvas
    DOM = mostcolor(I)
    BASE = canvas(DOM, shape(I))
    OBJS = objects(I, T, F, T)
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    INDS = mapply(toindices, OTH)
    M = merge(INDS)
    P = recolor(ONE, M)
    O = paint(BASE, P)
    return O

def dsl4(I):
    # Geometric: rotate cropped content by 180 degrees
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    O = rot180(A)
    return O

def dsl5(I):
    # Relational/structural: expand non-dominant objects to their backdrops and paint
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    DOM = mostcolor(A)
    BASE = canvas(DOM, shape(A))
    OBJS = objects(A, T, F, T)
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    COLS = totuple(apply(color, OTH))
    PATS = totuple(apply(compose(backdrop, toindices), OTH))
    COLORED = mpapply(recolor, COLS, PATS)
    EXP = merge(COLORED)
    O = paint(BASE, EXP)
    return O
