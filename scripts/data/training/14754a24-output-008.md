def dsl1(I):
    # Portions of 5-objects that lie on any 4-row or 4-column and touch a 4
    fou = ofcolor(I, FOUR)
    fiveobjs = colorfilter(objects(I, T, F, T), FIVE)
    adj5 = sfilter(fiveobjs, lbind(adjacent, fou))
    adj5idx = mapply(toindices, adj5)
    rows = mapply(hfrontier, fou)
    cols = mapply(vfrontier, fou)
    rowparts = prapply(intersection, rows, adj5idx)
    colparts = prapply(intersection, cols, adj5idx)
    target = merge(combine(rowparts, colparts))
    O = fill(I, TWO, target)
    return O

def dsl2(I):
    # Propagate from 5-seeds adjacent to 4 along pure axial 5-runs
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    nbh = mapply(neighbors, fou)
    seeds = intersection(nbh, fiv)
    segs = prapply(connect, seeds, fiv)
    axial = sfilter(segs, fork(either, hline, vline))
    pure = sfilter(axial, matcher(compose(size, rbind(difference, fiv)), ZERO))
    mask = merge(pure)
    O = fill(I, TWO, mask)
    return O

def dsl3(I):
    # 5s strictly between two 4s along pure axial segments
    fou = ofcolor(I, FOUR)
    fiv = ofcolor(I, FIVE)
    segs = prapply(connect, fou, fou)
    axial = sfilter(segs, fork(either, hline, vline))
    allowed = combine(fou, fiv)
    pure = sfilter(axial, matcher(compose(size, rbind(difference, allowed)), ZERO))
    mask = merge(pure)
    target = intersection(mask, fiv)
    O = fill(I, TWO, target)
    return O

def dsl4(I):
    # Cross of all 4-rows with all 4-columns restricted to line-shaped 5-objects
    fou = ofcolor(I, FOUR)
    rows = mapply(hfrontier, fou)
    cols = mapply(vfrontier, fou)
    cross = intersection(merge(rows), merge(cols))
    fiveobjs = colorfilter(objects(I, T, F, T), FIVE)
    lines = sfilter(fiveobjs, fork(either, hline, vline))
    linesidx = mapply(toindices, lines)
    allowed = merge(linesidx)
    target = intersection(cross, allowed)
    O = fill(I, TWO, target)
    return O

def dsl5(I):
    # Line-shaped 5-objects adjacent to 4s after removing 4s (splitting 5-components)
    fou = ofcolor(I, FOUR)
    obj4 = toobject(fou, I)
    cleared = cover(I, obj4)
    fiveobjs = colorfilter(objects(cleared, T, F, T), FIVE)
    touching = sfilter(fiveobjs, lbind(adjacent, fou))
    lines = sfilter(touching, fork(either, hline, vline))
    obj = mapply(lbind(recolor, TWO), lines)
    O = paint(I, obj)
    return O
