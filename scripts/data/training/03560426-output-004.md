def dsl1(I):
    BG = ofcolor(I, ZERO)
    FG = difference(asindices(I), BG)
    U = toobject(FG, I)
    G = subgrid(U, I)
    H = height(I)
    W = width(I)
    HR = height(G)
    WR = width(G)
    Zb = canvas(ZERO, astuple(subtract(H, HR), WR))
    T = vconcat(G, Zb)
    Zr = canvas(ZERO, astuple(H, subtract(W, WR)))
    O = hconcat(T, Zr)
    return O

def dsl2(I):
    A = objects(I, T, F, T)
    N = apply(normalize, A)
    NI = apply(toindices, N)
    C = apply(color, A)
    RO = mpapply(recolor, C, NI)
    U = merge(RO)
    Z = canvas(ZERO, shape(I))
    O = paint(Z, U)
    return O
