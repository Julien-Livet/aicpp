def dsl1(I):
    # Global: compress both axes via rotation
    A = compress(I)
    B = rot90(A)
    C = compress(B)
    O = rot270(C)
    return O

def dsl2(I):
    # Object extraction: keep the largest non-1 object as a filled box on 8 canvas
    OBJS = objects(I, T, F, T)
    NON1 = mfilter(OBJS, compose(flip, matcher(color, ONE)))
    MAIN = argmax(NON1, size)
    FILLED = recolor(color(MAIN), inbox(MAIN))
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, FILLED)
    return O

def dsl3(I):
    # Color filtering: remove 2 by turning it into 8, then crop to non-8 extent
    A = replace(I, TWO, EIGHT)
    ALL = asindices(A)
    BG = ofcolor(A, EIGHT)
    FG = difference(ALL, BG)
    O = subgrid(FG, A)
    return O

def dsl4(I):
    # Geometric: keep cells invariant under vertical mirroring, else 8
    A = vmirror(I)
    O = cellwise(I, A, EIGHT)
    return O

def dsl5(I):
    # Relational: recolor each object's bounding box outline by its center color on 8 canvas
    OBJS = objects(I, T, F, T)
    FCOL = compose(lbind(index, I), center)
    OUTS = apply(fork(recolor, FCOL, box), OBJS)
    ALL = merge(OUTS)
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, ALL)
    return O
