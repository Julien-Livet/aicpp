def dsl1(I):
    # Global: remove 5, then fill the overall bounding box (inbox) for each remaining color
    I2 = replace(I, FIVE, ZERO)
    cols = remove(ZERO, palette(I2))
    patcher = compose(inbox, lbind(ofcolor, I2))
    parts = apply(fork(recolor, identity, patcher), cols)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl2(I):
    # Object extraction: remove 5, rectangularize each detected object independently (inbox per object)
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    parts = apply(fork(recolor, color, inbox), objs)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl3(I):
    # Color filtering: keep colors with multiple objects; fill each kept color's overall bounding box
    objs = objects(I, T, F, T)
    cols = remove(FIVE, remove(ZERO, palette(I)))
    groups = apply(lbind(colorfilter, objs), cols)
    keepg = sfilter(groups, compose(rbind(greater, ONE), size))
    getcol = compose(color, merge)
    getbox = compose(inbox, compose(toindices, merge))
    parts = apply(fork(recolor, getcol, getbox), keepg)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl4(I):
    # Geometric: remove 5, then enforce vertical and horizontal symmetry by underpainting mirrored copies
    I2 = replace(I, FIVE, ZERO)
    O1 = underpaint(I2, asobject(vmirror(I2)))
    O2 = underpaint(O1, asobject(hmirror(O1)))
    return O2

def dsl5(I):
    # Relational: keep objects (excluding 5) that are adjacent to another object of the same color; rectangularize them
    objs = objects(I, T, F, T)
    not5 = compose(flip, matcher(color, FIVE))
    base = sfilter(objs, not5)
    samecolor = compose(lbind(colorfilter, objs), color)
    others = fork(remove, identity, samecolor)
    union_others = compose(merge, others)
    adj_same = fork(adjacent, identity, union_others)
    keep = sfilter(base, adj_same)
    parts = apply(fork(recolor, color, inbox), keep)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O
