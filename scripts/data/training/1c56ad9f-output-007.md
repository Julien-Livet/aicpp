def dsl1(I):
    u = merge(objects(I, T, F, T))
    ui = toindices(u)
    bd = backdrop(ui)
    inside = difference(bd, ui)
    col = color(u)
    stepw = hperiod(u)
    steph = vperiod(u)
    w = width(bd)
    h = height(bd)
    ks = interval(0, w, stepw)
    iset = interval(0, h, steph)
    ul = ulcorner(inside)
    top_starts = apply(lbind(add, ul), apply(tojvec, ks))
    left_starts = apply(lbind(add, ul), apply(toivec, iset))
    anchors = combine(top_starts, left_starts)
    diag1 = merge(apply(rbind(shoot, UNITY), anchors))
    diag2 = merge(apply(rbind(shoot, DOWN_LEFT), anchors))
    mask = intersection(combine(diag1, diag2), inside)
    O = underpaint(I, recolor(col, mask))
    return O
