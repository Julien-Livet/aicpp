def dsl1(I):
    # Global diagonal choice: choose the bbox diagonal (main vs counter) with more 5s -> 2, others -> 8
    S5 = ofcolor(I, FIVE)
    Dmain = connect(ulcorner(S5), lrcorner(S5))
    Dcount = connect(llcorner(S5), urcorner(S5))
    C1 = intersection(S5, Dmain)
    C2 = intersection(S5, Dcount)
    Best = argmax(insert(C2, initset(C1)), size)
    S2 = Best
    S8 = difference(S5, S2)
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O

def dsl2(I):
    # Object extraction via diagonal rays: start from bbox left/bottom borders, pick ray with max 5-overlap -> 2
    S5 = ofcolor(I, FIVE)
    Lcol = vfrontier(ulcorner(S5))
    Brow = hfrontier(lrcorner(S5))
    Starts = combine(intersection(S5, Lcol), intersection(S5, Brow))
    Rays = prapply(shoot, Starts, initset(UP_RIGHT))
    BestRay = argmax(Rays, compose(size, lbind(intersection, S5)))
    S2 = intersection(BestRay, S5)
    S8 = difference(S5, S2)
    O = fill(fill(I, EIGHT, S8), TWO, S2)
    return O
