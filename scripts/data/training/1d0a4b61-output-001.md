def dsl2(I):

def dsl4(I):
    # Half-to-half transfer: use left half as donor to fill right half zeros (object extraction over parts)
    L, R = vsplit(I, 2)
    idxL = asindices(L)
    nzL = difference(idxL, ofcolor(L, ZERO))
    objL = toobject(nzL, L)
    newR = underpaint(R, objL)
    O = hconcat(L, newR)
    return O

def dsl5(I):
    # Top-bottom mutual completion: exchange content between halves and reassemble (relational/structural)
    T, B = hsplit(I, 2)
    idxT = asindices(T)
    idxB = asindices(B)
    objT = toobject(difference(idxT, ofcolor(T, ZERO)), T)

    newT = underpaint(T, objB)
    newB = underpaint(B, objT)
    O = vconcat(newT, newB)
    return O
