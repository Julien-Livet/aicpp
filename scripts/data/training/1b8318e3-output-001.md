def dsl1(I):
    # Global (trim-and-reembed): trim zero borders and re-embed at top-left
    sub = trim(I)
    base = canvas(ZERO, shape(I))
    O = paint(base, asobject(sub))
    return O

def dsl2(I):
    # Object extraction (column-sharing): keep fives and objects that share any column with fives
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    others = difference(objs, fives)
    dest = mapply(toindices, fives)
    colmatch = rbind(vmatching, dest)
    sel = sfilter(others, colmatch)
    keepcells = combine(merge(fives), merge(sel))
    base = canvas(ZERO, shape(I))
    O = paint(base, keepcells)
    return O

def dsl3(I):
    # Color filtering (adjacent-to-5): keep fives and only those non-5 objects adjacent to any five-cluster
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    dest = mapply(toindices, fives)
    others = difference(objs, fives)
    isadj = rbind(adjacent, dest)
    near = sfilter(others, isadj)
    keep = combine(merge(fives), merge(near))
    base = canvas(ZERO, shape(I))
    O = paint(base, keep)
    return O

def dsl4(I):
    # Geometric (vertical alignment): move only column-matching non-5 objects vertically until adjacent to five-clusters
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    dest = mapply(toindices, fives)
    others = difference(objs, fives)
    colmatch = rbind(vmatching, dest)
    sel = sfilter(others, colmatch)
    rest = difference(others, sel)
    dirf = rbind(gravitate, dest)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    movedcells = mapply(moved, sel)
    restcells = merge(rest)
    fivecells = merge(fives)
    allcells = combine(fivecells, combine(movedcells, restcells))
    base = canvas(ZERO, shape(I))
    O = paint(base, allcells)
    return O

def dsl5(I):
    # Relational (anchor to five-box outline): gravitate every non-5 to the outline of the five bounding box
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    dest = mapply(toindices, fives)
    bb = backdrop(dest)
    outline = box(bb)
    others = difference(objs, fives)
    dirf = rbind(gravitate, outline)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    movedcells = mapply(moved, others)
    fivecells = merge(fives)
    allcells = combine(movedcells, fivecells)
    base = canvas(ZERO, shape(I))
    O = paint(base, allcells)
    return O
