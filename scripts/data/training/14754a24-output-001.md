def dsl1(I):
    fours = ofcolor(I, FOUR)
    fiveobjs = colorfilter(objects(I, T, F, T), FIVE)
    adjfiveobjs = sfilter(fiveobjs, lbind(adjacent, fours))
    target = merge(mapply(toindices, adjfiveobjs))
    O = fill(I, TWO, target)
    return O

def dsl2(I):
    fourobjs = colorfilter(objects(I, T, F, T), FOUR)
    centers = mapply(center, fourobjs)
    segs = prapply(connect, centers, centers)
    hsegs = sfilter(segs, hline)
    vsegs = sfilter(segs, vline)
    lines = merge(combine(hsegs, vsegs))
    target = intersection(lines, ofcolor(I, FIVE))
    O = fill(I, TWO, target)
    return O

def dsl3(I):
    fours = ofcolor(I, FOUR)
    n1 = mapply(neighbors, fours)
    n2 = mapply(neighbors, n1)
    fives = ofcolor(I, FIVE)
    direct = intersection(n1, fives)
    dist2 = intersection(n2, fives)
    target = difference(dist2, direct)
    O = fill(I, TWO, target)
    return O

def dsl4(I):
    fours = ofcolor(I, FOUR)

    rays = prapply(shoot, fours, dirs)
    rowmask = merge(rays)
    target = intersection(rowmask, ofcolor(I, FIVE))
