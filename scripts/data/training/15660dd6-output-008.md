def dsl1(I):
    # Global: convert 1->8, then compress both axes via rotation
    A = replace(I, ONE, EIGHT)
    B = compress(A)
    C = rot90(B)
    D = compress(C)
    O = rot270(D)
    return O

def dsl2(I):
    # Object extraction: paint colored bounding boxes (outlines) of all non-bg objects on 8 background
    OBJS = objects(I, T, F, T)
    BOXOBJ = apply(fork(recolor, color, compose(box, identity)), OBJS)
    ALL = merge(BOXOBJ)
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, ALL)
    return O

def dsl3(I):
    # Color filtering: remove 1 by turning it into 8, then crop to non-8 extent
    A = replace(I, ONE, EIGHT)
    ALL = asindices(A)
    BG = ofcolor(A, EIGHT)
    FG = difference(ALL, BG)
    O = subgrid(FG, A)
    return O

def dsl4(I):
    # Geometric: keep cells invariant under 180-rotation, else 8
    A = rot180(I)
    O = cellwise(I, A, EIGHT)
    return O

def dsl5(I):
    # Relational: replace each object by its filled bounding box with its own color on 8 canvas
    OBJS = objects(I, T, F, T)
    FILLED = apply(fork(recolor, color, inbox), OBJS)
    ALL = merge(FILLED)
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, ALL)
    return O
