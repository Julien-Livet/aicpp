def dsl1(I):
    # Global periodic mask via color-pair periods (hperiod/vperiod) and frontier shifts
    OS = objects(I, T, F, T)
    o1 = first(OS)
    o2 = last(OS)
    c1 = color(o1)
    c2 = color(o2)
    p1 = first(toindices(o1))
    p2 = first(toindices(o2))
    P = combine(toindices(o1), toindices(o2))
    O2 = toobject(P, I)
    vr = vperiod(O2)
    hr = hperiod(O2)
    H = height(I)
    W = width(I)
    stepR = branch(equality(vr, ZERO), ONE, vr)
    stepC = branch(equality(hr, ZERO), ONE, hr)
    offsRpos = apply(toivec, interval(ZERO, H, stepR))
    offsR = combine(offsRpos, apply(invert, offsRpos))
    offsCpos = apply(tojvec, interval(ZERO, W, stepC))
    offsC = combine(offsCpos, apply(invert, offsCpos))
    baseR1 = hfrontier(p1)
    baseR2 = hfrontier(p2)
    HR1 = mapply(lbind(shift, baseR1), offsR)
    HR2 = mapply(lbind(shift, baseR2), offsR)
    HOBJ = combine(recolor(c1, HR1), recolor(c2, HR2))
    baseC1 = vfrontier(p1)
    baseC2 = vfrontier(p2)
    VC1 = mapply(lbind(shift, baseC1), offsC)
    VC2 = mapply(lbind(shift, baseC2), offsC)
    VOBJ = combine(recolor(c1, VC1), recolor(c2, VC2))
    hsel = either(equality(hr, ZERO), flip(either(equality(vr, ZERO), greater(vr, hr))))
    OOBJ = branch(hsel, HOBJ, VOBJ)
    O = paint(canvas(ZERO, shape(I)), OOBJ)
    return O
