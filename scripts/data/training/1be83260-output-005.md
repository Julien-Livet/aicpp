def dsl1(I):
    # Global compression/downscale: crop nonzero region, compress, downscale, then fill background with dominant color
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    C = compress(A)
    D = downscale(C, 2)
    O = underfill(D, mostcolor(D), asindices(D))
    return O

def dsl2(I):
    # Object extraction: crop nonzero region, reconstitute objects exactly on a zero canvas
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    OBJS = objects(A, T, F, T)
    TOT = totuple(OBJS)
    PATS = totuple(apply(toindices, TOT))
    REOBJ = totuple(apply(rbind(toobject, A), PATS))
    EXP = merge(REOBJ)
    BASE = canvas(ZERO, shape(A))
    O = paint(BASE, EXP)
    return O

def dsl3(I):
    # Color filtering: keep non-dominant objects on a dominant-color canvas
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    DOM = mostcolor(A)
    BASE = canvas(DOM, shape(A))
    OBJS = objects(A, T, F, T)
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    TOT = totuple(OTH)
    PATS = totuple(apply(toindices, TOT))
    REOBJ = totuple(apply(rbind(toobject, A), PATS))
    EXP = merge(REOBJ)
    O = paint(BASE, EXP)
    return O

def dsl4(I):
    # Geometric symmetry: crop and compress, mirror left half horizontally to form right half
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    C = compress(A)
    L = lefthalf(C)
    MR = vmirror(L)
    O = hconcat(L, MR)
    return O

def dsl5(I):
    # Relational expansion: expand each non-dominant object to its box outline, recolored by the object's color
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    A = subgrid(NZ, I)
    DOM = mostcolor(A)
    BASE = canvas(DOM, shape(A))
    OBJS = objects(A, T, F, T)
    PURE = colorfilter(OBJS, DOM)
    OTH = difference(OBJS, PURE)
    TOT = totuple(OTH)
    COLS = totuple(apply(color, TOT))
    PATS = totuple(apply(compose(box, toindices), TOT))
    COLORED = mpapply(recolor, COLS, PATS)
    EXP_MERGED = merge(COLORED)
    EMPTYOBJ = intersection(initset((ZERO, ORIGIN)), initset((ONE, ORIGIN)))
    HAS = positive(size(OTH))
    EXP = branch(HAS, EXP_MERGED, EMPTYOBJ)
    O = paint(BASE, EXP)
    return O
