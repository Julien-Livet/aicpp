def dsl1(I):
    # Global: crop to foreground, compress, then unify background with the modal foreground color
    fgobjs = objects(I, T, F, T)
    fg = merge(fgobjs)
    G = subgrid(fg, I)
    C = compress(G)
    O = replace(C, ZERO, mostcolor(fg))
    return O

def dsl2(I):
    # Object extraction: place non-modal foreground objects onto a canvas of the modal foreground color
    fgobjs = objects(I, T, F, T)
    fg = merge(fgobjs)
    G = subgrid(fg, I)
    base = canvas(mostcolor(fg), shape(G))
    modalobjs = colorfilter(fgobjs, mostcolor(fg))
    others = normalize(difference(merge(fgobjs), merge(modalobjs)))
    O = paint(base, others)
    return O

def dsl3(I):
    # Color filtering: fill background inside the crop with modal foreground color, then restore non-modal colors
    fgobjs = objects(I, T, F, T)
    fg = merge(fgobjs)
    G = subgrid(fg, I)
    bgfilled = replace(G, ZERO, mostcolor(fg))
    modalobjs = colorfilter(fgobjs, mostcolor(fg))
    others = normalize(difference(merge(fgobjs), merge(modalobjs)))
    O = underpaint(bgfilled, others)
    return O

def dsl4(I):
    # Geometric reasoning: split the cropped region into halves and fill each half's background by its own modal color
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    L = lefthalf(G)
    R = righthalf(G)
    fL = merge(objects(L, T, F, T))
    fR = merge(objects(R, T, F, T))
    L2 = replace(L, ZERO, mostcolor(fL))
    R2 = replace(R, ZERO, mostcolor(fR))
    O = hconcat(L2, R2)
    return O

def dsl5(I):
    # Relational/structural: use the delta (bbox minus foreground) as a mask to underfill with the modal foreground color
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    bgmask = normalize(delta(fg))
    O = underfill(G, mostcolor(fg), bgmask)
    return O
