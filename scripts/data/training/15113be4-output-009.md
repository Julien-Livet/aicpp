def dsl1(I):
    # Global: remove frontiers, select 1-objects sharing a row or column with least-color, then fill and restore
    val = leastcolor(I)
    fours = ofcolor(I, FOUR)
    J = cover(I, recolor(FOUR, fours))
    vp = ofcolor(J, val)
    ones = colorfilter(fgpartition(J), ONE)
    idx = toindices
    hitc = compose(lbind(vmatching, vp), idx)
    hitr = compose(lbind(hmatching, vp), idx)
    pred = fork(either, hitc, hitr)
    hits = sfilter(ones, pred)
    target = merge(apply(idx, hits))
    O = fill(I, val, target)
    return O

def dsl2(I):
    # Object extraction: recolor 1-objects adjacent to any least-color cell
    val = leastcolor(I)
    vp = ofcolor(I, val)
    ones = colorfilter(fgpartition(I), ONE)
    idx = toindices
    pred = compose(lbind(adjacent, vp), idx)
    hits = sfilter(ones, pred)
    target = merge(apply(idx, hits))
    O = fill(I, val, target)
    return O

def dsl3(I):
    # Color filtering: recolor ones on the outline (box) of the least-color region
    val = leastcolor(I)
    ring = box(ofcolor(I, val))
    target = intersection(ring, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl4(I):
    # Geometric: draw cross using row/column frontiers through least-color center; recolor intersecting ones
    val = leastcolor(I)
    vp = ofcolor(I, val)
    c = centerofmass(vp)
    hr = hfrontier(c)
    vr = vfrontier(c)
    s0 = initset(hr)
    s1 = insert(vr, s0)
    cross = merge(s1)
    target = intersection(cross, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl5(I):
    # Relational: recolor 1-objects sharing a row or column with the least-color outline (box)
    val = leastcolor(I)
    rect = box(ofcolor(I, val))
    ones = colorfilter(fgpartition(I), ONE)
    idx = toindices
    rowhit = compose(lbind(hmatching, rect), idx)
    colhit = compose(lbind(vmatching, rect), idx)
    pred = fork(either, rowhit, colhit)
    hits = sfilter(ones, pred)
    target = merge(apply(idx, hits))
    O = fill(I, val, target)
    return O
