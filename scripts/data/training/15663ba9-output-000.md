def dsl5(I):
    c = leastcolor(I)
    S = ofcolor(I, c)
    lh = intersection(S, shift(S, RIGHT))
    rh = intersection(S, shift(S, LEFT))
    h_int = intersection(lh, rh)
    uh = intersection(S, shift(S, DOWN))
    dh = intersection(S, shift(S, UP))
    v_int = intersection(uh, dh)
    interior = combine(h_int, v_int)
    endpoints = difference(S, interior)
    n1 = shift(endpoints, LEFT)
    n2 = shift(endpoints, RIGHT)
    n3 = shift(endpoints, UP)
    n4 = shift(endpoints, DOWN)
    neigh = combine(combine(n1, n2), combine(n3, n4))
    inner = intersection(S, neigh)
    G1 = fill(I, FOUR, endpoints)
    O = fill(G1, TWO, inner)
