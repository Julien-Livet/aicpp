def dsl1(I):
    # Global via compression + coordinate comparison
    Ic = compress(I)
    P5c = ofcolor(Ic, FIVE)
    Bc = inbox(P5c)
    Gc = replace(Ic, SIX, NINE)
    P9nc = difference(ofcolor(Gc, NINE), ofcolor(Ic, NINE))
    Di = sign(subtract(uppermost(P9nc), uppermost(Bc)))
    Dj = sign(subtract(leftmost(P9nc), leftmost(Bc)))
    Dc = astuple(Di, Dj)
    P5 = ofcolor(I, FIVE)
    B = inbox(P5)
    J = intersection(ofcolor(I, NINE), B)
    G = replace(I, SIX, NINE)
    O = paint(fill(G, FIVE, J), recolor(NINE, shift(J, Dc)))
    return O
