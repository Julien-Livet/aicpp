def dsl1(I):
    # Global morphological: endpoints are degree-1 cells; neighbors and bends get 2
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    H = intersection(L, R)
    V = intersection(U, D)
    S1 = combine(combine(L, R), combine(U, D))
    LU = intersection(L, U)
    LD = intersection(L, D)
    RU = intersection(R, U)
    RD = intersection(R, D)
    B2 = combine(combine(LU, LD), combine(RU, RD))
    U2 = combine(H, V)
    U2p = combine(U2, B2)
    E = difference(S1, U2p)
    N1 = combine(shift(E, LEFT), shift(E, RIGHT))
    N2 = combine(shift(E, UP), shift(E, DOWN))
    N = intersection(combine(N1, N2), S)
    bends = difference(intersection(combine(L, R), combine(U, D)), combine(H, V))
    Nall = combine(N, bends)
    G = fill(I, FOUR, E)
    O = fill(G, TWO, Nall)
    return O

def dsl3(I):
    # Color filtering: strip leastcolor, write endpoints and immediate neighbors, then restore leastcolor
    c = leastcolor(I)
    S = ofcolor(I, c)
    L = intersection(S, shift(S, RIGHT))
    R = intersection(S, shift(S, LEFT))
    U = intersection(S, shift(S, DOWN))
    D = intersection(S, shift(S, UP))
    H = intersection(L, R)
    V = intersection(U, D)
    E = combine(difference(combine(L, R), H), difference(combine(U, D), V))
    N1 = combine(shift(E, LEFT), shift(E, RIGHT))
    N2 = combine(shift(E, UP), shift(E, DOWN))
    N = intersection(combine(N1, N2), S)
    G0 = replace(I, c, ZERO)
    G1 = fill(G0, FOUR, E)
    G2 = fill(G1, TWO, N)
    O = underfill(G2, c, S)
    return O
