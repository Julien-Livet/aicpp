def dsl1(I):
    P = difference(asindices(I), ofcolor(I, ZERO))
    G = subgrid(P, I)
    P2 = difference(asindices(G), ofcolor(G, ZERO))
    O = toobject(P2, G)
    V = vmirror(O)
    H = hmirror(O)
    VH = vmirror(H)
    U1 = combine(O, V)
    U2 = combine(U1, H)
    U3 = combine(U2, VH)
    start = ulcorner(P)
    OUg = shift(U3, start)
    base = canvas(ZERO, shape(I))
    Ogrid = paint(base, OUg)
    return Ogrid
