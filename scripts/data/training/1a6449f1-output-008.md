def dsl1(I):
    # Remove most common color globally and trim to content
    mc = mostcolor(I)
    O = trim(replace(I, mc, ZERO))
    return O

def dsl2(I):
    # Interior of bbox of the largest border-touching object of the dominant nonzero color
    mc = argmax(remove(ZERO, palette(I)), lbind(colorcount, I))
    objs = colorfilter(objects(I, T, F, T), mc)
    bord = sfilter(objs, lbind(bordering, I))
    big = argmax(bord, size)
    O = subgrid(delta(box(backdrop(big))), I)
    return O

def dsl3(I):
    # Color filtering: remove the dominant nonzero color and trim
    mc = argmax(remove(ZERO, palette(I)), lbind(colorcount, I))
    u = toobject(ofcolor(I, mc), I)
    core = cover(I, u)
    O = trim(core)
    return O

def dsl4(I):
    # Geometric: union of border-touching dominant-color objects, crop away 1-pixel frame, then trim
    mc = argmax(remove(ZERO, palette(I)), lbind(colorcount, I))
    os = colorfilter(objects(I, T, F, T), mc)
    bord = sfilter(os, lbind(bordering, I))
    frame = merge(bord)
    bbgrid = subgrid(backdrop(frame), I)
    core = crop(bbgrid, UNITY, subtract(shape(bbgrid), TWO_BY_TWO))
    O = trim(core)
    return O

def dsl5(I):
    # Relational: take the inbox of the merged-per-color object for the dominant nonzero color
    mc = argmax(remove(ZERO, palette(I)), lbind(colorcount, I))
    g = first(colorfilter(fgpartition(I), mc))
    inn = inbox(backdrop(g))
    O = subgrid(inn, I)
    return O
