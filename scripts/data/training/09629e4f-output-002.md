def dsl1(I):
    G = replace(I, EIGHT, ZERO)
    C = compress(G)
    L = downscale(C, THREE)
    U = upscale(L, THREE)
    L3 = crop(U, ORIGIN, astuple(NINE, THREE))
    M3 = crop(U, astuple(ZERO, THREE), astuple(NINE, THREE))
    R3 = crop(U, astuple(ZERO, SIX), astuple(NINE, THREE))
    VBAR = canvas(FIVE, astuple(NINE, ONE))
    H1 = hconcat(hconcat(L3, VBAR), hconcat(M3, VBAR))
    H2 = hconcat(H1, R3)
    W = width(H2)
    T3 = crop(H2, ORIGIN, astuple(THREE, W))
    MM = crop(H2, astuple(THREE, ZERO), astuple(THREE, W))
    B3 = crop(H2, astuple(SIX, ZERO), astuple(THREE, W))
    HBAR = canvas(FIVE, astuple(ONE, W))
    O1 = vconcat(vconcat(T3, HBAR), vconcat(MM, HBAR))
    O = vconcat(O1, B3)
    return O

def dsl2(I):
    Z = canvas(ZERO, shape(I))
    P = ofcolor(I, FIVE)
    O = fill(Z, FIVE, P)
    return O

def dsl3(I):
    V = vmirror(I)
    H = hmirror(I)
    O = cellwise(V, H, ZERO)
    return O

def dsl4(I):
    R = rot180(I)
    O = cellwise(I, R, ZERO)
    return O

def dsl5(I):
    R1 = rot90(I)
    R2 = rot270(I)
    O = cellwise(R1, R2, ZERO)
    return O
