def dsl1(I):
    # Global: decide color from bottom half geometry (corner-touch), then recolor 8 and remove 1
    B = bottomhalf(I)
    obs = colorfilter(objects(B, T, F, T), ONE)
    n = size(obs)
    o = first(obs)
    hit = intersection(toindices(o), corners(o))
    anyhit = positive(size(hit))
    c1 = branch(anyhit, SEVEN, TWO)
    col = branch(greater(n, ONE), THREE, c1)
    S = switch(I, EIGHT, col)
    O = replace(S, ONE, ZERO)
    return O
