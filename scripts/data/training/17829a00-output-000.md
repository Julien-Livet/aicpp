def dsl1(I):
    # Relational/structural: gravitate each color’s non-border cells toward its border band, then repaint
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)

    cT = color(TOP)
    cB = color(BOT)

    pAllT = ofcolor(I, cT)
    pBorT = toindices(TOP)
    pNonT = difference(pAllT, pBorT)
    dirT = gravitate(pNonT, TOP)
    movedT = shift(pNonT, dirT)

    pAllB = ofcolor(I, cB)
    pBorB = toindices(BOT)
    pNonB = difference(pAllB, pBorB)
    dirB = gravitate(pNonB, BOT)
    movedB = shift(pNonB, dirB)

    G1 = cover(I, toobject(pNonT, I))
    G2 = cover(G1, toobject(pNonB, I))
    G3 = fill(G2, cT, movedT)
    O  = fill(G3, cB, movedB)
    return O

def dsl2(I):
    # Global/geometric: overlay vertical and horizontal mirrors into background
    A = underpaint(I, asobject(vmirror(I)))
    O = underpaint(A, asobject(hmirror(I)))
    return O

def dsl3(I):
    # Color filtering: keep only colors that appear on borders; others become background
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    BG = mostcolor(I)
    C = canvas(BG, shape(I))
    O = fill(fill(C, cT, ofcolor(I, cT)), cB, ofcolor(I, cB))
    return O

def dsl4(I):
    # Object extraction: duplicate border bands one step inward
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    O1 = fill(I, cT, shift(toindices(TOP), DOWN))
    O = fill(O1, cB, shift(toindices(BOT), UP))
    return O

def dsl5(I):
    # Geometric (per-object abstraction): fill each object’s bounding rectangle (inbox) with its color
    OB = objects(I, T, F, T)
    RECTFUN = fork(recolor, color, inbox)
    RECTS = apply(RECTFUN, OB)
    O = paint(canvas(mostcolor(I), shape(I)), merge(RECTS))
    return O
