def dsl1(I):
    T0 = trim(I)
    P = dmirror(T0)
    hI = height(I)
    wI = width(I)
    hP = height(P)
    wP = width(P)
    rightpad = canvas(ZERO, astuple(hP, subtract(wI, wP)))
    toprow = hconcat(P, rightpad)
    bottompad = canvas(ZERO, astuple(subtract(hI, hP), wI))
    O = vconcat(toprow, bottompad)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    g = fork(recolor, color, compose(box, normalize))
    parts = apply(g, objs)
    packed = merge(parts)
    base = canvas(ZERO, shape(I))
    O = paint(base, packed)
    return O

def dsl3(I):
    mc = mostcolor(I)
    lc = leastcolor(I)
    O = switch(I, mc, lc)
    return O

def dsl5(I):
    nz = difference(asindices(I), ofcolor(I, ZERO))
    P = subgrid(nz, I)
