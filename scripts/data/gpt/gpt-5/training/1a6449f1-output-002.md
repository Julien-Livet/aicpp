def dsl1(I):
    # Crop to the interior of the largest foreground object
    objs = objects(I, T, F, T)
    big = argmax(objs, size)
    O = subgrid(delta(big), I)
    return O

def dsl2(I):
    # Interior of the largest object of the most common nonzero color
    nz = remove(ZERO, palette(I))
    comp = compose(size, lbind(ofcolor, I))
    mc = valmax(nz, comp)
    objs = colorfilter(objects(I, T, F, T), mc)
    big = argmax(objs, size)
    ul = ulcorner(big)
    lr = lrcorner(big)
    dims = add(subtract(lr, ul), UNITY)
    innerdims = subtract(dims, multiply(UNITY, TWO))
    start = add(ul, UNITY)
    O = crop(I, start, innerdims)
    return O

def dsl3(I):
    # Choose object with largest bounding box; take its interior
    objs = objects(I, T, F, T)
    big = argmax(objs, compose(size, backdrop))
    O = subgrid(delta(big), I)
    return O

def dsl4(I):
    # Use the bounding rectangle outline of the largest object; take inside it
    objs = objects(I, T, F, T)
    big = argmax(objs, size)
    ring = box(big)
    O = subgrid(delta(ring), I)
    return O

def dsl5(I):
    # Interior via (backdrop minus outline) of the largest object
    objs = objects(I, T, F, T)
    big = argmax(objs, size)
    bb = backdrop(big)
    br = box(big)
    inside = difference(bb, br)
    O = subgrid(inside, I)
    return O
