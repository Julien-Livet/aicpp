def dsl1(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    COL = lbind(index, I)
    S1 = rbind(shoot, UP_RIGHT)
    S2 = rbind(shoot, DOWN_LEFT)
    DIAG = fork(combine, S1, S2)
    OBJF = fork(recolor, COL, DIAG)
    BIG = merge(apply(OBJF, NZ))
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O

def dsl2(I):
    COLORS = remove(ZERO, palette(I))
    INDC = lbind(ofcolor, I)
    SUP = rbind(shoot, UP_RIGHT)
    SDL = rbind(shoot, DOWN_LEFT)
    LUP = compose(merge, compose(lbind(apply, SUP), INDC))
    LDL = compose(merge, compose(lbind(apply, SDL), INDC))
    DIAGC = fork(combine, LUP, LDL)
    OBJF = fork(recolor, identity, DIAGC)
    BIG = merge(apply(OBJF, COLORS))
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O

def dsl3(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    COL = lbind(index, I)
    SUP = rbind(shoot, UP_RIGHT)
    OBJF = fork(recolor, COL, SUP)
    HALF = merge(apply(OBJF, NZ))
    S0 = initset(HALF)
    S1 = insert(hmirror(HALF), S0)
    S2 = insert(vmirror(HALF), S1)
    S3 = insert(vmirror(hmirror(HALF)), S2)
    BIG = merge(S3)
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O

def dsl4(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    BASE = toobject(NZ, I)
    H = height(I)
    W = width(I)
    RO = interval(0, H, ONE)
    CO = interval(0, W, ONE)
    RV = apply(toivec, RO)
    CV = apply(tojvec, CO)
    OFFS = prapply(subtract, RV, CV)
    SH = apply(lbind(shift, BASE), OFFS)
    BIG = merge(SH)
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O
