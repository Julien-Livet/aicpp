def dsl1(I):
    # Global stripes: use doubled inter-seed gap to avoid overlaps; replicate full row/col rays
    OS = objects(I, T, F, T)
    o1 = first(OS); o2 = last(OS)
    c1 = color(o1); c2 = color(o2)
    l1 = first(toindices(o1)); l2 = first(toindices(o2))
    P = combine(toindices(o1), toindices(o2))
    dv = subtract(lowermost(P), uppermost(P))
    dh = subtract(rightmost(P), leftmost(P))
    H = height(I); W = width(I)
    baseR = branch(equality(dv, ZERO), ONE, dv)
    baseC = branch(equality(dh, ZERO), ONE, dh)
    stepR = double(baseR)
    stepC = double(baseC)
    offsR = apply(toivec, interval(ZERO, H, stepR))
    offsC = apply(tojvec, interval(ZERO, W, stepC))
    RL1 = recolor(c1, combine(shoot(l1, LEFT), shoot(l1, RIGHT)))
    RL2 = recolor(c2, combine(shoot(l2, LEFT), shoot(l2, RIGHT)))
    CL1 = recolor(c1, combine(shoot(l1, UP), shoot(l1, DOWN)))
    CL2 = recolor(c2, combine(shoot(l2, UP), shoot(l2, DOWN)))
    HOBJ = combine(merge(prapply(shift, initset(RL1), offsR)),
                   merge(prapply(shift, initset(RL2), offsR)))
    VOBJ = combine(merge(prapply(shift, initset(CL1), offsC)),
                   merge(prapply(shift, initset(CL2), offsC)))
    hsel = either(equality(dh, ZERO), flip(greater(dv, dh)))
    O = paint(canvas(ZERO, shape(I)), branch(hsel, HOBJ, VOBJ))
    return O

def dsl5(I):
    # Relational interleaving: even/odd phase offsets using base gap; avoid overlaps by staggering
    OS = objects(I, T, F, T)
    o1 = first(OS); o2 = last(OS)
    c1 = color(o1); c2 = color(o2)
    l1 = first(toindices(o1)); l2 = first(toindices(o2))
    P = combine(toindices(o1), toindices(o2))
    dv = subtract(lowermost(P), uppermost(P))
    dh = subtract(rightmost(P), leftmost(P))
    H = height(I); W = width(I)
    baseR = branch(equality(dv, ZERO), ONE, dv)
    baseC = branch(equality(dh, ZERO), ONE, dh)
    evenR = apply(toivec, interval(ZERO, H, double(baseR)))
    oddR  = mapply(lbind(add, toivec(baseR)), evenR)
    evenC = apply(tojvec, interval(ZERO, W, double(baseC)))
    oddC  = mapply(lbind(add, tojvec(baseC)), evenC)
    RL1 = recolor(c1, combine(shoot(l1, LEFT), shoot(l1, RIGHT)))
    RL2 = recolor(c2, combine(shoot(l2, LEFT), shoot(l2, RIGHT)))
    CL1 = recolor(c1, combine(shoot(l1, UP), shoot(l1, DOWN)))
    CL2 = recolor(c2, combine(shoot(l2, UP), shoot(l2, DOWN)))
    HOBJ = combine(merge(prapply(shift, initset(RL1), evenR)),
                   merge(prapply(shift, initset(RL2), oddR)))
    VOBJ = combine(merge(prapply(shift, initset(CL1), evenC)),
                   merge(prapply(shift, initset(CL2), oddC)))
    hsel = either(equality(dh, ZERO), flip(greater(dv, dh)))
    O = paint(canvas(ZERO, shape(I)), branch(hsel, HOBJ, VOBJ))
