def dsl1(I):
    B = bottomhalf(I)
    C = compress(B)
    P = rot90(C)
    base = canvas(ZERO, shape(I))
    O = paint(base, asobject(P))
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    up = compose(invert, uppermost)
    vvec = compose(toivec, up)
    sh = fork(shift, identity, vvec)
    parts = apply(sh, objs)
    packed = merge(parts)
    base = canvas(ZERO, shape(I))
    O = paint(base, packed)
    return O

def dsl5(I):

    leftaxis = shoot(ORIGIN, DOWN)

    gleft = rbind(gravitate, leftaxis)
    vvec = compose(gtop, identity)
    hvec = compose(gleft, identity)
    off = fork(add, vvec, hvec)
    sh = fork(shift, identity, off)

    packed = merge(parts)
