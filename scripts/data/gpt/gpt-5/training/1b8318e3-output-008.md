def dsl1(I):
    # O = keep only cells in any row or column that contains a 5
    fidxs = ofcolor(I, FIVE)
    rows = merge(apply(hfrontier, fidxs))
    cols = merge(apply(vfrontier, fidxs))
    mask = combine(rows, cols)
    O = paint(canvas(ZERO, shape(I)), toobject(mask, I))
    return O

def dsl2(I):
    # O = keep 5-objects and any objects sharing rows or columns with the union of 5-rows and 5-columns
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    others = difference(objs, fives)
    fidxs = ofcolor(I, FIVE)
    rows = merge(apply(hfrontier, fidxs))
    cols = merge(apply(vfrontier, fidxs))
    band = combine(rows, cols)
    ph = lbind(hmatching, band)
    pv = lbind(vmatching, band)
    selh = sfilter(others, ph)
    selv = sfilter(others, pv)
    kept = merge(combine(selh, selv))
    O = paint(canvas(ZERO, shape(I)), combine(fiveobj, kept))
    return O

def dsl3(I):
    # O = keep all colors that occur in objects adjacent to per-5 bounding-box outlines
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    crnf = compose(box, toindices)
    perboxes = merge(apply(crnf, fives))
    others = difference(objs, fives)
    adjf = lbind(adjacent, perboxes)
    adjobjs = sfilter(others, adjf)
    keepcolors = palette(merge(adjobjs))
    isallowed = compose(rbind(contained, keepcolors), color)
    keepobjs = sfilter(objs, isallowed)
    O = paint(canvas(ZERO, shape(I)), merge(keepobjs))
    return O

def dsl4(I):
    # O = move each non-5 toward the union of rows and columns containing any 5
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    others = difference(objs, fives)
    fidxs = ofcolor(I, FIVE)
    rows = merge(apply(hfrontier, fidxs))
    cols = merge(apply(vfrontier, fidxs))
    band = combine(rows, cols)
    dirf = rbind(gravitate, band)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    O = paint(canvas(ZERO, shape(I)), combine(fiveobj, mapply(moved, others)))
    return O

def dsl5(I):
    # O = move each non-5 toward the union of per-5 inboxes (their bounding box interiors)
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    others = difference(objs, fives)
    perin = merge(apply(compose(inbox, toindices), fives))
    dirf = rbind(gravitate, perin)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    O = paint(canvas(ZERO, shape(I)), combine(fiveobj, mapply(moved, others)))
    return O
