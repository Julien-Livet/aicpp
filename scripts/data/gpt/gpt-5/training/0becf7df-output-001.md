    BOT = last(hsplit(TL, TWO))
    c00 = mostcolor(first(vsplit(TOP, TWO)))
    c01 = mostcolor(last(vsplit(TOP, TWO)))

    c11 = mostcolor(last(vsplit(BOT, TWO)))

    r1 = hfrontier(DOWN)
    c0 = vfrontier(ORIGIN)
    c1 = vfrontier(RIGHT)
    rows01 = combine(r0, r1)
    cols01 = combine(c0, c1)

    M01 = intersection(intersection(M, r0), c1)
    M10 = intersection(intersection(M, r1), c0)
