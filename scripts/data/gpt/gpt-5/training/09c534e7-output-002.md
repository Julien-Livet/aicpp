def dsl1(I):
    H = hmirror(I)
    O1 = underpaint(I, asobject(H))
    V = vmirror(I)
    O = underpaint(O1, asobject(V))
    return O

def dsl2(I):
    FG = fgpartition(I)
    Seeds = difference(FG, colorfilter(FG, ONE))
    ones = ofcolor(I, ONE)
    ringf = compose(delta, toindices)
    maskf = compose(lbind(intersection, ones), ringf)
    recolorf = fork(recolor, color, maskf)
    Fills = mapply(recolorf, Seeds)
    O = paint(I, Fills)
    return O

def dsl3(I):
    lc = leastcolor(I)
    O = replace(I, ONE, lc)
    return O

def dsl4(I):
    L = lefthalf(I)
    R = hmirror(L)
    O = hconcat(L, R)
    return O

def dsl5(I):
    FG = fgpartition(I)
    Seeds = difference(FG, colorfilter(FG, ONE))
    ones = ofcolor(I, ONE)
    rowf = compose(hfrontier, center)
    colf = compose(vfrontier, center)
    linesf = fork(combine, rowf, colf)
    maskf = compose(lbind(intersection, ones), linesf)
    recolorf = fork(recolor, color, maskf)
    Paints = mapply(recolorf, Seeds)
    O = paint(I, Paints)
    return O
