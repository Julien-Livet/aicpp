def dsl1(I):
    P = remove(ZERO, palette(I))
    mc = argmax(P, lbind(colorcount, I))
    crossobj = toobject(ofcolor(I, mc), I)
    O = compress(cover(I, crossobj))
    return O

def dsl2(I):
    P = remove(ZERO, palette(I))
    mc = argmax(P, lbind(colorcount, I))
    cross = ofcolor(I, mc)
    crossobj = toobject(cross, I)
    c = centerofmass(cross)
    parts = objects(I, T, F, T)
    others = difference(parts, colorfilter(parts, mc))
    tocent = compose(rbind(subtract, c), centerofmass)
    quad = compose(sign, tocent)
    TL = merge(sfilter(others, matcher(quad, NEG_UNITY)))
    TR = merge(sfilter(others, matcher(quad, UP_RIGHT)))
    BL = merge(sfilter(others, matcher(quad, DOWN_LEFT)))
    BR = merge(sfilter(others, matcher(quad, UNITY)))
    fb = toobject(initset(ulcorner(crossobj)), I)
    TLp = branch(greater(size(TL), ZERO), TL, fb)
    TRp = branch(greater(size(TR), ZERO), TR, fb)
    BLp = branch(greater(size(BL), ZERO), BL, fb)
    BRp = branch(greater(size(BR), ZERO), BR, fb)
    TLg = compress(subgrid(TLp, I))
    TRg = compress(subgrid(TRp, I))
    BLg = compress(subgrid(BLp, I))
    BRg = compress(subgrid(BRp, I))
    O = compress(vconcat(hconcat(TLg, TRg), hconcat(BLg, BRg)))
    return O
