def dsl1(I):
    Sh = shape(I)
    Z = canvas(ZERO, Sh)
    M1 = ofcolor(I, ONE)
    M8 = ofcolor(I, EIGHT)
    G1 = fill(Z, ONE, M1)
    G = fill(G1, EIGHT, M8)
    S0 = objects(I, T, F, T)
    L = argmax(S0, size)
    R = remove(L, S0)
    Seeds = sizefilter(R, 1)
    Targets = objects(G, T, F, T)
    B = lbind(lbind, manhattan)
    N = compose(lbind(argmin, Seeds), B)
    CF = fork(recolor, compose(color, N), toindices)
    U = merge(apply(CF, Targets))
    O = paint(I, U)
    return O
