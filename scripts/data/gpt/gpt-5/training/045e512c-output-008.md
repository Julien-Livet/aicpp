def dsl1(I):
    # Global: horizontal and vertical mirroring of cropped foreground, placed back at original location
    nz = difference(asindices(I), ofcolor(I, ZERO))
    sub = subgrid(nz, I)
    base = asobject(sub)
    hsym = combine(base, hmirror(base))
    sym = combine(hsym, vmirror(hsym))
    placed = shift(sym, ulcorner(nz))
    O = paint(canvas(ZERO, shape(I)), placed)
    return O

def dsl2(I):
    # Object extraction: draw each object's outline with its own color
    objs = objects(I, T, F, T)
    mk = fork(recolor, color, box)
    outlines = apply(mk, objs)
    allout = merge(outlines)
    O = paint(canvas(ZERO, shape(I)), allout)
    return O

def dsl3(I):
    # Color filtering: cast rays from the most frequent nonzero color in four directions
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    starts = ofcolor(I, c)
    up = mapply(lbind(shoot, UP), starts)
    down = mapply(lbind(shoot, DOWN), starts)
    left = mapply(lbind(shoot, LEFT), starts)
    right = mapply(lbind(shoot, RIGHT), starts)
    rays = combine(combine(up, down), combine(left, right))
    obj = recolor(c, rays)
    O = paint(canvas(ZERO, shape(I)), obj)
    return O

def dsl4(I):
    # Geometric: place a 4-rotation symmetry of the cropped foreground back at original location
    nz = difference(asindices(I), ofcolor(I, ZERO))
    sub = subgrid(nz, I)
    o0 = asobject(sub)
    o1 = asobject(rot90(sub))
    o2 = asobject(rot180(sub))
    o3 = asobject(rot270(sub))
    quad = combine(combine(o0, o1), combine(o2, o3))
    placed = shift(quad, ulcorner(nz))
    O = paint(canvas(ZERO, shape(I)), placed)
    return O

def dsl5(I):
    # Relational: connect centers of all objects with lines colored by the dominant nonzero color
    objs = objects(I, T, F, T)
    cents = apply(center, objs)
    segs = prapply(connect, cents, cents)
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    obj = recolor(c, segs)
    O = paint(canvas(ZERO, shape(I)), obj)
    return O
