def dsl1(I):
    top = tophalf(I)
    O = vconcat(top, top)

def dsl2(I):
    objs = objects(I, T, F, T)
    edgeobjs = sfilter(objs, rbind(bordering, I))

def dsl3(I):
    lc = leastcolor(I)

    O = fill(canvas(ZERO, shape(I)), lc, idx)

def dsl4(I):
