def dsl1(I):
    # Global crop to the tight bounding box of all nonzero cells
    nz = difference(asindices(I), ofcolor(I, ZERO))
    O = subgrid(nz, I)
    return O

def dsl2(I):
    # Remove the most common nonzero color and crop remaining content
    nz = remove(ZERO, palette(I))
    comp = compose(size, lbind(ofcolor, I))
    mc = valmax(nz, comp)
    mask = ofcolor(I, mc)
    obj = toobject(mask, I)
    G = cover(I, obj)
    keep = difference(asindices(G), ofcolor(G, ZERO))
    O = subgrid(keep, G)
    return O

def dsl3(I):
    # Keep only nonzero cells not of the dominant nonzero color; crop to them
    nz = remove(ZERO, palette(I))
    comp = compose(size, lbind(ofcolor, I))
    mc = valmax(nz, comp)
    nonmc = difference(asindices(I), ofcolor(I, mc))
    keep = difference(nonmc, ofcolor(I, ZERO))
    O = subgrid(keep, I)
    return O

def dsl4(I):
    # Use the object with the largest outline; take the inside of its bounding box
    objs = objects(I, T, F, T)
    big = argmax(objs, compose(size, box))
    inner = difference(backdrop(big), box(big))
    O = subgrid(inner, I)
    return O

def dsl5(I):
    # Remove all objects touching the border; crop remaining interior content
    objs = objects(I, T, F, T)
    cond = rbind(bordering, I)
    borderobjs = sfilter(objs, cond)
    G = cover(I, merge(borderobjs))
    keep = difference(asindices(G), ofcolor(G, ZERO))
    O = subgrid(keep, G)
    return O
