def dsl1(I):
    top = tophalf(I)
    cnt1 = colorcount(I, ONE)
    val = halve(cnt1)
    top2 = replace(top, EIGHT, val)
    botz = canvas(ZERO, shape(bottomhalf(I)))
    O = vconcat(top2, botz)
    return O

def dsl2(I):
    bh = bottomhalf(I)
    cntb = colorcount(bh, ONE)
    val = halve(cntb)
    inds8 = ofcolor(I, EIGHT)
    obj8 = toobject(inds8, I)
    base = canvas(ZERO, shape(I))
    recol = recolor(val, obj8)
    O = paint(base, recol)
    return O
