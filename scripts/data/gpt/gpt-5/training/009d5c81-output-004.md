def dsl1(I):
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    ones = toobject(ofcolor(I, ONE), I)
    no_ones = cover(I, ones)
    O = replace(no_ones, EIGHT, val)
    return O

def dsl2(I):
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    base = canvas(ZERO, shape(I))
    allobjs = objects(I, T, F, T)
    eobjs = colorfilter(allobjs, EIGHT)
    recol = merge(apply(lbind(recolor, val), eobjs))
    O = paint(base, recol)
    return O

def dsl4(I):
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    inds8 = ofcolor(I, EIGHT)
    obj8 = toobject(inds8, I)
    ul = ulcorner(obj8)
    norm8 = normalize(obj8)
    rec = recolor(val, norm8)
    base = canvas(ZERO, shape(I))
    O = move(base, rec, ul)
    return O

def dsl5(I):
    bh = bottomhalf(I)
    f = lbind(ofcolor, ONE)
    g = size
    h = halve
    val = chain(h, g, f)(bh)
    obj8 = toobject(ofcolor(I, EIGHT), I)
    rec8 = recolor(val, obj8)
    base = canvas(ZERO, shape(I))
    O = underpaint(base, rec8)
