def dsl1(I):
    # Global tiling: compress separators, split into 3 horizontal bands, compress each, and join with 8-column separators
    C = compress(I)
    P = hsplit(C, THREE)
    A = first(P)
    Z = last(P)
    M = first(remove(Z, remove(A, P)))
    A2 = compress(A)
    M2 = compress(M)
    Z2 = compress(Z)
    H = height(A2)
    SEP = canvas(EIGHT, astuple(H, ONE))
    O = hconcat(hconcat(hconcat(A2, SEP), M2), hconcat(SEP, Z2))
    return O

def dsl2(I):
    # Object extraction: crop outer 8-background
    O = trim(I)
    return O

def dsl3(I):
    # Color filtering: remove the least frequent color by turning it into 8
    c = leastcolor(I)
    O = replace(I, c, EIGHT)
    return O

def dsl4(I):
    # Geometric: reflect along the counter-diagonal
    O = cmirror(I)
    return O

def dsl5(I):
    # Relational/structural: fill each object's bounding box with its color on an 8 canvas
    OBJS = objects(I, T, F, T)
    BDS = apply(fork(recolor, color, backdrop), OBJS)
    ALL = merge(BDS)
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, ALL)
