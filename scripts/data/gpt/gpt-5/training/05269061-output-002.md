def dsl1(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    BASE = toobject(NZ, I)
    H = height(I)
    W = width(I)
    HP0 = hperiod(BASE)
    VP0 = vperiod(BASE)
    HP = branch(greater(HP0, ZERO), HP0, ONE)
    VP = branch(greater(VP0, ZERO), VP0, ONE)
    RO = interval(0, H, HP)
    CO = interval(0, W, VP)
    RV = apply(toivec, RO)
    CV = apply(tojvec, CO)
    OFFS = prapply(add, RV, CV)
    SH = apply(lbind(shift, BASE), OFFS)
    BIG = merge(SH)
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O

def dsl2(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    TILE = subgrid(NZ, I)
    H = height(I)
    W = width(I)
    TH = height(TILE)
    TW = width(TILE)
    RO = interval(0, H, TH)
    CO = interval(0, W, TW)
    RV = apply(toivec, RO)
    CV = apply(tojvec, CO)
    OFFS = prapply(add, RV, CV)
    R0 = asobject(TILE)
    R1 = asobject(rot90(TILE))
    R2 = asobject(rot180(TILE))
    R3 = asobject(rot270(TILE))
    S0 = apply(lbind(shift, R0), OFFS)
    S1 = apply(lbind(shift, R1), OFFS)
    S2 = apply(lbind(shift, R2), OFFS)
    S3 = apply(lbind(shift, R3), OFFS)
    BIG = merge(combine(combine(S0, S1), combine(S2, S3)))
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O

def dsl3(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    UL = ulcorner(NZ)
    c = index(I, UL)
    O = replace(I, ZERO, c)
    return O

def dsl4(I):
    O0 = canvas(ZERO, shape(I))
    A0 = asobject(I)
    A1 = asobject(rot90(I))
    A2 = asobject(rot180(I))
    A3 = asobject(rot270(I))
    M1 = asobject(hmirror(I))
    M2 = asobject(vmirror(I))
    SET = insert(A0, initset(A1))
    SET = insert(A2, SET)
    SET = insert(A3, SET)
    SET = insert(M1, SET)
    SET = insert(M2, SET)
    BIG = merge(SET)
    O = underpaint(O0, BIG)
    return O

def dsl5(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    BASE = toobject(NZ, I)
    H = height(I)
    W = width(I)
    RO = interval(0, H, ONE)
    RV = apply(toivec, RO)
    CV = apply(tojvec, RO)
    OFF1 = prapply(add, RV, CV)
    NRV = apply(invert, RV)
    OFF2 = prapply(add, NRV, CV)
    OFFS = combine(OFF1, OFF2)
    SH = apply(lbind(shift, BASE), OFFS)
    BIG = merge(SH)
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O
