def dsl1(I):
    # Geometric (gravitate all non-5 to the 1-ring around all fives)
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fidxs = merge(mapply(toindices, fives))
    nbrs = mapply(neighbors, fidxs)
    ring = difference(merge(nbrs), fidxs)
    others = difference(objs, fives)
    dirf = rbind(gravitate, ring)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    keep = combine(merge(fives), mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl2(I):
    # Relational (snap each non-5 to the ring, restricted to its row/column)
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fidxs = merge(mapply(toindices, fives))
    nbrs = mapply(neighbors, fidxs)
    ring = difference(merge(nbrs), fidxs)
    others = difference(objs, fives)
    rowline = compose(hfrontier, ulcorner)
    colline = compose(vfrontier, ulcorner)
    inter = lbind(intersection, ring)
    rowring = compose(inter, rowline)
    colring = compose(inter, colline)
    target = fork(combine, rowring, colring)
    dirf = fork(gravitate, toindices, target)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    keep = combine(merge(fives), mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl3(I):
    # Color filtering (keep fives and non-5 inside the global five-bbox)
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    bb = backdrop(merge(mapply(toindices, fives)))
    others = difference(objs, fives)
    interbb = lbind(intersection, bb)
    hits = compose(positive, compose(size, compose(interbb, toindices)))
    sel = sfilter(others, hits)
    keep = combine(merge(fives), merge(sel))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl4(I):
    # Global (rotate the five-bbox subgrid by 180 degrees)
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    bb = backdrop(merge(mapply(toindices, fives)))
    sg = subgrid(bb, I)
    rot = rot180(sg)
    obj = asobject(rot)
    base = canvas(ZERO, shape(I))
    O = move(base, obj, ulcorner(bb))
    return O

def dsl5(I):
    # Structural (gravitate to union of corners of each five-cluster)
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
