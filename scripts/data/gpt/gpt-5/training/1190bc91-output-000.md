def dsl2(I):
    OBJS = fgpartition(I)
    BIG = argmax(OBJS, size)
    O = subgrid(BIG, I)

def dsl3(I):
    FG = fgpartition(I)
    ALL = merge(FG)
    C = mostcolor(ALL)
    O = replace(I, ZERO, C)

def dsl4(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    MIR = vmirror(OBJ)
    O = underpaint(I, MIR)
    return O

def dsl5(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    UL = ulcorner(OBJ)
    LR = lrcorner(OBJ)
    UR = urcorner(OBJ)
    LL = llcorner(OBJ)
    L1 = connect(UL, LR)
    L2 = connect(UR, LL)
    L = combine(L1, L2)
    COL = mostcolor(OBJ)
    DIAG = recolor(COL, L)
    O = underpaint(I, DIAG)
    return O
