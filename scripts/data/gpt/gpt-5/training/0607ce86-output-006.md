def dsl1(I):
    top = tophalf(I)
    O = vconcat(top, top)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    f = fork(recolor, color, box)
    colbxs = mapply(f, objs)
    O = paint(canvas(ZERO, shape(I)), colbxs)
    return O

def dsl3(I):
    p = palette(I)
    nz = remove(ZERO, p)
    toobj = compose(rbind(toobject, I), lbind(ofcolor, I))
    score = compose(hperiod, toobj)
    keep = valmin(nz, score)
    mask = ofcolor(I, keep)
    O = fill(canvas(ZERO, shape(I)), keep, mask)
    return O

def dsl4(I):
    left = lefthalf(I)
    right = hmirror(left)
    O = hconcat(left, right)
    return O

def dsl5(I):
    frs = frontiers(I)
    allf = merge(frs)
    O = cover(I, allf)
