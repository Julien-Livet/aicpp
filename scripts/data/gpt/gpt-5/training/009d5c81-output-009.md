def dsl1(I):
    # Global transformation: directly remap colors on full grid
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    O = switch(switch(I, ONE, ZERO), EIGHT, val)
    return O

def dsl2(I):
    # Object extraction: recolor extracted 8-objects by horizontal periodicity
    parts = fgpartition(I)
    eobjs = colorfilter(parts, EIGHT)
    obj8 = merge(eobjs)
    val = hperiod(obj8)
    O = paint(canvas(ZERO, shape(I)), recolor(val, obj8))
    return O

def dsl3(I):
    # Color filtering: fill 8-mask with value from bottom-half 1s
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    mask8 = ofcolor(I, EIGHT)
    O = fill(canvas(ZERO, shape(I)), val, mask8)
    return O

def dsl4(I):
    # Geometric reasoning: recolor by vertical periodicity of the 8-shape
    obj8 = toobject(ofcolor(I, EIGHT), I)
    val = vperiod(obj8)
    O = fill(canvas(ZERO, shape(I)), val, ofcolor(I, EIGHT))
    return O

def dsl5(I):
    # Relational/structural: use relation between horizontal and vertical periodicities
    obj8 = toobject(ofcolor(I, EIGHT), I)
    hp = hperiod(obj8)
    vp = vperiod(obj8)
    vals = insert(vp, initset(hp))
    val = minimum(vals)
    O = fill(canvas(ZERO, shape(I)), val, ofcolor(I, EIGHT))
    return O
