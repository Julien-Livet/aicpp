def dsl1(I):
    # Global: cross through the center of mass of ones
    val = leastcolor(I)
    com = centerofmass(ofcolor(I, ONE))
    lines = combine(hfrontier(com), vfrontier(com))
    target = intersection(lines, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl2(I):
    # Object extraction: fill entire one-objects touching the least-color box outline
    val = leastcolor(I)
    rect = box(ofcolor(I, val))
    ones = colorfilter(fgpartition(I), ONE)
    idx = toindices
    intr = compose(lbind(intersection, rect), idx)
    pred = chain(positive, size, intr)
    hits = sfilter(ones, pred)
    target = merge(apply(idx, hits))
    O = fill(I, val, target)
    return O

def dsl3(I):
    # Color filtering: fill ones on the outbox of the least-color region
    val = leastcolor(I)
    border = outbox(ofcolor(I, val))
    target = intersection(border, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl4(I):
    # Geometric: rays from the center of the least-color bounding box
    val = leastcolor(I)
    rect = backdrop(ofcolor(I, val))
    c = center(rect)
    rays = combine(combine(shoot(c, UP), shoot(c, DOWN)), combine(shoot(c, LEFT), shoot(c, RIGHT)))
    target = intersection(rays, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl5(I):
    # Relational: connect centers of one-objects to the least-color center of mass
    val = leastcolor(I)
    vp = ofcolor(I, val)
    cv = centerofmass(vp)
    objs = colorfilter(fgpartition(I), ONE)
    comobj = compose(centerofmass, toindices)
    pathfunc = compose(lbind(connect, cv), comobj)
    paths = merge(apply(pathfunc, objs))
    target = intersection(paths, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O
