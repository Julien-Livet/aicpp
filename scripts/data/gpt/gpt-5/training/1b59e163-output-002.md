def dsl1(I):
    # Global symmetry: remove 5s, then overlay horizontal and vertical mirrors onto the grid
    J = replace(I, FIVE, FOUR)
    H = hmirror(J)
    V = vmirror(J)
    objH = asobject(H)
    objV = asobject(V)
    tmp = underpaint(J, objH)
    O = underpaint(tmp, objV)
    return O

def dsl2(I):
    # Object extraction and recentering: center the bounding subgrid of all non-4 (with 5s removed)
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    bg = ofcolor(J, FOUR)
    fg = difference(allidx, bg)
    S = subgrid(fg, J)
    allS = asindices(S)
    bgS = ofcolor(S, FOUR)
    fgS = difference(allS, bgS)
    objS = toobject(fgS, S)
    cS = center(fgS)
    cJ = center(allidx)
    off = subtract(cJ, cS)
    base = canvas(FOUR, shape(J))
    O = move(base, objS, off)
    return O

def dsl3(I):
    # Color filtering: keep only color 1, remove all other colors
    base = canvas(FOUR, shape(I))
    ones = toobject(ofcolor(I, ONE), I)
    O = paint(base, ones)
    return O

def dsl4(I):
    # Geometric reasoning on a single color: mirror 1s across the vertical axis and overlay
    J = replace(I, FIVE, FOUR)
    M = vmirror(J)
    obj1 = toobject(ofcolor(J, ONE), J)
    obj2 = toobject(ofcolor(M, ONE), M)
    base = canvas(FOUR, shape(J))
    tmp = underpaint(base, obj1)
    O = underpaint(tmp, obj2)
    return O

def dsl5(I):
    # Relational/structural: swap and recenter foreground of left and right halves (after removing 5s)
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    bg = ofcolor(J, FOUR)
    fg = difference(allidx, bg)
    idxL = asindices(lefthalf(J))
    idxR = asindices(righthalf(J))
    leftFG = intersection(fg, idxL)
    rightFG = intersection(fg, idxR)
    leftObj = toobject(leftFG, J)
    rightObj = toobject(rightFG, J)
    cL = center(idxL)
    cR = center(idxR)
    offLR = subtract(cR, cL)
    offRL = subtract(cL, cR)
    movedL = shift(leftObj, offLR)
    movedR = shift(rightObj, offRL)
    base = canvas(FOUR, shape(J))
    tmp = underpaint(base, movedL)
    O = underpaint(tmp, movedR)
    return O
