def dsl1(I):
    # Relational unification: color 5s on straight, pure paths connecting a 4 to either a 4 or a zero that touches 5s.
    p4 = ofcolor(I, FOUR)
    p5 = ofcolor(I, FIVE)
    z0 = ofcolor(I, ZERO)
    dirs = insert(LEFT, insert(RIGHT, insert(UP, initset(DOWN))))
    n5 = merge(apply(lbind(shift, p5), dirs))
    znear5 = intersection(z0, n5)
    allowed = merge(insert(p4, insert(p5, initset(znear5))))
    paths44 = prapply(connect, p4, p4)
    paths40 = prapply(connect, p4, znear5)
    paths = combine(paths44, paths40)
    pure = compose(lbind(equality, ZERO), compose(size, rbind(difference, allowed)))
    linear = fork(either, hline, vline)
    cond = fork(both, pure, linear)
    good = mfilter(paths, cond)
    mask = intersection(merge(good), p5)
    O = fill(I, TWO, mask)
    return O

def dsl4(I):
