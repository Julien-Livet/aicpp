def dsl1(I):
    # Global quadrant recoloring: TL->4, TR->2, BL->3, BR->1
    TH = tophalf(I)
    BH = bottomhalf(I)
    TL = lefthalf(TH)
    TR = righthalf(TH)
    BL = lefthalf(BH)
    BR = righthalf(BH)
    TLc = replace(TL, EIGHT, FOUR)
    TRc = replace(TR, EIGHT, TWO)
    BLc = replace(BL, EIGHT, THREE)
    BRc = replace(BR, EIGHT, ONE)
    top = hconcat(TLc, TRc)
    bot = hconcat(BLc, BRc)
    O = vconcat(top, bot)
    return O

def dsl2(I):
    # Object extraction by quadrant membership: TL->4, TR->2, BL->3, BR->1
    objs = colorfilter(objects(I, T, F, T), EIGHT)
    th8 = ofcolor(tophalf(I), EIGHT)
    lh8 = ofcolor(lefthalf(I), EIGHT)
    cond_top = compose(lbind(hmatching, th8), toindices)
    cond_left = compose(lbind(vmatching, lh8), toindices)
    tops = sfilter(objs, cond_top)
    lefts = sfilter(objs, cond_left)
    tl = intersection(tops, lefts)
    tr = difference(tops, tl)
    bl = difference(lefts, tl)
    bots = difference(objs, tops)
    br = difference(bots, bl)
    tl_r = merge(apply(lbind(recolor, FOUR), tl))
    tr_r = merge(apply(lbind(recolor, TWO), tr))
    bl_r = merge(apply(lbind(recolor, THREE), bl))
    br_r = merge(apply(lbind(recolor, ONE), br))
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(paint(paint(base, tl_r), tr_r), bl_r), br_r)
    return O

def dsl3(I):
    # Color filtering by vertical halves: top->4, bottom->1
    top = replace(tophalf(I), EIGHT, FOUR)
    bot = replace(bottomhalf(I), EIGHT, ONE)
    O = vconcat(top, bot)
    return O

def dsl4(I):
    # Geometric symmetry: 8s symmetric under horizontal mirror ->2, others ->1
    p = ofcolor(I, EIGHT)
    ph = ofcolor(hmirror(I), EIGHT)
    common = intersection(p, ph)
    unique = difference(p, common)
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(base, recolor(ONE, unique)), recolor(TWO, common))
    return O

def dsl5(I):
    # Relational: classify 8s by center-of-mass axes; on V-axis->2, on H-axis (excluding V)->3, rest->1
    p = ofcolor(I, EIGHT)
    c = centerofmass(p)
    vax = vfrontier(c)
    hax = hfrontier(c)
    onv = intersection(p, vax)
    rem = difference(p, onv)
    onh = intersection(rem, hax)
    rest = difference(rem, onh)
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(paint(base, recolor(ONE, rest)), recolor(THREE, onh)), recolor(TWO, onv))
    return O
