def dsl1(I):
    bg = ofcolor(I, ZERO)
    U = asindices(I)
    nz = difference(U, bg)
    leftEP = intersection(nz, shift(bg, RIGHT))
    rightEP = intersection(nz, shift(bg, LEFT))
    endpoints = merge(insert(leftEP, initset(rightEP)))
    bothUD = intersection(intersection(endpoints, shift(bg, UP)), shift(bg, DOWN))
    O = paint(I, recolor(FOUR, endpoints))
    O = paint(O, recolor(TWO, bothUD))
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    tinds = toindices
    shL = rbind(shift, LEFT)
    shR = rbind(shift, RIGHT)
    hasR_fn = fork(intersection, tinds, compose(shL, tinds))
    hasL_fn = fork(intersection, tinds, compose(shR, tinds))
    noR_fn = fork(difference, tinds, hasR_fn)
    noL_fn = fork(difference, tinds, hasL_fn)
    leftEP_fn = fork(intersection, noL_fn, hasR_fn)
    rightEP_fn = fork(intersection, noR_fn, hasL_fn)
    LE_sets = apply(leftEP_fn, objs)
    RE_sets = apply(rightEP_fn, objs)
    LE = merge(LE_sets)
    RE = merge(RE_sets)
    O = paint(I, recolor(FOUR, LE))
    O = paint(O, recolor(TWO, RE))
    return O

def dsl3(I):
    bg = ofcolor(I, ZERO)
    pal = palette(I)
    pal_nz = remove(ZERO, pal)
    getc = lbind(ofcolor, I)
    sets = apply(getc, pal_nz)
    SR = shift(bg, RIGHT)
    SL = shift(bg, LEFT)
    l_fn = lbind(intersection, SR)
    r_fn = lbind(intersection, SL)
    LE_sets = apply(l_fn, sets)
    RE_sets = apply(r_fn, sets)
    LE = merge(LE_sets)
    RE = merge(RE_sets)
    O = paint(I, recolor(FOUR, LE))
    O = paint(O, recolor(TWO, RE))
    return O

    shL = rbind(shift, LEFT)

def dsl5(I):

    hasL = intersection(nz, shift(nz, LEFT))
    hasR = intersection(nz, shift(nz, RIGHT))
