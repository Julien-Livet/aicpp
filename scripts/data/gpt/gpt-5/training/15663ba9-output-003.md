def dsl1(I):
    bg = ofcolor(I, ZERO)
    U = asindices(I)
    nz = difference(U, bg)
    lExp = intersection(nz, shift(bg, RIGHT))
    rExp = intersection(nz, shift(bg, LEFT))
    hasR = intersection(nz, shift(nz, LEFT))
    hasL = intersection(nz, shift(nz, RIGHT))
    leftEP = intersection(lExp, hasR)
    rightEP = intersection(rExp, hasL)
    vt = intersection(rightEP, shift(bg, DOWN))
    vb = intersection(rightEP, shift(bg, UP))
    tipsR = intersection(vt, vb)
    o4 = recolor(FOUR, merge(insert(leftEP, initset(rightEP))))
    o2 = recolor(TWO, tipsR)
    O = paint(I, o4)
    O = paint(O, o2)
    return O

    bg = ofcolor(I, ZERO)
    objs = objects(I, T, F, T)
    comp = lbind(difference, U)
    shR = rbind(shift, RIGHT)
    shL = rbind(shift, LEFT)
    le_fn = fork(intersection, toindices, compose(shR, compose(comp, toindices)))
    hr_fn = fork(intersection, toindices, compose(shL, toindices))
    leftEP_fn = fork(intersection, le_fn, hr_fn)
