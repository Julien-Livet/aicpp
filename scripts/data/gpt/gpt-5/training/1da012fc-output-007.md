    S = objects(I, T, F, T)
    L = argmax(S, size)
    R = remove(L, S)
    T1 = colorfilter(S, ONE)
    T8 = colorfilter(S, EIGHT)
    Targets = combine(T1, T8)
    NonT = difference(R, Targets)
    Seeds0 = sizefilter(NonT, 1)
    AdjF = rbind(adjacent, L)
    Seeds = sfilter(Seeds0, AdjF)
    M0 = lbind(lbind, manhattan)
