def dsl1(I):
    DIR = insert(DOWN_LEFT, initset(UP_RIGHT))
    COLORS = remove(ZERO, palette(I))
    SFN = lbind(ofcolor, I)
    R0 = lbind(prapply, shoot)
    R1 = rbind(R0, DIR)
    PFN = compose(merge, R1)
    STARTS = compose(PFN, SFN)
    COLORIZE = fork(recolor, identity, STARTS)
    PARTS = apply(COLORIZE, COLORS)
    OBJ = merge(PARTS)
    O = paint(canvas(ZERO, shape(I)), OBJ)
    return O

def dsl2(I):
    FG = fgpartition(I)
    EXPAND = fork(recolor, color, backdrop)
    PARTS = apply(EXPAND, FG)
    OBJ = merge(PARTS)
    O = paint(canvas(ZERO, shape(I)), OBJ)
    return O

def dsl3(I):
    FG = fgpartition(I)
    ALL = merge(FG)
    LC = leastcolor(ALL)
    O = replace(I, ZERO, LC)
    return O

def dsl4(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    H = hmirror(OBJ)
    V = vmirror(OBJ)
    S = combine(H, V)
    O = underpaint(I, S)
    return O

def dsl5(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    UL = ulcorner(OBJ)
    LR = lrcorner(OBJ)
    UR = urcorner(OBJ)
    LL = llcorner(OBJ)
    D1 = connect(UL, LR)
    D2 = connect(UR, LL)
    DL = combine(D1, D2)
    MC = mostcolor(OBJ)
    LINES = recolor(MC, DL)
    O = underpaint(I, LINES)
    return O
