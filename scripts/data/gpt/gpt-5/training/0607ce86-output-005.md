def dsl1(I):
    left = lefthalf(I)
    O = hconcat(left, left)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    big = argmax(objs, size)
    O = paint(canvas(ZERO, shape(I)), big)
    return O

def dsl3(I):
    p = palette(I)
    nz = remove(ZERO, p)
    mc = valmax(nz, lbind(colorcount, I))
    mask = ofcolor(I, mc)
    O = fill(canvas(ZERO, shape(I)), mc, mask)
    return O

def dsl4(I):
    O = paint(hmirror(I), asobject(I))
    return O
