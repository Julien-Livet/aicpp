def dsl1(I):
    # Global midlines over full grid: recolor 8s by quadrant with palette TL=1, TR=2, BL=3, BR=1
    P = ofcolor(I, EIGHT)
    hi = height(I); wi = width(I)
    mi = halve(hi); mj = halve(wi)
    TL = asindices(crop(I, ORIGIN, astuple(mi, mj)))
    TR = shift(asindices(crop(I, astuple(ZERO, mj), astuple(mi, subtract(wi, mj)))), astuple(ZERO, mj))
    BL = shift(asindices(crop(I, astuple(mi, ZERO), astuple(subtract(hi, mi), mj))), astuple(mi, ZERO))
    BR = shift(asindices(crop(I, astuple(mi, mj), astuple(subtract(hi, mi), subtract(wi, mj)))), astuple(mi, mj))
    base = replace(I, EIGHT, ZERO)
    O = fill(base, ONE, intersection(P, TL))
    O = fill(O, TWO, intersection(P, TR))
    O = fill(O, THREE, intersection(P, BL))
    O = fill(O, ONE, intersection(P, BR))
    return O

def dsl2(I):
    # Objectwise COM vs 8-box center: TL=1, TR=2, BL=3, BR(+axes/center)=1
    P = ofcolor(I, EIGHT)
    l = leftmost(P); r = rightmost(P)
    u = uppermost(P); d = lowermost(P)
    gc = astuple(halve(add(u, d)), halve(add(l, r)))
    objs = colorfilter(objects(I, T, F, T), EIGHT)
    vec = compose(rbind(subtract, gc), centerofmass)
    sgn = compose(sign, vec)
    is_tl = compose(lbind(equality, NEG_UNITY), sgn)
    is_tr = compose(lbind(equality, UP_RIGHT), sgn)
    is_bl = compose(lbind(equality, DOWN_LEFT), sgn)
    is_br = compose(lbind(equality, UNITY), sgn)
    is_up = compose(lbind(equality, UP), sgn)
    is_dn = compose(lbind(equality, DOWN), sgn)
    is_lf = compose(lbind(equality, LEFT), sgn)
    is_rt = compose(lbind(equality, RIGHT), sgn)
    is_ct = compose(lbind(equality, ORIGIN), sgn)
    tl = sfilter(objs, is_tl)
    tr = combine(sfilter(objs, is_tr), sfilter(objs, is_up))
    bl = sfilter(objs, is_bl)
    br = combine(sfilter(objs, is_br), combine(sfilter(objs, is_rt), combine(sfilter(objs, is_dn), sfilter(objs, is_ct))))
    tl_r = merge(apply(lbind(recolor, ONE), tl))
    tr_r = merge(apply(lbind(recolor, TWO), tr))
    bl_r = merge(apply(lbind(recolor, THREE), bl))
    br_r = merge(apply(lbind(recolor, ONE), br))
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(paint(paint(base, tl_r), tr_r), bl_r), br_r)
    return O

def dsl3(I):
    # Color filtering with 8-subgrid halves: default 1, then TR=2, BL=3 (BR stays 1)
    P = ofcolor(I, EIGHT)
    base = replace(I, EIGHT, ZERO)
    O = fill(base, ONE, P)
    S = subgrid(P, I)
    hi = height(S); wi = width(S)
    mi = halve(hi); mj = halve(wi)
    UL = ulcorner(P)
    TR = crop(S, astuple(ZERO, mj), astuple(mi, subtract(wi, mj)))
    BL = crop(S, astuple(mi, ZERO), astuple(subtract(hi, mi), mj))
    TRm = shift(asindices(TR), add(UL, astuple(ZERO, mj)))
    BLm = shift(asindices(BL), add(UL, astuple(mi, ZERO)))
    O = fill(O, TWO, intersection(P, TRm))
    O = fill(O, THREE, intersection(P, BLm))
    return O

def dsl4(I):
    # Geometric 2x2 canvas on 8-subgrid: TL=4, TR=2, BL=3, BR=1
    P = ofcolor(I, EIGHT)
    S = subgrid(P, I)
    hi = height(S); wi = width(S)
    mi = halve(hi); mj = halve(wi)
    TLg = canvas(FOUR, astuple(mi, mj))
    TRg = canvas(TWO, astuple(mi, subtract(wi, mj)))
    BLg = canvas(THREE, astuple(subtract(hi, mi), mj))
    BRg = canvas(ONE, astuple(subtract(hi, mi), subtract(wi, mj)))
    Sc = vconcat(hconcat(TLg, TRg), hconcat(BLg, BRg))
    UL = ulcorner(P)
    base = replace(I, EIGHT, ZERO)
    C4 = shift(ofcolor(Sc, FOUR), UL)
    C2 = shift(ofcolor(Sc, TWO), UL)
    C3 = shift(ofcolor(Sc, THREE), UL)
    C1 = shift(ofcolor(Sc, ONE), UL)
    O = fill(base, FOUR, intersection(P, C4))
    O = fill(O, TWO, intersection(P, C2))
    O = fill(O, THREE, intersection(P, C3))
    O = fill(O, ONE, intersection(P, C1))
    return O

def dsl5(I):
    # Relational: recolor each 8-object by overlap with 8-subgrid quadrants (TL=2, TR=1, BL=3, BR=1)
    P = ofcolor(I, EIGHT)
    S = subgrid(P, I)
    hi = height(S); wi = width(S)
    mi = halve(hi); mj = halve(wi)
    UL = ulcorner(P)
    TLm = shift(asindices(crop(S, ORIGIN, astuple(mi, mj))), UL)
    TRm = shift(asindices(crop(S, astuple(ZERO, mj), astuple(mi, subtract(wi, mj)))), add(UL, astuple(ZERO, mj)))
    BLm = shift(asindices(crop(S, astuple(mi, ZERO), astuple(subtract(hi, mi), mj))), add(UL, astuple(mi, ZERO)))
    BRm = shift(asindices(crop(S, astuple(mi, mj), astuple(subtract(hi, mi), subtract(wi, mj)))), add(UL, astuple(mi, mj)))
    objs = colorfilter(objects(I, T, F, T), EIGHT)
    has = lambda M: chain(positive, size, compose(lbind(intersection, M), toindices))
    tl = sfilter(objs, has(TLm))
    tr = sfilter(objs, has(TRm))
    bl = sfilter(objs, has(BLm))
    br = sfilter(objs, has(BRm))
    tl_r = merge(apply(lbind(recolor, TWO), tl))
    tr_r = merge(apply(lbind(recolor, ONE), tr))
    bl_r = merge(apply(lbind(recolor, THREE), bl))
    br_r = merge(apply(lbind(recolor, ONE), br))
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(paint(paint(base, tl_r), tr_r), bl_r), br_r)
    return O
