def dsl1(I):
    P5 = ofcolor(I, FIVE)
    B = inbox(P5)
    C = center(B)
    PC = connect(C, C)
    S6 = ofcolor(I, SIX)
    RL = insert(C, combine(shoot(C, LEFT), shoot(C, RIGHT)))
    CL = insert(C, combine(shoot(C, UP), shoot(C, DOWN)))
    DR = manhattan(RL, S6)
    DC = manhattan(CL, S6)
    Gv = greater(DR, DC)
    Gh = greater(DC, DR)
    TIE = both(flip(Gv), flip(Gh))
    Dcard = gravitate(PC, S6)
    Ddiag = sign(position(PC, S6))
    D = branch(TIE, Ddiag, Dcard)
    PN = shift(PC, D)
    G = replace(I, SIX, NINE)
    G = fill(G, FIVE, PC)
    O = fill(G, NINE, PN)
    return O
