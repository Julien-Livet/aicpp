def dsl1(I):
    nz = difference(asindices(I), ofcolor(I, ZERO))
    bg = ofcolor(I, ZERO)
    left_edge = intersection(nz, shift(bg, RIGHT))
    right_edge = intersection(nz, shift(bg, LEFT))
    top_edge = intersection(nz, shift(bg, DOWN))
    bottom_edge = intersection(nz, shift(bg, UP))
    oL = recolor(FOUR, left_edge)
    oR = recolor(FOUR, right_edge)
    oT = recolor(TWO, top_edge)
    oB = recolor(TWO, bottom_edge)
    O = paint(I, oL)
    O = paint(O, oR)
    O = paint(O, oT)
    O = paint(O, oB)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    oidx = apply(toindices, objs)
    uls = apply(ulcorner, objs)
    urs = apply(urcorner, objs)
    vlefts = apply(vfrontier, uls)
    vrights = apply(vfrontier, urs)
    left_parts = mapply(intersection, oidx, vlefts)
    right_parts = mapply(intersection, oidx, vrights)
    left_edges = merge(left_parts)
    right_edges = merge(right_parts)
    oL = recolor(FOUR, left_edges)
    oR = recolor(TWO, right_edges)
    O = paint(I, oL)
    O = paint(O, oR)
    return O

def dsl3(I):
    objs = objects(I, T, F, T)
    uls = apply(ulcorner, objs)
    lrs = apply(lrcorner, objs)
    urs = apply(urcorner, objs)
    lls = apply(llcorner, objs)
    d1 = mpapply(connect, uls, lrs)
    d2 = mpapply(connect, urs, lls)
    diag1 = merge(d1)
    diag2 = merge(d2)
    o1 = recolor(FOUR, diag1)
    o2 = recolor(TWO, diag2)
    O = paint(I, o1)
    O = paint(O, o2)
    return O

def dsl4(I):
    nz = difference(asindices(I), ofcolor(I, ZERO))
    bg = ofcolor(I, ZERO)
    hL = intersection(nz, shift(bg, RIGHT))
    hR = intersection(nz, shift(bg, LEFT))
    vT = intersection(nz, shift(bg, DOWN))
    vB = intersection(nz, shift(bg, UP))
    h_edges = merge(insert(hR, initset(hL)))
    v_edges = merge(insert(vB, initset(vT)))
    only_v = difference(v_edges, h_edges)
    oH = recolor(FOUR, h_edges)
    oV = recolor(TWO, only_v)
    O = paint(I, oH)
    O = paint(O, oV)
    return O

def dsl5(I):
    fs = frontiers(I)
    hfs = sfilter(fs, hline)
    vfs = sfilter(fs, vline)
    hi = merge(apply(toindices, hfs))
    vi = merge(apply(toindices, vfs))
    intsct = intersection(hi, vi)
    allf = merge(insert(vi, initset(hi)))
    sides = difference(allf, intsct)
    oS = recolor(FOUR, sides)
    oI = recolor(TWO, intsct)
    O = paint(I, oS)
    O = paint(O, oI)
    return O
