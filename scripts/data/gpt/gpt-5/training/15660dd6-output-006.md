def dsl1(I):
    # Global: remove outer 8-border, suppress 1 as 8, then remove both vertical and horizontal 8-separators
    A = trim(I)
    B = replace(A, ONE, EIGHT)
    C = rot90(B)
    D = compress(C)
    E = rot270(D)
    O = compress(E)
    return O

def dsl2(I):
    # Object extraction: keep the largest non-1 and non-8 object and crop to it
    OBJS = objects(I, T, F, T)
    NOTONE = compose(flip, matcher(color, ONE))
    NOTEIGHT = compose(flip, matcher(color, EIGHT))
    PRED = fork(both, NOTONE, NOTEIGHT)
    GOOD = sfilter(OBJS, PRED)
    BIG = argmax(GOOD, size)
    O = subgrid(BIG, I)
    return O

def dsl3(I):
    # Color filtering: isolate the least frequent color and crop to it
    C = leastcolor(I)
    P = ofcolor(I, C)
    OBJ = toobject(P, I)
    O = subgrid(OBJ, I)
    return O

def dsl4(I):
    # Geometric: keep cells invariant under both horizontal and vertical mirrors, else 8
    A = hmirror(I)
    B = cellwise(I, A, EIGHT)
    R = vmirror(I)
    O = cellwise(B, R, EIGHT)
    return O

def dsl5(I):
    # Relational: outline each object's bounding box (box) with its own color on an 8 canvas
    OBJS = objects(I, T, F, T)
    OUT = apply(fork(recolor, color, box), OBJS)
    ALL = merge(OUT)
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, ALL)
    return O
