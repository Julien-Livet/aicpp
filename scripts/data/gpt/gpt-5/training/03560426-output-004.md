def dsl1(I):
    T = trim(I)
    R = rot90(T)
    C = compress(R)
    hI = height(I)
    wI = width(I)
    hC = height(C)
    wC = width(C)
    rightpad = canvas(ZERO, astuple(hC, subtract(wI, wC)))
    toprow = hconcat(C, rightpad)
    bottompad = canvas(ZERO, astuple(subtract(hI, hC), wI))
    O = vconcat(toprow, bottompad)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    norms = apply(normalize, objs)
    packed = merge(norms)
    base = canvas(ZERO, shape(I))
    O = paint(base, packed)
    return O

def dsl3(I):
    cols = remove(ZERO, palette(I))
    comp = lbind(colorcount, I)
    c = valmax(cols, comp)
    mask = ofcolor(I, c)
    obj = recolor(c, mask)
    base = canvas(ZERO, shape(I))
    onlyc = paint(base, obj)
    C = compress(onlyc)
    hI = height(I)
    wI = width(I)
    hC = height(C)
    wC = width(C)
    rightpad = canvas(ZERO, astuple(hC, subtract(wI, wC)))
    toprow = hconcat(C, rightpad)
    bottompad = canvas(ZERO, astuple(subtract(hI, hC), wI))

def dsl5(I):
    allidx = asindices(I)
    zeros = ofcolor(I, ZERO)
    nonzeros = difference(allidx, zeros)
    cols = remove(ZERO, palette(I))
    comp = lbind(colorcount, I)
    c1 = valmax(cols, comp)
    m1 = ofcolor(I, c1)
    o1 = toobject(m1, I)
    m2 = difference(nonzeros, m1)
    o2 = toobject(m2, I)
    n1 = normalize(o1)
    n2 = normalize(o2)
    h1 = height(n1)
    shifted_n2 = shift(n2, toivec(h1))
    base = canvas(ZERO, shape(I))
    O = paint(paint(base, n1), shifted_n2)
    return O
