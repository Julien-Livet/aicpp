def dsl1(I):
    S = compress(trim(I))
    P = ofcolor(S, TWO)
    C = centerofmass(P)
    A = astuple(THREE, THREE)
    ST = subtract(C, A)
    O = crop(S, ST, astuple(SEVEN, SEVEN))
    return O

def dsl2(I):
    P = toobject(ofcolor(I, TWO), I)
    N = normalize(P)
    R = recolor(TWO, N)
    G = canvas(ZERO, astuple(SEVEN, SEVEN))
    O = paint(G, R)
    return O

def dsl3(I):
    G0 = canvas(ZERO, shape(I))
    R = recolor(TWO, toobject(ofcolor(I, TWO), I))
    S = paint(G0, R)
    C = compress(S)
    H = height(C)
    W = width(C)
    FH = branch(greater(H, SEVEN), divide(H, SEVEN), ONE)
    FW = branch(greater(W, SEVEN), divide(W, SEVEN), ONE)
    F = branch(greater(FH, FW), FH, FW)
    D = downscale(C, F)
    O = crop(D, ORIGIN, astuple(SEVEN, SEVEN))
    return O

def dsl4(I):
    S = subgrid(ofcolor(I, TWO), I)
    M = cmirror(S)
    C = compress(M)
    O = crop(C, ORIGIN, astuple(SEVEN, SEVEN))
    return O

def dsl5(I):
    P = toobject(ofcolor(I, TWO), I)
    N = normalize(P)
    R = recolor(TWO, N)
    C = centerofmass(toindices(R))
    T = astuple(THREE, THREE)
    OFS = subtract(T, C)
    G = canvas(ZERO, astuple(SEVEN, SEVEN))
    G1 = paint(G, R)
    O = move(G1, R, OFS)
    return O
