def dsl1(I):
    ones = ofcolor(I, ONE)
    obj = toobject(ones, I)
    d = hperiod(obj)
    vec = multiply(UNITY, d)
    ul = ulcorner(ones)
    periodic = shoot(ul, vec)
    inside = intersection(periodic, inbox(ones))
    cont = difference(periodic, inside)
    twos = recolor(TWO, cont)
    O = underpaint(I, twos)
    return O
