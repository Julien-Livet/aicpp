def dsl1(I):
    # Global: zero out ones, then swap 8 with computed value
    bh = bottomhalf(I)
    cnt = size(ofcolor(bh, ONE))
    val = divide(cnt, TWO)
    base = replace(I, ONE, ZERO)
    O = switch(base, EIGHT, val)
    return O

def dsl2(I):
    # Object extraction via partition: recolor 8-objects and paint on blank canvas
    bh = bottomhalf(I)
    f = compose(size, rbind(ofcolor, ONE))
    val = halve(f(bh))
    parts = partition(I)
    eobjs = colorfilter(parts, EIGHT)
    recol = merge(apply(lbind(recolor, val), eobjs))
    O = paint(canvas(ZERO, shape(I)), recol)
    return O

def dsl3(I):
    # Color filtering: clear non-8s, then underfill 8-mask with value
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    cleared8 = switch(I, EIGHT, ZERO)
    base = replace(cleared8, ONE, ZERO)
    mask8 = ofcolor(I, EIGHT)
    O = underfill(base, val, mask8)
    return O

def dsl4(I):
    # Geometric/structural: remove 8-object from grid, then repaint it recolored
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    inds8 = ofcolor(I, EIGHT)
    obj8 = toobject(inds8, I)
    base0 = replace(I, ONE, ZERO)
    base = cover(base0, obj8)
    recol = recolor(val, obj8)
    O = paint(base, recol)
    return O

def dsl5(I):
    # Relational/mask-based: recolor 8-object and underpaint onto zero canvas
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    inds8 = ofcolor(I, EIGHT)
    obj8 = toobject(inds8, I)
    recol = recolor(val, obj8)
    O = underpaint(canvas(ZERO, shape(I)), recol)
    return O
