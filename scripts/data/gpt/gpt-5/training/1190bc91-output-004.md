def dsl1(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    D = dmirror(OBJ)
    C = cmirror(OBJ)
    S = combine(combine(OBJ, D), C)
    O = paint(canvas(ZERO, shape(I)), S)
    return O

def dsl2(I):
    COLORS = remove(ZERO, palette(I))
    SFN = lbind(ofcolor, I)
    DIR = insert(UP_RIGHT, initset(DOWN_LEFT))
    R0 = lbind(prapply, shoot)
    R1 = rbind(R0, DIR)
    PFN = compose(merge, R1)
    STARTS = compose(PFN, SFN)
    COLORIZE = fork(recolor, identity, STARTS)
    PARTS = apply(COLORIZE, COLORS)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O

def dsl3(I):
    FG = fgpartition(I)
    ALL = merge(FG)
    MC = mostcolor(ALL)
    S = ofcolor(I, MC)
    DIR = insert(UP_LEFT, insert(UP_RIGHT, insert(DOWN_LEFT, initset(DOWN_RIGHT))))
    R0 = lbind(prapply, shoot)
    R1 = rbind(R0, DIR)
    PFN = compose(merge, R1)
    LINES = PFN(S)
    COLORED = recolor(MC, LINES)
    O = underpaint(I, COLORED)
    return O

def dsl4(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    B = box(OBJ)
    MC = mostcolor(OBJ)
    LINES = recolor(MC, B)
    O = underpaint(I, LINES)
    return O

def dsl5(I):
    FG = fgpartition(I)
    ALL = merge(FG)
    C = centerofmass(ALL)
    CST = lbind(connect, C)
    MID = compose(CST, center)
    RC = fork(recolor, color, MID)
    PARTS = apply(RC, FG)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O
