def dsl1(I):
    # Global: enforce 180-degree symmetry by keeping only cells that match their 180-rotated counterparts, then rectangularize
    sym = cellwise(I, rot180(I), ZERO)
    sobjs = objects(sym, T, F, T)
    parts = apply(fork(recolor, color, backdrop), sobjs)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl2(I):
    # Object extraction: for each interior color, keep the largest object and fill its backdrop
    objs = objects(I, T, F, T)
    inner = sfilter(objs, compose(flip, rbind(bordering, I)))
    cols = dedupe(totuple(apply(color, inner)))
    percolor = apply(lbind(colorfilter, inner), cols)
    pickmax = rbind(argmax, compose(size, toindices))
    winners = apply(pickmax, percolor)
    parts = apply(fork(recolor, color, backdrop), winners)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl3(I):
    # Color filtering: keep only colors present in both top and bottom halves, then rectangularize their objects
    top = tophalf(I)
    bot = bottomhalf(I)
    com = intersection(palette(top), palette(bot))
    cols = dedupe(totuple(com))
    objs = objects(I, T, F, T)
    groups = apply(lbind(colorfilter, objs), cols)
    selobjs = merge(groups)
    parts = apply(fork(recolor, color, backdrop), selobjs)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl4(I):
    # Geometric: rectangularize every detected object by filling its bounding box with its color
    objs = objects(I, T, F, T)
    parts = apply(fork(recolor, color, backdrop), objs)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl5(I):
    # Relational: keep objects that are adjacent to another object of the same color; then fill their backdrops
    objs = objects(I, T, F, T)
    samecolor = compose(lbind(colorfilter, objs), color)
    others = fork(remove, identity, samecolor)
    union_others = compose(merge, others)
    adj_same = fork(adjacent, identity, union_others)
    keep = sfilter(objs, adj_same)
    parts = apply(fork(recolor, color, backdrop), keep)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O
