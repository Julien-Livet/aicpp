def dsl1(I):
    # Global: remove color 5 everywhere, then fill the overall bounding box of each remaining color
    cols = remove(ZERO, palette(I))
    patcher = compose(backdrop, lbind(ofcolor, I))
    parts = apply(fork(recolor, identity, patcher), cols)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    O = replace(O, FIVE, ZERO)
    return O

def dsl2(I):

    objs = objects(I, T, F, T)
    cols = remove(ZERO, palette(I))
    groups = apply(lbind(colorfilter, objs), cols)
    getcol = compose(color, merge)
    getbox = compose(backdrop, compose(toindices, merge))
    parts = apply(fork(recolor, getcol, getbox), groups)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    O = replace(O, FIVE, ZERO)
    return O

def dsl3(I):

    top = tophalf(I)
    bot = bottomhalf(I)
    ptop = remove(FIVE, remove(ZERO, palette(top)))

    cols = intersection(ptop, pbot)
    patcher = compose(backdrop, lbind(ofcolor, I))
    parts = apply(fork(recolor, identity, patcher), cols)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl5(I):
