def dsl1(I):
    objs = objects(I, T, F, T)
    b5s = colorfilter(objs, FIVE)
    b5 = first(b5s)
    non5 = difference(objs, b5s)
    seeds = merge(mapply(toindices, non5))
    B = toindices(b5)
    pairs = product(seeds, B)
    f_ifst = compose(initset, first)
    f_ilast = compose(initset, last)
    pred_h = fork(hmatching, f_ifst, f_ilast)
    pred_v = fork(vmatching, f_ifst, f_ilast)
    pred = fork(either, pred_h, pred_v)
    aligned = sfilter(pairs, pred)
    f_conn = fork(connect, first, last)
    f_col = compose(lbind(index, I), first)
    f_recol = fork(recolor, f_col, f_conn)
    segs = mapply(f_recol, aligned)
    base = paint(canvas(ZERO, shape(I)), b5)
    O = underpaint(base, segs)
    return O
