def dsl1(I):
    # Per-quadrant adaptive downscale to 3x3, tile into 7x7 corners, carve central cross
    S = compress(subgrid(ofcolor(I, TWO), I))
    T = first(hsplit(S, TWO))
    B = last(hsplit(S, TWO))
    TLg = first(vsplit(T, TWO))
    TRg = last(vsplit(T, TWO))
    BLg = first(vsplit(B, TWO))
    BRg = last(vsplit(B, TWO))

    Htl = height(TLg); Wtl = width(TLg)
    FHtl = divide(add(Htl, TWO), THREE); FWtl = divide(add(Wtl, TWO), THREE)
    Ftl = branch(greater(FHtl, FWtl), FHtl, FWtl)
    Dtl = downscale(TLg, Ftl); Etl = compress(Dtl)
    OTL = recolor(TWO, toobject(ofcolor(Etl, TWO), Etl))

    Htr = height(TRg); Wtr = width(TRg)
    FHtr = divide(add(Htr, TWO), THREE); FWtr = divide(add(Wtr, TWO), THREE)
    Ftr = branch(greater(FHtr, FWtr), FHtr, FWtr)
    Dtr = downscale(TRg, Ftr); Etr = compress(Dtr)
    OTR = recolor(TWO, toobject(ofcolor(Etr, TWO), Etr))

    Hbl = height(BLg); Wbl = width(BLg)
    FHbl = divide(add(Hbl, TWO), THREE); FWbl = divide(add(Wbl, TWO), THREE)
    Fbl = branch(greater(FHbl, FWbl), FHbl, FWbl)
    Dbl = downscale(BLg, Fbl); Ebl = compress(Dbl)
    OBL = recolor(TWO, toobject(ofcolor(Ebl, TWO), Ebl))

    Hbr = height(BRg); Wbr = width(BRg)
    FHbr = divide(add(Hbr, TWO), THREE); FWbr = divide(add(Wbr, TWO), THREE)
    Fbr = branch(greater(FHbr, FWbr), FHbr, FWbr)
    Dbr = downscale(BRg, Fbr); Ebr = compress(Dbr)
    OBR = recolor(TWO, toobject(ofcolor(Ebr, TWO), Ebr))

    T1 = astuple(ONE, ONE); OX = tojvec(FOUR); OY = toivec(FOUR); OD = add(OX, OY)
    STL = shift(OTL, T1); STR = shift(OTR, add(OX, T1))
    SBL = shift(OBL, add(OY, T1)); SBR = shift(OBR, add(OD, T1))
    U = combine(combine(STL, STR), combine(SBL, SBR))

    G7 = canvas(ZERO, astuple(SEVEN, SEVEN))

    R0 = astuple(THREE, ZERO); R6 = astuple(THREE, SIX)
    C0 = astuple(ZERO, THREE); C6 = astuple(SIX, THREE)
    ROW = connect(R0, R6); COL = connect(C0, C6)
    O = fill(P, ZERO, combine(ROW, COL))
    return O

def dsl4(I):
    # Global downscale to <=7x7, center, carve central cross
    S = compress(subgrid(ofcolor(I, TWO), I))
    H = height(S); W = width(S)
    FH = divide(add(H, SIX), SEVEN); FW = divide(add(W, SIX), SEVEN)
    F = branch(greater(FH, FW), FH, FW)
    D = downscale(S, F); E = compress(D)
    OO = recolor(TWO, toobject(ofcolor(E, TWO), E))
    CEN = center(OO); MID = astuple(THREE, THREE)
    OFF = subtract(MID, CEN)
    SO = shift(OO, OFF)
    G7 = canvas(ZERO, astuple(SEVEN, SEVEN))
    P = paint(G7, SO)
    R0 = astuple(THREE, ZERO); R6 = astuple(THREE, SIX)
