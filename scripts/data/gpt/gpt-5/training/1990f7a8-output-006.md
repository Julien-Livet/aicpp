def dsl1(I):
    # Global: compress two-color region, downscale with ceil factor to fit, center into 7x7, carve zero cross
    S = compress(subgrid(ofcolor(I, TWO), I))
    H = height(S)
    W = width(S)
    FHc = divide(add(H, SIX), SEVEN)
    FWc = divide(add(W, SIX), SEVEN)
    F = branch(greater(FHc, FWc), FHc, FWc)
    D = downscale(S, F)
    E = compress(D)
    OOBJ = recolor(TWO, toobject(ofcolor(E, TWO), E))
    CEN = center(OOBJ)
    TGT = astuple(THREE, THREE)
    OFF = subtract(TGT, CEN)
    OO = shift(OOBJ, OFF)
    G7 = canvas(ZERO, astuple(SEVEN, SEVEN))
    P = paint(G7, OO)
    R0 = astuple(THREE, ZERO)
    R6 = astuple(THREE, SIX)
    C0 = astuple(ZERO, THREE)
    C6 = astuple(SIX, THREE)
    ROW = connect(R0, R6)
    COL = connect(C0, C6)
    CROSS = combine(ROW, COL)
    O = fill(P, ZERO, CROSS)
    return O

def dsl2(I):

    S = compress(subgrid(ofcolor(I, TWO), I))

    B = last(hsplit(S, TWO))
    TLg = first(vsplit(T, TWO))
    TRg = last(vsplit(T, TWO))
    BLg = first(vsplit(B, TWO))
    BRg = last(vsplit(B, TWO))

    CTR = compress(subgrid(ofcolor(TRg, TWO), TRg))

    CBR = compress(subgrid(ofcolor(BRg, TWO), BRg))

    HTR = height(CTR); WTR = width(CTR)
    HBL = height(CBL); WBL = width(CBL)
    HBR = height(CBR); WBR = width(CBR)

    FHTL = divide(add(HTL, TWO), THREE)
    FWTL = divide(add(WTL, TWO), THREE)
    FHTR = divide(add(HTR, TWO), THREE)
    FWTR = divide(add(WTR, TWO), THREE)
    FHBL = divide(add(HBL, TWO), THREE)
    FWBL = divide(add(WBL, TWO), THREE)
    FHBR = divide(add(HBR, TWO), THREE)
    FWBR = divide(add(WBR, TWO), THREE)

    FBR = branch(greater(FHBR, FWBR), FHBR, FWBR)

    OBR = recolor(TWO, toobject(ofcolor(DBR, TWO), DBR))

    OX = tojvec(FOUR)
    OY = toivec(FOUR)
    OD = add(OX, OY)

    STL = shift(OTL, T1)
    STR = shift(OTR, add(OX, T1))
    SBL = shift(OBL, add(OY, T1))
    SBR = shift(OBR, add(OD, T1))

    U = combine(combine(STL, STR), combine(SBL, SBR))
