def dsl1(I):
    zeros8 = replace(I, ZERO, EIGHT)
    z1 = replace(zeros8, ONE, ZERO)
    z2 = replace(z1, NINE, ZERO)
    mir = vmirror(z2)
    both = cellwise(z2, mir, ZERO)
    axis = ofcolor(I, ONE)
    L1 = shift(axis, LEFT)
    L2 = shift(axis, multiply(LEFT, TWO))
    L3 = shift(axis, multiply(LEFT, THREE))
    leftarea = combine(L1, combine(L2, L3))
    dims = astuple(height(I), THREE)
    BL = crop(both, ulcorner(leftarea), dims)
    R1 = shift(axis, RIGHT)
    R2 = shift(axis, multiply(RIGHT, TWO))
    R3 = shift(axis, multiply(RIGHT, THREE))
    rightarea = combine(R1, combine(R2, R3))
    RG = crop(z2, ulcorner(rightarea), dims)
    cols = hsplit(RG, THREE)
    c3 = last(cols)
    Pboth = ofcolor(BL, EIGHT)
    Pc3 = ofcolor(c3, EIGHT)
    Pc3s = shift(Pc3, multiply(RIGHT, TWO))
    P = combine(Pboth, Pc3s)
    O = fill(canvas(ZERO, dims), EIGHT, P)
    return O
