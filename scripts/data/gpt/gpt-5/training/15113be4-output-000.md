    val = leastcolor(I)
    mask = ofcolor(I, ONE)
    O = fill(I, val, mask)

def dsl2(I):
    top = tophalf(I)
    bot = bottomhalf(I)
    valtop = leastcolor(top)
    masktop = ofcolor(top, ONE)
    newtop = fill(top, valtop, masktop)
    O = vconcat(newtop, bot)
    return O

def dsl3(I):
    left = lefthalf(I)
    right = righthalf(I)
    valr = leastcolor(right)
    maskr = ofcolor(right, ONE)
    newright = fill(right, valr, maskr)
    O = hconcat(left, newright)
    return O

def dsl5(I):
    val = leastcolor(I)
    m = ofcolor(I, val)
    u = combine(m, shift(m, UP))
    u = combine(u, shift(m, DOWN))
