def dsl1(I):
    DIR0 = initset(UNITY)
    DIR1 = insert(NEG_UNITY, DIR0)
    DIR2 = insert(UP_RIGHT, DIR1)
    DIR = insert(DOWN_LEFT, DIR2)
    COLORS = remove(ZERO, palette(I))
    SFN = lbind(ofcolor, I)
    R0 = lbind(prapply, shoot)
    R1 = rbind(R0, DIR)
    PFN = compose(merge, R1)
    STARTS = compose(PFN, SFN)
    COLORIZE = fork(recolor, identity, STARTS)
    PARTS = apply(COLORIZE, COLORS)
    OBJ = merge(PARTS)
    Z = canvas(ZERO, shape(I))
    O = paint(Z, OBJ)
    return O

def dsl2(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    D = dmirror(OBJ)
    C = cmirror(OBJ)
    S = combine(D, C)
    O = underpaint(I, S)
    return O

def dsl3(I):
    FG = fgpartition(I)
    ALL = merge(FG)
    MC = mostcolor(ALL)
    O = replace(I, ZERO, MC)
    return O

def dsl4(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    S1 = shift(OBJ, UNITY)
    S2 = shift(OBJ, NEG_UNITY)
    S3 = shift(OBJ, UP_RIGHT)
    S4 = shift(OBJ, DOWN_LEFT)
    U1 = combine(OBJ, S1)
    U2 = combine(U1, S2)
    U3 = combine(U2, S3)
    S = combine(U3, S4)
    O = underpaint(I, S)
    return O

def dsl5(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    SG = subgrid(OBJ, I)
