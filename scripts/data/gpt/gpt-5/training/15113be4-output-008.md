def dsl1(I):
    # Global: propagate least-color along rook lines; recolor intersecting ones
    val = leastcolor(I)
    starts = ofcolor(I, val)
    d0 = initset(UP)
    d1 = insert(DOWN, d0)
    d2 = insert(LEFT, d1)
    dirs = insert(RIGHT, d2)
    rays = prapply(shoot, starts, dirs)
    paths = merge(rays)
    target = intersection(paths, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl2(I):
    # Object extraction: recolor one-objects sharing a column with any least-color cell
    val = leastcolor(I)
    vp = ofcolor(I, val)
    ones = colorfilter(fgpartition(I), ONE)
    idx = toindices
    colmatch = compose(lbind(vmatching, vp), idx)
    hits = sfilter(ones, colmatch)
    target = merge(apply(idx, hits))
    O = fill(I, val, target)
    return O

def dsl3(I):
    # Color filtering: recolor ones inside the rectangular hull (delta) of least-color region
    val = leastcolor(I)
    ring = delta(ofcolor(I, val))
    target = intersection(ring, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl4(I):
    # Geometric: draw cross through least-color center of mass; recolor intersecting ones
    val = leastcolor(I)
    vp = ofcolor(I, val)
    c = centerofmass(vp)
    d0 = initset(UP)
    d1 = insert(DOWN, d0)
    d2 = insert(LEFT, d1)
    dirs = insert(RIGHT, d2)
    rays = apply(lbind(shoot, c), dirs)

    O = fill(I, val, target)
    return O

def dsl5(I):
    # Relational: recolor one-objects sharing a row or a column with the least-color bounding box
    val = leastcolor(I)
    rect = backdrop(ofcolor(I, val))
    ones = colorfilter(fgpartition(I), ONE)
    idx = toindices
    rowhit = compose(lbind(hmatching, rect), idx)
    colhit = compose(lbind(vmatching, rect), idx)
    pred = fork(either, rowhit, colhit)
    hits = sfilter(ones, pred)
    target = merge(apply(idx, hits))
    O = fill(I, val, target)
    return O
