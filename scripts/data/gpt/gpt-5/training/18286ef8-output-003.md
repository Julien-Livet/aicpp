def dsl1(I):
    # Global/centroidal: steer by center-of-mass of all 9s after 6->9
    P5 = ofcolor(I, FIVE)
    B = inbox(P5)
    G = replace(I, SIX, NINE)
    P9 = ofcolor(G, NINE)
    J = intersection(ofcolor(I, NINE), B)
    P9out = difference(P9, B)
    Cj = centerofmass(J)
    Cx = centerofmass(P9out)
    V = subtract(Cx, Cj)
    D = sign(V)
    M = shift(J, D)
    G2 = fill(G, FIVE, J)
    O = paint(G2, recolor(NINE, M))
    return O

def dsl2(I):
    # Object extraction + move: move the internal 9 toward outer 9s and restore 5 at source
    P5 = ofcolor(I, FIVE)
    B = inbox(P5)
    G = replace(I, SIX, NINE)
    J = intersection(ofcolor(I, NINE), B)
    P9out = difference(ofcolor(G, NINE), B)
    D = sign(position(J, P9out))
    MG = move(G, toobject(J, I), D)
    O = fill(MG, FIVE, J)
    return O

def dsl3(I):
