def dsl1(I):
    # Geometric: gravitate all non-5 objects to the 8-neighborhood ring around all 5s
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    fidxs = toindices(fiveobj)
    ring = difference(merge(mapply(neighbors, fidxs)), fidxs)
    others = difference(objs, fives)
    dirf = rbind(gravitate, ring)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    keep = combine(fiveobj, mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl2(I):
    # Relational: snap each non-5 to the ring cells sharing its row or column
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    fidxs = toindices(fiveobj)
    ring = difference(merge(mapply(neighbors, fidxs)), fidxs)
    others = difference(objs, fives)
    rowline = compose(hfrontier, ulcorner)
    colline = compose(vfrontier, ulcorner)
    rowring = compose(lbind(intersection, ring), rowline)
    colring = compose(lbind(intersection, ring), colline)
    target = fork(combine, rowring, colring)
    dirf = fork(gravitate, toindices, target)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    keep = combine(fiveobj, mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl3(I):
    # Color filtering: keep all 5s and only non-5 objects that already intersect the ring
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    fidxs = toindices(fiveobj)
    ring = difference(merge(mapply(neighbors, fidxs)), fidxs)
    others = difference(objs, fives)
    hits = compose(positive, compose(size, compose(lbind(intersection, ring), toindices)))
    sel = sfilter(others, hits)
    keep = combine(fiveobj, merge(sel))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl4(I):
    # Global: rotate the subgrid within the bounding box of all 5s by 180° and place it back
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    bb = backdrop(fiveobj)
    sg = subgrid(bb, I)
    rot = rot180(sg)
    base = canvas(ZERO, shape(I))
    O = move(base, asobject(rot), ulcorner(bb))
    return O
