def dsl1(I):
    # Interior of the bounding box of the largest object
    objs = objects(I, T, F, T)
    big = argmax(objs, size)
    bb = backdrop(big)
    inner = delta(box(bb))
    O = subgrid(inner, I)
    return O

def dsl2(I):
    # Interior of the bounding box of the most common nonzero color
    cols = remove(ZERO, palette(I))
    mc = argmax(cols, lbind(colorcount, I))
    bb = backdrop(ofcolor(I, mc))
    inner = delta(box(bb))
    O = subgrid(inner, I)
    return O

def dsl3(I):
    # Crop to largest object bbox, remove its color, then trim zeros
    objs = objects(I, T, F, T)
    big = argmax(objs, size)
    bbgrid = subgrid(backdrop(big), I)
    O = trim(replace(bbgrid, color(big), ZERO))
    return O

def dsl4(I):
    # Color-anchored crop, remove border by 1, then trim zeros
    cols = remove(ZERO, palette(I))
    mc = argmax(cols, lbind(colorcount, I))
    bb = subgrid(backdrop(toobject(ofcolor(I, mc), I)), I)
    core = crop(bb, UNITY, subtract(shape(bb), TWO_BY_TWO))
    O = trim(core)
    return O

def dsl5(I):
    # Merge all objects of the dominant color, take interior of its bbox
    cols = remove(ZERO, palette(I))
    mc = argmax(cols, lbind(colorcount, I))
    os = colorfilter(objects(I, T, F, T), mc)
    uni = merge(os)
    bb = backdrop(uni)
    inner = delta(box(bb))
    O = subgrid(inner, I)
    return O
