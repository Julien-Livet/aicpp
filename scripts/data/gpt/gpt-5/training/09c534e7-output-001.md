def dsl1(I):
    # Relational multi-step growth: expand non-1 colors into adjacent 1-cells twice
    FG = fgpartition(I)
    C1 = colorfilter(FG, ONE)
    Seeds = difference(FG, C1)
    idxf = toindices
    nbf = lbind(mapply, dneighbors)
    nb_of_obj = compose(nbf, idxf)
    ringf = fork(difference, nb_of_obj, idxf)
    ones = ofcolor(I, ONE)
    adj1f = compose(lbind(intersection, ones), ringf)
    recolorf = fork(recolor, color, adj1f)
    G1 = paint(I, mapply(recolorf, Seeds))
    FGb = fgpartition(G1)
    C1b = colorfilter(FGb, ONE)
    Seedsb = difference(FGb, C1b)
    onesb = ofcolor(G1, ONE)
    adj1fb = compose(lbind(intersection, onesb), ringf)
    recolorfb = fork(recolor, color, adj1fb)
    O = paint(G1, mapply(recolorfb, Seedsb))
    return O

def dsl2(I):
    # Global transformation: overlay 180-rotated grid into background (zeros)
    R = rot180(I)
    O = underpaint(I, asobject(R))
    return O

def dsl3(I):
    # Color filtering: binarize non-1 foreground to 1
    FG = fgpartition(I)
    Not1 = difference(FG, colorfilter(FG, ONE))
    ToOnes = mapply(lbind(recolor, ONE), Not1)
    O = paint(I, ToOnes)
    return O

def dsl4(I):
    # Geometric reasoning: enforce vertical symmetry from the top half
    T = tophalf(I)
    O = vconcat(T, vmirror(T))
    return O

def dsl5(I):
    # Object extraction + relational fill: for each non-1 object, fill 1s in its bounding box with that color
    FG = fgpartition(I)
    C1 = colorfilter(FG, ONE)
    Seeds = difference(FG, C1)
    ones = ofcolor(I, ONE)
    boxf = compose(backdrop, toindices)
    inbox_ones = compose(lbind(intersection, ones), boxf)
    recolorf = fork(recolor, color, inbox_ones)
    Fills = mapply(recolorf, Seeds)
    O = paint(I, Fills)
    return O
