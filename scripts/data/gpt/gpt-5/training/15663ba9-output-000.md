def dsl1(I):
    fs = frontiers(I)
    hfs = sfilter(fs, hline)
    vfs = sfilter(fs, vline)
    hi = merge(apply(toindices, hfs))
    vi = merge(apply(toindices, vfs))
    hobj = recolor(FOUR, hi)
    vobj = recolor(TWO, vi)
    O = paint(I, hobj)
    O = paint(O, vobj)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    bbox = merge(apply(box, objs))
    crns = merge(apply(corners, objs))
    o_box = recolor(FOUR, bbox)
    o_crn = recolor(TWO, crns)
    O = paint(I, o_box)
    O = paint(O, o_crn)
    return O

def dsl3(I):
    top = tophalf(I)
    bot = bottomhalf(I)
    tnz = difference(asindices(top), ofcolor(top, ZERO))
    bnz = difference(asindices(bot), ofcolor(bot, ZERO))
    tobj = recolor(FOUR, tnz)
    bobj = recolor(TWO, bnz)
    topo = paint(top, tobj)
    boto = paint(bot, bobj)
    O = vconcat(topo, boto)
    return O

def dsl4(I):
    nz = difference(asindices(I), ofcolor(I, ZERO))
    leftnbr = intersection(nz, shift(nz, RIGHT))
    rightnbr = intersection(nz, shift(nz, LEFT))
    leftedge = difference(nz, leftnbr)
    rightedge = difference(nz, rightnbr)
    oL = recolor(FOUR, leftedge)
    oR = recolor(TWO, rightedge)
    O = paint(I, oL)
    O = paint(O, oR)
    return O

def dsl5(I):
    objs = objects(I, T, F, T)
    centers = apply(center, objs)
    urs = apply(urcorner, objs)
    lls = apply(llcorner, objs)
    purs = mpapply(connect, centers, urs)
    plls = mpapply(connect, centers, lls)
    u = merge(purs)
    l = merge(plls)
    oU = recolor(FOUR, u)
    oL = recolor(TWO, l)
    O = paint(I, oU)
    O = paint(O, oL)
    return O
