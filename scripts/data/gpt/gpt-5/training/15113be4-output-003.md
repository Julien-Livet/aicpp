def dsl1(I):
    # Global row/column propagation from least color
    val = leastcolor(I)
    m = ofcolor(I, val)
    hrows = merge(apply(hfrontier, m))
    vcols = merge(apply(vfrontier, m))
    lines = combine(hrows, vcols)
    target = intersection(lines, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl2(I):
    # Object extraction with both horizontal and vertical alignment
    val = leastcolor(I)
    m = ofcolor(I, val)
    parts = fgpartition(I)
    onesobjs = colorfilter(parts, ONE)
    hm = sfilter(onesobjs, lbind(hmatching, m))
    vm = sfilter(onesobjs, lbind(vmatching, m))
    aligned = combine(hm, vm)
    target = merge(apply(toindices, aligned))
    O = fill(I, val, target)
    return O

def dsl3(I):
    # Geometric propagation via 4-directional shooting
    val = leastcolor(I)
    idx = ofcolor(I, val)
    rshots = merge(apply(rbind(shoot, RIGHT), idx))
    lshots = merge(apply(rbind(shoot, LEFT), idx))
    ushots = merge(apply(rbind(shoot, UP), idx))
    dshots = merge(apply(rbind(shoot, DOWN), idx))
    lines = combine(combine(rshots, lshots), combine(ushots, dshots))
    target = intersection(lines, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O
