def dsl1(I):
    # Geometric: gravitate each non-5 object until adjacent to the union of all 5s
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    others = difference(objs, fives)
    dirf = rbind(gravitate, fiveobj)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    keep = combine(fiveobj, mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl2(I):
    # Relational: gravitate each non-5 to the ring (8-neighborhood) of all 5s, preferring same row/column
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    fidxs = toindices(fiveobj)
    ring = difference(merge(mapply(neighbors, fidxs)), fidxs)
    ringobj = recolor(ZERO, ring)
    others = difference(objs, fives)
    rowline = compose(hfrontier, centerofmass)
    colline = compose(vfrontier, centerofmass)
    rowring = compose(lbind(intersection, ring), rowline)
    colring = compose(lbind(intersection, ring), colline)
    anchor = fork(combine, rowring, colring)
    targetidx = compose(lbind(combine, ring), anchor)
    targetobj = compose(lbind(recolor, ZERO), targetidx)
    dirf = fork(gravitate, identity, targetobj)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    keep = combine(fiveobj, mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl4(I):
    # Global: shift the union of all non-5 objects together until adjacent to the 5s
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    others = difference(objs, fives)
    unionothers = merge(others)
    d = gravitate(unionothers, fiveobj)
    dfun = lbind(identity, d)
    shifted = fork(shift, toindices, dfun)
    moved = fork(recolor, color, shifted)
    keep = combine(fiveobj, mapply(moved, others))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O

def dsl5(I):
    # Color filtering: keep all 5s and only non-5 objects sharing a row or column with the 5s
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    others = difference(objs, fives)
    samecol = lbind(vmatching, fiveobj)
    samerow = lbind(hmatching, fiveobj)
    keepcond = fork(either, samecol, samerow)
    sel = sfilter(others, keepcond)
    keep = combine(fiveobj, merge(sel))
    O = paint(canvas(ZERO, shape(I)), keep)
    return O
