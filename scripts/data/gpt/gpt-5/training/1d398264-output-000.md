def dsl1(I):
    # Cross propagation from unique center: extend colors along four cardinal directions
    colors = palette(I)
    nz = remove(ZERO, colors)
    countf = lbind(colorcount, I)
    ctrcol = valmin(nz, countf)
    ctrset = ofcolor(I, ctrcol)
    ctr = first(ctrset)
    L = add(ctr, LEFT)
    R = add(ctr, RIGHT)
    U = add(ctr, UP)
    D = add(ctr, DOWN)
    lc = index(I, L)
    rc = index(I, R)
    uc = index(I, U)
    dc = index(I, D)
    leftseg = shoot(ctr, LEFT)
    rightseg = shoot(ctr, RIGHT)
    upseg = shoot(ctr, UP)
    downseg = shoot(ctr, DOWN)
    O = canvas(ZERO, shape(I))
    O = fill(O, lc, leftseg)
    O = fill(O, rc, rightseg)
    O = fill(O, uc, upseg)
    O = fill(O, dc, downseg)
    O = fill(O, ctrcol, initset(ctr))
    return O

def dsl3(I):
    # Assemble rays as recolored objects and paint them
    colors = palette(I)
    nz = remove(ZERO, colors)
    countf = lbind(colorcount, I)
    ctrcol = valmin(nz, countf)
    ctr = first(ofcolor(I, ctrcol))
    L = add(ctr, LEFT); R = add(ctr, RIGHT); U = add(ctr, UP); D = add(ctr, DOWN)
    UL = add(U, LEFT); UR = add(U, RIGHT); DL = add(D, LEFT); DR = add(D, RIGHT)
    leftobj = recolor(index(I, L), shoot(ctr, LEFT))
    rightobj = recolor(index(I, R), shoot(ctr, RIGHT))
    upobj = recolor(index(I, U), shoot(ctr, UP))
    downobj = recolor(index(I, D), shoot(ctr, DOWN))
    ulobj = recolor(index(I, add(ctr, UL)), shoot(add(ctr, UL), UP_LEFT))
    urobj = recolor(index(I, add(ctr, UR)), shoot(add(ctr, UR), UP_RIGHT))
    dlobj = recolor(index(I, add(ctr, DL)), shoot(add(ctr, DL), DOWN_LEFT))
    drobj = recolor(index(I, add(ctr, DR)), shoot(add(ctr, DR), add(DOWN, RIGHT)))
    centerobj = recolor(ctrcol, initset(ctr))
    O = canvas(ZERO, shape(I))
    O = paint(O, leftobj)
    O = paint(O, rightobj)
    O = paint(O, upobj)
    O = paint(O, downobj)
    O = paint(O, ulobj)
    O = paint(O, urobj)
    O = paint(O, dlobj)
    O = paint(O, drobj)
    O = paint(O, centerobj)
    return O

def dsl4(I):
    # Use row/column frontiers and half-masks via set difference (geometric row/col split)
    colors = palette(I)
    nz = remove(ZERO, colors)
    countf = lbind(colorcount, I)
    ctrcol = valmin(nz, countf)
    ctr = first(ofcolor(I, ctrcol))

    lc = index(I, L); rc = index(I, R); uc = index(I, U); dc = index(I, D)
    rowmask = hfrontier(ctr)
    colmask = vfrontier(ctr)
    rightseg = shoot(ctr, RIGHT)
    leftseg = difference(rowmask, rightseg)
    downseg = shoot(ctr, DOWN)
    upseg = difference(colmask, downseg)
    O = canvas(ZERO, shape(I))
    O = fill(O, lc, leftseg)
    O = fill(O, rc, rightseg)
    O = fill(O, uc, upseg)
    O = fill(O, dc, downseg)
    O = fill(O, ctrcol, initset(ctr))
    return O
