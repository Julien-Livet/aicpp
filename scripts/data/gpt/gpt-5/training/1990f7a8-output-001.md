def dsl1(I):
    S = subgrid(ofcolor(I, TWO), I)
    H = height(S)
    W = width(S)
    FH = branch(greater(H, SEVEN), divide(H, SEVEN), ONE)
    FW = branch(greater(W, SEVEN), divide(W, SEVEN), ONE)
    F = branch(greater(FH, FW), FH, FW)
    D = downscale(S, F)
    O = crop(D, ORIGIN, astuple(SEVEN, SEVEN))
    return O

def dsl2(I):
    P = ofcolor(I, TWO)
    S = subgrid(P, I)
    C = compress(S)
    HC = height(C)
    WC = width(C)
    FV = branch(greater(HC, SEVEN), ONE, branch(greater(SEVEN, HC), divide(SEVEN, HC), ONE))
    FH = branch(greater(WC, SEVEN), ONE, branch(greater(SEVEN, WC), divide(SEVEN, WC), ONE))
    HU = hupscale(C, FH)
    VU = vupscale(HU, FV)
    O = crop(VU, ORIGIN, astuple(SEVEN, SEVEN))
    return O
