    S = hupscale(I, THREE)
    M = hmirror(S)
    O = vconcat(vconcat(S, M), S)

def dsl2(I):
    C = vsplit(I, TWO)
    L = first(C)
    R = last(C)
    P = hconcat(L, R)
    P3 = hconcat(hconcat(P, P), P)
    Q = hconcat(R, L)
    Q3 = hconcat(hconcat(Q, Q), Q)
    O = vconcat(vconcat(P3, Q3), P3)
    return O

def dsl4(I):

    O = vconcat(vconcat(A, B), A)
