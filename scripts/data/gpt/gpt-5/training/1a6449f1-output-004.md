def dsl1(I):
    # Interior of the most common nonzero color's bounding box
    nz = remove(ZERO, palette(I))
    comp = compose(size, lbind(ofcolor, I))
    mc = valmax(nz, comp)
    obj = toobject(ofcolor(I, mc), I)
    inner = delta(obj)
    O = subgrid(inner, I)
    return O

def dsl2(I):
    # Use the object whose bounding-box interior is largest
    objs = objects(I, T, F, T)
    big = argmax(objs, compose(size, delta))
    inner = delta(big)
    O = subgrid(inner, I)
    return O

def dsl3(I):
    # Largest frontier's bounding-box interior
    frs = frontiers(I)
    bigf = argmax(frs, size)
    inner = difference(backdrop(bigf), box(bigf))
    O = subgrid(inner, I)
    return O

def dsl4(I):
    # Interior region defined by the union of all border-touching objects
    objs = objects(I, T, F, T)
    borderobjs = sfilter(objs, rbind(bordering, I))
    bd = merge(borderobjs)
    inner = delta(bd)
    O = subgrid(inner, I)
    return O

def dsl5(I):
    # Global normalization: remove separators and trim outer border
    O = trim(compress(I))
    return O
