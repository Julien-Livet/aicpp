def dsl1(I):
    # Global: overlay all four rotations
    A = asobject(I)
    B = asobject(rot90(I))
    C = asobject(rot180(I))
    D = asobject(rot270(I))
    AB = combine(A, B)
    CD = combine(C, D)
    OBJ = combine(AB, CD)
    O = underpaint(I, OBJ)
    return O

def dsl2(I):
    # Object extraction: draw box and both diagonals for each object
    FG = fgpartition(I)
    IDX = toindices
    UL = compose(ulcorner, IDX)
    UR = compose(urcorner, IDX)
    LL = compose(llcorner, IDX)
    LR = compose(lrcorner, IDX)
    D1 = fork(connect, UL, LR)
    D2 = fork(connect, UR, LL)
    BX = compose(box, IDX)
    D12 = fork(combine, D1, D2)
    SHP = fork(combine, BX, D12)
    RC = fork(recolor, color, SHP)
    PARTS = apply(RC, FG)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O

def dsl3(I):
    # Color filtering: propagate least frequent nonzero color along anti-diagonals
    FG = fgpartition(I)
    ALL = merge(FG)
    C = leastcolor(ALL)
    S = ofcolor(I, C)
    D0 = initset(UP_RIGHT)
    DIR = insert(DOWN_LEFT, D0)
    R0 = lbind(prapply, shoot)
    FN = rbind(R0, DIR)
    RAW = FN(S)
    LINES = merge(RAW)
    COLORED = recolor(C, LINES)
    O = underpaint(I, COLORED)
    return O

def dsl4(I):
    # Geometric: overlay both diagonal mirrors
    A = asobject(I)
    B = asobject(dmirror(I))
    C = asobject(cmirror(I))
    AB = combine(A, B)
    OBJ = combine(AB, C)
    O = underpaint(I, OBJ)
    return O

def dsl5(I):
    # Relational: from each object's center, shoot toward UL and LR corners
    FG = fgpartition(I)
    GRID = asindices(I)
    UL = ulcorner(GRID)
    LR = lrcorner(GRID)
    PUL = initset(UL)
    PLR = initset(LR)
    DIR_UL = compose(sign, rbind(position, PUL))
    DIR_LR = compose(sign, rbind(position, PLR))
    SD0 = compose(initset, DIR_UL)
    DIRS = fork(insert, DIR_LR, SD0)
    START = compose(initset, compose(center, toindices))
    R0 = lbind(prapply, shoot)
    RAW = fork(R0, START, DIRS)
    PFN = compose(merge, RAW)
    RC = fork(recolor, color, PFN)
    PARTS = apply(RC, FG)
    OBJ = merge(PARTS)
    O = underpaint(I, OBJ)
    return O
