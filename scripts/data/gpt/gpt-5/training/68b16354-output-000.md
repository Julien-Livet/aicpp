def dsl1(I):
    O = hmirror(I)
    return O

def dsl2(I):
    dims = shape(I)
    base = canvas(ZERO, dims)
    obj = asobject(I)
    mobj = hmirror(obj)
    O = paint(base, mobj)
    return O

def dsl3(I):
    cols = palette(I)
    percol = lbind(ofcolor, I)
    mirlayer = compose(hmirror, percol)
    build = fork(recolor, identity, mirlayer)
    objs = apply(build, cols)
    merged = merge(objs)
    O = paint(canvas(ZERO, shape(I)), merged)
    return O

def dsl4(I):
    O = rot270(vmirror(rot90(I)))
    return O

def dsl5(I):
    O = dmirror(vmirror(dmirror(I)))
    return O
