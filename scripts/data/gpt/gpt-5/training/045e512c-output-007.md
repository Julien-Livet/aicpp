def dsl1(I):
    # Global tiling of the trimmed foreground across the full grid
    sub = trim(I)
    base = asobject(sub)
    ws = width(sub)
    hs = height(sub)
    wI = width(I)
    hI = height(I)
    offsH = interval(0, wI, ws)
    vecsH = apply(tojvec, offsH)
    tileH = mapply(lbind(shift, base), vecsH)
    offsV = interval(0, hI, hs)
    vecsV = apply(toivec, offsV)
    tileHV = mapply(lbind(shift, tileH), vecsV)
    O = paint(canvas(ZERO, shape(I)), tileHV)
    return O

def dsl2(I):
    # Object extraction: fill each object's inbox with its own color
    objs = objects(I, T, F, T)
    recol_in = fork(recolor, color, inbox)
    fills = apply(recol_in, objs)
    allfills = merge(fills)
    O = paint(canvas(ZERO, shape(I)), allfills)
    return O

def dsl3(I):
    # Color filtering: draw all rows and columns through the most frequent nonzero color
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    starts = ofcolor(I, c)
    rows = mapply(hfrontier, starts)
    cols = mapply(vfrontier, starts)
    lines = combine(rows, cols)
    obj = recolor(c, lines)
    O = paint(canvas(ZERO, shape(I)), obj)
    return O

def dsl4(I):
    # Geometric: diagonal and counter-diagonal symmetrization of cropped foreground
    nz = difference(asindices(I), ofcolor(I, ZERO))
    sub = subgrid(nz, I)
    base = asobject(sub)
    symd = combine(base, dmirror(base))
    sym2 = combine(symd, cmirror(symd))
    O = paint(canvas(ZERO, shape(I)), sym2)
    return O

def dsl5(I):
    # Relational: connect centers of all objects with lines colored by the dominant nonzero color
    objs = objects(I, T, F, T)
    cents = apply(center, objs)
    pairs = product(cents, cents)
    segs = mapply(connect, pairs)
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    obj = recolor(c, segs)
    O = paint(canvas(ZERO, shape(I)), obj)
    return O
