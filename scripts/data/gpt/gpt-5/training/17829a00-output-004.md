def dsl1(I):
    # Relational projection: gravitate each top/bottom-colored object to its border, then keep only the inner-adjacent row
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    TO = colorfilter(OB, cT)
    BO = colorfilter(OB, cB)
    R0 = hfrontier(astuple(ZERO, ZERO))
    H = height(I)
    Hm1 = subtract(H, ONE)
    Rb = hfrontier(astuple(Hm1, ZERO))
    R1 = hfrontier(astuple(ONE, ZERO))
    Hm2 = subtract(H, TWO)
    Rm2 = hfrontier(astuple(Hm2, ZERO))
    ST = fork(shift, identity, rbind(gravitate, R0))
    SB = fork(shift, identity, rbind(gravitate, Rb))
    Tmov = apply(ST, TO)
    Bmov = apply(SB, BO)
    Pt = intersection(toindices(merge(Tmov)), R1)
    Pb = intersection(toindices(merge(Bmov)), Rm2)
    BG = mostcolor(I)
    C = canvas(BG, shape(I))
    O0 = fill(fill(C, cT, Pt), cB, Pb)
    O = paint(paint(O0, TOP), BOT)
    return O

def dsl2(I):
    # Geometric (rotation): rotate, gravitate horizontally to side borders, keep only inner-adjacent columns, rotate back
    J = rot90(I)
    OB = objects(J, T, F, T)

    LFT = argmin(BOR, leftmost)
    RGT = argmax(BOR, rightmost)

    cR = color(RGT)
    LO = colorfilter(OB, cL)
    RO = colorfilter(OB, cR)
    CL0 = vfrontier(astuple(ZERO, ZERO))
    W = width(J)
    Wm1 = subtract(W, ONE)
    CR0 = vfrontier(astuple(ZERO, Wm1))
    SL = fork(shift, identity, rbind(gravitate, CL0))
    SR = fork(shift, identity, rbind(gravitate, CR0))
    Lmov = apply(SL, LO)
    Rmov = apply(SR, RO)
    C1 = vfrontier(astuple(ZERO, ONE))
    Wm2 = subtract(W, TWO)
    Cw2 = vfrontier(astuple(ZERO, Wm2))

    Pright = intersection(toindices(merge(Rmov)), Cw2)
    BGJ = mostcolor(J)

    J0 = fill(fill(CJ, cL, Pleft), cR, Pright)

    O = rot270(J1)

def dsl4(I):
    # Symmetric decomposition: solve top in original, bottom via vertical mirror, then merge
    # Top part on I
    OB1 = objects(I, T, F, T)
    BOR1 = sfilter(OB1, rbind(bordering, I))
    TOP1 = argmin(BOR1, uppermost)
    c1 = color(TOP1)
    R01 = hfrontier(astuple(ZERO, ZERO))
    R11 = hfrontier(astuple(ONE, ZERO))
    TO1 = colorfilter(OB1, c1)
    ST1 = fork(shift, identity, rbind(gravitate, R01))
    Tmov1 = apply(ST1, TO1)
    P1 = intersection(toindices(merge(Tmov1)), R11)
    BG = mostcolor(I)
    Ct = canvas(BG, shape(I))
    Gt = paint(fill(Ct, c1, P1), TOP1)
    # Bottom part via vertical mirror
    V = vmirror(I)
    OB2 = objects(V, T, F, T)
    BOR2 = sfilter(OB2, rbind(bordering, V))
    TOP2 = argmin(BOR2, uppermost)
    c2 = color(TOP2)
    R02 = hfrontier(astuple(ZERO, ZERO))
    R12 = hfrontier(astuple(ONE, ZERO))
    TO2 = colorfilter(OB2, c2)
    ST2 = fork(shift, identity, rbind(gravitate, R02))
    Tmov2 = apply(ST2, TO2)
    P2 = intersection(toindices(merge(Tmov2)), R12)
    Cv = canvas(mostcolor(V), shape(V))
    Gv = paint(fill(Cv, c2, P2), TOP2)
    Gb = vmirror(Gv)
    # Merge results
    O0 = canvas(BG, shape(I))
    O1 = underpaint(O0, asobject(Gt))
    O = underpaint(O1, asobject(Gb))
    return O

def dsl5(I):
    # Relational + reconstruction: align non-border objects to inner-adjacent rows; keep only those rows; restore others
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    TO = colorfilter(OB, cT)
    BO = colorfilter(OB, cB)
    Tbor = sfilter(TO, rbind(bordering, I))
    Bbor = sfilter(BO, rbind(bordering, I))
    Tnon = difference(TO, Tbor)
    Bnon = difference(BO, Bbor)
    H = height(I)
    Hm2 = subtract(H, TWO)
    drT = compose(lbind(subtract, ONE), uppermost)
    vecT = compose(toivec, drT)
    drB = compose(lbind(subtract, Hm2), lowermost)
    vecB = compose(toivec, drB)
    ST = fork(shift, identity, vecT)
    SB = fork(shift, identity, vecB)
    Tmov = apply(ST, Tnon)
    Bmov = apply(SB, Bnon)
    R1 = hfrontier(astuple(ONE, ZERO))
    Hm1 = subtract(H, ONE)
    Rb = hfrontier(astuple(Hm1, ZERO))
    Rm2 = hfrontier(astuple(Hm2, ZERO))
    Pt = intersection(toindices(merge(Tmov)), R1)
    Pb = intersection(toindices(merge(Bmov)), Rm2)
    BG = mostcolor(I)
    C = canvas(BG, shape(I))
    O0 = fill(fill(C, cT, Pt), cB, Pb)
    O1 = paint(paint(O0, TOP), BOT)
    pT = ofcolor(I, cT)
    pB = ofcolor(I, cB)
    S = initset(pT)
    S2 = insert(pB, S)
    pTB = merge(S2)
    pAll = asindices(I)
    pOther = difference(pAll, pTB)
    O = underpaint(O1, toobject(pOther, I))
    return O
