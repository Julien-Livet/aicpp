def dsl1(I):
    cg = compress(I)
    hI = height(I)
    wI = width(I)
    hC = height(cg)
    wC = width(cg)
    rightpad = canvas(ZERO, astuple(hC, subtract(wI, wC)))
    top = hconcat(cg, rightpad)
    bottompad = canvas(ZERO, astuple(subtract(hI, hC), wI))
    O = vconcat(top, bottompad)
    return O

def dsl2(I):
    nonzeros = difference(asindices(I), ofcolor(I, ZERO))
    obj = toobject(nonzeros, I)
    norm = normalize(obj)
    mir = dmirror(norm)
    base = canvas(ZERO, shape(I))
    O = paint(base, mir)
    return O

def dsl3(I):
    colors = remove(ZERO, palette(I))
    comp = lbind(colorcount, I)
    c = valmax(colors, comp)
    mask = ofcolor(I, c)
    obj = toobject(mask, I)
    norm = normalize(obj)
    base = canvas(ZERO, shape(I))
    O = paint(base, norm)
    return O

def dsl4(I):
    allidx = asindices(I)
    zeros = ofcolor(I, ZERO)
    nonzeros = difference(allidx, zeros)
    sg = subgrid(nonzeros, I)
    R = rot90(sg)
    hI = height(I)
    wI = width(I)
    hR = height(R)
    wR = width(R)
    rightpad = canvas(ZERO, astuple(hR, subtract(wI, wR)))
    toprow = hconcat(R, rightpad)
    bottompad = canvas(ZERO, astuple(subtract(hI, hR), wI))
    O = vconcat(toprow, bottompad)
    return O
