def dsl1(I):
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    bg = ofcolor(J, FOUR)
    fg = difference(allidx, bg)
    obj = toobject(fg, J)
    c_obj = center(fg)
    c_grid = center(allidx)
    off = subtract(c_grid, c_obj)
    base = canvas(FOUR, shape(J))
    O = move(base, obj, off)
    return O

def dsl2(I):
    J = replace(I, FIVE, FOUR)
    ones = ofcolor(J, ONE)
    O = subgrid(ones, J)
    return O

def dsl3(I):
    allidx = asindices(I)
    bg = ofcolor(I, FOUR)
    idx5 = ofcolor(I, FIVE)
    tmp = difference(allidx, bg)
    fg = difference(tmp, idx5)
    obj = toobject(fg, I)
    base = canvas(FOUR, shape(I))
    O = paint(base, obj)
    return O

def dsl4(I):
    J = replace(I, FIVE, FOUR)
    M1 = dmirror(J)
    M2 = cmirror(J)
    obj1 = asobject(M1)
    obj2 = asobject(M2)
    tmp = underpaint(J, obj1)
    O = underpaint(tmp, obj2)
    return O

def dsl5(I):
    J = replace(I, FIVE, FOUR)
    idxL = asindices(lefthalf(J))
    idxR = asindices(righthalf(J))
    ones = ofcolor(J, ONE)
    leftOnes = intersection(ones, idxL)
    leftObj = toobject(leftOnes, J)
    cL = center(idxL)
    cR = center(idxR)
    off = subtract(cR, cL)
    moved = shift(leftObj, off)
    O = underpaint(J, moved)
    return O
