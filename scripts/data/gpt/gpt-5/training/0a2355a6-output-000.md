def dsl1(I):
    top = replace(tophalf(I), EIGHT, FOUR)
    bot = replace(bottomhalf(I), EIGHT, ONE)
    O = vconcat(top, bot)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    eights = colorfilter(objs, EIGHT)
    th = ofcolor(tophalf(I), EIGHT)
    cond_top = compose(lbind(hmatching, th), toindices)
    tops = sfilter(eights, cond_top)
    rest = difference(eights, tops)
    tops_r = merge(apply(lbind(recolor, FOUR), tops))
    rest_r = merge(apply(lbind(recolor, ONE), rest))
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(base, rest_r), tops_r)
    return O

def dsl3(I):
    base = replace(I, EIGHT, ZERO)
    p = ofcolor(I, EIGHT)
    obj = recolor(ONE, p)
    O = paint(base, obj)
    return O

def dsl4(I):
    base = replace(I, EIGHT, ONE)
    outline = box(ofcolor(I, EIGHT))
    O = fill(base, TWO, outline)
    return O

def dsl5(I):
    objs = colorfilter(objects(I, T, F, T), EIGHT)
    bord = sfilter(objs, rbind(bordering, I))
    inner = difference(objs, bord)
    bord_r = merge(apply(lbind(recolor, TWO), bord))
    inner_r = merge(apply(lbind(recolor, ONE), inner))
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(base, inner_r), bord_r)
    return O
