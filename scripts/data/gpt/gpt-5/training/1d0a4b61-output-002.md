def dsl1(I):
    # Period-guided lattice propagation: use discovered horizontal/vertical periods to underpaint via multi-offset shifts (global periodic completion)
    idx = asindices(I)
    nz = difference(idx, ofcolor(I, ZERO))
    obj = toobject(nz, I)
    hp = hperiod(obj)
    vp = vperiod(obj)
    r = tojvec(hp)
    l = invert(r)
    d = toivec(vp)
    u = invert(d)
    r2 = double(r); l2 = invert(r2)
    r4 = double(r2); l4 = invert(r4)
    r8 = double(r4); l8 = invert(r8)
    d2 = double(d); u2 = invert(d2)
    d4 = double(d2); u4 = invert(d4)
    d8 = double(d4); u8 = invert(d8)
    S = shift(obj, r)
    S = combine(S, shift(obj, l))
    S = combine(S, shift(obj, r2))
    S = combine(S, shift(obj, l2))
    S = combine(S, shift(obj, r4))
    S = combine(S, shift(obj, l4))
    S = combine(S, shift(obj, r8))
    S = combine(S, shift(obj, l8))
    S = combine(S, shift(obj, d))
    S = combine(S, shift(obj, u))
    S = combine(S, shift(obj, d2))
    S = combine(S, shift(obj, u2))
    S = combine(S, shift(obj, d4))
    S = combine(S, shift(obj, u4))
    S = combine(S, shift(obj, d8))
    S = combine(S, shift(obj, u8))
    O = underpaint(I, S)
    return O

def dsl3(I):

    idx = asindices(I)
    nz = difference(idx, ofcolor(I, ZERO))
    obj = toobject(nz, I)
    # Horizontal pass with multi-offsets
    r1 = RIGHT; l1 = LEFT
    r2 = ZERO_BY_TWO; l2 = invert(r2)
    r4 = double(r2); l4 = invert(r4)
    r8 = double(r4); l8 = invert(r8)
    S1 = shift(obj, r1)
    S1 = combine(S1, shift(obj, l1))
    S1 = combine(S1, shift(obj, r2))
    S1 = combine(S1, shift(obj, l2))
    S1 = combine(S1, shift(obj, r4))
    S1 = combine(S1, shift(obj, l4))
    S1 = combine(S1, shift(obj, r8))
    S1 = combine(S1, shift(obj, l8))
    O1 = underpaint(I, S1)
    # Recompute and do vertical pass
    idx1 = asindices(O1)
    nz1 = difference(idx1, ofcolor(O1, ZERO))
    obj1 = toobject(nz1, O1)
    d1 = DOWN; u1 = UP
    d2 = TWO_BY_ZERO; u2 = invert(d2)
    d4 = double(d2); u4 = invert(d4)
    d8 = double(d4); u8 = invert(d8)
    S2 = shift(obj1, d1)
    S2 = combine(S2, shift(obj1, u1))
    S2 = combine(S2, shift(obj1, d2))
    S2 = combine(S2, shift(obj1, u2))
    S2 = combine(S2, shift(obj1, d4))

    S2 = combine(S2, shift(obj1, u8))
    O2 = underpaint(O1, S2)
    return O2

def dsl4(I):
    # Color-filtered propagation: propagate only vivid (non-1) colors across periodic lattice to fill zeros (color filtering strategy)
    idx = asindices(I)
    nz = difference(idx, ofcolor(I, ZERO))
    obj = toobject(nz, I)
    hp = hperiod(obj)
    vp = vperiod(obj)
    r = tojvec(hp); l = invert(r)
    d = toivec(vp); u = invert(d)
    # collect only colors {2,3,4,5}
    parts = objects(I, T, F, T)
    c2 = merge(colorfilter(parts, TWO))
    c3 = merge(colorfilter(parts, THREE))
    c4 = merge(colorfilter(parts, FOUR))
    c5 = merge(colorfilter(parts, FIVE))
    C = combine(c2, combine(c3, combine(c4, c5)))
    r2 = double(r); l2 = invert(r2)
    d2 = double(d); u2 = invert(d2)
    S = shift(C, r)
    S = combine(S, shift(C, l))
    S = combine(S, shift(C, r2))
    S = combine(S, shift(C, l2))
    S = combine(S, shift(C, d))
    S = combine(S, shift(C, u))
    S = combine(S, shift(C, d2))
    S = combine(S, shift(C, u2))
    O = underpaint(I, S)
    return O

def dsl5(I):
    # Mutual half-completion and fusion: exchange content between halves then fuse (relational/structural over parts)
    # Horizontal exchange
    T, B = hsplit(I, 2)
    objT = toobject(difference(asindices(T), ofcolor(T, ZERO)), T)
    objB = toobject(difference(asindices(B), ofcolor(B, ZERO)), B)
    newT = underpaint(T, objB)
    newB = underpaint(B, objT)
    U = vconcat(newT, newB)
    # Vertical exchange
    L, R = vsplit(I, 2)
    objL = toobject(difference(asindices(L), ofcolor(L, ZERO)), L)
    objR = toobject(difference(asindices(R), ofcolor(R, ZERO)), R)
    newL = underpaint(L, objR)
    newR = underpaint(R, objL)
    V = hconcat(newL, newR)
    # Fuse vertical result onto horizontal result
    SV = toobject(difference(asindices(V), ofcolor(V, ZERO)), V)
    O = underpaint(U, SV)
    return O
