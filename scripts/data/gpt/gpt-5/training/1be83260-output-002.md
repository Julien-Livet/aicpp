def dsl1(I):
    G = trim(I)
    O = compress(G)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    fg = merge(objs)
    O = subgrid(fg, I)
    return O

def dsl3(I):
    objs = objects(I, T, F, T)
    rare = colorfilter(objs, leastcolor(I))
    ro = merge(rare)
    base = canvas(ZERO, shape(I))
    O = paint(base, ro)
    return O

def dsl4(I):
    G = trim(I)
    L = lefthalf(G)
    R = vmirror(L)
    O = hconcat(L, R)
    return O

def dsl5(I):
    objs = objects(I, T, F, T)
    fg = merge(objs)
    p0 = ulcorner(fg)
    p1 = lrcorner(fg)
    line = recolor(color(first(objs)), connect(p0, p1))
    base = canvas(ZERO, shape(I))
    O = paint(base, line)
    return O
