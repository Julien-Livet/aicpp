def dsl1(I):
    # Global: remove 5s, overlay original with its 180-degree rotation
    J = replace(I, FIVE, FOUR)
    base = canvas(FOUR, shape(J))
    O1 = underpaint(base, asobject(J))
    O = underpaint(O1, asobject(rot180(J)))
    return O

def dsl2(I):
    # Object extraction: take largest foreground object and overlay its vertical mirror
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    big = argmax(objs, size)
    base = canvas(FOUR, shape(J))
    g = paint(base, big)
    m = vmirror(g)
    O1 = underpaint(base, big)
    O = underpaint(O1, asobject(m))
    return O

def dsl3(I):
    # Color filtering: keep most common foreground color and add its vertical mirror
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    cols = apply(color, objs)
    cstar = mostcommon(cols)
    foc = colorfilter(objs, cstar)
    merged = merge(foc)
    base = canvas(FOUR, shape(J))
    G = paint(base, merged)
    M = vmirror(G)
    O = underpaint(G, asobject(M))
    return O

def dsl4(I):
    # Geometric: gravitate all foreground toward central cross (both axes)
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    fg = difference(allidx, ofcolor(J, FOUR))
    c = center(allidx)
    vln = combine(shoot(c, UP), shoot(c, DOWN))
    hln = combine(shoot(c, LEFT), shoot(c, RIGHT))
    cross = combine(vln, hln)
    d = gravitate(fg, cross)
    O = move(canvas(FOUR, shape(J)), toobject(fg, J), d)
    return O

def dsl5(I):
    # Relational: remove 5s, swap vertical halves, then swap horizontal halves
    J = replace(I, FIVE, FOUR)
    S = vconcat(bottomhalf(J), tophalf(J))
    O = hconcat(righthalf(S), lefthalf(S))
    return O
