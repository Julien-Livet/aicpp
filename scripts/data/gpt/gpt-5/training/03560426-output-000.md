def dsl1(I):
    O = rot270(I)
    return O

def dsl2(I):
    allidx = asindices(I)
    zeros = ofcolor(I, ZERO)
    nonzeros = difference(allidx, zeros)
    sg = subgrid(nonzeros, I)
    hI = height(I)
    wI = width(I)
    hS = height(sg)
    wS = width(sg)
    wdiff = subtract(wI, wS)
    hdiff = subtract(hI, hS)
    rightpad = canvas(ZERO, astuple(hS, wdiff))
    toprow = hconcat(sg, rightpad)
    bottompad = canvas(ZERO, astuple(hdiff, wI))
    O = vconcat(toprow, bottompad)
    return O

def dsl3(I):
    c = leastcolor(I)
    mask = ofcolor(I, c)
    obj = toobject(mask, I)
    base = canvas(ZERO, shape(I))
    O = paint(base, obj)
    return O

def dsl4(I):
    O = dmirror(I)
    return O

def dsl5(I):
    nonzeros = difference(asindices(I), ofcolor(I, ZERO))
    obj = toobject(nonzeros, I)
    normobj = normalize(obj)
    base = canvas(ZERO, shape(I))
    O = paint(base, normobj)
    return O
