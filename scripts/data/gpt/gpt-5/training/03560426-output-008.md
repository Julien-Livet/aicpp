def dsl1(I):
    C = compress(I)
    P = rot180(C)
    hI = height(I)
    wI = width(I)
    hP = height(P)
    wP = width(P)
    rightpad = canvas(ZERO, astuple(hP, subtract(wI, wP)))
    top = hconcat(P, rightpad)
    bottompad = canvas(ZERO, astuple(subtract(hI, hP), wI))
    O = vconcat(top, bottompad)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    us = apply(uppermost, objs)
    ls = apply(leftmost, objs)
    minu = valmin(us, identity)
    minl = valmin(ls, identity)
    dyf = compose(rbind(subtract, minu), uppermost)
    dxf = compose(rbind(subtract, minl), leftmost)
    vvec = compose(toivec, dyf)
    hvec = compose(tojvec, dxf)
    off = fork(add, vvec, hvec)
    sh = fork(shift, identity, off)
    parts = apply(sh, objs)
    packed = merge(parts)
    base = canvas(ZERO, shape(I))
    O = paint(base, packed)
    return O
