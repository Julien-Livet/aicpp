def dsl1(I):
    EO = colorfilter(objects(I, T, F, T), EIGHT)
    IN = merge(mapply(inbox, EO))
    BO = merge(mapply(box, EO))
    INT = difference(IN, BO)
    P0 = ofcolor(vmirror(I), TWO)
    P1 = ofcolor(hmirror(I), TWO)
    P2 = ofcolor(rot180(I), TWO)
    P = intersection(combine(combine(P0, P1), P2), INT)
    O = underfill(I, TWO, P)
    return O

def dsl2(I):

    Av = chain(g2, vmirror, Gf)

    Pv = merge(mapply(Sv, EO))
    Ph = merge(mapply(Sh, EO))
    Pr = merge(mapply(Sr, EO))
    P = combine(Pv, combine(Ph, Pr))
    Q = intersection(P, INT)
    O = underfill(I, TWO, Q)
    return O

def dsl4(I):
    EO = colorfilter(objects(I, T, F, T), EIGHT)
    IN = merge(mapply(inbox, EO))
    BO = merge(mapply(box, EO))
    INT = difference(IN, BO)
    S = intersection(ofcolor(I, TWO), INT)
    SO = toobject(S, I)
    HP = hperiod(SO)
    VP = vperiod(SO)
    HD = tojvec(HP)
    VD = toivec(VP)
    nHD = invert(HD)
    nVD = invert(VD)
    P0 = combine(shift(S, HD), shift(S, nHD))
    P1 = combine(shift(S, VD), shift(S, nVD))
    P2 = combine(P0, P1)
    P3 = combine(shift(S, add(HD, VD)), shift(S, add(nHD, VD)))
    P4 = combine(shift(S, add(HD, nVD)), shift(S, add(nHD, nVD)))
    P = intersection(combine(P2, combine(P3, P4)), INT)
    O = underfill(I, TWO, P)
    return O
