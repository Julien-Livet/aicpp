def dsl1(I):
    # Global: quadrant recolor inside 8-bounding box via hsplit/vsplit
    P = ofcolor(I, EIGHT)
    S = subgrid(P, I)
    TH = first(hsplit(S, TWO))
    BH = last(hsplit(S, TWO))
    TL = first(vsplit(TH, TWO))
    TR = last(vsplit(TH, TWO))
    BL = first(vsplit(BH, TWO))
    BR = last(vsplit(BH, TWO))
    TLc = replace(TL, EIGHT, FOUR)
    TRc = replace(TR, EIGHT, TWO)
    BLc = replace(BL, EIGHT, THREE)
    BRc = replace(BR, EIGHT, ONE)
    top = hconcat(TLc, TRc)
    bot = hconcat(BLc, BRc)
    Sc = vconcat(top, bot)
    UL = ulcorner(P)
    base = replace(I, EIGHT, ZERO)
    O = paint(base, recolor(FOUR, shift(ofcolor(Sc, FOUR), UL)))
    O = paint(O, recolor(TWO, shift(ofcolor(Sc, TWO), UL)))
    O = paint(O, recolor(THREE, shift(ofcolor(Sc, THREE), UL)))
    O = paint(O, recolor(ONE, shift(ofcolor(Sc, ONE), UL)))
    return O

def dsl2(I):
    # Object extraction: classify by object-center vs global 8-box center
    P = ofcolor(I, EIGHT)
    l = leftmost(P)
    r = rightmost(P)
    u = uppermost(P)
    d = lowermost(P)
    mi = halve(add(u, d))
    mj = halve(add(l, r))
    gc = astuple(mi, mj)
    objs = colorfilter(objects(I, T, F, T), EIGHT)
    vec = compose(rbind(subtract, gc), center)
    sgn = compose(sign, vec)
    is_tl = compose(lbind(equality, NEG_UNITY), sgn)
    is_tr = compose(lbind(equality, UP_RIGHT), sgn)
    is_bl = compose(lbind(equality, DOWN_LEFT), sgn)
    is_br = compose(lbind(equality, UNITY), sgn)
    tl = sfilter(objs, is_tl)
    tr = sfilter(objs, is_tr)
    bl = sfilter(objs, is_bl)
    br = sfilter(objs, is_br)
    tl_r = merge(apply(lbind(recolor, FOUR), tl))
    tr_r = merge(apply(lbind(recolor, TWO), tr))
    bl_r = merge(apply(lbind(recolor, THREE), bl))
    br_r = merge(apply(lbind(recolor, ONE), br))
    base = replace(I, EIGHT, ZERO)
    O = paint(paint(paint(paint(base, tl_r), tr_r), bl_r), br_r)
    return O

def dsl3(I):
    # Color filtering: pixelwise recolor by quadrant masks using cover + fill
    P = ofcolor(I, EIGHT)
    S = subgrid(P, I)
    UL = ulcorner(P)
    TLm = shift(intersection(asindices(lefthalf(S)), asindices(tophalf(S))), UL)
    TRm = shift(intersection(asindices(righthalf(S)), asindices(tophalf(S))), UL)
    BLm = shift(intersection(asindices(lefthalf(S)), asindices(bottomhalf(S))), UL)
    BRm = shift(intersection(asindices(righthalf(S)), asindices(bottomhalf(S))), UL)
    base = cover(I, toobject(P, I))
    O = fill(base, FOUR, intersection(P, TLm))
    O = fill(O, TWO, intersection(P, TRm))
    O = fill(O, THREE, intersection(P, BLm))
    O = fill(O, ONE, intersection(P, BRm))
    return O
