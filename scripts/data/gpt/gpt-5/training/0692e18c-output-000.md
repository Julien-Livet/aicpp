def dsl1(I):
    dims = shape(I)
    z = ofcolor(I, ZERO)
    p_obj = recolor(ONE, z)
    p = paint(canvas(ZERO, dims), p_obj)
    row = hconcat(p, hconcat(p, p))
    t = vconcat(row, vconcat(row, row))
    b = upscale(I, THREE)
    s = ofcolor(t, ONE)
    obj = toobject(s, b)
    O = paint(canvas(ZERO, shape(b)), obj)
    return O

def dsl2(I):
    z = ofcolor(I, ZERO)
    mkpatch = rbind(recolor, z)
    getpatch = compose(mkpatch, color)
    getoff = compose(rbind(multiply, THREE), center)
    place = fork(shift, getpatch, getoff)
    objs = objects(I, T, F, T)
    u = mapply(place, objs)
    O = paint(canvas(ZERO, multiply(shape(I), THREE)), u)
    return O

def dsl3(I):
    z = ofcolor(I, ZERO)
    p_obj = recolor(ONE, z)
    p = paint(canvas(ZERO, shape(I)), p_obj)
    v = ofcolor(p, ONE)
    pcs = fgpartition(I)
    originsfunc = compose(rbind(apply, rbind(multiply, THREE)), toindices)
    addv = rbind(rbind(prapply, add), v)
    kfunc = compose(addv, originsfunc)
    build = fork(recolor, color, kfunc)
    u = mapply(build, pcs)
    O = paint(canvas(ZERO, multiply(shape(I), THREE)), u)
    return O

def dsl5(I):

    nz_grid = paint(canvas(ZERO, dims), recolor(ONE, nzI))
