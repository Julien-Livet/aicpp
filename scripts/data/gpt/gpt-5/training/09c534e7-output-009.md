def dsl1(I):
    ones = ofcolor(I, ONE)
    mirrored = dmirror(I)
    obj = toobject(ones, mirrored)
    O = paint(I, obj)
    return O

def dsl2(I):
    FG = fgpartition(I)
    Seeds = difference(FG, colorfilter(FG, ONE))
    ones = ofcolor(I, ONE)
    window = compose(backdrop, toindices)
    maskf = compose(lbind(intersection, ones), window)
    recolorf = fork(recolor, color, maskf)
    Paints = mapply(recolorf, Seeds)
    O = paint(I, Paints)
    return O
