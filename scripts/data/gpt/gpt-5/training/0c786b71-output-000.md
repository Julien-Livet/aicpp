def dsl1(I):
    top = hconcat(rot180(I), hmirror(I))
    bottom = hconcat(vmirror(I), I)
    O = vconcat(top, bottom)
    return O

def dsl2(I):
    bottom = hconcat(vmirror(I), I)
    top = hmirror(bottom)
    O = vconcat(top, bottom)
    return O

def dsl3(I):
    h = height(I)
    w = width(I)
    dims = astuple(double(h), double(w))
    base = canvas(ZERO, dims)
    obj = normalize(asobject(I))
    O1 = move(base, rot180(obj), ORIGIN)
    O2 = move(O1, hmirror(obj), astuple(ZERO, w))
    O3 = move(O2, vmirror(obj), astuple(h, ZERO))
    O = move(O3, obj, astuple(h, w))
    return O

def dsl4(I):
    h = height(I)
    w = width(I)
    dims = astuple(double(h), double(w))
    base = canvas(ZERO, dims)
    obj = normalize(asobject(I))
    Gbr = move(base, obj, astuple(h, w))
    Gtr = hmirror(Gbr)
    Gbrtr = underpaint(Gbr, asobject(Gtr))
    Gleft = vmirror(Gbrtr)
    O = underpaint(Gbrtr, asobject(Gleft))
    return O

def dsl5(I):
    right = vconcat(hmirror(I), I)
    left = vmirror(right)
    O = hconcat(left, right)
    return O
