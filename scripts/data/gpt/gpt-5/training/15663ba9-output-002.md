def dsl1(I):
    bg = ofcolor(I, ZERO)
    nz = difference(asindices(I), bg)
    lE = intersection(nz, shift(bg, RIGHT))
    rE = intersection(nz, shift(bg, LEFT))
    hE = merge(insert(lE, initset(rE)))
    oH = recolor(FOUR, hE)
    vT = intersection(nz, shift(bg, DOWN))
    vB = intersection(nz, shift(bg, UP))
    rt = intersection(rE, vT)
    rt = intersection(rt, vB)
    oRT = recolor(TWO, rt)
    O = paint(I, oH)
    O = paint(O, oRT)
    return O
