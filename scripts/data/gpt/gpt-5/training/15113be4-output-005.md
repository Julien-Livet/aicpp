def dsl1(I):
    # Global cross through the overall center of mass of the least-color
    val = leastcolor(I)
    m = ofcolor(I, val)
    com = centerofmass(m)
    rows = hfrontier(com)
    cols = vfrontier(com)
    lines = combine(rows, cols)
    target = intersection(lines, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl2(I):
    # Object extraction: fill ONE-objects entirely inside the least-color box
    val = leastcolor(I)
    rect = backdrop(ofcolor(I, val))
    onesobjs = colorfilter(fgpartition(I), ONE)
    idcs = toindices
    intr = compose(lbind(intersection, rect), idcs)
    pred = fork(equality, idcs, intr)
    matches = sfilter(onesobjs, pred)
    target = merge(apply(toindices, matches))
    O = fill(I, val, target)
    return O

def dsl3(I):
    # Color filtering: fill ones on the outline of the least-color region
    val = leastcolor(I)
    border = box(ofcolor(I, val))
    target = intersection(border, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl4(I):
    # Geometric reasoning: fill ones on the diagonals of the least-color bounding box
    val = leastcolor(I)
    rect = backdrop(ofcolor(I, val))
    ul = ulcorner(rect)
    lr = lrcorner(rect)
    ur = urcorner(rect)
    ll = llcorner(rect)
    d1 = connect(ul, lr)
    d2 = connect(ur, ll)
    lines = combine(d1, d2)
    target = intersection(lines, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O

def dsl5(I):
    # Relational: fill ones adjacent (4-neighborhood) to least-color pixels
    val = leastcolor(I)
    m = ofcolor(I, val)
    nbrs = merge(apply(neighbors, m))
    target = intersection(nbrs, ofcolor(I, ONE))
    O = fill(I, val, target)
    return O
