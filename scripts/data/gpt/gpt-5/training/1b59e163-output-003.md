def dsl1(I):
    # Global symmetry aggregation: remove 5s, overlay all axial/diagonal mirrors
    J = replace(I, FIVE, FOUR)
    H = hmirror(J)
    V = vmirror(J)
    D = dmirror(J)
    C = cmirror(J)
    base = canvas(FOUR, shape(J))
    tmp1 = underpaint(base, asobject(J))
    tmp2 = underpaint(tmp1, asobject(H))
    tmp3 = underpaint(tmp2, asobject(V))
    tmp4 = underpaint(tmp3, asobject(D))
    O = underpaint(tmp4, asobject(C))
    return O

def dsl2(I):
    # Object extraction + global recentering via compress
    J = replace(I, FIVE, FOUR)
    C = compress(J)
    objC = asobject(C)
    cC = center(asindices(C))
    cJ = center(asindices(J))
    off = subtract(cJ, cC)
    O = move(canvas(FOUR, shape(J)), objC, off)
    return O

def dsl3(I):
    # Color filtering with selective symmetry: keep only 1s and 3s, plus their vertical mirrors
    J = replace(I, FIVE, FOUR)
    M = vmirror(J)
    base = canvas(FOUR, shape(J))
    o1 = toobject(ofcolor(J, ONE), J)
    o3 = toobject(ofcolor(J, THREE), J)
    m1 = toobject(ofcolor(M, ONE), M)
    m3 = toobject(ofcolor(M, THREE), M)
    t1 = underpaint(base, o1)
    t2 = underpaint(t1, o3)
    t3 = underpaint(t2, m1)
    O = underpaint(t3, m3)
    return O

def dsl4(I):
    # Geometric gravitation: move all foreground toward the central vertical line
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    fg = difference(allidx, ofcolor(J, FOUR))
    c = center(allidx)
    up = shoot(c, UP)
    down = shoot(c, DOWN)
    vline = combine(up, down)
    d = gravitate(fg, vline)
    O = move(canvas(FOUR, shape(J)), toobject(fg, J), d)
    return O

def dsl5(I):
    # Relational: pull each half toward the center by half the inter-center offset and overlay
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    bg = ofcolor(J, FOUR)
    fg = difference(allidx, bg)
    idxL = asindices(lefthalf(J))
    idxR = asindices(righthalf(J))
    leftFG = intersection(fg, idxL)
    rightFG = intersection(fg, idxR)
    leftObj = toobject(leftFG, J)
    rightObj = toobject(rightFG, J)
    cL = center(idxL)
    cR = center(idxR)
    full = subtract(cR, cL)
    half = halve(full)
    movedL = shift(leftObj, half)
    movedR = shift(rightObj, invert(half))
    t = underpaint(canvas(FOUR, shape(J)), movedL)
    O = underpaint(t, movedR)
    return O
