def dsl1(I):
    # O = globally translate all non-5 objects toward the union of per-5 inboxes
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    others = difference(objs, fives)
    perin = merge(apply(compose(inbox, toindices), fives))
    dir = gravitate(toindices(merge(others)), perin)
    moved = move(canvas(ZERO, shape(I)), merge(others), dir)
    O = paint(moved, fiveobj)
    return O

def dsl2(I):
    # O = keep only content inside the bounding rectangle spanned by all 5-rows and 5-columns
    fidxs = ofcolor(I, FIVE)
    rows = merge(apply(hfrontier, fidxs))
    cols = merge(apply(vfrontier, fidxs))
    band = combine(rows, cols)
    rect = backdrop(band)
    O = paint(canvas(ZERO, shape(I)), toobject(rect, I))
    return O

def dsl3(I):
    # O = keep all objects whose colors occur within the per-5 inbox area
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    perin = merge(apply(compose(inbox, toindices), fives))
    cols = palette(toobject(perin, I))
    isok = compose(rbind(contained, cols), color)
    keep = sfilter(objs, isok)
    O = paint(canvas(ZERO, shape(I)), merge(keep))
    return O

def dsl4(I):
    # O = place original 5s and mirror the rest vertically, restricted to the 5-band
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    fidxs = ofcolor(I, FIVE)
    rows = merge(apply(hfrontier, fidxs))
    cols = merge(apply(vfrontier, fidxs))
    band = combine(rows, cols)
    mir = vmirror(I)
    objmir = toobject(band, mir)
    Gm = paint(canvas(ZERO, shape(I)), objmir)
    mirobjs = objects(Gm, T, F, T)
    non5 = difference(mirobjs, colorfilter(mirobjs, FIVE))
    mirkept = merge(non5)
    O = paint(canvas(ZERO, shape(I)), combine(fiveobj, mirkept))
    return O

def dsl5(I):
    # O = move each non-5 to the nearest intersection of 5-rows and 5-columns
    objs = objects(I, T, F, T)
    fives = colorfilter(objs, FIVE)
    fiveobj = merge(fives)
    others = difference(objs, fives)
    fidxs = ofcolor(I, FIVE)
    rows = merge(apply(hfrontier, fidxs))
    cols = merge(apply(vfrontier, fidxs))
    cross = intersection(rows, cols)
    dirf = rbind(gravitate, cross)
    shifted = fork(shift, toindices, dirf)
    moved = fork(recolor, color, shifted)
    O = paint(canvas(ZERO, shape(I)), combine(fiveobj, mapply(moved, others)))
    return O
