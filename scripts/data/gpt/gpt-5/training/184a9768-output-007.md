def dsl1(I):
    # Fill holes inside each object's bounding box without overwriting existing nonzero pixels
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    boxes = apply(fork(recolor, color, inbox), objs)
    O = underpaint(I2, merge(boxes))
    return O

def dsl2(I):
    # Keep only the largest object and rectangularize it
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    big = argmax(objs, size)
    rect = recolor(color(big), inbox(big))
    O = paint(canvas(ZERO, shape(I)), rect)
    return O

def dsl3(I):
    # For the least frequent nonzero color, fill the rectangle spanning all its occurrences
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    cols = remove(ZERO, palette(I2))
    grp = compose(merge, lbind(colorfilter, objs))
    sel = valmin(cols, lbind(colorcount, I2))
    rect = recolor(sel, inbox(grp(sel)))
    O = paint(canvas(ZERO, shape(I)), rect)
    return O

def dsl4(I):
    # Draw an 'X' (both diagonals) across each object's bounding box
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    d1 = fork(connect, ulcorner, lrcorner)
    d2 = fork(connect, urcorner, llcorner)
    parts1 = apply(fork(recolor, color, d1), objs)
    parts2 = apply(fork(recolor, color, d2), objs)
    O = paint(canvas(ZERO, shape(I)), merge(combine(parts1, parts2)))
    return O

def dsl5(I):
    # Keep only objects adjacent to any other object; rectangularize them
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    others_all = rbind(remove, objs)
    union_others = compose(merge, others_all)
    keep = sfilter(objs, fork(adjacent, identity, union_others))
    parts = apply(fork(recolor, color, inbox), keep)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O
