def dsl1(I):
    nz = difference(asindices(I), ofcolor(I, ZERO))
    sub = subgrid(nz, I)
    ws = width(sub)
    wI = width(I)
    objs = objects(I, T, F, T)
    norms = apply(normalize, objs)
    offsH = interval(0, wI, ws)
    vecsH = apply(tojvec, offsH)
    copies = prapply(shift, norms, vecsH)
    allcopies = merge(copies)
    O = paint(canvas(ZERO, shape(I)), allcopies)
    return O

def dsl3(I):
    nz = difference(asindices(I), ofcolor(I, ZERO))
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    obj = recolor(c, nz)
    O = paint(canvas(ZERO, shape(I)), obj)
    return O

def dsl4(I):
    nz = difference(asindices(I), ofcolor(I, ZERO))
    sub = subgrid(nz, I)
    ws = width(sub)
    hs = height(sub)
    wI = width(I)
    hI = height(I)
    base = asobject(sub)
    sym1 = combine(base, dmirror(base))
    sym2 = combine(sym1, cmirror(sym1))
    offsH = interval(0, wI, ws)
    vecsH = apply(tojvec, offsH)
    tileH = mapply(lbind(shift, sym2), vecsH)
    offsV = interval(0, hI, hs)
    vecsV = apply(toivec, offsV)
    tileHV = mapply(lbind(shift, tileH), vecsV)
    O = paint(canvas(ZERO, shape(I)), tileHV)
    return O

def dsl5(I):
    objs = objects(I, T, F, T)
    centers = apply(center, objs)
    rows = mapply(hfrontier, centers)
    cols = mapply(vfrontier, centers)
    lines = combine(rows, cols)
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    obj = recolor(c, lines)
    O = paint(canvas(ZERO, shape(I)), obj)
    return O
