def dsl1(I):
    dims = shape(I)
    blank = canvas(ZERO, dims)
    objs = objects(I, T, F, T)
    isline = fork(either, hline, vline)
    lines = sfilter(objs, isline)
    def per_line(l):
        c = color(l)
        ai = toindices(l)
        cobjs = colorfilter(objs, c)
        extras = remove(l, cobjs)
        def per_extra(e):
            ei = toindices(e)
            d = gravitate(ei, ai)
            mi = shift(ei, d)
            return recolor(c, mi)
        moved = merge(apply(per_extra, extras))
        return combine(l, moved)
    parts = apply(per_line, lines)
    O = paint(blank, merge(parts))
    return O

def dsl4(I):
    dims = shape(I)
    blank = canvas(ZERO, dims)
    objs = objects(I, T, F, T)
    isline = fork(either, hline, vline)
    lines = sfilter(objs, isline)
    def per_line(l):
        c = color(l)
        ai = toindices(l)
        cobjs = colorfilter(objs, c)
        extras = remove(l, cobjs)
        ei = merge(apply(toindices, extras))
        vmask = merge(apply(vfrontier, ei))
        hmask = merge(apply(hfrontier, ei))
        sides_h = combine(shift(ai, UP), shift(ai, DOWN))
        sides_v = combine(shift(ai, LEFT), shift(ai, RIGHT))
        tgt_h = intersection(sides_h, vmask)
        tgt_v = intersection(sides_v, hmask)
        tgt = branch(hline(l), tgt_h, tgt_v)
        recol = recolor(c, tgt)
        return combine(l, recol)
    parts = apply(per_line, lines)
    O = paint(blank, merge(parts))
    return O
