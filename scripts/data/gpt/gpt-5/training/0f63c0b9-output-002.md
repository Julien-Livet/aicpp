def dsl1(I):
    dims = shape(I)
    H = height(I)
    Z = canvas(ZERO, dims)
    objs = objects(I, T, F, T)

    idxs = asindices(I)
    jL = leftmost(idxs)
    jR = rightmost(idxs)
    vL = tojvec(jL)
    vR = tojvec(jR)

    ords = order(objs, uppermost)
    headless = remove(last(ords), ords)
    tailless = remove(first(ords), ords)
    pairs = pair(headless, tailless)

    ra = compose(uppermost, first)
    rb = compose(uppermost, last)
    ca = compose(color, first)
    cb = compose(color, last)
    mid = compose(halve, fork(add, ra, rb))
    mid1 = compose(increment, mid)

    aL = compose(lbind(add, vL), compose(toivec, ra))
    aR = compose(lbind(add, vR), compose(toivec, ra))
    bL = compose(lbind(add, vL), compose(toivec, rb))
    bR = compose(lbind(add, vR), compose(toivec, rb))
    mL = compose(lbind(add, vL), compose(toivec, mid))
    mR = compose(lbind(add, vR), compose(toivec, mid))
    mL1 = compose(lbind(add, vL), compose(toivec, mid1))
    mR1 = compose(lbind(add, vR), compose(toivec, mid1))

    topRow = fork(recolor, ca, fork(connect, aL, aR))
    botRow = fork(recolor, cb, fork(connect, bL, bR))
    segLT = fork(recolor, ca, fork(connect, aL, mL))
    segRT = fork(recolor, ca, fork(connect, aR, mR))
    segLB = fork(recolor, cb, fork(connect, mL1, bL))
    segRB = fork(recolor, cb, fork(connect, mR1, bR))

    band = fork(combine, fork(combine, fork(combine, topRow, botRow), fork(combine, segLT, segRT)), fork(combine, segLB, segRB))
    borders = mapply(band, pairs)

    topobj = argmin(objs, uppermost)
    botobj = argmax(objs, lowermost)
    H1 = decrement(H)
    topL0 = astuple(ZERO, jL)
    topR0 = astuple(ZERO, jR)
    botL0 = astuple(H1, jL)
    botR0 = astuple(H1, jR)
    toprow = recolor(color(topobj), connect(topL0, topR0))
    botrow = recolor(color(botobj), connect(botL0, botR0))

    O = paint(Z, combine(borders, combine(toprow, botrow)))
    return O
