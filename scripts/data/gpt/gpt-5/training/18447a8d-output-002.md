def dsl1(I):
    G = rot180(I)
    E = ofcolor(G, EIGHT)
    S = ofcolor(G, SEVEN)
    OI = difference(asindices(G), combine(E, S))
    OO = toobject(OI, G)
    B = cover(G, OO)
    D = gravitate(OI, E)
    M = shift(OO, D)
    O = underpaint(B, M)
    return rot180(O)

def dsl2(I):
    E = ofcolor(I, EIGHT)
    S = ofcolor(I, SEVEN)
    OI = difference(asindices(I), combine(E, S))
    OO = toobject(OI, I)
    DJ = subtract(tojvec(add(rightmost(E), ONE)), tojvec(leftmost(OI)))
    B = cover(I, OO)
    M = shift(OO, DJ)
    O = underpaint(B, M)
    return O

def dsl3(I):
    E = ofcolor(I, EIGHT)
    S = ofcolor(I, SEVEN)
    OI = difference(asindices(I), combine(E, S))
    OO = toobject(OI, I)
    C = leastcolor(OO)
    O = fill(I, C, OI)
    return O

def dsl4(I):
    E = ofcolor(I, EIGHT)
    RE = vmirror(E)
    EO = recolor(EIGHT, RE)
    O = underpaint(I, EO)
    return O
