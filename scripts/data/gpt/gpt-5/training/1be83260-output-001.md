def dsl1(I):
    G = trim(I)
    O = downscale(G, 2)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    fg = merge(mapply(toindices, objs))
    O = subgrid(fg, I)
    return O

def dsl3(I):
    objs = objects(I, T, F, T)
    edgesets = mapply(box, objs)
    edgeidx = merge(edgesets)
    edgeobj = recolor(mostcolor(I), edgeidx)
    base = canvas(leastcolor(I), shape(I))
    O = paint(base, edgeobj)
    return O

def dsl4(I):
    G = trim(I)
    T0 = tophalf(G)
    B = vmirror(T0)
    O = vconcat(T0, B)
    return O

def dsl5(I):
    objs = objects(I, T, F, T)
    f = fork(recolor, color, backdrop)
    rect_objs = mapply(f, objs)
    rects = merge(rect_objs)
    base = canvas(leastcolor(I), shape(I))
