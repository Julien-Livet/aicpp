def dsl4(I):
    dims = shape(I)
    blank = canvas(ZERO, dims)
    objs = objects(I, T, F, T)
    cols = remove(ZERO, palette(I))
    percolor = apply(lbind(colorfilter, objs), cols)
    anchors = apply(rbind(argmax, size), percolor)
    pairs = pair(percolor, anchors)
    def per_pair(pc_a):
        cobjs = first(pc_a)
        anch = last(pc_a)
        ai = toindices(anch)
        ob = outbox(ai)
        extras = remove(anch, cobjs)
        def per_extra(o):
            ce = centerofmass(o)
            ca = centerofmass(anch)
            path = connect(ce, ca)
            hit = intersection(ob, path)
            return recolor(color(anch), hit)
        moved = merge(apply(per_extra, extras))
        return combine(anch, moved)
    parts = apply(per_pair, pairs)
    O = paint(blank, merge(parts))
    return O

def dsl5(I):
    dims = shape(I)
    blank = canvas(ZERO, dims)
    objs = objects(I, T, F, T)
    cols = remove(ZERO, palette(I))
    percolor = apply(lbind(colorfilter, objs), cols)
    anchors = apply(rbind(argmax, size), percolor)
    pairs = pair(percolor, anchors)
    isline = fork(either, hline, vline)
    def per_pair(pc_a):
        cobjs = first(pc_a)
        anch = last(pc_a)
        ai = toindices(anch)
        extras = remove(anch, cobjs)
        signpos = compose(sign, rbind(position, ai))
        is_up = compose(matcher(identity, UP), signpos)
        is_down = compose(matcher(identity, DOWN), signpos)
        is_left = compose(matcher(identity, LEFT), signpos)
        is_right = compose(matcher(identity, RIGHT), signpos)
        above = sfilter(extras, is_up)
        below = sfilter(extras, is_down)
        lefts = sfilter(extras, is_left)
        rights = sfilter(extras, is_right)
        ab_is = merge(apply(toindices, above))
        bl_is = merge(apply(toindices, below))
        lf_is = merge(apply(toindices, lefts))
        rt_is = merge(apply(toindices, rights))
        side_up = shift(ai, UP)
        side_down = shift(ai, DOWN)
        side_left = shift(ai, LEFT)
        side_right = shift(ai, RIGHT)
        cols_ab = merge(apply(vfrontier, ab_is))
        cols_bl = merge(apply(vfrontier, bl_is))
        rows_lf = merge(apply(hfrontier, lf_is))
        rows_rt = merge(apply(hfrontier, rt_is))
        tgt_h = combine(intersection(side_up, cols_ab), intersection(side_down, cols_bl))
        tgt_v = combine(intersection(side_left, rows_lf), intersection(side_right, rows_rt))
        tgt = branch(hline(anch), tgt_h, tgt_v)
        recol = recolor(color(anch), tgt)
        return combine(anch, recol)
    parts = apply(per_pair, pairs)
    O = paint(blank, merge(parts))
    return O
