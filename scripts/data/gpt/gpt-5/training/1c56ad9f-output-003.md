def dsl1(I):
    u = merge(objects(I, T, F, T))
    ui = toindices(u)
    bd = backdrop(ui)
    inside = difference(bd, ui)
    col = color(u)
    g = paint(canvas(ZERO, shape(I)), recolor(col, inside))
    cells = objects(g, T, F, T)
    ulf = compose(ulcorner, toindices)
    lrf = compose(lrcorner, toindices)
    urf = compose(urcorner, toindices)
    llf = compose(llcorner, toindices)
    line1 = fork(connect, ulf, lrf)
    line2 = fork(connect, urf, llf)
    crossf = fork(combine, line1, line2)
    crosses = merge(apply(crossf, cells))
    mask = intersection(crosses, inside)
    O = underpaint(I, recolor(col, mask))
    return O
