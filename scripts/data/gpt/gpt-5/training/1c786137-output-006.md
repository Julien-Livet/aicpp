    OBJS = objects(I, T, F, T)
    area = fork(multiply, compose(width, box), compose(height, box))
    A = argmax(OBJS, area)
    O = subgrid(delta(A), I)

def dsl2(I):
    FR = frontiers(I)
    area = fork(multiply, compose(width, box), compose(height, box))
    A = argmax(FR, area)
    G = subgrid(box(A), I)
    DIMS = subtract(shape(G), TWO_BY_TWO)
    O = crop(G, UNITY, DIMS)
    return O
