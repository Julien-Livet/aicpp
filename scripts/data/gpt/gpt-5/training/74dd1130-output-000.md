def dsl1(I):
    c0 = crop(I, ORIGIN, astuple(THREE, ONE))
    c1 = crop(I, tojvec(ONE), astuple(THREE, ONE))
    c2 = crop(I, ZERO_BY_TWO, astuple(THREE, ONE))
    r0 = rot270(c0)
    r1 = rot270(c1)
    r2 = rot270(c2)
    O = vconcat(vconcat(r0, r1), r2)
    return O

def dsl5(I):
