def dsl1(I):
    # Global trim of non-5 content, placed at top-left of original canvas
    I2 = replace(I, FIVE, ZERO)
    Tgrid = trim(I2)
    O = paint(canvas(ZERO, shape(I)), asobject(Tgrid))
    return O

def dsl2(I):
    # Keep only edge-touching objects and rectangularize them (fixed bordering binding)
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    keep = sfilter(objs, rbind(bordering, I2))
    rects = apply(fork(recolor, color, inbox), keep)
    O = paint(canvas(ZERO, shape(I)), merge(rects))
    return O

def dsl3(I):
    # Rectangularize the union of the most frequent nonzero color
    I2 = replace(I, FIVE, ZERO)
    cols = remove(ZERO, palette(I2))
    maj = valmax(cols, lbind(colorcount, I2))
    grp = compose(merge, lbind(colorfilter, objects(I2, T, F, T)))
    rect = recolor(maj, inbox(grp(maj)))
    O = paint(canvas(ZERO, shape(I)), rect)
    return O

def dsl4(I):
    # Geometric closing: fill each object's bounding box into the original (no 5s) by underpainting
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    rects = apply(fork(recolor, color, inbox), objs)
    R = merge(rects)
    O = underpaint(I2, R)
    return O

def dsl5(I):
    # Relational: keep objects aligned (share any row or column) with others; rectangularize them
    I2 = replace(I, FIVE, ZERO)
    objs = objects(I2, T, F, T)
    others = rbind(remove, objs)
    union_others = compose(merge, others)
    aligned_cols = fork(vmatching, identity, union_others)
    aligned_rows = fork(hmatching, identity, union_others)
    aligned = sfilter(objs, fork(either, aligned_cols, aligned_rows))
    rects = apply(fork(recolor, color, inbox), aligned)
    O = paint(canvas(ZERO, shape(I)), merge(rects))
    return O
