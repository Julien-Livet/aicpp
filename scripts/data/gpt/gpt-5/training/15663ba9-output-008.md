def dsl1(I):
    bg = ofcolor(I, ZERO)
    allc = asindices(I)
    fg = difference(allc, bg)
    SR = shift(bg, RIGHT)
    SL = shift(bg, LEFT)
    hasR = shift(fg, LEFT)
    hasL = shift(fg, RIGHT)
    LB = intersection(fg, intersection(SR, hasR))
    RB = intersection(fg, intersection(SL, hasL))
    O = paint(I, recolor(FOUR, LB))
    O = paint(O, recolor(TWO, RB))
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    tinds = toindices
    shL = rbind(shift, LEFT)
    shR = rbind(shift, RIGHT)
    hasR_fn = fork(intersection, tinds, compose(shL, tinds))
    hasL_fn = fork(intersection, tinds, compose(shR, tinds))
    noR_fn = fork(difference, tinds, hasR_fn)
    noL_fn = fork(difference, tinds, hasL_fn)
    LE_sets = apply(fork(intersection, noL_fn, hasR_fn), objs)
    RE_sets = apply(fork(intersection, noR_fn, hasL_fn), objs)
    LE = merge(LE_sets)
    RE = merge(RE_sets)
    O = paint(I, recolor(FOUR, LE))
    O = paint(O, recolor(TWO, RE))
    return O

def dsl3(I):
    pal = remove(ZERO, palette(I))
    bg = ofcolor(I, ZERO)
    getc = lbind(ofcolor, I)
    sets = apply(getc, pal)
    SR = shift(bg, RIGHT)
    SL = shift(bg, LEFT)
    idf = identity
    shL = rbind(shift, LEFT)
    shR = rbind(shift, RIGHT)
    hasR_fn = fork(intersection, idf, compose(shL, idf))
    hasL_fn = fork(intersection, idf, compose(shR, idf))
    interSR = lbind(intersection, SR)
    interSL = lbind(intersection, SL)
    LE_sets = apply(fork(intersection, hasR_fn, interSR), sets)
    RE_sets = apply(fork(intersection, hasL_fn, interSL), sets)
    LE = merge(LE_sets)
    RE = merge(RE_sets)
    O = paint(I, recolor(FOUR, LE))
    O = paint(O, recolor(TWO, RE))
    return O

def dsl4(I):
    edges_objs = frontiers(I)
    edges_sets = apply(toindices, edges_objs)
    edges = merge(edges_sets)
    bg = ofcolor(I, ZERO)

    SL = shift(bg, LEFT)
