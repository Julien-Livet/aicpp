def dsl1(I):
    top = tophalf(I)
    bot = bottomhalf(I)
    val = halve(colorcount(bot, ONE))
    top2 = replace(top, EIGHT, val)
    botz = canvas(ZERO, shape(bot))
    O = vconcat(top2, botz)
    return O

def dsl2(I):
    bh = bottomhalf(I)
    val = halve(colorcount(bh, ONE))
    inds8 = ofcolor(I, EIGHT)
    obj8 = toobject(inds8, I)
    base = canvas(ZERO, shape(I))
    recol = recolor(val, obj8)
    O = paint(base, recol)
    return O

def dsl3(I):
    bh = bottomhalf(I)
    val = halve(colorcount(bh, ONE))
    tmp = replace(I, ONE, ZERO)
    O = replace(tmp, EIGHT, val)
    return O

def dsl4(I):
    inds1 = ofcolor(I, ONE)
    obj1 = toobject(inds1, I)
    ul = ulcorner(obj1)
    lr = lrcorner(obj1)
    p1 = initset(ul)
    p2 = initset(lr)
    val = halve(manhattan(p1, p2))
    inds8 = ofcolor(I, EIGHT)
    obj8 = toobject(inds8, I)
    base = canvas(ZERO, shape(I))
    recol = recolor(val, obj8)
    O = paint(base, recol)
    return O

def dsl5(I):
    bh = bottomhalf(I)
    inds1 = ofcolor(bh, ONE)
    cnt = size(inds1)
    val = halve(cnt)
    inds8 = ofcolor(I, EIGHT)
    obj8 = toobject(inds8, I)
    base = canvas(ZERO, shape(I))
    recol = recolor(val, obj8)
    O = underpaint(base, recol)
    return O
