    E = ofcolor(I, EIGHT)
    OA = objects(I, T, F, T)
    O8 = colorfilter(OA, EIGHT)
    ON = difference(OA, O8)
    U = merge(ON)
    rE = rightmost(E)
    jt = increment(rE)
    Vt = tojvec(jt)
    js = leftmost(U)
    Vs = tojvec(js)
    Off = subtract(Vt, Vs)
    M = shift(U, Off)
    B = cover(I, U)
    O = underpaint(B, M)
    return O
