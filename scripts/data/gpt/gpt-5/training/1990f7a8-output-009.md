def dsl1(I):
    # Global adaptive downscale to <=7x7, center by mass, carve central cross
    S = compress(subgrid(ofcolor(I, TWO), I))
    H = height(S); W = width(S)
    FH = divide(add(H, SIX), SEVEN); FW = divide(add(W, SIX), SEVEN)
    F = branch(greater(FH, FW), FH, FW)
    D = downscale(S, F); E = compress(D)
    P = ofcolor(E, TWO)
    CEN = centerofmass(P); MID = astuple(THREE, THREE)
    OFF = subtract(MID, CEN)
    PS = shift(P, OFF)
    G7 = canvas(ZERO, astuple(SEVEN, SEVEN))
    G = fill(G7, TWO, PS)
    R0 = astuple(THREE, ZERO); R6 = astuple(THREE, SIX)
    C0 = astuple(ZERO, THREE); C6 = astuple(SIX, THREE)
    ROW = connect(R0, R6); COL = connect(C0, C6)
    O = fill(G, ZERO, combine(ROW, COL))
    return O
