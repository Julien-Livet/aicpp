def dsl1(I):
    # Global: remove dominant nonzero color, compress frontiers, then trim
    mc = argmax(remove(ZERO, palette(I)), lbind(colorcount, I))
    O = trim(compress(replace(I, mc, ZERO)))
    return O

def dsl2(I):
    # Object extraction: interior of bbox of the largest border-touching object of the dominant nonzero color
    mc = argmax(remove(ZERO, palette(I)), lbind(colorcount, I))
    objs = colorfilter(objects(I, T, F, T), mc)
    bord = sfilter(objs, rbind(bordering, I))
    big = argmax(bord, size)
    inner = delta(box(backdrop(big)))
    O = subgrid(inner, I)
    return O

def dsl3(I):
    # Color filtering: remove the dominant nonzero color and trim
    mc = argmax(remove(ZERO, palette(I)), lbind(colorcount, I))
    u = toobject(ofcolor(I, mc), I)
    core = cover(I, u)
    O = trim(core)
    return O

def dsl4(I):
    # Geometric: union of border-touching dominant-color objects, crop away 1-pixel frame, then trim
    mc = argmax(remove(ZERO, palette(I)), lbind(colorcount, I))
    os = colorfilter(objects(I, T, F, T), mc)
    bord = sfilter(os, rbind(bordering, I))
    frame = merge(bord)
    bbgrid = subgrid(backdrop(frame), I)
    core = crop(bbgrid, UNITY, subtract(shape(bbgrid), TWO_BY_TWO))
    O = trim(core)
    return O

def dsl5(I):
    # Relational: intersection of backdrops of the two most common nonzero colors
    colors = remove(ZERO, palette(I))
    mc = argmax(colors, lbind(colorcount, I))
    rest = remove(mc, colors)
    sc = argmax(rest, lbind(colorcount, I))
    p1 = backdrop(ofcolor(I, mc))
    p2 = backdrop(ofcolor(I, sc))
    inter = intersection(p1, p2)
    O = subgrid(inter, I)
    return O
