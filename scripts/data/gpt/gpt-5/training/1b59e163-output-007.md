def dsl1(I):
    # Global: union with horizontal mirror after removing 5s
    J = replace(I, FIVE, FOUR)
    fg = difference(asindices(J), ofcolor(J, FOUR))
    base = canvas(FOUR, shape(J))
    G = paint(base, toobject(fg, J))
    M = hmirror(G)
    O = underpaint(G, asobject(M))
    return O

def dsl2(I):
    # Object extraction: mirror the leftmost object vertically
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    s = argmin(objs, leftmost)
    base = canvas(FOUR, shape(J))
    G = paint(base, s)
    M = vmirror(G)
    O = underpaint(G, asobject(M))
    return O

def dsl3(I):
    # Color filtering: move most-common non-bg color to central horizontal frontier
    J = replace(I, FIVE, FOUR)
    objs = objects(J, T, F, T)
    cols = apply(color, objs)
    cstar = mostcommon(totuple(cols))
    foc = colorfilter(objs, cstar)
    merged = merge(foc)
    src = toobject(merged, J)
    allidx = asindices(J)
    dest = hfrontier(center(allidx))
    v = gravitate(merged, dest)
    O = move(canvas(FOUR, shape(J)), src, v)
    return O

def dsl4(I):
    # Geometric: overlay with 180-rotation and counterdiagonal mirror
    J = replace(I, FIVE, FOUR)
    fg = difference(asindices(J), ofcolor(J, FOUR))
    base = canvas(FOUR, shape(J))
    G = paint(base, toobject(fg, J))
    R = rot180(G)
    C = cmirror(G)
    O1 = underpaint(G, asobject(R))
    O = underpaint(O1, asobject(C))
    return O

def dsl5(I):
    # Relational: gravitate foreground to the cross through its own center
    J = replace(I, FIVE, FOUR)
    allidx = asindices(J)
    fg = difference(allidx, ofcolor(J, FOUR))
    src = toobject(fg, J)
    c = center(fg)
    cross = combine(hfrontier(c), vfrontier(c))
    v = gravitate(fg, cross)
    O = move(canvas(FOUR, shape(J)), src, v)
    return O
