def dsl1(I):
    # Global: crop to foreground then compress to remove internal frontiers
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    O = compress(G)
    return O

def dsl2(I):
    # Object extraction: crop to foreground then fill all zeros with the band filler (leastcolor)
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    O = fill(G, leastcolor(G), ofcolor(G, ZERO))
    return O

def dsl3(I):
    # Color filtering: repaint foreground on a canvas filled with the band filler to eliminate internal zeros
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    base = canvas(leastcolor(G), shape(G))
    O = paint(base, asobject(G))
    return O

def dsl4(I):
    # Geometric: crop then mirror the left half to the right to fill the central gap
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    L = lefthalf(G)
    R = vmirror(L)
    O = hconcat(L, R)
    return O

def dsl5(I):
    # Relational: fill the background inside the cropped bounding box using delta/backdrop
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    bg = delta(asobject(G))
    O = fill(G, leastcolor(G), bg)
    return O
