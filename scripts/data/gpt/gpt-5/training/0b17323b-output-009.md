def dsl1(I):
    rI = dmirror(I)
    ones = ofcolor(rI, ONE)
    ul = ulcorner(ones)
    lr = lrcorner(ones)
    seg = connect(ul, lr)
    k = decrement(size(ones))
    per = divide(decrement(size(seg)), k)
    dir = sign(subtract(lr, ul))
    step = multiply(dir, per)
    start = add(lr, step)
    path = intersection(shoot(start, step), asindices(rI))
    rO = underpaint(rI, recolor(TWO, path))
    O = dmirror(rO)
    return O
