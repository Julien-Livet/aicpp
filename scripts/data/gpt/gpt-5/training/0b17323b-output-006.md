def dsl1(I):
    ones = ofcolor(I, ONE)
    ul = ulcorner(ones)
    lr = lrcorner(ones)
    k = decrement(size(ones))
    step = divide(subtract(lr, ul), k)
    seq = intersection(shoot(ul, step), asindices(I))
    future = difference(seq, ones)
    O = underpaint(I, recolor(TWO, future))
    return O

def dsl2(I):
    ones = ofcolor(I, ONE)
    obj = toobject(ones, I)
    d = vperiod(obj)
    dir = gravitate(initset(ulcorner(ones)), initset(lrcorner(ones)))
    vec = multiply(dir, multiply(UNITY, d))
    start = add(lrcorner(ones), vec)
    path = intersection(shoot(start, vec), asindices(I))
    O = underpaint(I, recolor(TWO, path))
    return O

def dsl3(I):
    rI = rot90(I)

def dsl4(I):

def dsl5(I):

    ones = ofcolor(I, ONE)
    ul = ulcorner(ones)
    lr = lrcorner(ones)
