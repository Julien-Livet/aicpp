def dsl1(I):
    # Global: filled bounding-rectangle of all twos, ceil-downscale to fit 7x7, center, carve zero cross
    S = compress(subgrid(ofcolor(I, TWO), I))
    R = inbox(ofcolor(S, TWO))
    G = subgrid(R, S)
    H = height(G)
    W = width(G)
    FHc = divide(add(H, SIX), SEVEN)
    FWc = divide(add(W, SIX), SEVEN)
    F = branch(greater(FHc, FWc), FHc, FWc)
    D = downscale(G, F)
    E = compress(D)
    OO = recolor(TWO, toobject(ofcolor(E, TWO), E))
    CEN = center(OO)
    MID = astuple(THREE, THREE)
    OFF = subtract(MID, CEN)
    SO = shift(OO, OFF)
    G7 = canvas(ZERO, astuple(SEVEN, SEVEN))
    P = paint(G7, SO)
    R0 = astuple(THREE, ZERO)
    R6 = astuple(THREE, SIX)
    C0 = astuple(ZERO, THREE)
    C6 = astuple(SIX, THREE)
    ROW = connect(R0, R6)
    COL = connect(C0, C6)
    CROSS = combine(ROW, COL)
    O = fill(P, ZERO, CROSS)
    return O
