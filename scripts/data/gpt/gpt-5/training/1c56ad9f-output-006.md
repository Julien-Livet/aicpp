def dsl1(I):
    u = merge(objects(I, T, F, T))
    ui = toindices(u)
    bd = backdrop(ui)
    inside = difference(bd, ui)
    col = color(u)
    stepw = hperiod(u)
    steph = vperiod(u)
    w = width(bd)
    h = height(bd)
    ks = interval(0, w, stepw)
    iset = interval(0, h, steph)
    ul = ulcorner(inside)
    hstarts = apply(lbind(add, ul), apply(toivec, iset))
    vstarts = apply(lbind(add, ul), apply(tojvec, ks))
    hlines = merge(apply(rbind(shoot, RIGHT), hstarts))
    vlines = merge(apply(rbind(shoot, DOWN), vstarts))
    mask = intersection(combine(hlines, vlines), inside)
    O = underpaint(I, recolor(col, mask))
    return O
