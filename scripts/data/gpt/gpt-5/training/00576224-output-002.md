def dsl1(I):
    V = vconcat(vconcat(I, vmirror(I)), I)
    O = hconcat(hconcat(V, V), V)
    return O

def dsl2(I):
    dims = multiply(shape(I), THREE)
    base = canvas(ZERO, dims)
    obj = asobject(I)
    mobj = asobject(vmirror(I))
    h0 = ORIGIN
    h1 = ZERO_BY_TWO
    h2 = add(ZERO_BY_TWO, ZERO_BY_TWO)
    v0 = ORIGIN
    v1 = TWO_BY_ZERO
    v2 = add(TWO_BY_ZERO, TWO_BY_ZERO)
    top = paint(paint(paint(base, shift(obj, add(v0, h0))), shift(obj, add(v0, h1))), shift(obj, add(v0, h2)))
    mid = paint(paint(paint(top, shift(mobj, add(v1, h0))), shift(mobj, add(v1, h1))), shift(mobj, add(v1, h2)))
    bot = paint(paint(paint(mid, shift(obj, add(v2, h0))), shift(obj, add(v2, h1))), shift(obj, add(v2, h2)))
    O = bot
    return O

def dsl3(I):
    C = vsplit(I, TWO)
    L = first(C)
    R = last(C)
    P = hconcat(L, R)
    Q = hconcat(R, L)
    H = hconcat(hconcat(P, P), P)
    M = hconcat(hconcat(Q, Q), Q)
    O = vconcat(vconcat(H, M), H)
    return O

def dsl4(I):
    M = hmirror(rot180(I))
    H = hconcat(hconcat(I, I), I)
    N = hconcat(hconcat(M, M), M)
    O = vconcat(vconcat(H, N), H)
    return O

def dsl5(I):
    M = dmirror(hmirror(dmirror(I)))
    H = hconcat(hconcat(I, I), I)
    N = hconcat(hconcat(M, M), M)
    O = vconcat(vconcat(H, N), H)
    return O
