def dsl1(I):
    # Global: crop the foreground, then downscale and remove remaining background by filling with modal FG color
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    M = mostcolor(fg)
    D = downscale(G, TWO)
    O = replace(D, ZERO, M)
    return O

def dsl2(I):
    # Object extraction: fill each object's bounding box with its color on a modal-filled crop
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    objs = objects(G, T, F, T)
    modal = mostcolor(fg)
    base = replace(G, ZERO, modal)
    B = totuple(apply(box, objs))
    C = totuple(apply(color, objs))
    colored_boxes = mpapply(recolor, C, B)
    M = merge(colored_boxes)
    O = paint(base, M)
    return O

def dsl3(I):
    # Color filtering: emphasize least-common color via backdrops on a modal base
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    objs = objects(G, T, F, T)
    modal = mostcolor(fg)
    base = replace(G, ZERO, modal)
    rare = leastcolor(fg)
    rareobjs = colorfilter(objs, rare)
    patches = apply(backdrop, rareobjs)
    P = merge(patches)
    O = fill(base, rare, P)
    return O

def dsl4(I):
    # Geometric: accentuate frontiers (boundaries) on a modal-filled crop using the least-common color
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    modal = mostcolor(fg)
    base = replace(G, ZERO, modal)
    edges = merge(frontiers(G))
    E = toindices(edges)
    O = fill(base, leastcolor(fg), E)
    return O

def dsl5(I):
    # Relational: fill interior gaps (delta of FG) with modal color, then draw FG outline with least-common color
    fg = merge(objects(I, T, F, T))
    G = subgrid(fg, I)
    holes = delta(fg)
    B = fill(G, mostcolor(fg), holes)
    outline = box(fg)
    O = fill(B, leastcolor(fg), outline)
    return O
