def dsl1(I):
    # Global periodicity: use hperiod/vperiod from the two seeds; propagate forward only
    OS = objects(I, T, F, T)
    o1 = first(OS); o2 = last(OS)
    c1 = color(o1); c2 = color(o2)
    l1 = first(toindices(o1)); l2 = first(toindices(o2))
    P = combine(toindices(o1), toindices(o2))
    O2 = toobject(P, I)
    vr = vperiod(O2); hr = hperiod(O2)
    H = height(I); W = width(I)
    stepR = branch(equality(vr, ZERO), ONE, vr)
    stepC = branch(equality(hr, ZERO), ONE, hr)
    offsR = apply(toivec, interval(ZERO, H, stepR))
    offsC = apply(tojvec, interval(ZERO, W, stepC))
    HR1 = mapply(lbind(shift, hfrontier(l1)), offsR)
    HR2 = mapply(lbind(shift, hfrontier(l2)), offsR)
    HOBJ = combine(recolor(c1, HR1), recolor(c2, HR2))
    VC1 = mapply(lbind(shift, vfrontier(l1)), offsC)
    VC2 = mapply(lbind(shift, vfrontier(l2)), offsC)
    VOBJ = combine(recolor(c1, VC1), recolor(c2, VC2))
    hsel = either(equality(hr, ZERO), flip(either(equality(vr, ZERO), greater(vr, hr))))
    O = paint(canvas(ZERO, shape(I)), branch(hsel, HOBJ, VOBJ))
    return O

    offsR = apply(toivec, interval(ZERO, H, stepR))
    offsC = apply(tojvec, interval(ZERO, W, stepC))
    BR1 = recolor(c1, hfrontier(l1))
    BR2 = recolor(c2, hfrontier(l2))
    HR1set = prapply(shift, initset(BR1), offsR)
    HR2set = prapply(shift, initset(BR2), offsR)

    HR2 = merge(HR2set)
    HOBJ = combine(HR1, HR2)
    BC1 = recolor(c1, vfrontier(l1))
    BC2 = recolor(c2, vfrontier(l2))
    VC1set = prapply(shift, initset(BC1), offsC)
    VC2set = prapply(shift, initset(BC2), offsC)
    VC1 = merge(VC1set)
    VC2 = merge(VC2set)
    VOBJ = combine(VC1, VC2)
