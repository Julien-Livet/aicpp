def dsl1(I):
    # Global connectors: mark 5s on any straight pure 0-to-4 or 4-to-4 path.
    p0 = ofcolor(I, ZERO)
    p4 = ofcolor(I, FOUR)
    p5 = ofcolor(I, FIVE)
    linear = fork(either, hline, vline)
    # 0 <-> 4
    paths04 = prapply(connect, p0, p4)
    allowed04 = merge(insert(p0, insert(p5, initset(p4))))
    pure04 = compose(lbind(equality, ZERO), compose(size, rbind(difference, allowed04)))
    good04 = mfilter(paths04, fork(both, pure04, linear))
    # 4 <-> 4
    paths44 = prapply(connect, p4, p4)
    allowed44 = merge(insert(p4, initset(p5)))
    pure44 = compose(lbind(equality, ZERO), compose(size, rbind(difference, allowed44)))
    good44 = mfilter(paths44, fork(both, pure44, linear))
    # union of good lines, keep the 5s on them
    lines = merge(combine(good04, good44))
    mask = intersection(lines, p5)
    O = fill(I, TWO, mask)
    return O

def dsl2(I):
    # Object extraction: expand 5-components intersecting pure straight 0-to-4 paths.
    p0 = ofcolor(I, ZERO)
    p4 = ofcolor(I, FOUR)
    p5 = ofcolor(I, FIVE)
    linear = fork(either, hline, vline)
    # 0 <-> 4 paths (pure in {0,4,5})
    paths04 = prapply(connect, p0, p4)
    allowed04 = merge(insert(p0, insert(p5, initset(p4))))
    pure04 = compose(lbind(equality, ZERO), compose(size, rbind(difference, allowed04)))
    good04 = mfilter(paths04, fork(both, pure04, linear))
    line04mask = merge(good04)
    seed = intersection(line04mask, p5)
    # select 5-objects that intersect the seed
    objs = objects(I, T, F, T)
    fiveobjs = colorfilter(objs, FIVE)
    inter_with_seed = compose(rbind(intersection, seed), toindices)
    has_seed = compose(positive, compose(size, inter_with_seed))
    chosen = mfilter(fiveobjs, has_seed)
    mask = merge(apply(toindices, chosen))
    O = fill(I, TWO, mask)
    return O

def dsl3(I):
    # Local color filtering: 5s flanked by 4s or by a 0 and a 4 (any order), horizontally or vertically.
    p0 = ofcolor(I, ZERO)
    p4 = ofcolor(I, FOUR)
    p5 = ofcolor(I, FIVE)
    # 4-5-4
    midH44 = intersection(intersection(p5, shift(p4, LEFT)), shift(p4, RIGHT))
    midV44 = intersection(intersection(p5, shift(p4, UP)), shift(p4, DOWN))
    # 0-5-4 and 4-5-0
    midH04 = intersection(intersection(p5, shift(p0, LEFT)), shift(p4, RIGHT))
    midH40 = intersection(intersection(p5, shift(p4, LEFT)), shift(p0, RIGHT))
    midV04 = intersection(intersection(p5, shift(p0, UP)), shift(p4, DOWN))
    midV40 = intersection(intersection(p5, shift(p4, UP)), shift(p0, DOWN))
    mask = combine(combine(midH44, midV44), combine(combine(midH04, midH40), combine(midV04, midV40)))
    O = fill(I, TWO, mask)
    return O
