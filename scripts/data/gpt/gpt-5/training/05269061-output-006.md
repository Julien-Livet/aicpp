def dsl1(I):
    COLORS = remove(ZERO, palette(I))
    INDC = lbind(ofcolor, I)
    SUP = rbind(shoot, UP_RIGHT)
    SDL = rbind(shoot, DOWN_LEFT)
    LUP = compose(merge, compose(lbind(apply, SUP), INDC))
    LDL = compose(merge, compose(lbind(apply, SDL), INDC))
    DIAGC = fork(combine, LUP, LDL)
    OBJF = fork(recolor, identity, DIAGC)
    BASE = merge(apply(OBJF, COLORS))
    H = height(I)
    W = width(I)
    RO = interval(0, H, THREE)
    CO = interval(0, W, THREE)
    DV = apply(toivec, RO)
    DH = apply(tojvec, CO)
    NDV = apply(invert, DV)
    NDH = apply(invert, DH)
    VOFFS = combine(DV, NDV)
    HOFFS = combine(DH, NDH)
    OFFS = prapply(add, VOFFS, HOFFS)
    SH = apply(lbind(shift, BASE), OFFS)
    BIG = merge(SH)
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O
