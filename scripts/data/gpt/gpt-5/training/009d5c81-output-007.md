def dsl1(I):
    # Global mapping: zero out ones, swap 8 with computed value
    bh = bottomhalf(I)
    val = divide(size(ofcolor(bh, ONE)), TWO)
    base = replace(I, ONE, ZERO)
    O = switch(base, EIGHT, val)
    return O

def dsl2(I):
    # Object extraction: recolor 8-objects from fgpartition and paint on blank canvas
    bh = bottomhalf(I)
    f = compose(size, rbind(ofcolor, ONE))
    val = halve(f(bh))
    parts = fgpartition(I)
    eobjs = colorfilter(parts, EIGHT)
    recol = merge(apply(lbind(recolor, val), eobjs))
    O = paint(canvas(ZERO, shape(I)), recol)
    return O

def dsl3(I):
    # Color filtering: mask-based fill of 8-indices onto zero canvas
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    mask8 = ofcolor(I, EIGHT)
    O = fill(canvas(ZERO, shape(I)), val, mask8)
    return O

def dsl4(I):
    # Geometric/structural: remove 8-object then repaint it recolored
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    inds8 = ofcolor(I, EIGHT)
    obj8 = toobject(inds8, I)
    base = cover(replace(I, ONE, ZERO), obj8)
    O = paint(base, recolor(val, obj8))
    return O

def dsl5(I):
    # Relational/mask-based: underfill 8-mask with computed value on zero canvas
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    mask8 = ofcolor(I, EIGHT)
    O = underfill(canvas(ZERO, shape(I)), val, mask8)
    return O
