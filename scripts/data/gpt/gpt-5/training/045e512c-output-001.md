def dsl1(I):
    nz = difference(asindices(I), ofcolor(I, ZERO))
    sub = subgrid(nz, I)
    base = toobject(asindices(sub), sub)
    wI = width(I)
    hI = height(I)
    ws = width(sub)
    hs = height(sub)
    offsH = interval(0, wI, ws)
    vecsH = apply(tojvec, offsH)
    tileH = mapply(lbind(shift, base), vecsH)
    offsV = interval(0, hI, hs)
    vecsV = apply(toivec, offsV)
    tileHV = mapply(lbind(shift, tileH), vecsV)
    O = paint(canvas(ZERO, shape(I)), tileHV)
    return O

def dsl2(I):
    objs = objects(I, T, F, T)
    biggest = argmax(objs, size)
    base = normalize(biggest)
    wI = width(I)
    step = width(base)
    offsH = interval(0, wI, step)
    vecsH = apply(tojvec, offsH)
    tile = mapply(lbind(shift, base), vecsH)
    O = paint(canvas(ZERO, shape(I)), tile)
    return O

def dsl3(I):
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    nz = difference(asindices(I), ofcolor(I, ZERO))
    obj = recolor(c, nz)
    O = paint(canvas(ZERO, shape(I)), obj)
    return O

def dsl4(I):
    A = I
    H = hmirror(I)
    V = vmirror(I)
    D = dmirror(I)
    C = cmirror(I)
    nA = difference(asindices(A), ofcolor(A, ZERO))
    oA = toobject(nA, A)
    nH = difference(asindices(H), ofcolor(H, ZERO))
    oH = toobject(nH, H)
    nV = difference(asindices(V), ofcolor(V, ZERO))
    oV = toobject(nV, V)
    nD = difference(asindices(D), ofcolor(D, ZERO))
    oD = toobject(nD, D)
    nC = difference(asindices(C), ofcolor(C, ZERO))
    oC = toobject(nC, C)
    u1 = combine(oA, oH)
    u2 = combine(u1, oV)
    u3 = combine(u2, oD)
    u4 = combine(u3, oC)
    O = paint(canvas(ZERO, shape(I)), u4)
    return O
