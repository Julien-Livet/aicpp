def dsl1(I):
    MIR = cmirror(I)
    O = rot90(MIR)
    return O

def dsl2(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    O = subgrid(OBJ, I)
    return O

def dsl3(I):
    FG = fgpartition(I)
    ALL = merge(FG)
    C = leastcolor(ALL)
    O = replace(I, ZERO, C)
    return O

def dsl4(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    H = hmirror(OBJ)
    V = vmirror(OBJ)
    S = combine(H, V)
    O = underpaint(I, S)
    return O

def dsl5(I):
    FG = fgpartition(I)
    OBJ = merge(FG)
    B = box(OBJ)
    C = mostcolor(OBJ)
    OUT = recolor(C, B)
    S = shape(I)
    Z = canvas(ZERO, S)
    O = paint(Z, OUT)
    return O
