def dsl1(I):
    # Global: for each nonzero color, fill the backdrop of all its pixels (one rectangle per color)
    pal = remove(ZERO, palette(I))
    color2patch = compose(backdrop, lbind(ofcolor, I))
    parts = apply(fork(recolor, identity, color2patch), pal)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl2(I):
    # Object extraction: keep only non-bordering objects and replace each by its filled backdrop
    objs = objects(I, T, F, T)
    nonborder = sfilter(objs, compose(flip, rbind(bordering, I)))
    parts = apply(fork(recolor, color, backdrop), nonborder)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl3(I):
    # Color filtering: recolor every nonzero cell to the most common color
    host = mostcolor(I)
    nz = difference(asindices(I), ofcolor(I, ZERO))
    O = fill(canvas(ZERO, shape(I)), host, nz)
    return O

def dsl4(I):
    # Geometric: keep cells that agree with both their horizontal and vertical mirrors
    symh = cellwise(I, hmirror(I), ZERO)
    O = cellwise(symh, vmirror(I), ZERO)
    return O

def dsl5(I):
    # Relational: keep only border-touching objects and replace each by its filled backdrop
    objs = objects(I, T, F, T)
    borders = sfilter(objs, rbind(bordering, I))
    parts = apply(fork(recolor, color, backdrop), borders)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O
