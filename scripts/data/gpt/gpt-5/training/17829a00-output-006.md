def dsl1(I):
    # Half-wise object gravitation: move all non-background objects in the top/bottom halves to the nearest border row, recolor by that border color, keep full moved shapes except the border rows, then redraw border rows
    OB = objects(I, T, F, T)
    BG = mostcolor(I)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    H = height(I)
    Half = halve(H)
    TT = interval(ZERO, Half, ONE)
    TB = interval(Half, H, ONE)
    FT = mapply(hfrontier, apply(toivec, TT))
    FB = mapply(hfrontier, apply(toivec, TB))
    R0 = hfrontier(astuple(ZERO, ZERO))
    Hm1 = subtract(H, ONE)
    Rb = hfrontier(astuple(Hm1, ZERO))
    condT = compose(positive, compose(size, compose(rbind(intersection, FT), toindices)))
    condB = compose(positive, compose(size, compose(rbind(intersection, FB), toindices)))
    TO = sfilter(OB, condT)
    BO = sfilter(OB, condB)
    ST = fork(shift, identity, rbind(gravitate, R0))
    SB = fork(shift, identity, rbind(gravitate, Rb))
    Tmov = merge(apply(ST, TO))
    Bmov = merge(apply(SB, BO))
    Tproj = difference(toindices(Tmov), R0)
    Bproj = difference(toindices(Bmov), Rb)
    C = canvas(BG, shape(I))
    O0 = fill(C, cT, Tproj)
    O1 = fill(O0, cB, Bproj)
    O2 = fill(O1, cT, R0)
    O = fill(O2, cB, Rb)
    return O

def dsl3(I):
    # Rotational symmetry with half-wise non-background packing: solve "compress-to-top" on I and on rot180(I), then rotate back and merge
    BG = mostcolor(I)
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    H = height(I)
    Half = halve(H)
    TT = interval(ZERO, Half, ONE)
    FT = mapply(hfrontier, apply(toivec, TT))
    # Top-only on I
    ALL1 = asindices(I)
    NB1 = difference(ALL1, ofcolor(I, BG))
    Tcells1 = intersection(NB1, FT)
    Tobj1 = toobject(Tcells1, I)
    R01 = hfrontier(astuple(ZERO, ZERO))
    ST1 = fork(shift, identity, rbind(gravitate, R01))
    M1 = merge(apply(ST1, initset(Tobj1)))
    P1 = difference(toindices(M1), R01)
    Ct = canvas(BG, shape(I))
    Gt0 = fill(Ct, cT, P1)
    Gt = fill(Gt0, cT, R01)
    # Top-only on rot180(I), then rotate back
    V = rot180(I)
    HV = height(V)
    HalfV = halve(HV)
    TTV = interval(ZERO, HalfV, ONE)
    FTV = mapply(hfrontier, apply(toivec, TTV))
    ALL2 = asindices(V)
    NB2 = difference(ALL2, ofcolor(V, mostcolor(V)))
    Tcells2 = intersection(NB2, FTV)
    Tobj2 = toobject(Tcells2, V)
    R02 = hfrontier(astuple(ZERO, ZERO))
    ST2 = fork(shift, identity, rbind(gravitate, R02))
    M2 = merge(apply(ST2, initset(Tobj2)))
    P2 = difference(toindices(M2), R02)
    Cv = canvas(mostcolor(V), shape(V))
    Gv0 = fill(Cv, cB, P2)
    Gv = fill(Gv0, cB, R02)
    Gb = rot180(Gv)
    # Merge layers by underpainting over a background canvas
    O0 = canvas(BG, shape(I))
    O1 = underpaint(O0, asobject(Gt))
    O = underpaint(O1, asobject(Gb))
    return O
