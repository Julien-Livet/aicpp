    NZ = difference(asindices(I), ofcolor(I, ZERO))
    TILE = subgrid(NZ, I)
    TH = height(TILE)
    TW = width(TILE)
    TOP = crop(TILE, ORIGIN, astuple(ONE, TW))
    SEEDS = fgpartition(TOP)
    SHIFT0 = ulcorner(NZ)
    ABS = apply(rbind(shift, SHIFT0), SEEDS)
    S_UP = compose(rbind(shoot, UP_RIGHT), compose(first, toindices))
    S_DL = compose(rbind(shoot, DOWN_LEFT), compose(first, toindices))
    LINES = apply(fork(recolor, color, fork(combine, S_UP, S_DL)), ABS)
    BIG = merge(LINES)
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O

    SHIFT0 = ulcorner(NZ)

    H = height(I)
    W = width(I)
    RO = interval(0, H, ONE)
    CO = interval(0, W, ONE)
    RV = apply(toivec, RO)
    CV = apply(tojvec, RO)

    HOFF = apply(tojvec, CO)
    OFFS = prapply(add, OFFD, HOFF)
    SH = apply(lbind(shift, BASE), OFFS)
    BIG = merge(SH)
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O
