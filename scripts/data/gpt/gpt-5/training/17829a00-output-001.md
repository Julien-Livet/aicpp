def dsl1(I):
    # Relational per-object gravitation of non-border top/bottom colors toward their border bands
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    TO = colorfilter(OB, cT)
    BO = colorfilter(OB, cB)
    Tbor = sfilter(TO, rbind(bordering, I))
    Bbor = sfilter(BO, rbind(bordering, I))
    Tnon = difference(TO, Tbor)
    Bnon = difference(BO, Bbor)
    ST = fork(shift, identity, rbind(gravitate, TOP))
    SB = fork(shift, identity, rbind(gravitate, BOT))
    Tmov = apply(ST, Tnon)
    Bmov = apply(SB, Bnon)
    G = cover(cover(I, merge(Tnon)), merge(Bnon))
    O = paint(paint(G, merge(Tmov)), merge(Bmov))
    return O

def dsl2(I):
    # Global move of union of non-border top/bottom colors toward their border bands
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    pT = ofcolor(I, cT)
    pB = ofcolor(I, cB)
    pNonT = difference(pT, toindices(TOP))
    pNonB = difference(pB, toindices(BOT))
    oNonT = toobject(pNonT, I)
    oNonB = toobject(pNonB, I)
    dT = gravitate(pNonT, TOP)
    dB = gravitate(pNonB, BOT)
    G = cover(cover(I, oNonT), oNonB)
    O = move(move(G, oNonT, dT), oNonB, dB)
    return O

def dsl3(I):
    # Color filtering to keep only border colors, then gravitate unions toward their borders
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    BG = mostcolor(I)
    C = canvas(BG, shape(I))
    J = fill(fill(C, cT, ofcolor(I, cT)), cB, ofcolor(I, cB))
    pNonT = difference(ofcolor(J, cT), toindices(TOP))
    pNonB = difference(ofcolor(J, cB), toindices(BOT))
    oNonT = toobject(pNonT, J)
    oNonB = toobject(pNonB, J)
    dT = gravitate(pNonT, TOP)
    dB = gravitate(pNonB, BOT)
    G = cover(cover(J, oNonT), oNonB)
    O = move(move(G, oNonT, dT), oNonB, dB)
    return O

def dsl4(I):
    # Geometric: duplicate the top/bottom border bands one step inward
    OB = objects(I, T, F, T)
    BOR = sfilter(OB, rbind(bordering, I))
    TOP = argmin(BOR, uppermost)
    BOT = argmax(BOR, lowermost)
    cT = color(TOP)
    cB = color(BOT)
    O1 = fill(I, cT, shift(toindices(TOP), DOWN))
    O = fill(O1, cB, shift(toindices(BOT), UP))
    return O
