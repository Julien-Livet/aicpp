def dsl1(I):
    # Global: replace every object by its filled bounding box (backdrop) with its own color
    objs = objects(I, T, F, T)
    parts = apply(fork(recolor, color, backdrop), objs)
    union = merge(parts)
    O = paint(canvas(ZERO, shape(I)), union)
    return O

def dsl2(I):
    # Object extraction: keep only the most common color, fill its objects' backdrops
    host = mostcolor(I)
    objs = objects(I, T, F, T)
    hostobjs = colorfilter(objs, host)
    parts = apply(lbind(recolor, host), apply(backdrop, hostobjs))
    union = merge(parts)
    O = paint(canvas(ZERO, shape(I)), union)
    return O

def dsl3(I):
    # Color filtering: remove color 5 everywhere
    O = replace(I, FIVE, ZERO)
    return O

def dsl4(I):
    # Geometric: keep cells that match their horizontal mirror (enforce horizontal symmetry)
    O = cellwise(I, hmirror(I), ZERO)
    return O

def dsl5(I):
    # Relational: keep only border-touching objects and replace each by its filled backdrop
    objs = objects(I, T, F, T)
    borderobjs = mfilter(objs, rbind(bordering, I))
    parts = apply(fork(recolor, color, backdrop), borderobjs)
    union = merge(parts)
    O = paint(canvas(ZERO, shape(I)), union)
    return O
