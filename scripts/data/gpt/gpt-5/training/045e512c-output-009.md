def dsl1(I):
    # Global: extract and paint all frontiers (edges between colors)
    allfront = merge(frontiers(I))
    O = paint(canvas(ZERO, shape(I)), allfront)
    return O

def dsl2(I):
    # Object extraction: fill each object's bounding box (backdrop) with its own color
    objs = objects(I, T, F, T)
    mk = fork(recolor, color, backdrop)
    fills = apply(mk, objs)
    O = paint(canvas(ZERO, shape(I)), merge(fills))
    return O

def dsl3(I):
    # Color filtering: expand the most frequent nonzero color to its 4-neighborhood
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    starts = ofcolor(I, c)
    nbs = combine(combine(shift(starts, UP), shift(starts, DOWN)),
                  combine(shift(starts, LEFT), shift(starts, RIGHT)))
    O = paint(canvas(ZERO, shape(I)), recolor(c, nbs))
    return O

def dsl4(I):
    # Geometric: diagonal symmetrization of the cropped foreground placed back
    nz = difference(asindices(I), ofcolor(I, ZERO))
    sub = subgrid(nz, I)
    base = asobject(sub)
    o1 = dmirror(base)
    o2 = cmirror(base)
    quad = combine(combine(base, o1), o2)
    placed = shift(quad, ulcorner(nz))
    O = paint(canvas(ZERO, shape(I)), placed)
    return O

def dsl5(I):
    # Relational: connect centers of all objects with lines colored by dominant nonzero color
    objs = objects(I, T, F, T)
    cents = apply(center, objs)
    segs = prapply(connect, cents, cents)
    lines = merge(segs)
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    O = paint(canvas(ZERO, shape(I)), recolor(c, lines))
    return O
