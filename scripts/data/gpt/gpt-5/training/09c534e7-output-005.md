def dsl1(I):
    H = hmirror(I)
    V = vmirror(I)
    mask = ofcolor(I, ONE)
    ObjH = toobject(mask, H)
    ObjV = toobject(mask, V)
    Obj = combine(ObjH, ObjV)
    O = paint(I, Obj)
    return O

def dsl2(I):
    FG = fgpartition(I)
    Seeds = difference(FG, colorfilter(FG, ONE))
    ones = ofcolor(I, ONE)
    maskf = compose(lbind(intersection, ones), compose(backdrop, toindices))
    recolorf = fork(recolor, color, maskf)
    Paints = mapply(recolorf, Seeds)
    O = paint(I, Paints)
    return O

def dsl4(I):
    FG = fgpartition(I)
    Seeds = difference(FG, colorfilter(FG, ONE))
    rshift = rbind(shift, RIGHT)
    lshift = rbind(shift, LEFT)
    ushift = rbind(shift, UP)
    dshift = rbind(shift, DOWN)
    horiz = fork(combine, rshift, lshift)
    vert = fork(combine, ushift, dshift)
    unionf = fork(combine, horiz, vert)
    Dilates = mapply(unionf, Seeds)
    O = paint(I, Dilates)
    return O

def dsl5(I):
    FG = fgpartition(I)
    Seeds = difference(FG, colorfilter(FG, ONE))
    ones = ofcolor(I, ONE)
    rray = compose(rbind(shoot, RIGHT), center)
    lray = compose(rbind(shoot, LEFT), center)
    uray = compose(rbind(shoot, UP), center)
    dray = compose(rbind(shoot, DOWN), center)
    cross = fork(combine, fork(combine, rray, lray), fork(combine, uray, dray))
    maskf = compose(lbind(intersection, ones), cross)
    recolorf = fork(recolor, color, maskf)
    Paints = mapply(recolorf, Seeds)
    O = paint(I, Paints)
