def dsl1(I):
    ones = ofcolor(I, ONE)
    ul = ulcorner(ones)
    lr = lrcorner(ones)
    n = size(ones)
    k = decrement(n)
    step = divide(subtract(lr, ul), k)
    allidx = asindices(I)
    diag = connect(ulcorner(allidx), lrcorner(allidx))
    start = add(lr, step)
    path = intersection(shoot(start, step), diag)
    O = underpaint(I, recolor(TWO, path))
    return O
