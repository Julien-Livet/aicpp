def dsl1(I):
    O0 = canvas(ZERO, shape(I))
    S1 = asobject(I)
    S2 = asobject(rot90(I))
    S3 = asobject(rot180(I))
    S4 = asobject(rot270(I))
    S5 = asobject(hmirror(I))
    S6 = asobject(vmirror(I))
    S7 = asobject(dmirror(I))
    S8 = asobject(cmirror(I))
    SET = insert(S1, initset(S2))
    SET = insert(S3, SET)
    SET = insert(S4, SET)
    SET = insert(S5, SET)
    SET = insert(S6, SET)
    SET = insert(S7, SET)
    SET = insert(S8, SET)
    BIG = merge(SET)
    O = underpaint(O0, BIG)
    return O

def dsl2(I):
    NZ = difference(asindices(I), ofcolor(I, ZERO))
    TILE = subgrid(NZ, I)
    H = height(I)
    W = width(I)
    TH = height(TILE)
    TW = width(TILE)
    RO = interval(0, H, TH)
    CO = interval(0, W, TW)
    RV = apply(toivec, RO)
    CV = apply(tojvec, CO)
    OFFS = prapply(add, RV, CV)
    BASE = asobject(TILE)
    ROT = asobject(rot180(TILE))
    SH1 = mapply(lbind(shift, BASE), OFFS)
    SH2 = mapply(lbind(shift, ROT), OFFS)
    BIG = merge(combine(SH1, SH2))
    O = paint(canvas(ZERO, shape(I)), BIG)
    return O

def dsl5(I):

    A = first(OBJS)
    OBJS2 = remove(A, OBJS)
    B = first(OBJS2)
    OBJS3 = remove(B, OBJS2)

    IA = toindices(A)
    HA = merge(apply(hfrontier, IA))
    VA = merge(apply(vfrontier, IA))

    VB = merge(apply(vfrontier, IB))
    SB = recolor(color(B), combine(HB, VB))
    IC = toindices(C)
    HC = merge(apply(hfrontier, IC))
    VC = merge(apply(vfrontier, IC))
    SC = recolor(color(C), combine(HC, VC))
    SET = insert(SA, initset(SB))
