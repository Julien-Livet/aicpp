def dsl1(I):
    # Global: zero out ones, swap 8 with computed value from bottom-half ones
    bh = bottomhalf(I)
    val = divide(size(ofcolor(bh, ONE)), TWO)
    base = replace(I, ONE, ZERO)
    O = switch(base, EIGHT, val)
    return O

def dsl2(I):
    # Object extraction: recolor all 8-objects and paint onto blank canvas
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    parts = fgpartition(I)
    eobjs = colorfilter(parts, EIGHT)
    recol = merge(apply(lbind(recolor, val), eobjs))
    O = paint(canvas(ZERO, shape(I)), recol)
    return O

def dsl3(I):
    # Color filtering: use 8-mask and fill with computed value
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    mask8 = ofcolor(I, EIGHT)
    O = fill(canvas(ZERO, shape(I)), val, mask8)
    return O

def dsl4(I):
    # Geometric: form 8-object, recolor, and paint over zero canvas
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    obj8 = toobject(ofcolor(I, EIGHT), I)
    O = paint(canvas(ZERO, shape(I)), recolor(val, obj8))
    return O

def dsl5(I):
    # Relational: underfill 8-mask with computed value on zero canvas
    bh = bottomhalf(I)
    val = halve(size(ofcolor(bh, ONE)))
    mask8 = ofcolor(I, EIGHT)
    O = underfill(canvas(ZERO, shape(I)), val, mask8)
