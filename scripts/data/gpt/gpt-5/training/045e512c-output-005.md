def dsl1(I):
    # Global tiling of the compressed content across the full grid
    C = compress(I)
    base = asobject(C)
    ws = width(C)
    hs = height(C)
    wI = width(I)
    hI = height(I)
    offsH = interval(0, wI, ws)
    vecsH = apply(tojvec, offsH)
    tileH = mapply(lbind(shift, base), vecsH)
    offsV = interval(0, hI, hs)
    vecsV = apply(toivec, offsV)
    tileHV = mapply(lbind(shift, tileH), vecsV)
    O = paint(canvas(ZERO, shape(I)), tileHV)
    return O

def dsl2(I):
    # Object-period tiling: tile all normalized objects by minimal object bbox sizes
    objs = objects(I, T, F, T)
    bbs = apply(lbind(subgrid, I), objs)
    ws = apply(width, bbs)
    hs = apply(height, bbs)
    pH = minimum(ws)
    pV = minimum(hs)
    norms = apply(normalize, objs)
    wI = width(I)
    hI = height(I)
    offsH = interval(0, wI, pH)
    vecsH = apply(tojvec, offsH)
    copiesH = prapply(shift, norms, vecsH)
    tileH = merge(copiesH)
    offsV = interval(0, hI, pV)
    vecsV = apply(toivec, offsV)
    tileHV = mapply(lbind(shift, tileH), vecsV)
    O = paint(canvas(ZERO, shape(I)), tileHV)
    return O

def dsl3(I):
    # Color filtering: draw all rows and columns passing through the dominant nonzero color
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    starts = ofcolor(I, c)
    rows = mapply(hfrontier, starts)
    cols = mapply(vfrontier, starts)
    lines = combine(rows, cols)
    obj = recolor(c, lines)
    O = paint(canvas(ZERO, shape(I)), obj)
    return O

def dsl4(I):
    # Geometric: tile the outline of the nonzero bounding box across the grid using main color
    nz = difference(asindices(I), ofcolor(I, ZERO))
    sub = subgrid(nz, I)
    ws = width(sub)
    hs = height(sub)
    pal = remove(ZERO, palette(I))
    c = argmax(pal, lbind(colorcount, I))
    outline = box(nz)
    base = recolor(c, normalize(outline))
    wI = width(I)
    hI = height(I)
    offsH = interval(0, wI, ws)
    vecsH = apply(tojvec, offsH)
    tileH = mapply(lbind(shift, base), vecsH)
    offsV = interval(0, hI, hs)
    vecsV = apply(toivec, offsV)
    tileHV = mapply(lbind(shift, tileH), vecsV)
    O = paint(canvas(ZERO, shape(I)), tileHV)
    return O

def dsl5(I):
    # Global symmetry: combine both diagonal mirrors of the entire foreground
    base = asobject(I)
    sym = combine(base, dmirror(base))
    sym2 = combine(sym, cmirror(sym))
    O = paint(canvas(ZERO, shape(I)), sym2)
    return O
