def dsl1(I):
    # Global: remove 8-separators by alternating compress across axes after trimming
    A = trim(I)
    B = compress(A)
    C = rot90(B)
    D = compress(C)
    O = rot270(D)
    return O

def dsl2(I):
    # Object extraction: crop to the union of border-touching foreground objects
    OBJS = objects(I, T, F, T)
    BORD = sfilter(OBJS, rbind(bordering, I))
    MER = merge(BORD)
    O = subgrid(MER, I)
    return O

def dsl3(I):
    # Color filtering: turn the least frequent color into 8
    LC = leastcolor(I)
    O = replace(I, LC, EIGHT)
    return O

def dsl4(I):
    # Geometric: keep only cells invariant under 90° rotation, else 8
    R = rot90(I)
    O = cellwise(I, R, EIGHT)
    return O

def dsl5(I):
    # Relational/structural: paint each object's bounding-box complement (delta) in its color on an 8 canvas
    OBJS = objects(I, T, F, T)
    DL = apply(fork(recolor, color, delta), OBJS)
    ALL = merge(DL)
    BASE = canvas(EIGHT, shape(I))
    O = paint(BASE, ALL)
    return O
