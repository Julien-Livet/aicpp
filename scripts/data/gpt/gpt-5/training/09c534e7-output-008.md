def dsl1(I):
    FG = fgpartition(I)
    Seeds = difference(FG, colorfilter(FG, ONE))
    ones = ofcolor(I, ONE)
    base = toindices
    hm = compose(hmirror, base)
    vm = compose(vmirror, base)
    rot = compose(vmirror, hm)
    orb1 = fork(combine, base, hm)
    orb2 = fork(combine, vm, rot)
    orbit = fork(combine, orb1, orb2)
    maskf = compose(lbind(intersection, ones), orbit)
    recolorf = fork(recolor, color, maskf)
    Paints = mapply(recolorf, Seeds)
    O = paint(I, Paints)
    return O
