def dsl1(I):
    # Global per-color consolidation of interior parts into filled rectangles
    objs = objects(I, T, F, T)
    nonborder = sfilter(objs, compose(flip, rbind(bordering, I)))
    cols = dedupe(totuple(apply(color, nonborder)))
    groups = apply(compose(merge, lbind(colorfilter, nonborder)), cols)
    bigs = sfilter(groups, compose(lbind(greater, ONE), compose(size, toindices)))
    parts = apply(fork(recolor, color, backdrop), bigs)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl2(I):
    # Object extraction: keep only non-bordering objects of size > 1 and fill each backdrop
    objs = objects(I, T, F, T)
    nonborder = sfilter(objs, compose(flip, rbind(bordering, I)))
    bigobjs = sfilter(nonborder, compose(lbind(greater, ONE), compose(size, toindices)))
    parts = apply(fork(recolor, color, backdrop), bigobjs)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl3(I):
    # Color filtering: retain only colors that appear in interior multi-cell objects (no rectangles)
    objs = objects(I, T, F, T)
    nonborder = sfilter(objs, compose(flip, rbind(bordering, I)))
    bigobjs = sfilter(nonborder, compose(lbind(greater, ONE), compose(size, toindices)))
    O = paint(canvas(ZERO, shape(I)), merge(bigobjs))
    return O

def dsl4(I):
    # Geometric: enforce global two-axis symmetry then rectangularize resulting parts
    sym = cellwise(cellwise(I, hmirror(I), ZERO), vmirror(I), ZERO)
    sobjs = objects(sym, T, F, T)
    parts = apply(fork(recolor, color, backdrop), sobjs)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O

def dsl5(I):
    # Relational: keep interior objects that are adjacent to another same-colored object; fill their backdrops
    objs = objects(I, T, F, T)
    nonborder = sfilter(objs, compose(flip, rbind(bordering, I)))
    samecolor = compose(lbind(colorfilter, nonborder), color)
    others = fork(remove, identity, samecolor)
    union_others = compose(merge, others)
    adj_same = compose(rbind(equality, ONE), fork(manhattan, identity, union_others))
    keep = sfilter(nonborder, adj_same)
    parts = apply(fork(recolor, color, backdrop), keep)
    O = paint(canvas(ZERO, shape(I)), merge(parts))
    return O
