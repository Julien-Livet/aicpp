def dsl1(I):
    # Object-level orientation via corner-membership (asc if UL and LR are in the object)
    objs = colorfilter(objects(I, T, T, T), FIVE)
    isUL = fork(contained, ulcorner, toindices)
    isLR = fork(contained, lrcorner, toindices)
    isAsc = fork(both, isUL, isLR)
    asc = mfilter(objs, isAsc)
    desc = mfilter(objs, compose(flip, isAsc))
    col8 = mapply(lbind(recolor, EIGHT), asc)
    col2 = mapply(lbind(recolor, TWO), desc)
    colored = combine(col8, col2)
    O = paint(I, colored)
    return O

def dsl3(I):

    S = ofcolor(I, FIVE)
