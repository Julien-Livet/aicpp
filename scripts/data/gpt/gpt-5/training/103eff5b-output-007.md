def dsl1(I):
    pix8 = ofcolor(I, EIGHT)
    base = cover(I, pix8)
    ul8 = ulcorner(pix8)
    tile = asobject(crop(I, ul8, THREE_BY_THREE))
    small = downscale(I, THREE)
    small8 = ofcolor(small, EIGHT)
    scale3 = lbind(multiply, THREE_BY_THREE)
    to_ul = compose(scale3, identity)
    offset = compose(rbind(subtract, ul8), to_ul)
    place = compose(lbind(shift, tile), offset)
    objs = objects(I, T, F, T)
    seeds = difference(objs, colorfilter(objs, EIGHT))
    tile_ul_obj = compose(rbind(toobject, I), compose(initset, to_ul))
    comp_builder = compose(lbind(rbind, manhattan), tile_ul_obj)
    nearest_seed = chain(lbind(argmin, seeds), comp_builder, identity)
    recol = fork(recolor, compose(color, nearest_seed), place)
    recolored = apply(recol, small8)
    O = paint(base, merge(recolored))
    return O
