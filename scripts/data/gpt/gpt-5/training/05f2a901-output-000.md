def dsl1(I):
    # Relational (gravitate + move): move color-2 object until adjacent to color-8
    objs = objects(I, T, F, T)
    twos = colorfilter(objs, TWO)
    eights = colorfilter(objs, EIGHT)
    two = first(twos)
    eight = first(eights)
    off = gravitate(two, eight)
    O = move(I, two, off)
    return O

def dsl2(I):
    # Geometric edges (object extraction): use edge distances with row/column overlap
    objs = objects(I, T, F, T)
    two = first(colorfilter(objs, TWO))
    eight = first(colorfilter(objs, EIGHT))
    off_h = tojvec(decrement(subtract(leftmost(eight), rightmost(two))))
    off_v = toivec(decrement(subtract(uppermost(eight), lowermost(two))))
    off = branch(hmatching(two, eight), off_h, off_v)
    base = cover(I, two)
    shifted = shift(toindices(two), off)
    moved = recolor(TWO, shifted)
    O = paint(base, moved)
    return O

def dsl3(I):
    # Relational (axis selection via overlap + manhattan magnitude)
    objs = objects(I, T, F, T)
    two = first(colorfilter(objs, TWO))
    eight = first(colorfilter(objs, EIGHT))
    axis = branch(vmatching(two, eight), toivec(ONE), tojvec(ONE))
    dirv = sign(position(two, eight))
    mag = decrement(manhattan(two, eight))
    stepvec = multiply(dirv, mag)
    off = mpapply(multiply, stepvec, axis)
    base = cover(I, two)
    shifted = shift(toindices(two), off)
    moved = recolor(TWO, shifted)
    O = paint(base, moved)
    return O

def dsl4(I):
    # Color filtering (work directly with color-index patches)
    two_idx = ofcolor(I, TWO)
    eight_idx = ofcolor(I, EIGHT)
    off_h = tojvec(decrement(subtract(leftmost(eight_idx), rightmost(two_idx))))
    off_v = toivec(decrement(subtract(uppermost(eight_idx), lowermost(two_idx))))
    off = branch(hmatching(two_idx, eight_idx), off_h, off_v)
    cleared = fill(I, ZERO, two_idx)
    shifted = shift(two_idx, off)
    O = fill(cleared, TWO, shifted)
    return O

def dsl5(I):
    # Global geometric transform (rotate, solve by edges, rotate back)
    R = rot90(I)
    ro = objects(R, T, F, T)
    rtwo = first(colorfilter(ro, TWO))
    reight = first(colorfilter(ro, EIGHT))
    rh = tojvec(decrement(subtract(leftmost(reight), rightmost(rtwo))))
    rv = toivec(decrement(subtract(uppermost(reight), lowermost(rtwo))))
    roff = branch(hmatching(rtwo, reight), rh, rv)
    rbase = cover(R, rtwo)
    rshift = shift(toindices(rtwo), roff)
    rnew = recolor(TWO, rshift)
    OR = paint(rbase, rnew)
    O = rot270(OR)
    return O
