def dsl1(I):
    five = ofcolor(I, FIVE)
    c = corners(five)
    miss = first(difference(c, five))
    tl = ulcorner(five)
    tr = urcorner(five)
    bl = llcorner(five)
    br = lrcorner(five)
    cond = both(contained(tl, five), contained(br, five))
    diag = branch(cond, connect(tl, br), connect(tr, bl))
    pbox = box(five)
    rowedge = intersection(hfrontier(miss), pbox)
    coledge = intersection(vfrontier(miss), pbox)
    borders = combine(rowedge, coledge)
    patch = combine(diag, borders)
    O = underfill(I, EIGHT, patch)
    return O

def dsl4(I):

    cond = both(contained(tl, five), contained(br, five))
    dr = add(DOWN, RIGHT)
    ul = NEG_UNITY
    ur = UP_RIGHT
    dl = DOWN_LEFT
    diagA = combine(shoot(tl, dr), shoot(br, ul))
    diagB = combine(shoot(tr, dl), shoot(bl, ur))
    diag = branch(cond, diagA, diagB)
    rect = backdrop(five)
    diag = intersection(diag, rect)
    c = corners(five)
    miss = first(difference(c, five))
    rowedge = intersection(hfrontier(miss), box(five))
    coledge = intersection(vfrontier(miss), box(five))
    borders = combine(rowedge, coledge)

    colpeerset = difference(intersection(vfrontier(miss), c), initset(miss))
    rowpeer = first(rowpeerset)
    colpeer = first(colpeerset)
    borders = combine(connect(miss, rowpeer), connect(miss, colpeer))
    diag = connect(rowpeer, colpeer)
    patch = combine(borders, diag)
    O = underfill(I, EIGHT, patch)
