def dsl1(I):
    # Global – combine the original grid with its 90° rotation.
    # Original non‑background cells win; background cells are taken from the rotation.
    R = rot90(I)                         # 90° clockwise rotation
    O = cellwise(I, R, 4)                # keep original where it is not background
    return O


def dsl2(I):
    # Object extraction – isolate the most frequent non‑background colour
    # and move that whole coloured object to the top‑left corner.
    bg = 4
    nb = palette(I) - {bg}               # colours different from background
    tgt = argmax(nb, lambda c: colorcount(I, c))      # dominant colour
    obj = ofcolor(I, tgt)                 # cells of the dominant colour
    O   = fill(I, bg, obj)               # erase the original occurrence
    # shift the object so its upper‑most and left‑most cells become (0,0)
    shifted = shift(obj, (-uppermost(obj), -leftmost(obj)))
    O = fill(O, tgt, shifted)            # paste it at the origin
    return O


def dsl3(I):
    # Colour‑filtering – replace every non‑background colour by the
    # most common non‑background colour (background stays unchanged).
    bg = 4
    nb = palette(I) - {bg}
    tgt = argmax(nb, lambda c: colorcount(I, c))      # dominant non‑bg colour
    # a full‑grid of the dominant colour
    C = canvas(tgt, shape(I))
    # wherever the input is background we keep it; otherwise we use the dominant colour
    O = cellwise(C, I, bg)
    return O


def dsl4(I):
    # Geometric – reflect the picture across the vertical axis and then
    # overlay the reflection, giving priority to the original cells.
    V = vmirror(I)                        # vertical mirror
    O = cellwise(I, V, 4)                 # original wins, else mirrored colour
    return O


def dsl5(I):
    # Relational / structural – replicate the whole coloured pattern once
    # to the right using the intrinsic horizontal period.
    objs   = objects(I, False, False, True)            # all coloured objects
    patch  = {c for obj in objs for c in obj}          # union of their cells
    p      = hperiod(patch)                            # basic horizontal period
    shifted = shift(patch, (0, p))                      # copy one period to the right
    O = fill(I, 4, patch)                              # erase the original pattern
    O = fill(O, mostcolor(I), shifted)                 # paste the replicated copy
    return O
