def dsl1(I):
    # 1️⃣  Horizontal‑strip split (rows of colour 4 are separators) → column‑wise swap
    h, w = shape(I)
    sep_rows = {r for r in range(h) if all(index(I, (r, c)) == 4 for c in range(w))}
    strips   = hsplit(I, len(sep_rows) + 1)

    def proc(seg):
        ones   = ofcolor(seg, 1)
        fours  = ofcolor(seg, 4)                     # empty – separators are outside the strip
        others = sfilter(palette(seg), lambda c: c not in {0, 1, 4})

        up_to_c = lambda c: {
            cell for cell in ones
            if intersection(
                    {(r, cell[1]) for r in range(cell[0] + 1, len(seg))},
                    ofcolor(seg, c))
               and not intersection(
                    {(r, cell[1]) for r in range(cell[0] + 1, len(seg))},
                    fours)
        }

        down_to_one = lambda c: {
            cell for cell in ofcolor(seg, c)
            if intersection(
                    {(r, cell[1]) for r in range(0, cell[0])},
                    ones)
               and not intersection(
                    {(r, cell[1]) for r in range(0, cell[0])},
                    fours)
        }

        recol_up   = apply(lambda c: recolor(c, up_to_c(c)),   others)
        recol_down = apply(lambda c: recolor(1, down_to_one(c)), others)

        O = paint(seg, merge(recol_up))
        O = paint(O,   merge(recol_down))
        return O

    return vconcat(*apply(proc, strips))


def dsl2(I):
    # 2️⃣  Global colour‑swap: replace every 1 with the most frequent non‑0/1/4 colour and vice‑versa
    other_colors = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main        = argmax(other_colors, lambda c: colorcount(I, c))
    O = paint(I, recolor(main, ofcolor(I, 1)))
    O = paint(O, recolor(1,    ofcolor(I, main)))
    return O


def dsl3(I):
    # 3️⃣  Object‑wise copy: move the lowest non‑0/1/4 object onto the 1‑region and swap colours
    objs      = fgpartition(I)
    coloured = sfilter(objs, lambda o: color(o) not in {0, 1, 4})
    src_obj   = argmax(coloured, lowermost)                     # source object (the propagating colour)
    src_box   = backdrop(src_obj)
    tgt_cells = ofcolor(I, 1)

    # shift the source so that its UL corner coincides with the UL corner of the 1‑patch
    shift_vec = (min(r for r, _ in tgt_cells) - min(r for r, _ in src_box),

    moved_src = shift(src_obj, shift_vec)

    O = paint(I, moved_src)                                      # place source colour onto the 1‑region
    O = paint(O, recolor(1, ofcolor(I, color(src_obj))))       # bring the 1’s onto the former source cells
    return O


def dsl4(I):
    # 4️⃣  Mirror lower‑right quadrant onto upper‑left and replace 1’s with the mirrored colour
    h, w = shape(I)

    # extract the lower‑right quadrant
    br_coords = {(r, c) for r in range(h // 2, h) for c in range(w // 2, w)}
    br        = subgrid(br_coords, I)

    # mirror vertically then horizontally
    mir       = hmirror(vmirror(br))

    # move the mirrored quadrant to the upper‑left corner
    moved     = shift(mir, (-h // 2, -w // 2))

    # replace every 1 with the colour coming from the mirrored content
    O = paint(I, recolor(1, moved))
    return O


def dsl5(I):
    # 5️⃣  Column‑wise visibility using “gravitate” → swap 1 with the colour it sees
    ones   = ofcolor(I, 1)
    fours  = ofcolor(I, 4)
