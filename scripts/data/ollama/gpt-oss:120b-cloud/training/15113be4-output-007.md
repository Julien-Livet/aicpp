def dsl1(I):
    # GLOBAL rule – replace every 1 by the most frequent colour that is not 0,1 or 4
    other   = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main    = argmax(other, lambda c: colorcount(I, c))
    O       = paint(I, recolor(1, ofcolor(I, main)))
    return O


def dsl2(I):
    # PER‑HORIZONTAL‑STRIP rule – 4‑rows act as separators.
    # 1) erase the separator rows (turn them into 0) so the whole grid becomes one strip,
    # 2) apply the global rule inside each connected component that is bounded by 0‑cells,
    # 3) restore the original 4‑rows.
    no4          = paint(I, recolor(4, ofcolor(I, 4)))          # turn all 4‑rows into 0
    comps        = objects(no4, False, False, True)            # connected components (no 4)
    repl_one     = lambda comp:                                 \
        branch(contains(comp, (lambda p: index(I, p) == 1)),
               let(other = sfilter(palette(I), lambda c: c not in {0,1,4}),
                   main   = branch(size(other)==0,
                                 None,
                                 argmax(other, lambda c: colorcount(I, c))),
                   branch(main is None,
                         comp,
                         paint(comp, recolor(1, ofcolor(I, main))))),
               comp)
    new_comps    = apply(repl_one, comps)
    merged       = merge(new_comps)
    O            = paint(merged, recolor(0, ofcolor(I, 4)))     # put back the 4‑rows
    return O


def dsl3(I):
    # SWAP rule – every non‑{0,1,4} colour is exchanged with colour 1.
    others   = sfilter(palette(I), lambda c: c not in {0, 1, 4})

    # 1 → other
    to_other = apply(lambda c: recolor(c, ofcolor(I, 1)), others)

    # other → 1
    from_other = apply(lambda c: recolor(1, ofcolor(I, c)), others)

    O = paint(I, merge(to_other))
    O = paint(O, merge(from_other))
    return O


def dsl4(I):
    # QUADRANT‑MIRROR rule – mirror the lower‑right quadrant onto the upper‑left
    # and replace every 1 by the colour that appears at the same position in the
    # mirrored patch.
    h, w   = shape(I)

    # lower‑right quadrant as a set of indices
    br_idx = {(r, c) for r in interval(h//2, h) for c in interval(w//2, w)}
    quad   = subgrid(br_idx, I)                 # smallest sub‑grid that contains the quadrant

    # mirror vertically then horizontally (counter‑diagonal symmetry)
    mir    = hmirror(vmirror(quad))

    # move the mirrored quadrant to the upper‑left corner (now a patch of indices)
    moved  = shift(asindices(mir), (-h//2, -w//2))

    # replace 1’s by the colour that sits at the same coordinate in the moved patch
    O = paint(I, recolor(1, moved))
    return O


def dsl5(I):
    # COLUMN‑VISIBILITY rule – a 1 looks upward in its column; the first cell that is
    # a colour different from 0,1,4 and not blocked by a 4 becomes its new colour.
    ones   = ofcolor(I, 1)
    fours  = ofcolor(I, 4)
    others = sfilter(palette(I), lambda c: c not in {0, 1, 4})

    # cells that see a colour c above them (no 4 in between)
    sees_up = lambda c: {
        (r, col) for (r, col) in ones
        if intersection({(rr, col) for rr in interval(r+1, height(I))},
                       ofcolor(I, c))
           and not intersection({(rr, col) for rr in interval(r+1, height(I))},
                               fours)
    }

    # recolour those 1‑cells with the colour they see
    recol   = apply(lambda c: recolor(c, sees_up(c)), others)

    O = paint(I, merge(recol))
    return O
