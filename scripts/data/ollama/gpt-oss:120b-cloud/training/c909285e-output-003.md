def dsl1(I):
    # 1️⃣  Global‑bounding‑box extraction:
    #    Crop the minimal rectangle that contains every cell whose colour is **not** 8
    all_idx   = asindices(I)
    eight_idx = ofcolor(I, 8)
    target    = difference(all_idx, eight_idx)                # cells ≠ 8
    box       = backdrop(target)                              # bounding box of those cells
    tl        = ulcorner(box)                                 # top‑left corner of the box
    h         = add(subtract(lowermost(box), uppermost(box)), ONE)   # height  =  low‑top + 1
    w         = add(subtract(rightmost(box), leftmost(box)), ONE)     # width   =  right‑left + 1
    O         = crop(I, tl, (h, w))
    return O


def dsl2(I):
    # 2️⃣  Whole‑grid compression:
    #    Remove all uniform outer “frontier” rows/columns (the dominant colour 8)
    O = compress(I)
    return O


def dsl3(I):
    # 3️⃣  Trimming the uniform border:
    #    Trim away the outermost rows/columns that are completely filled with colour 8
    O = trim(I)                         # trim removes a one‑pixel border of uniform colour
    # repeat until the outer border is no longer colour 8
    O = branch(equality(colorcount(O, 8), height(O) * width(O)),
                lambda _: O,

    return O


def dsl4(I):
    # 4️⃣  Object‑centric extraction:
    #    Partition the grid into colour‑connected objects, keep the **largest** non‑8 object,
    #    and return its tight bounding box.
    objs      = fgpartition(I, False, False, True)               # objects, background ignored
    non8_objs = mfilter(objs, lambda o: not equal(color(o), 8))   # discard the 8‑background object
    biggest   = argmax(non8_objs, lambda o: size(o))             # largest remaining object
    bbox      = backdrop(biggest)
    tl        = ulcorner(bbox)
    h         = add(subtract(lowermost(bbox), uppermost(bbox)), ONE)
    w         = add(subtract(rightmost(bbox), leftmost(bbox)), ONE)
    O         = crop(I, tl, (h, w))
    return O


def dsl5(I):
