def dsl1(I):
    # colour‑filtering: recolour 1‑cells that lie inside the bounding box
    # of any colour > 1.
    seeds   = sfilter(palette(I), lambda c: c > 1)                     # colours to use
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)              # their boxes
    patches = prapply(lambda c, b: recolor(c,
                                          intersection(b,
                                                       ofcolor(I, 1))),
                     seeds, boxes)                                    # colour the 1‑mask
    O = paint(I, merge(patches))                                     # paint on top
    return O


def dsl2(I):
    # object‑extraction: treat every colour > 1 as a single merged object,
    # then recolour the 1‑cells that belong to its bounding box.
    seeds = sfilter(palette(I), lambda c: c > 1)
    objs  = objects(I, False, False, True)                            # all objects
    unions = apply(lambda c: merge(colorfilter(objs, c)), seeds)       # one object per colour
    boxes = apply(lambda o: box(o), unions)                           # its box
    patches = prapply(lambda c, b: recolor(c,
                                          intersection(b,
                                                       ofcolor(I, 1))),
                     seeds, boxes)
    O = paint(I, merge(patches))
    return O


def dsl3(I):
    # geometric – build a rectangle from the UL and LR corners of each colour>1
    # and recolour the 1‑cells that lie inside it.
    seeds = sfilter(palette(I), lambda c: c > 1)
    patches = apply(
        lambda c: recolor(
            c,
            intersection(
                box(connect(ulcorner(ofcolor(I, c)),
                             lrcorner(ofcolor(I, c))),
                ofcolor(I, 1))),
        seeds)
    O = paint(I, merge(patches))
    return O


def dsl4(I):
    # relational / structural: erase all 1‑cells, then underpaint the
    # recoloured masks that correspond to the boxes of colours > 1.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)
    masks   = prapply(lambda c, b: intersection(b, ofcolor(I, 1)), seeds, boxes)
    patches = prapply(lambda c, m: recolor(c, m), seeds, masks)
    cleared = cover(I, ofcolor(I, 1))                     # grid without any 1‑cell
    O = underpaint(cleared, merge(patches))
    return O


def dsl5(I):
    # global‑style transformation: create an empty canvas, paint the
    # colour‑specific boxes on it, then overlay the result on the original
    # grid keeping everything except the original 1‑cells unchanged.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)
    patches = prapply(lambda c, b: recolor(c, b), seeds, boxes)
    empty   = canvas(0, shape(I))                         # all‑zero background
    coloured = paint(empty, merge(patches))               # colour‑filled boxes
    # keep only the cells that were 1 in the input
