def dsl1(I):
    # global‑transformation: crop the lower‑right sub‑grid, draw its diagonal,
    # shift it back and paint with 2.
    ones      = ofcolor(I, 1)
    last_one  = astuple(lowermost(ones), rightmost(ones))
    start     = increment(last_one)                                    # first cell after the line of 1's
    h, w      = shape(I)
    end       = subtract(astuple(h, w), astuple(1, 1))                 # bottom‑right corner
    dims      = add(subtract(end, start), astuple(1, 1))                # size of the sub‑grid
    sub_diag  = connect(astuple(0, 0), subtract(dims, astuple(1, 1)))   # diagonal inside the sub‑grid
    proj_diag = shift(sub_diag, start)                                 # bring it to original coordinates
    O = fill(I, 2, proj_diag)
    return O


def dsl2(I):
    # object‑extraction: use the bounding box of the 1‑pixels,
    # start from its lower‑right corner and draw the continuation of the main diagonal.
    ones      = ofcolor(I, 1)
    bbox      = backdrop(ones)                                          # rectangular hull of the 1's
    corner    = lrcorner(bbox)                                         # lower‑right cell of the hull
    start     = increment(corner)                                     # first empty cell after the line
    h, w      = shape(I)
    end       = subtract(astuple(h, w), astuple(1, 1))                 # bottom‑right corner of the whole grid
    diag      = connect(start, end)                                    # remaining diagonal cells
    O = fill(I, 2, diag)
    return O


def dsl3(I):
    # relational/structural: compute the step vector from the first to the last 1,
    # then shoot from the cell after the last 1.
    ones      = ofcolor(I, 1)
    pts       = tuple(sorted(ones))                                    # ordered list of 1‑positions
    full_vec  = subtract(pts[-1], pts[0])                             # total displacement
    step_cnt  = add(size(pts), -1)                                     # number of intervals (len‑1)
    step      = divide(full_vec, step_cnt)                             # elementary (Δr,Δc)
    start     = increment(pts[-1])                                     # first empty cell after the line
    line      = shoot(start, step)                                     # continuation of the line
    O = fill(I, 2, line)
    return O


def dsl4(I):
    # colour‑filtering with cover & underpaint:
    # build a full canvas of 2’s, keep only the main diagonal,
    # erase the prefix up to the last 1, then underpaint onto the original grid.
    h, w          = shape(I)
    canvas2       = canvas(2, astuple(h, w))                           # whole grid filled with 2
    full_diag     = connect(astuple(0, 0), subtract(astuple(h, w), astuple(1, 1)))
    diag_only     = cover(canvas2, difference(ofcolor(canvas2, 2), full_diag))
    ones          = ofcolor(I, 1)
    last_one      = astuple(lowermost(ones), rightmost(ones))
    prefix_diag   = connect(astuple(0, 0), last_one)
    to_fill       = cover(diag_only, prefix_diag)                     # erase the occupied prefix
    O = underpaint(I, toobject(to_fill, I))                           # paint the remaining 2’s on background
    return O

    last_one = astuple(lowermost(ones), rightmost(ones))
    start  = increment(last_one)                                       # cell after the line of 1's
    line   = shoot(start, step)                                        # continuation of the diagonal
