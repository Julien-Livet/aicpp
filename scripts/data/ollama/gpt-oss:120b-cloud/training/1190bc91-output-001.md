def dsl1(I):
    # ------------------------------------------------------------
    # Strategy 1 – object extraction + translation
    #   Extract every non‑zero object and shift it 2 cells up‑left.
    #   The shifted copies are then merged with the original grid.
    # ------------------------------------------------------------
    objs   = fgpartition(I, False, False, True)          # all coloured objects (no background)
    shift  = lambda o: move(I, o, (-2, -2))              # translate the object
    moved  = apply(shift, objs)                         # apply the translation to every object
    O      = combine(I, moved)                          # overlay original and moved objects
    return O


def dsl2(I):
    # ------------------------------------------------------------
    # Strategy 2 – most frequent non‑zero colour background fill
    #   Find the colour that appears most often (ignoring 0) and
    #   paint every background cell (value 0) with that colour.
    # ------------------------------------------------------------
    nonzero = difference(palette(I), initset(0))         # set of colours different from 0
    fillcol = mostcommon(nonzero)                      # most frequent non‑zero colour
    bg      = ofcolor(I, 0)                             # indices of the background
    O       = underfill(I, fillcol, bg)                  # paint only the background cells
    return O


def dsl3(I):
    # ------------------------------------------------------------
    # Strategy 3 – periodic tiling using vertical period
    #   Detect the vertical period of the 1‑colour objects and
    #   repeat the pattern across the whole grid by stacking copies.
    # ------------------------------------------------------------
    objs      = fgpartition(I, False, False, True)      # coloured objects
    single    = colorfilter(objs, mostcommon(palette(I)))# keep objects of the most common colour
    period    = vperiod(single)                         # vertical period of those objects
    tiled     = combine(*[move(I, single, (k*period, 0)) for k in range(10//period)])
    O         = combine(I, tiled)                       # merge with original grid
    return O


def dsl4(I):
    # ------------------------------------------------------------
    # Strategy 4 – gravitate every coloured cell to the top‑left corner
    #   For each non‑zero cell, compute the direction that brings it
    #   as close as possible to the (0,0) corner and move it once.
    # ------------------------------------------------------------
    srcs = ofcolor(I, mostcommon(palette(I)))           # all cells of the dominant colour
    dir  = gravitate(srcs, initset((0, 0)))             # direction towards the corner
    O    = move(I, srcs, dir)                          # shift all those cells
    return O


def dsl5(I):
    # ------------------------------------------------------------
    # Strategy 5 – relative positioning of two colour groups
    #   Take the two most frequent colours, compute the vector from
    #   the first group to the second, and translate the first group
    #   by that vector, then merge the result.
    # ------------------------------------------------------------
    colours   = mostcommon(palette(I))                  # two most common non‑zero colours
    col1, col2 = colours[0], colours[1]
    patch1    = ofcolor(I, col1)
    patch2    = ofcolor(I, col2)
    vec       = position(patch1, patch2)                # relative offset from col1 to col2
    moved1    = move(I, patch1, vec)                    # translate the first colour
    O         = combine(I, moved1)                     # overlay the moved cells
    return O
