def dsl1(I):
    # Relational – erase the right half, then copy the left half into it.
    bg    = ofcolor(I, 4)                                   # background cells
    w     = add(1, subtract(rightmost(bg), leftmost(bg)))   # grid width
    half  = divide(w, 2)                                   # half‑width
    all_i = asindices(I)                                    # all cell indices
    left  = sfilter(all_i, lambda p: p[1] < half)           # cells left of centre
    right = sfilter(all_i, lambda p: p[1] >= half)          # cells right of centre
    left_o= toobject(left, I)                               # left‑side coloured object
    O     = fill(I, 4, right)                              # clear right half
    O     = move(O, left_o, (0, half))                     # paste left side shifted right
    return O


def dsl2(I):
    # Global – trim the picture, mirror it across the counter‑diagonal,
    # and paint the mirrored picture onto the original background.
    T = trim(I)                     # remove outer background
    M = dmirror(T)                 # counter‑diagonal mirror
    O = underpaint(I, M)           # overlay mirrored picture on background
    return O


def dsl3(I):
    # Colour‑statistics – swap the background colour (4) with the most
    # frequent non‑background colour.
    bg   = 4
    nonbg = palette(I) - {bg}                                 # colours ≠ background
    tgt  = argmax(nonbg, lambda c: colorcount(I, c))          # dominant colour
    O = recolor(bg,   ofcolor(I, tgt))                        # background → dominant
    O = recolor(tgt,  ofcolor(I, bg))                         # dominant → background
    return O


def dsl4(I):
    # Geometric – vertical mirror overlay, keeping original cells where they differ
    # from the mirror (fallback colour 4).
    V = vmirror(I)                     # vertical flip
    O = cellwise(I, V, 4)             # keep I‑pixels, otherwise background
    return O


def dsl5(I):
    # Relational / structural – replicate the pattern one vertical period downwards.
    objs   = objects(I, False, False, True)
    patch  = {c for obj in objs for c in obj}   # all coloured cells
    p      = vperiod(patch)                     # basic vertical period
    shifted= shift(patch, (p, 0))               # copy one period down
    O = fill(I, 4, patch)                       # erase original pattern
    O = fill(O, mostcolor(I), shifted)         # paste the replicated copy
    return O
