def dsl5(I):

    # and top‑most in their column (the “corner” cells) one step right
    cells          = asobject(I)
    left_adj_row   = intersection(cells, shift(cells, LEFT))
    leftmost_row   = difference(cells, left_adj_row)          # left‑most per row
    up_adj_col     = intersection(cells, shift(cells, UP))
    topmost_col    = difference(cells, up_adj_col)           # top‑most per column
    corner         = intersection(leftmost_row, topmost_col)   # cells that satisfy both
    moved          = shift(corner, RIGHT)
    stay           = difference(cells, corner)
    empty          = canvas(0, shape(I))
    return paint(paint(empty, moved), stay)
