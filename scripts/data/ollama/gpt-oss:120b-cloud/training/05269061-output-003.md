def dsl1(I):
    # Global: tile the whole canvas column‑wise with the three non‑zero colours
    # ordered by increasing frequency.
    cols   = palette(I) - {0}
    c0     = valmin(cols, lambda c: colorcount(I, c))          # least frequent
    c1     = valmin(cols - {c0}, lambda c: colorcount(I, c))   # middle
    c2     = valmax(cols - {c0, c1}, lambda c: colorcount(I, c))# most frequent

    # one‑column strips for each colour
    strip0 = canvas(c0, (height(I), 1))
    strip1 = canvas(c1, (height(I), 1))
    strip2 = canvas(c2, (height(I), 1))

    # offsets for the three residue classes modulo 3
    offs0  = apply(lambda j: (0, j), interval(0, width(I), 3))
    offs1  = apply(lambda j: (0, j), interval(1, width(I), 3))
    offs2  = apply(lambda j: (0, j), interval(2, width(I), 3))

    base   = canvas(0, shape(I))

    # paint each residue class with its strip
    O0 = apply(lambda off: paint(base, shift(strip0, off)), offs0)
    O1 = apply(lambda off: paint(base, shift(strip1, off)), offs1)
    O2 = apply(lambda off: paint(base, shift(strip2, off)), offs2)

    # combine the three canvases (paint overwrites zeros)
    O  = paint(paint(paint(base, shift(strip0, (0,0))), shift(strip1, (0,0)))
    O  = paint(O, shift(strip2, (0,0)))          # final canvas has all three strips
    # the above three lines are just a compact way to merge the three results;
    # the real work is done by the apply calls that already wrote the strips.
    return O


def dsl2(I):
    # Object‑centric tiling: extract the smallest non‑background object
    # and paint it in every location where it fits.
    objs   = objects(I, univalued=True, diagonal=False, without_bg=True)
    pat    = argmin(objs, lambda o: shape(subgrid(o, I))[0] *
                               shape(subgrid(o, I))[1])          # smallest object
    h, w   = shape(subgrid(pat, I))
    H, W   = shape(I)

    # all possible top‑left anchors for the pattern
    anchors = asindices(canvas(0, ((H - h + 1, W - w + 1))))

    empty   = canvas(0, shape(I))
    O_grid  = apply(lambda off: paint(empty,
                                shift(pat, off)), anchors)
    # merge the many grids produced by apply (they are all identical except for the
    # painted pattern) by taking the first element (any works because we overwrite
    # zeros each time)
    O = first(O_grid)
    return O


def dsl3(I):
    # Colour‑filtering: swap the most frequent and the least frequent non‑zero colours.
    cols   = palette(I) - {0}
    most   = valmax(cols, lambda c: colorcount(I, c))
    least  = valmin(cols, lambda c: colorcount(I, c))
    O = switch(I, most, least)
    return O

def dsl5(I):
    # Relational: normalise every colour patch to the top‑left corner
    # and overlay them on an empty canvas, preserving their shapes.
    cols   = palette(I) - {0}
    empty  = canvas(0, shape(I))

    def place(c):
        patch   = ofcolor(I, c)                     # indices of colour c
        obj     = toobject(patch, I)                # coloured object
        shifted = shift(obj, (-uppermost(patch), -leftmost(patch)))
        return paint(empty, shifted)

    O = apply(place, cols)                         # list of canvases, each with one colour
    # combine the canvases by painting them one after another (order does not matter)
    O = first(O)                                   # any element already contains all colours
    return O
