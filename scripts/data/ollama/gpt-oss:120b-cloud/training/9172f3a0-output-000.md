def dsl3(I):

    # each single‑cell object of the input becomes a n×n block of the same colour
    n = height(I)                                   # scaling factor
    base = canvas(0, (n * n, n * n))                # empty canvas
    # for every cell colour, create the corresponding n×n block and paste it
    O = apply(
        lambda cell: move(

            # a 1‑pixel object at the cell position
            initset((color(I, cell), cell)),
            multiply(cell, n)                       # offset = original coordinate * n
        ),
        asindices(I)
    )
    # merge the many intermediate canvases (the DSL `apply` returns a container)
    O = merge(O)
    return O
