def dsl1(I):
    # 1️⃣  Column‑wise colour swap (1 ↔ other colour) -----------------------
    ones   = ofcolor(I, 1)
    fours  = ofcolor(I, 4)

    # colours that may take part in a swap (everything except 0,1,4)
    others = mfilter(palette(I), lambda c: c not in {0, 1, 4})

    # 1 → c  : a 1‑cell that sees a c‑cell below it, with no 4 in‑between
    def up_to_c(c):
        below = lambda cell: {(r, cell[1]) for r in range(cell[0] + 1, len(I))}
        return {cell for cell in ones
                if intersection(below(cell), ofcolor(I, c))
                and not intersection(below(cell), fours)}

    # c → 1  : a c‑cell that sees a 1‑cell above it, with no 4 in‑between
    def down_to_one(c):
        above = lambda cell: {(r, cell[1]) for r in range(0, cell[0])}
        cs    = ofcolor(I, c)
        return {cell for cell in cs
                if intersection(above(cell), ones)
                and not intersection(above(cell), fours)}

    # recolour every eligible 1‑cell and every eligible c‑cell
    recol_up   = apply(lambda c: recolor(c, up_to_c(c)),   others)
    recol_down = apply(lambda c: recolor(1, down_to_one(c)), others)

    O = paint(I, merge(recol_up))
    O = paint(O, merge(recol_down))
    O = paint(O, recolor(4, fours))                # keep the 4‑cells
    return O

    h, w = shape(I)
