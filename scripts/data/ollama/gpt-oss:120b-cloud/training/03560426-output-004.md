def dsl1(I):
    # GLOBAL – remove all outer empty rows/cols and then close every interior gap.
    #   trim   : cut away the outer border that is completely background.
    #   compress: collapse every completely‑empty row or column that remains.
    T = trim(I)                     # outer empty border
    C = compress(T)                 # interior empty rows / columns
    O = canvas(0, shape(I))        # blank canvas of original size
    return paint(O, asobject(C))    # render the compacted pattern



def dsl2(I):
    # OBJECT‑LEVEL – isolate each coloured object, shift it to the
    #   origin of its own bounding box, then unite all shifted objects
    #   on a fresh canvas.
    objs     = fgpartition(I)                       # all coloured objects
    normed   = apply(normalize, objs)               # each to its top‑left corner
    # fold the set of objects pair‑wise with ‘combine’
    united   = combine(*normed)                      # union of all normalised objects
    empty    = canvas(0, shape(I))
    return paint(empty, united)



def dsl3(I):
    # COLOUR‑FILTER – keep the three most frequent non‑zero colours,
    #   discard everything else and then compact the remaining pattern.
    colset   = palette(I) - {0}                     # non‑zero colours
    # order colours by decreasing frequency
    ordered  = order(colset, lambda c: -colorcount(I, c))
    top3     = {first(apply(lambda _: ordered, interval(0, 3, 1)))}
    keep     = ofcolor(I, top3)                     # cells of the three colours
    filt     = underpaint(I, keep)                  # delete all other colours
    compact  = compress(trim(filt))                 # remove outer & interior gaps
    empty    = canvas(0, shape(I))
    return paint(empty, asobject(compact))



def dsl4(I):
    # GEOMETRIC – rotate clockwise, then erase every empty row/column
    #   (both external and internal) and finally paint the result.
    rot      = rot90(I)                             # 90° clockwise
    trimmed  = trim(rot)                            # cut outer empty border
    compact  = compress(trimmed)                     # collapse interior gaps
    empty    = canvas(0, shape(I))
    return paint(empty, asobject(compact))



def dsl5(I):
    # RELATIONAL – compute a single translation that brings the
    #   upper‑most, left‑most coloured object to the origin and
    #   apply that very same translation to **every** coloured object.
    objs      = fgpartition(I)                       # all coloured objects
    # reference object: one with the smallest (row, col) coordinate
    ref       = argmin(objs, lambda o: add(uppermost(o), leftmost(o)))
    # vector that moves the reference to the top border
    up_vec    = gravitate(ref,
                         {(0, c) for c in interval(0, width(I), 1)})
    # then moves it to the left border
    left_vec  = gravitate(move(I, ref, up_vec),
                         {(r, 0) for r in interval(0, height(I), 1)})
    offset    = add(up_vec, left_vec)               # total translation
    shifted   = apply(lambda o: shift(o, offset), objs)  # move *all* objects
    united    = combine(*shifted)                   # union of shifted objects
    empty     = canvas(0, shape(I))
    return paint(empty, united)
