def dsl1(I):
    # Global – mirror the left half of the picture onto the right half.
    bg = 4
    L = lefthalf(I)               # left‑hand side sub‑grid
    M = vmirror(L)                # its horizontal mirror (appears on the right side)
    O = fill(I, bg, L)            # erase coloured cells that were on the left
    O = underfill(O, bg, M)       # paste the mirrored pattern on the right

def dsl2(I):
    # Object‑extraction – take the most frequent non‑background colour,
    # remove it and paste its horizontally mirrored copy.
    bg = 4
    nb   = palette(I) - {bg}
    tgt  = argmax(nb, lambda c: colorcount(I, c))   # dominant colour ≠ background
    obj  = ofcolor(I, tgt)                          # cells of that colour
    O    = fill(I, bg, obj)                         # erase the original occurrence
    M    = vmirror(obj)                             # mirror the set of indices
    Mobj = toobject(M, I)                           # re‑attach colour information
    Mcol = recolor(tgt, Mobj)                       # colour the mirrored cells
    O    = paint(O, Mcol)                           # paint them back
    return O


def dsl3(I):
    # Colour‑filtering – replace every non‑background cell by the most common
    # non‑background colour, keeping background cells unchanged.
    bg = 4
    nb   = palette(I) - {bg}
    tgt  = argmax(nb, lambda c: colorcount(I, c))   # dominant non‑background colour
    C    = canvas(tgt, shape(I))                    # full grid of that colour
    O    = cellwise(C, I, bg)                       # keep background where it occurs
    return O


def dsl4(I):
    # Geometric – rotate the whole picture 180° (both axes) and keep the
    # original background colour where the rotation leaves empty cells.
    bg = 4
    R = rot180(I)               # half‑turn rotation
    O = cellwise(R, I, bg)      # fill gaps with the original background colour
    return O


def dsl5(I):
    # Relational – treat all coloured cells as a single patch, erase them,
    # then shift the whole patch right by one horizontal period and paint the
    # shifted cells with their original colours.
    bg = 4
    objs   = objects(I, False, False, True)                 # all coloured objects
    patch  = {c for obj in objs for c in obj}               # union of their cells
    p      = hperiod(patch)                                # basic horizontal period
    O      = fill(I, bg, patch)                             # clear original coloured cells
    shifted = shift(patch, (0, p))                          # move them one period to the right
    O      = fill(O, bg, shifted)                           # place the shifted patch
    O      = paint(O, toobject(shifted, I))                # restore the original colours
    return O
