def dsl1(I):
    # GLOBAL – trim, recolour, find topmost pattern row, its vertical period
    # and crop exactly one period (full width).
    c   = trim(I)
    c   = replace(c, 8, 5)
    c   = replace(c, 1, 9)
    c   = replace(c, 3, 8)
    G   = replace(c, 2, 7)
    pat = asobject(G)                     # all coloured cells after recolouring
    start = uppermost(pat)                 # first row that contains the pattern
    per   = vperiod(pat)                   # height of one repeating block
    w     = width(G)
    O = crop(G, (start, 0), (per, w))
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION – trim & recolour, isolate the three pattern colours,
    # merge the three colour objects, take its bounding box and split it
    # vertically by the period, keep the first block.
    c   = trim(I)
    c   = replace(c, 8, 5)
    c   = replace(c, 1, 9)
    c   = replace(c, 3, 8)
    G   = replace(c, 2, 7)
    objs = fgpartition(G)                              # all colour‑connected objects
    col9 = colorfilter(objs, 9)
    col8 = colorfilter(objs, 8)
    col7 = colorfilter(objs, 7)
    pat  = combine(combine(col9, col8), col7)          # union of the three colour objects
    box  = subgrid(pat, G)                            # minimal sub‑grid enclosing them
    per  = vperiod(pat)                               # vertical period of the pattern
    parts = vsplit(box, per)                          # vertical slices of height = period
    O = first(parts)                                  # keep the first slice
    return O


def dsl3(I):
    # COLOR‑FILTER – trim & recolour, build a mask of the three pattern
    # colours, compute its period and topmost row, then crop one period.
    c   = trim(I)
    c   = replace(c, 8, 5)
    c   = replace(c, 1, 9)
    c   = replace(c, 3, 8)
    G   = replace(c, 2, 7)
    mask = ofcolor(G, 9) | ofcolor(G, 8) | ofcolor(G, 7)
    pat  = asobject(mask)                 # treat the mask as an object
    start = uppermost(pat)                 # first row where any pattern colour appears
    per   = vperiod(pat)                   # height of one repeating block
    w     = width(G)
    O = crop(G, (start, 0), (per, w))
    return O


def dsl4(I):
    # GEOMETRIC – rotate so that vertical repetitions become horizontal,
    # trim, compute the horizontal period, split that rectangle, keep the
    # first block and rotate back.
    c   = trim(I)
    c   = replace(c, 8, 5)
    c   = replace(c, 1, 9)
    c   = replace(c, 3, 8)
    G   = replace(c, 2, 7)
    R   = rot90(G)                         # height ↔ width
    R   = trim(R)                         # remove the former top/bottom borders
    pat = asobject(R)
    per = hperiod(pat)                    # horizontal period = original vertical period
    parts = hsplit(R, per)                # horizontal slices of width = period
    part = first(parts)                   # keep the first slice
    O = rot270(part)                      # rotate back to original orientation
    return O


def dsl5(I):
