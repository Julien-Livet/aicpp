    # GLOBAL – remove empty border, rotate 90° clockwise, paste into a zero canvas
    C = trim(I)                     # crop outer rows / columns that are all 0
    R = rot90(C)                    # 90° clockwise rotation
    O = canvas(0, shape(I))        # blank canvas of original size
    O = cover(O, R)                 # place the rotated core at the top‑left corner
    return O

def dsl2(I):
    # OBJECT – keep the largest non‑zero object unchanged,
    #          translate every other object so that its centre coincides
    #          with the centre of the largest object.
    objs      = fgpartition(I)                                 # all colour‑based objects
    big       = argmax(objs, lambda o: size(o))                # the biggest object
    ctr_big   = center(big)                                    # its centre
    others    = difference(objs, initset(big))                 # all objects except the biggest

    def move_to_big(o):
        d = subtract(ctr_big, center(o))                       # vector needed to align centres
        return move(I, o, d)                                    # shift object by d

    moved = apply(move_to_big, others)                         # translate the smaller objects
    O = canvas(0, shape(I))                                    # start from a blank canvas
    O = cover(O, big)                                          # paste the biggest one
    O = cover(O, combine(*moved))                              # paste all the moved ones
    return O
```
