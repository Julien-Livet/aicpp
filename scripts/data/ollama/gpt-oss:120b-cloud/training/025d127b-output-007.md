def dsl1(I):
    # GLOBAL – relational, using set‑difference to get the topmost cells of each column
    all_cells = asobject(I)
    left_adj   = intersection(all_cells, shift(all_cells, RIGHT))
    leftmost   = difference(all_cells, left_adj)                     # left‑most column of every object
    topmost    = difference(leftmost, shift(leftmost, UP))          # cells with no neighbour above
    shifted    = shift(topmost, RIGHT)                               # move those cells right
    static     = difference(all_cells, topmost)                      # everything else stays
    empty      = canvas(0, shape(I))
    return paint(paint(empty, shifted), static)


def dsl2(I):
    # OBJECT – work on each connected component independently
    objs = fgpartition(I)

    def proc(o):
        # isolate the left‑most column of the object
        left_adj   = intersection(o, shift(o, RIGHT))
        leftmost   = difference(o, left_adj)
        # pick the topmost cells of that column
        topmost    = difference(leftmost, shift(leftmost, UP))
        shifted    = shift(topmost, RIGHT)
        keep       = difference(o, topmost)
        return combine(shifted, keep)

    transformed = apply(proc, objs)
    return underpaint(canvas(0, shape(I)), merge(transformed))
