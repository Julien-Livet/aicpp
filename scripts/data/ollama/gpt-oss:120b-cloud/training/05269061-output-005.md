def dsl1(I):
    # GLOBAL PATTERN: 3‑color (row+col) mod 3 tiling.

    o1     = argmin(difference(objs, {o0}), sum_of)
    o2     = argmin(difference(objs, {o0, o1}), sum_of)

def dsl2(I):

    # 1) objects of each colour
    objs   = objects(I, univalued=True, diagonal=False, without_bg=True)
    # 2) longest object (the diagonal)

    h, w   = shape(subgrid(pat, I))
    H, W   = shape(I)
    # 4) all top‑left anchors where it fits
    anchors = asindices(canvas(0, ((H - h + 1, W - w + 1))))

    O       = first(grids)          # any order works – all cells become the pattern
    return O

def dsl5(I):

    cols   = palette(I) - {0}

        obj     = toobject(patch, I)                 # coloured object
