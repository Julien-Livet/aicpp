def dsl1(I):
    # OBJECT‑EXTRACTION + BOUNDING‑BOX edges
    idx      = asindices(I)
    objs     = objects(I, True, False, False)                     # all colour objects
    full_obj = lambda o: either(
        both(equality(leftmost(o),   leftmost(idx)),
             equality(rightmost(o),  rightmost(idx))),            # spans whole width
        both(equality(uppermost(o),  uppermost(idx)),
             equality(lowermost(o),   lowermost(idx)))           # spans whole height
    target = first(sfilter(objs, full_obj))
    return canvas(color(target), (1, 1))


def dsl2(I):
    # FG‑PARTITION + EDGE‑CHECK
    idx      = asindices(I)
    parts    = fgpartition(I)                                      # each coloured object
    bg       = mostcolor(I)
    non_bg   = sfilter(parts, lambda o: flip(equality(color(o), bg)))
    full_obj = lambda o: either(
        both(equality(leftmost(o),   leftmost(idx)),
             equality(rightmost(o),  rightmost(idx))),
        both(equality(uppermost(o),  uppermost(idx)),
             equality(lowermost(o),   lowermost(idx))))
    target = first(sfilter(non_bg, full_obj))
    return canvas(color(target), (1, 1))


def dsl3(I):
    # SUBGRID + SHAPE test (height = 1 or width = 1)
    objs = objects(I, True, False, False)
    line = lambda o: either(
        equality(height(subgrid(o, I)), 1),     # a single row (full‑width line)
        equality(width(subgrid(o, I)), 1))      # a single column (full‑height line)
    target = first(sfilter(objs, line))
    return canvas(color(target), (1, 1))


def dsl4(I):
    # BOX (outline) size compared with grid dimensions
    idx = asindices(I)
    w   = add(subtract(rightmost(idx), leftmost(idx)), (0, 1))[1]   # grid width
    h   = add(subtract(lowermost(idx), uppermost(idx)), (1, 0))[0]   # grid height
    objs = objects(I, True, False, False)
    full_obj = lambda o: either(
        equality(width(box(o)), w),      # box spans whole width
        equality(height(box(o)), h))     # box spans whole height
    target = first(sfilter(objs, full_obj))
    return canvas(color(target), (1, 1))


def dsl5(I):
    # APPLY on coordinates → distinct‑row / distinct‑column detection
    idx    = asindices(I)
    colors = palette(I)
    bg     = mostcolor(I)
    w      = add(subtract(rightmost(idx), leftmost(idx)), (0, 1))[1]
    h      = add(subtract(lowermost(idx), uppermost(idx)), (1, 0))[0]

    # number of distinct rows / columns occupied by a patch
    nrows  = lambda p: size(apply(lambda pt: pt[0], p))
    ncols  = lambda p: size(apply(lambda pt: pt[1], p))

    good_c = lambda c: either(
        both(equality(nrows(ofcolor(I, c)), 1),
             equality(width(ofcolor(I, c)), w)),
        both(equality(ncols(ofcolor(I, c)), 1),
             equality(height(ofcolor(I, c)), h)))

    target_c = first(sfilter(colors,
                lambda c: both(not equality(c, bg), good_c(c))))
    return canvas(target_c, (1, 1))
