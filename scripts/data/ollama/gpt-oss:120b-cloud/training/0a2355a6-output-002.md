def dsl1(I):

def dsl2(I):
    # OBJECT‑LEVEL: border / interior / isolated classification
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    def colour_obj(patch):
        is_border = bordering(patch, I)                     # touches outer frame ?
        # interior = cells that have the four orthogonal neighbours inside the patch
        interior = intersection(
            patch,
            intersection(shift(patch, UP),
                          intersection(shift(patch, DOWN),
                                        intersection(shift(patch, LEFT),
                                                    shift(patch, RIGHT)))))
        has_interior = size(interior) > 0

        col = branch(is_border, 1,
               branch(has_interior, 2, 3))
        return recolor(col, patch)

    coloured = apply(colour_obj, objs)                     # recoloured objects
    canvas = replace(I, 8, 0)                             # clean background
    O = paint(canvas, merge(coloured))                    # overlay
    return O


def dsl3(I):
    # COLOR‑FILTERING: keep the largest 8‑component as 2, others as 3
    comps = objects(I, univalued=True, diagonal=False, without_bg=True)

    sizes = apply(lambda o: size(o), comps)               # size of each component
    biggest = argmax(sizes, lambda x: x)                  # maximal size

    def recol(col_obj):
        col = branch(size(col_obj) == biggest, 2, 3)
        return recolor(col, col_obj)

    recoloured = apply(recol, comps)
    O = paint(canvas(0, shape(I)), merge(recoloured))
    return O

def dsl5(I):
    # RELATIONAL / STRUCTURAL: objects that touch another object become 3,
    # otherwise become 1
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    # all ordered pairs of distinct objects that are orthogonally adjacent
    pairs = product(objs, objs)
    close = mfilter(pairs,
                    lambda p: p[0] != p[1] and adjacent(p[0], p[1]))

    touching = apply(lambda p: p[0], close)                # objects that touch someone
    touching_set = merge(touching)                        # unified set

    def recolour(o):
        col = branch(contains(touching_set, o), 3, 1)
        return recolor(col, o)

    coloured = apply(recolour, objs)
    O = paint(canvas(0, shape(I)), merge(coloured))
    return O
