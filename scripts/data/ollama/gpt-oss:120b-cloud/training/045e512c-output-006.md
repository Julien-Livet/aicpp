def dsl1(I):
    # GLOBAL – crop to the tight box, double its size, rotate 90° clockwise,
    # then paste the result at the top‑left corner of a fresh canvas.
    sub   = compress(I)                           # minimal non‑zero rectangle
    big   = upscale(sub, 2)                       # 2× enlargement
    rot   = rot90(big)                            # quarter‑clockwise rotation
    base  = canvas(0, shape(I))                   # empty canvas, original size
    O     = paint(base, asobject(rot))            # embed at (0,0)
    return O


def dsl2(I):
    # OBJECT‑LEVEL – for every colour, extract its exact bounding box,
    # enlarge it 2× and translate it to (2·ulcorner) before painting everything together.
    cols = palette(I)                                            # distinct non‑zero colours
    objs = apply(
        lambda c: shift(
            asobject(
                upscale(
                    subgrid(backdrop(ofcolor(I, c)), I), 2)),    # 2× enlarged bounding box
            multiply(ulcorner(ofcolor(I, c)), (TWO, TWO))        # shift = 2·ulcorner
        ),
        cols)
    base = canvas(0, shape(I))
    O    = paint(base, merge(objs))                             # paint the union
    return O


def dsl3(I):
    # GLOBAL – strip the outer zero border, double the size,
    # then centre the enlarged pattern on the original canvas.
    trimmed = trim(I)                                            # remove empty outer rows/cols
    big     = upscale(trimmed, 2)                                # 2‑fold enlargement

    base    = canvas(0, shape(I))
    O       = paint(base, shifted)
    return O
