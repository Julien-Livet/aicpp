def dsl1(I):
    # ---- Global periodicity (no colour filter) ----
    # Use the whole grid as a mask, compute its vertical and horizontal periods,
    # and crop the grid to that elementary tile.
    obj   = toobject(asindices(I), I)          # all cells as one object
    vsize = vperiod(obj)                      # vertical period
    hsize = hperiod(obj)                      # horizontal period
    O     = crop(I, (0, 0), (vsize, hsize))   # elementary repeating tile
    return O


def dsl2(I):
    # ---- Most‑common object extraction ----
    # Break the picture into objects, pick the shape that occurs most often,
    # and return its bounding box as the output tile.
    objs   = objects(I, False, False, True)    # all objects (any colour)
    tile_o = mostcommon(objs)                 # the most frequent object
    bbox   = box(toindices(tile_o))            # its bounding box (indices)
    O      = subgrid(bbox, I)                 # extract that region from the input
    return O


def dsl3(I):
    # ---- Dominant‑colour mask + trim ----
    # Keep only the dominant colour, replace everything else by a uniform background
    # (value 0), then trim the resulting grid to remove empty borders.
    dom    = mostcolor(I)
    mask   = ofcolor(I, dom)                   # indices of dominant colour
    tmp    = cover(I, complement(mask, asindices(I)))  # set non‑dom cells to 0
    O      = trim(tmp)                         # remove outer rows/cols of 0s
    return O


def dsl4(I):
    # ---- Symmetry core without period computation ----
    # Cells that stay unchanged after both horizontal and vertical mirroring
    # form a core; that core itself is the desired output.
    H      = hmirror(I)
    V      = vmirror(I)
    eq_h   = cellwise(I, H, 0)                # equal to horizontal mirror
    core   = cellwise(eq_h, V, 0)              # also equal to vertical mirror
    O      = core
    return O


def dsl5(I):
    # ---- Relational tile from first two occurrences ----
    # Find the first two occurrences of the dominant colour,
    # use their displacement as the tile size, and crop the grid
    # from the first occurrence using that size.
    dom     = mostcolor(I)
    mask    = ofcolor(I, dom)                 # indices of dominant colour
    obj     = toobject(mask, I)
    occs    = occurrences(I, obj)             # all occurrence origins
    first1  = first(occs)                     # top‑left of the first tile
    rest    = difference(occs, initset(first1))
    first2  = first(rest)                     # origin of the second tile
    vsize   = add(first2, (-first1[0], 0))[0] # vertical distance
    hsize   = add(first2, (0, -first1[1]))[1]  # horizontal distance
    O       = crop(I, first1, (vsize, hsize))
    return O
