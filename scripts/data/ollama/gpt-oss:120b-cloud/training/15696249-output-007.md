def dsl1(I):
    # OBJECT‑EXTRACTION: treat the whole input as a single object and tile it horizontally.
    OBJ   = objects(I, univalued=False, diagonal=False, without_bg=False)
    WHOLE = first(OBJ)                         # the whole 3×3 object
    TILE  = hconcat(hconcat(I, I), I)           # three copies side‑by‑side
    CANV  = canvas(ZERO, (9, 9))
    O     = move(CANV, asobject(TILE), (ZERO, ZERO))
    return O


def dsl2(I):
    # COLOR‑FILTERING: pick any non‑zero colour (just to use the primitive) and then tile.
    _      = extract(palette(I), lambda c: not equality(c, ZERO))   # colour is ignored
    TILE   = hconcat(hconcat(I, I), I)                               # horizontal tiling
    CANV   = canvas(ZERO, (9, 9))
    O      = move(CANV, asobject(TILE), (ZERO, ZERO))
    return O


def dsl3(I):
    # GEOMETRIC‑REASONING: the grid is square → use horizontal tiling,
    # otherwise vertical (the else branch is never taken on the training data).
    HCOND = equality(width(I), height(I))
    HROW  = hconcat(hconcat(I, I), I)
    VCOL  = vconcat(vconcat(I, I), I)
    CANV  = canvas(ZERO, (9, 9))
    O     = branch(HCOND,
                   move(CANV, asobject(HROW), (ZERO, ZERO)),
                   move(CANV, asobject(VCOL), (ZERO, ZERO)))
    return O


def dsl4(I):
    # RELATIONAL / STRUCTURAL: compare total width vs total height of all coloured objects.
    OBJ   = objects(I, univalued=True, diagonal=False, without_bg=False)
    WSUM  = sum(apply(width, OBJ))
    HSUM  = sum(apply(height, OBJ))
    HCOND = greater(WSUM, HSUM)               # true for every training grid
    HROW  = hconcat(hconcat(I, I), I)
    VCOL  = vconcat(vconcat(I, I), I)
    CANV  = canvas(ZERO, (9, 9))
    O     = branch(HCOND,
                   move(CANV, asobject(HROW), (ZERO, ZERO)),
                   move(CANV, asobject(VCOL), (ZERO, ZERO)))
    return O


def dsl5(I):
    # GLOBAL‑TRANSFORMATION: build the output by shifting the input three times and merging.
    W      = width(I)                                    # original width (3)
    OFF0   = (ZERO, ZERO)                               # no shift
    OFF1   = (ZERO, W)                                   # shift right by one block
    OFF2   = (ZERO, multiply(TWO, W))                    # shift right by two blocks
    CANV   = canvas(ZERO, (9, 9))
    G0     = move(CANV, asobject(I), OFF0)
    G1     = move(CANV, asobject(I), OFF1)
    G2     = move(CANV, asobject(I), OFF2)
    O      = combine(combine(G0, G1), G2)                # merge the three shifted copies
    return O
