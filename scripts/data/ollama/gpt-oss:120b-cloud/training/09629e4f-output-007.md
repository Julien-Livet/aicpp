def dsl1(I):
    # ---------- global – cut the grid at every full‑row/column of colour 5 ----------
    # rows that are completely filled with 5
    rows = mfilter(
        interval(0, shape(I)[0], 1),
        lambda i: size(
            intersection(
                ofcolor(I, 5),
                apply(lambda j: (i, j), interval(0, shape(I)[1], 1))
            )
        ) == shape(I)[1]
    )
    # columns that are completely filled with 5
    cols = mfilter(
        interval(0, shape(I)[1], 1),
        lambda j: size(
            intersection(
                ofcolor(I, 5),
                apply(lambda i: (i, j), interval(0, shape(I)[0], 1))
            )
        ) == shape(I)[0]
    )
    # cut positions (include outer borders)
    rcuts = order(combine({0}, combine(set(rows), {shape(I)[0]})), lambda a, b: a < b)
    ccuts = order(combine({0}, combine(set(cols), {shape(I)[1]})), lambda a, b: a < b)

    # recolour each rectangular block delimited by two consecutive cuts
    blocks = apply(
        lambda r: apply(
            lambda c: recolor(
                mostcolor(
                    crop(
                        I,
                        (list(rcuts)[r], list(ccuts)[c]),
                        (list(rcuts)[r + 1] - list(rcuts)[r],
                         list(ccuts)[c + 1] - list(ccuts)[c])
                ),
                shoot(
                    (list(rcuts)[r], list(ccuts)[c]),
                    (list(rcuts)[r + 1] - list(rcuts)[r],
                     list(ccuts)[c + 1] - list(ccuts)[c])
                )
            ),
            interval(0, size(ccuts) - 1, 1)
        ),
        interval(0, size(rcuts) - 1, 1)
    )
    O = paint(I, merge(flatten(blocks)))
    return O


def dsl2(I):
    # ---------- object extraction – turn separators into background ----------
    # replace colour 5 with 0 (background)
    S = switch(I, 5, 0)

    # each connected component in S is a rectangular block bounded by the separators
    blocks = objects(S, False, False, False)

    # recolour the whole block with the most frequent non‑5 colour appearing in the
    # original input inside the same rectangle
    recoloured = apply(
        lambda o: recolor(
            mostcolor(subgrid(o, I)),
            o
        ),
        blocks
    )
    O = paint(I, merge(recoloured))
    return O


def dsl3(I):
    # ---------- colour‑filtering – work on each block individually ----------
    # obtain the indices of every separator cell (colour 5)
    sep = ofcolor(I, 5)

    # rows and columns that contain a separator line (full line of 5)
    rows = apply(ulcorner, mfilter(objects(I, True, False, False), hline))
    cols = apply(ulcorner, mfilter(objects(I, True, False, False), vline))

    # cut positions (including outer borders)
    rcuts = order(combine({0}, combine(set(rows), {shape(I)[0]})), lambda a, b: a < b)
    ccuts = order(combine({0}, combine(set(cols), {shape(I)[1]})), lambda a, b: a < b)

    # for every block compute the dominant colour (ignoring the separator colour 5)
    blocks = apply(
        lambda r: apply(
            lambda c: recolor(
                mostcolor(

                        I,
                        (list(rcuts)[r], list(ccuts)[c]),
                        (list(rcuts)[r + 1] - list(rcuts)[r],
                         list(ccuts)[c + 1] - list(ccuts)[c])
                    )
                ),
                shoot(
                    (list(rcuts)[r], list(ccuts)[c]),

            ),
            interval(0, size(ccuts) - 1, 1)
        ),

    O = paint(I, merge(flatten(blocks)))

def dsl4(I):
    # ---------- geometric – rotate, cut, rotate back ----------
    # after a 90° rotation the horizontal separator rows become vertical lines
    R = rot90(I)

    # objects that are whole separator lines in the rotated grid
    sep_objs = objects(R, True, False, False)
    rows = apply(ulcorner, mfilter(sep_objs, hline))   # vertical lines → rows after rotation
    cols = apply(ulcorner, mfilter(sep_objs, vline))   # horizontal lines → cols after rotation

    # cut positions in the rotated view
    rcuts = order(combine({0}, combine(set(rows), {shape(R)[0]})), lambda a, b: a < b)
    ccuts = order(combine({0}, combine(set(cols), {shape(R)[1]})), lambda a, b: a < b)

    # recolour each block in the rotated grid
    blocks = apply(
        lambda r: apply(
            lambda c: recolor(
                mostcolor(
                    crop(
                        R,
                        (list(rcuts)[r], list(ccuts)[c]),
                        (list(rcuts)[r + 1] - list(rcuts)[r],
                         list(ccuts)[c + 1] - list(ccuts)[c])
                ),
                shoot(
                    (list(rcuts)[r], list(ccuts)[c]),
                    (list(rcuts)[r + 1] - list(rcuts)[r],
                     list(ccuts)[c + 1] - list(ccuts)[c])
                )
            ),
            interval(0, size(ccuts) - 1, 1)
        ),
        interval(0, size(rcuts) - 1, 1)
    )
    Orot = paint(R, merge(flatten(blocks)))
    O = rot270(Orot)          # back to original orientation
    return O


def dsl5(I):
    # ---------- relational / structural – nearest separators ----------
    # set of all separator cells (colour 5)
    sep = ofcolor(I, 5)

    # every non‑separator cell
    non5 = mfilter(asindices(I), lambda p: index(I, p) != 5)

    # for each such cell locate the rectangle bounded by the nearest separators
    # in the four cardinal directions and recolour the cell with the dominant colour
    # of that rectangle (taken from the original input).
    rects = apply(
        lambda p: recolor(
            mostcolor(
                subgrid(
                    mfilter(asindices(I), lambda q:
                        q[0] >= gravitate({p}, sep)[0] + p[0] and
                        q[0] <= -gravitate({p}, sep)[0] + p[0] and
                        q[1] >= gravitate({p}, sep)[1] + p[1] and
                        q[1] <= -gravitate({p}, sep)[1] + p[1]
                    ),
                    I
                )
            ),
            {p}
        ),
        non5
    )
    O = paint(I, merge(rects))
    return O
