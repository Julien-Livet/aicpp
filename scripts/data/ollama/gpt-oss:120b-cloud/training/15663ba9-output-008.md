def dsl1(I):
    # GLOBAL EXTREME COLUMNS – colour the leftmost non‑zero column with 4
    # and the rightmost non‑zero column with 2 (the whole picture at once).
    nz_cols = {c for r in I for c in range(len(r)) if I[r][c] != 0}
    left  = min(nz_cols)
    right = max(nz_cols)
    left_mask  = recolor(4, sfilter(asindices(I), lambda rc: rc[1] == left))
    right_mask = recolor(2, sfilter(asindices(I), lambda rc: rc[1] == right))
    O = paint(I, left_mask)
    O = paint(O, right_mask)
    return O


def dsl2(I):
    # OBJECT‑LEVEL EXTREME COLUMNS – for every connected component
    # recolour its own leftmost column with 4 and its own rightmost column with 2.
    objs      = fgpartition(I)                                   # all objects (incl. bg)
    coloured  = mfilter(objs, lambda o: color(o) > 0)              # drop background
    lefts = merge(apply(
        lambda o: recolor(
            4,
            sfilter(toindices(o), lambda rc: rc[1] == leftmost(toindices(o)))
        ), coloured))
    rights = merge(apply(
        lambda o: recolor(
            2,
            sfilter(toindices(o), lambda rc: rc[1] == rightmost(toindices(o)))
        ), coloured))
    O = paint(I, lefts)
    O = paint(O, rights)
    return O


def dsl3(I):
    # RECTANGLE‑SIDES PER OBJECT – keep only those border cells of an object's
    # bounding box that already belong to the object, colour left side 4 and right side 2.
    objs      = fgpartition(I)
    coloured  = mfilter(objs, lambda o: color(o) > 0)
    lefts = merge(apply(
        lambda o: recolor(
            4,
            intersection(

        ), coloured))
    O = paint(I, lefts)
    O = paint(O, rights)
    return O
