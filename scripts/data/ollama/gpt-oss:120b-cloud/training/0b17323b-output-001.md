def dsl1(I):
    # geometric – extend the diagonal of 1’s with a 2‑line using the next cell after the last 1
    ones      = ofcolor(I, 1)
    last_row  = lowermost(ones)
    last_col  = rightmost(ones)
    start     = increment(astuple(last_row, last_col))
    direction = add(DOWN, RIGHT)                     # (1,1)
    line      = shoot(start, direction)              # all cells to the border
    O = fill(I, 2, line)
    return O


def dsl2(I):
    # periodicity – use the horizontal period of the 1‑objects as the step vector
    ones      = ofcolor(I, 1)
    # the three 1‑pixels form a line; its horizontal period equals the row‑step
    step      = hperiod(ones)                         # ∆c  (equals ∆r)
    v         = astuple(step, step)                   # (step,step)
    last_row  = lowermost(ones)
    last_col  = rightmost(ones)
    start     = increment(astuple(last_row, last_col))
    line      = shoot(start, v)                       # continue with the same period
    O = fill(I, 2, line)
    return O


def dsl3(I):
    # whole‑diagonal – take the complete main diagonal, then keep only the still‑empty part
    h, w      = shape(I)
    full_diag = connect(astuple(0, 0), astuple(h - 1, w - 1))
    ones_set  = ofcolor(I, 1)
    fresh     = difference(full_diag, ones_set)       # cells on the diagonal that are 0
    O = fill(I, 2, fresh)
    return O


def dsl4(I):
    # object‑based – shift the 1‑object repeatedly by its own displacement until the border is hit
    ones      = ofcolor(I, 1)
    pts       = tuple(sorted(ones))
    v         = subtract(pts[1], pts[0])               # displacement between successive 1’s
    last_pt   = pts[-1]
    start     = add(last_pt, v)                       # first cell after the existing line
    direction = v
    line      = shoot(start, direction)                # same as “keep shifting”
    O = fill(I, 2, line)
    return O


def dsl5(I):
    # relational – cells on the main diagonal whose Manhattan distance from (0,0)
    # is larger than the farthest 1‑pixel’s distance
    ones          = ofcolor(I, 1)
    max_dist      = valmax(ones, lambda p: manhattan(initset((0, 0)), initset(p)))
    h, w          = shape(I)
    full_diag     = connect(astuple(0, 0), astuple(h - 1, w - 1))
    far_diag      = {p for p in full_diag
                     if manhattan(initset((0, 0)), initset(p)) > max_dist}
    O = fill(I, 2, far_diag)
    return O
