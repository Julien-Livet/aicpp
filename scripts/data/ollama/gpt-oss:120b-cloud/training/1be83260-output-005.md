def dsl1(I):
    # Global transformation: crop outer zero border, turn interior 0 → 1,
    # then swap colours 2 ↔ 3 (colour 4 stays unchanged).
    G = compress(I)                     # remove outer zero border
    B = replace(G, 0, 1)                # interior background → 1
    O = switch(B, 2, 3)                 # swap 2 and 3
    return O


def dsl2(I):
    # Purely geometric trimming: trim exactly one outer layer,
    # recolour the remaining background and swap 2 ↔ 3.
    G = trim(I)                         # cut off one outer row/col
    B = replace(G, 0, 1)                # background → 1
    O = switch(B, 2, 3)                 # swap 2 and 3
    return O


def dsl3(I):
    # Object‑level recolouring: extract every coloured object,
    # recolour background (0) to 1, swap 2 ↔ 3 and repaint.
    G   = compress(I)                                            # remove outer zeros
    BG  = replace(G, 0, 1)                                      # provisional background
    OBJ = fgpartition(G)                                         # all non‑zero objects
    SET2 = colorfilter(OBJ, 2)                                   # objects of colour 2
    SET3 = colorfilter(OBJ, 3)                                   # objects of colour 3
    SET4 = colorfilter(OBJ, 4)                                   # objects of colour 4
    SW2 = apply(lambda o: recolor(3, o), SET2)                  # 2 → 3
    SW3 = apply(lambda o: recolor(2, o), SET3)                  # 3 → 2
    ALL = combine(combine(SW2, SW3), SET4)                       # union of recoloured objects
    O   = paint(BG, ALL)                                          # paint them on the new background
    return O


def dsl4(I):

    # swap 2 ↔ 3 and rotate back.
    R   = rot90(I)                     # clockwise rotation

    B   = replace(C, 0, 1)             # background → 1
    S   = switch(B, 2, 3)              # swap 2 and 3
    O   = rot270(S)                    # rotate back to original orientation
    return O


def dsl5(I):
