def dsl1(I):
    # ---------- GLOBAL ----------
    # 1.  Crop to the tight non‑zero bounding box.
    # 2.  Fill every remaining background cell with the most frequent colour
    #     that appears inside that box (ignoring colour 0).
    T   = trim(I)                                 # non‑zero bounding box
    O   = canvas(0, shape(T))                    # empty canvas of trimmed size
    O   = paint(O, asobject(T))                  # copy the trimmed pattern
    C   = mostcolor(T)                           # dominant colour inside the box
    O   = fill(O, C, ofcolor(O, 0))               # recolour all zeros
    return O


def dsl2(I):
    # ---------- OBJECT EXTRACTION ----------
    # Keep only the largest connected component (by number of cells),
    # irrespective of its colour.
    OBJ = objects(I, univalued=False, diagonal=False, without_bg=True)
    BIG = argmax(OBJ, lambda o: size(o))          # biggest component
    O   = canvas(0, shape(I))
    O   = paint(O, BIG)                           # render the chosen component
    return O


def dsl3(I):
    # ---------- COLOUR FILTERING ----------
    # Remove the three “marker’’ colours 3, 5 and 8, and also delete any colour
    # that occurs only once in the whole grid.
    BAD_MARK   = combine(combine(ofcolor(I, 5), ofcolor(I, 8)), ofcolor(I, 3))
    ONE_OCC   = ofcolor(I, branch(equality(colorcount(I, mostcolor(I)), 1),
                         mostcolor(I), 0)        # cells of a colour that appears once
    BAD       = combine(BAD_MARK, ONE_OCC)
    O         = fill(I, 0, BAD)                   # replace them by background
    return O


def dsl4(I):
    # ---------- GEOMETRIC REASONING ----------
    # Create horizontal and vertical mirrors, overlay them (original cells dominate).
    H = hmirror(I)                                 # horizontal flip
    V = vmirror(I)                                 # vertical flip
    M = cellwise(H, V, fallback=0)                # union of the two mirrors
    O = cellwise(I, M, fallback=0)                # original cells keep precedence
    return O


def dsl5(I):
    # ---------- RELATIONAL / STRUCTURAL ----------
    # For every coloured region, recolour its outline (the box surrounding it)
    # with the region's own colour.
    OBJ = objects(I, univalued=False, diagonal=False, without_bg=True)
    O   = canvas(0, shape(I))
    O   = apply(lambda o: paint(O,
                               recolor(color(o), box(o))), OBJ)
    return O
