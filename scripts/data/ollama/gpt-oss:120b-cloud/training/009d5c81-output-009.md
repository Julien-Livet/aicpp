def dsl1(I):
    # GLOBAL – size + interior density decides new colour,
    # then erase any 1‑pixels that lie inside the 8‑bbox.
    p8        = ofcolor(I, 8)
    sz        = size(p8)
    interior  = intersection(p8, shift(p8, UP))
    interior  = intersection(interior, shift(p8, DOWN))
    interior  = intersection(interior, shift(p8, LEFT))
    interior  = intersection(interior, RIGHT)
    cnt_int   = size(interior)
    col = branch(greater(sz, 50), 7,
           branch(greater(cnt_int, 0), 3, 2))
    O = replace(I, 8, col)                     # recolour the 8‑object
    bbox      = backdrop(p8)                    # bounding rectangle of the 8‑object
    ones      = ofcolor(I, 1)
    to_clear  = intersection(ones, bbox)      # 1‑pixels inside the bbox
    O = cover(O, to_clear)                    # erase them
    return O


def dsl2(I):

    # recolour it, and clear any other colour that touches the object.
    p8        = ofcolor(I, 8)
    interior  = intersection(p8, shift(p8, UP))
    interior  = intersection(interior, shift(p8, DOWN))
    interior  = intersection(interior, shift(p8, LEFT))
    interior  = intersection(interior, RIGHT)
    cnt_int   = size(interior)
    sz        = size(p8)
    col = branch(greater(sz, 50), 7,
           branch(greater(cnt_int, 0), 3, 2))
    O = replace(I, 8, col)
    # find every non‑zero cell that is adjacent (4‑neighbour) to the 8‑object
    nbrs      = union(shift(p8, UP), shift(p8, DOWN), shift(p8, LEFT), shift(p8, RIGHT))

    O = cover(O, other)                       # erase those neighbours
