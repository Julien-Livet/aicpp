    vp = vperiod(asindices(I))                 # vertical period of the full index set
    hp = hperiod(asindices(I))                 # horizontal period of the full index set

## DSL
```python
def dsl2(I):
    # Object extraction – tight box around the part of the picture that stays unchanged
    # after a horizontal mirror (the invariant cells)
    mirrored   = hmirror(I)                                    # horizontally mirrored picture
    invariant  = intersection(asobject(I), asobject(mirrored))   # cells that are identical in both
    O          = subgrid(toindices(invariant), I)              # minimal rectangle enclosing them
    return O
```

def dsl5(I):
    # Relational – region formed by the rarest colour together with its
    # orthogonal neighbours; if that region is empty, return the whole picture
    rare      = leastcolor(I)                                 # colour occurring the fewest times
    rc        = ofcolor(I, rare)                              # cells of that colour
    neigh     = combine(rc,
                        combine(shift(rc, DOWN),
                                combine(shift(rc, UP),
                                        combine(shift(rc, LEFT),
                                                shift(rc, RIGHT)))))
    O         = subgrid(neigh, I)                             # tight box around the neighbourhood
    return O
