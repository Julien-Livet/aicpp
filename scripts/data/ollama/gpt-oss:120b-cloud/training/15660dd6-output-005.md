def dsl1(I):
    # GLOBAL – trim the uniform outer 8‑border, then recolour the interior.
    c   = trim(I)                # remove the whole outer frame of 8s
    c1  = replace(c, 8, 5)       # original interior 8 → 5
    c2  = replace(c1, 1, 9)      # 1 → 9
    c3  = replace(c2, 3, 8)      # 3 → 8
    O   = replace(c3, 2, 7)      # 2 → 7
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION – select the largest non‑8 object, crop to it,
    # fill the rest with 9 and recolour.
    non8   = ofcolor(I, 1) | ofcolor(I, 2) | ofcolor(I, 3) | ofcolor(I, 8)
    objs   = fgpartition(I, False, False, True)          # all objects, background excluded
    target = argmax(objs, lambda o: size(o))              # biggest object (the pattern)
    pat    = target & non8                                # keep only the needed colors
    bbox   = backdrop(pat)                                 # minimal rectangle containing the pattern
    sub    = subgrid(bbox, I)                             # crop to that rectangle
    base   = underfill(sub, 9, pat)                       # background → 9
    o1     = replace(base, 8, 5)
    o2     = replace(o1,   1, 9)
    o3     = replace(o2,   3, 8)
    O      = replace(o3,   2, 7)
    return O


def dsl3(I):
