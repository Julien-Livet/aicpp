def dsl1(I):
    # -------- GLOBAL‑TRANSFORM ----------
    # For every colour > 1 build the orthogonal bounding box of *all* its cells,
    # keep only the 1‑cells that lie inside it and recolour them with that colour.
    # The whole operation is performed by one parallel “apply‑mapply” pass.
    seeds   = sfilter(palette(I), lambda c: c > 1)                 # colours to treat
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)           # their boxes
    patches = mapply(
                lambda c, b: recolor(
                                c,
                                intersection(b, ofcolor(I, 1))   # 1‑cells inside the box
                              ),
                seeds,
                boxes)
    O = paint(I, merge(patches))
    return O


def dsl2(I):
    # -------- OBJECT‑EXTRACTION ----------
    # Treat every *object* (connected component) of colour > 1 as a separate entity.
    # For each object take its orthogonal out‑box, intersect it with the set of 1‑cells
    # and recolour that intersection with the object’s colour.
    objs    = sfilter(fgpartition(I), lambda o: color(o) > 1)      # objects of colour >1
    patches = apply(
                lambda o: recolor(
                            color(o),
                            intersection(
                                outbox(o),                     # orthogonal box of the object
                                ofcolor(I, 1)                  # all 1‑cells
                            )
                         ),
                objs)
    O = paint(I, merge(patches))
    return O


def dsl3(I):
    # -------- COLOUR‑FILTERING ----------
    # Directly work on colours (no objects).  For each colour > 1 take the bounding box
    # of *all* its cells, intersect with the 1‑cells and recolour the intersection.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                box(ofcolor(I, c)),   # full box of colour c
                                ofcolor(I, 1)        # all 1‑cells
                            )
                         ),
                seeds)
    O = paint(I, merge(patches))
    return O


def dsl4(I):
    # -------- GEOMETRIC‑REASONING ----------
    # For every colour > 1 locate its extreme corners (UL and LR).  Connect the two
    # corners, turn the resulting segment into a rectangle with `box`, intersect with
    # the 1‑cells and recolour the result.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                box(
                                    connect(
                                        ulcorner(ofcolor(I, c)),   # upper‑left corner
                                        lrcorner(ofcolor(I, c))    # lower‑right corner
                                    )
                                ),
                                ofcolor(I, 1)                   # all 1‑cells
                            )
                         ),
                seeds)
    O = paint(I, merge(patches))
    return O


def dsl5(I):
    # -------- RELATIONAL‑STRUCTURAL ----------
    # First erase all 1‑cells (`cover`).  Then, for each colour > 1, paint the part of
    # its bounding box that originally contained a 1‑cell.  The mask is built by
    # intersecting the box with the original set of 1‑cells, ensuring that no other
    # colour is overwritten.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)
    mask    = apply(lambda b: intersection(b, ofcolor(I, 1)), boxes)   # keep only original 1‑cells
    patches = mapply(lambda c, m: recolor(c, m), seeds, mask)          # recolour those cells
    base    = cover(I, ofcolor(I, 1))                                 # grid without any 1‑cell
    O = paint(base, merge(patches))
    return O
