def dsl1(I):
    # GLOBAL‑TRANSFORMATION: compress the shape, then tile it by shifting the
    # whole (compressed) object to every foreground cell.
    N      = height(I)                                           # block size
    fg     = argmax(difference(palette(I), initset(0)), lbind(colorcount, I))
    # keep only the colored shape, throw away empty border
    shape  = compress(I)
    Pobj   = asobject(shape)                                    # object of the shape
    cells  = ofcolor(I, fg)                                      # foreground anchors
    shifted = apply(lambda p: shift(Pobj, (p[0]*N, p[1]*N)), cells)
    O = paint(canvas(0, (N*N, N*N)), merge(shifted))
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION: treat each foreground pixel as an independent object,
    # copy the whole inverse pattern to the position of each pixel.
    N      = height(I)
    fg     = argmax(difference(palette(I), initset(0)), lbind(colorcount, I))
    inv    = replace(replace(I, fg, 0), 0, fg)                   # colour swap
    # every foreground pixel becomes a one‑cell object
    pixobj = objects(I, True, False, True)
    cells  = ofcolor(I, fg)
    shifted = apply(lambda p: shift(pixobj, (p[0]*N, p[1]*N)), cells)
    O = paint(canvas(0, (N*N, N*N)), merge(shifted))
    return O


def dsl3(I):
    # GEOMETRIC‑REASONING: mirror the layout before replication.
    N      = height(I)
    fg     = argmax(difference(palette(I), initset(0)), lbind(colorcount, I))
    inv    = replace(replace(I, fg, 0), 0, fg)
    # diagonal mirror (rotate 90 after a horizontal flip) gives a new layout
    layout = rot90(hmirror(I))
    cells  = ofcolor(layout, fg)
    Pobj   = asobject(inv)
    shifted = apply(lambda p: shift(Pobj, (p[0]*N, p[1]*N)), cells)
    O = paint(canvas(0, (N*N, N*N)), merge(shifted))
    return O

    shifted = apply(lambda p: shift(Pobj, (p[0]*N, p[1]*N)), target)

def dsl5(I):
    # COLOR‑FILTERING: build a binary mask of the foreground, upscale it,
    # tile the original (non‑inverse) pattern, then keep only the tiled blocks
    # where the mask is 1.
    N      = height(I)
    fg     = argmax(difference(palette(I), initset(0)), lbind(colorcount, I))
    # binary mask (1 = foreground, 0 = background)
    mask   = replace(I, fg, 1)
    # upscale the mask to N×N blocks
    bigmask = upscale(mask, N)
    # tile the original input pattern N times horizontally and vertically
    row    = apply(lambda _: I, interval(0, N, 1))               # list of N rows
    tiled_row = fold(row, lambda a,b: hconcat(a,b), I)           # one full row
    tiled   = apply(lambda _: tiled_row, interval(0, N, 1))        # N identical rows
    tiled_grid = fold(tiled, lambda a,b: vconcat(a,b), tiled_row)
    # keep only cells whose bigmask is 1 (i.e. foreground blocks)
    O = fill(tiled_grid, 0, ofcolor(bigmask, 0))
    return O
