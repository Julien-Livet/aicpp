def dsl1(I):
    # GLOBAL‑TILING:  place the compressed foreground shape at *every* cell of the
    # input (including background).  The N×N blocks therefore overlap and create
    # the thick diagonal bands seen in the examples.
    N      = height(I)                                            # block size
    fg     = argmax(difference(palette(I), initset(0)),
                    lbind(colorcount, I))                         # dominant colour
    shape  = compress(I)                                          # remove empty border
    Pobj   = asobject(shape)                                      # object of the shape
    cells  = asindices(I)                                         # all input coordinates
    shifted = apply(lambda p: shift(Pobj, (p[0]*N, p[1]*N)), cells)
    O      = paint(canvas(0, (N*N, N*N)), merge(shifted))
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION + INVERSION:  each foreground cell becomes a 1‑cell
    # object; the colour‑swapped whole pattern is copied into the N×N block that
    # originates at that cell.
    N      = height(I)
    fg     = argmax(difference(palette(I), initset(0)),
                    lbind(colorcount, I))
    inv    = replace(replace(I, fg, 0), 0, fg)                     # swap fg ↔ 0
    Pobj   = asobject(inv)                                       # whole inverted pattern
    cells  = ofcolor(I, fg)                                       # foreground anchors only
    shifted = apply(lambda p: shift(Pobj, (p[0]*N, p[1]*N)), cells)
    O      = paint(canvas(0, (N*N, N*N)), merge(shifted))
    return O


def dsl3(I):
    # COLOR‑FILTERING + MASKED TILING:  build a binary mask of the dominant colour,
    # upscale it, tile the original (non‑inverted) pattern with hconcat/vconcat,
    # then keep only the blocks whose mask cell is 1.
    N      = height(I)
    fg     = argmax(difference(palette(I), initset(0)),
                    lbind(colorcount, I))
    mask   = replace(I, fg, 1)                                     # 1 = foreground
    bigmask = upscale(mask, N)                                    # each mask cell → N×N block
    # tile the whole input pattern N times horizontally and vertically
    row    = apply(lambda _: I, interval(0, N, 1))               # N copies of the row‑list
    tiled_row = fold(row, lambda a,b: hconcat(a,b), I)              # one full horizontal strip
    tiled   = apply(lambda _: tiled_row, interval(0, N, 1))       # N copies of the strip
    tiled_grid = fold(tiled, lambda a,b: vconcat(a,b), tiled_row) # full N‑by‑N tiling
    # zero‑out everything where the up‑scaled mask is 0
    O = fill(tiled_grid, 0, ofcolor(bigmask, 0))
    return O
