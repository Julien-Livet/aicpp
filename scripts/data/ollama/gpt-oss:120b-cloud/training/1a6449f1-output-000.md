```python
def dsl1(I):
    # global transformation: remove the most frequent colour and trim empty border
    mc = mostcolor(I)                         # most common colour (usually background)
    O  = trim(replace(I, mc, 0))             # set it to 0 then trim surrounding zeros
    return O


def dsl2(I):
    # object extraction: take the tight sub‑grid around the least frequent colour
    lc = leastcolor(I)                       # colour that appears the fewest times
    O  = subgrid(ofcolor(I, lc), I)          # smallest rectangle that contains all its cells
    return O


def dsl3(I):
    # colour‑filtering: erase the dominant colour and keep the rest, then trim
    mc = mostcolor(I)
    O  = trim(cover(I, ofcolor(I, mc)))      # delete the dominant colour, trim border
    return O


def dsl4(I):
    # geometric reasoning: crop to the bounding box that contains both the most
    # and the least frequent colours
    mc = mostcolor(I)
    lc = leastcolor(I)
    O  = subgrid(combine(ofcolor(I, mc), ofcolor(I, lc)), I)  # rectangle around both groups
    return O


def dsl5(I):
    # relational / structural: centre a fixed 5×5 window on the least frequent colour
    lc   = leastcolor(I)
    ctr  = center(ofcolor(I, lc))                       # centre of that colour’s cells
    start = add(ctr, astuple(-2, -2))                    # top‑left corner of the window
    O    = crop(I, start, astuple(5, 5))                # extract the 5×5 neighbourhood
    return O
```
