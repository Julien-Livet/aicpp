def dsl1(I):
    # GLOBAL‑TILING + MASKING
    # 1️⃣  dominant non‑zero colour
    fg = argmax(difference(palette(I), initset(0)),
                lbind(colorcount, I))
    N  = height(I)                                    # block size
    # 2️⃣  colour swap  (0 ↔ fg)  using a temporary colour (‑1)
    tmp = replace(I, 0, NEG_ONE)                       # 0 → -1
    tmp = replace(tmp, fg, 0)                          # fg → 0
    inv = replace(tmp, NEG_ONE, fg)                    # -1 → fg   →  inv is the inverted pattern
    # 3️⃣  tile the inverted pattern at **every** cell (including background)
    all_cells = asindices(I)
    tiled_obj = apply(lambda p: shift(asobject(inv),
                                      (p[0] * N, p[1] * N)), all_cells)
    tiled = paint(canvas(0, (N * N, N * N)), merge(tiled_obj))
    # 4️⃣  build a binary mask of the foreground and upscale it to block size
    mask = replace(I, fg, 1)                           # 1 = foreground
    bigmask = upscale(mask, N)                         # each mask cell → N×N block
    # 5️⃣  keep only blocks whose mask cell is 1
    O = fill(tiled, 0, ofcolor(bigmask, 0))
    return O
