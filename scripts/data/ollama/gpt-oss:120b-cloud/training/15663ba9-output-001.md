def dsl1(I):
    # Object‑based endpoint recoloring
    #  – for every non‑background object recolor the cells that lie on its
    #    leftmost column with 4,
    #    the cells that lie on its rightmost column with 2.
    #  – left‑side (4) is painted after right‑side (2) so that width‑1
    #    objects keep colour 4.
    objs      = fgpartition(I)                                      # all objects
    non_bg   = mfilter(objs, lambda o: color(o) > 0)                # ignore bg
    # left‑most cells of every object
    lefts = apply(
        lambda o: recolor(
            4,
            { (r, c) for (col, (r, c)) in o if c == leftmost(o) }
        ),
        non_bg
    )
    # right‑most cells of every object
    rights = apply(
        lambda o: recolor(
            2,
            { (r, c) for (col, (r, c)) in o if c == rightmost(o) }
        ),
        non_bg
    )
    # paint right side first, then left side (4 overrides 2 on width‑1 objects)
    O = I
    for p in rights:   O = paint(O, p)
    for p in lefts:    O = paint(O, p)
    return O


def dsl2(I):
    # Colour‑frequency replacement
    #  – most frequent non‑zero colour → 4,
    #    second most frequent non‑zero colour → 2.
    #  – 4 is painted after 2 so it has priority where the two colours meet.
    nonzero_palette = palette(I) - {0}
    most   = mostcommon(totuple(nonzero_palette))                      # most frequent
    second = other(totuple(nonzero_palette - {most}), most)             # second most
    # patches for the two colours
    p_most   = recolor(4, ofcolor(I, most))
    p_second = recolor(2, ofcolor(I, second))
    O = I
    O = paint(O, p_second)   # 2 first
    O = paint(O, p_most)    # 4 on top
    return O


def dsl3(I):

def dsl4(I):
    # Relational – distance from the outer border
    #  – cells at Manhattan distance 1 from the outer border become 4,
    #    cells at distance 2 become 2, everything else unchanged.
    h, w      = shape(I)
    border    = ({ (0, c) for c in range(w) } |
                { (h-1, c) for c in range(w) } |
                { (r, 0) for r in range(h) } |
                { (r, w-1) for r in range(h) })
    d1        = { p for b in border for p in neighbors(b)
                 if 0 <= p[0] < h and 0 <= p[1] < w }
    d2        = { p for d in d1 for p in neighbors(d)
                 if 0 <= p[0] < h and 0 <= p[1] < w and p not in border and p not in d1 }
    O = recolor(4, d1)
    O = recolor(2, d2)
    return O


def dsl5(I):
