def dsl1(I):
    # GLOBAL: build the whole repeated grid, then replace the middle
    # vertical third with its horizontal mirror.
    full   = vupscale(hupscale(I, 3), 3)                # 6×6, original everywhere
    top    = crop(full, (0, 0), (2, 6))                # rows 0‑1
    middle = crop(full, (2, 0), (2, 6))                # rows 2‑3
    bottom = crop(full, (4, 0), (2, 6))                # rows 4‑5
    middle = hmirror(middle)                          # mirror only the middle part
    O = vconcat(top, vconcat(middle, bottom))
    return O


def dsl2(I):
    # OBJECT‑LEVEL: treat each cell as an object, upscale it,
    # and place three copies (original, mirrored, original) at the
    # proper vertical offsets.
    orig_objs = objects(I, True, False, False)
    orig_up   = apply(lambda o: shift(upscale(toobject(o, I), 3), (0, 0)), orig_objs)

    mirr_grid = hmirror(I)
    mirr_objs = objects(mirr_grid, True, False, False)
    mirr_up   = apply(lambda o: shift(upscale(toobject(o, mirr_grid), 3), (3, 0)), mirr_objs)

    all_blocks = combine(orig_up, mirr_up)
    O = paint(canvas(0, (6, 6)), all_blocks)
    return O


def dsl3(I):
    # COLOR‑FILTERING: start from a blank canvas, paste the three
    # horizontal‑scaled parts (original‑mirrored‑original) and finally
    # replace any remaining background (0) with the most frequent colour.
    blank = canvas(0, (6, 6))
    A = hupscale(I, 3)               # original orientation
    B = hupscale(hmirror(I), 3)       # mirrored orientation
    O1 = paint(blank, asobject(A))
    O2 = paint(O1,    asobject(shift(B, (2, 0))))   # middle block
    O  = paint(O2,    asobject(shift(A, (4, 0))))   # bottom block
    bgcol = mostcolor(O)
    O = underfill(O, bgcol, ofcolor(O, 0))
    return O


def dsl4(I):
    # GEOMETRIC REASONING: rotate the input, apply the basic three‑part
    # tiling, then rotate the result back to the original orientation.
    R      = rot90(I)                         # 90° clockwise
    A = hupscale(R, 3)
    B = hupscale(hmirror(R), 3)
    tiled  = vconcat(A, vconcat(B, A))
    O = rot270(tiled)                         # rotate back (270°)
    return O


def dsl5(I):

    # using explicit shifts (0 rows, 2 rows, 4 rows).
