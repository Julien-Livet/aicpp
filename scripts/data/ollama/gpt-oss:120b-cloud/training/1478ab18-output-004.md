def dsl1(I):
    # GLOBAL‑ROTATE: rotate, colour the 5‑bounding box with 8, restore 5’s,
    # then rotate back to the original orientation.
    R        = rot90(I)                                            # turn the whole grid
    five     = ofcolor(R, 5)                                      # 5‑pixels in the rotated view
    box      = backdrop(five)                                     # their bounding rectangle
    O1       = fill(R, 8, box)                                    # paint the rectangle 8
    O2       = paint(O1, toobject(five, O1))                       # restore the original 5’s
    O        = rot270(O2)                                          # undo the rotation
    return O


def dsl2(I):
    # OBJECT‑MIRROR‑H: extract the sub‑grid that encloses the 5’s,
    # mirror it horizontally, recolour the mirrored 5‑pixels with 8,
    # and overlay the result on the original grid.
    five     = ofcolor(I, 5)                                      # positions of all 5’s
    box      = backdrop(five)                                     # their bounding box
    sub      = subgrid(box, I)                                    # the enclosing rectangle
    mir      = hmirror(sub)                                       # horizontal mirror of that rectangle
    mir_five = ofcolor(mir, 5)                                    # where the mirrored 5’s appear
    O        = fill(I, 8, mir_five)                               # colour those cells 8
    O        = paint(O, toobject(five, O))                        # put the original 5’s back
    return O


def dsl3(I):
    # ADJACENT‑FILTER: recolour every 7 that is orthogonally adjacent to a 5.
    five          = ofcolor(I, 5)                                 # the 5‑pixels
    # for each 5 produce its four direct neighbours and unite them all
    neigh_sets    = apply(lambda p: neighbors(p), five)            # a set of neighbour‑sets
    all_neigh     = combine(*neigh_sets)                          # union of all neighbour cells
    target        = intersection(all_neigh, ofcolor(I, 7))           # those neighbours that are 7’s
    O             = fill(I, 8, target)                            # recolour them 8
    return O


def dsl4(I):
    # CENTER‑CROSS: compute the centre of mass of the 5’s,
    # draw a full horizontal and vertical line through that centre,
    # and colour the intersecting 7‑cells with 8.
    five          = ofcolor(I, 5)                                 # positions of the 5’s
    cm            = centerofmass(five)                             # (row, col) of the centre
    row_line      = shoot((cm[0], 0), RIGHT)                        # whole row through centre
    col_line      = shoot((0, cm[1]), DOWN)                        # whole column through centre
    cross         = combine(row_line, col_line)                    # the cross
    target        = intersection(cross, ofcolor(I, 7))              # 7‑cells that lie on the cross
    O             = fill(I, 8, target)                            # recolour them 8

def dsl5(I):
