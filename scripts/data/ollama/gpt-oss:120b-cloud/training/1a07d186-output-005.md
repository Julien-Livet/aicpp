def dsl1(I):
    # ---------- GLOBAL – compress the pattern to the top‑left corner ----------
    C      = compress(I)                         # cut away empty outer rows/cols
    OBJ    = asobject(C)                         # turn the compact grid into an object
    O      = underpaint(canvas(0, shape(I)), OBJ) # render it on a blank canvas of original size
    return O


def dsl2(I):
    # ---------- OBJECT EXTRACTION – keep only line‑shaped objects ----------
    OBJS   = fgpartition(I)                                         # all coloured objects
    LINES  = mfilter(OBJS, lambda o:   hline(toindices(o))           # keep horizontal lines
                              or   vline(toindices(o)))           # or vertical lines
    COMB   = merge(LINES)                                            # union of the selected objects
    O      = underpaint(canvas(0, shape(I)), COMB)                    # paint on a blank canvas
    return O


def dsl3(I):
    # ---------- COLOR FILTERING – retain the most frequent non‑zero colour ----------
    MCOL   = mostcolor(I)                                 # colour that occurs most often
    IDX    = ofcolor(I, MCOL)                             # its indices
    OBJ    = recolor(MCOL, IDX)                           # patch coloured with MCOL
    O      = underpaint(canvas(0, shape(I)), OBJ)         # render on empty canvas
    return O


def dsl4(I):
    # ---------- GEOMETRIC REASONING – overlay the 180° rotation ----------
    ORIG   = asobject(I)                                 # original as an object
    ROT    = asobject(rot180(I))                         # rotate 180°
    COMB   = merge({ORIG, ROT})                          # union of both objects
    O      = underpaint(canvas(0, shape(I)), COMB)         # paint on a blank canvas
    return O


def dsl5(I):
    # ---------- RELATIONAL / STRUCTURAL – align every object vertically
    #               with the topmost row of the largest object ----------
    OBJS       = fgpartition(I)                                              # all objects
    LARGEST    = argmax(OBJS, lambda o: size(o))                              # object with most cells
    REF_TOP    = topmost(toindices(LARGEST))                                   # its top row
    SHIFTED    = apply(
                    lambda o:
                        shift(
                            toindices(o),
                            subtract( astuple(REF_TOP, 0), astuple(topmost(toindices(o)), 0)
                        ),
                    OBJS)
    COMBINED   = merge(SHIFTED)                                               # reunite shifted objects
    O          = underpaint(canvas(0, shape(I)), COMBINED)                    # paint on blank canvas
    return O
