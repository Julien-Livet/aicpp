def dsl1(I):
    # --- Object‑extraction / placement strategy ---
    # foreground colour = most common non‑zero colour
    fg = argmax(
            difference(palette(I), initset(0)),
            lbind(colorcount, I)
         )
    # inverse pattern (swap 0 and fg)
    Z  = canvas(0, shape(I))
    Z1 = fill(Z, fg, ofcolor(I, 0))
    P  = fill(Z1, 0, ofcolor(I, fg))
    # empty output canvas (N·N)
    N   = height(I)                     # = width(I)
    O   = canvas(0, (N*N, N*N))
    # place a copy of P at every fg cell
    patches = apply(
                lambda idx: move(O, asobject(P), (idx[0]*N, idx[1]*N)),
                ofcolor(I, fg)
              )
    O = fold(patches, lambda acc, g: paint(acc, asobject(g)), O)
    return O


def dsl2(I):

    fg = argmax(
            difference(palette(I), initset(0)),
            lbind(colorcount, I)
         )
    # build inverse pattern

    return O
