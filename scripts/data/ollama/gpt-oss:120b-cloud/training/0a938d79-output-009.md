def dsl1(I):
    # ── GLOBAL‑TRANSFORM ──  compress → pattern → tile both axes
    slice_ = compress(I)                                   # tight bounding box
    objs   = objects(slice_, False, False, True)           # objects already relative
    hb, wb = height(slice_), width(slice_)

    base = canvas(0, (hb, wb))

    rows_pat = merge(apply(
        lambda o: fill(base,
                       color(o),
                       {(r, c) for (r, c) in asindices(base)
                        if r == center(o)[0]}),
        objs))

    cols_pat = merge(apply(
        lambda o: fill(base,
                       color(o),
                       {(r, c) for (r, c) in asindices(base)
                        if c == center(o)[1]}),
        objs))

    pat = branch(greater(hb, wb), rows_pat, cols_pat)      # orientation by aspect
    tmp = vupscale(pat, divide(height(I), hb))             # repeat vertically
    result = hupscale(tmp, divide(width(I), wb))            # repeat horizontally
    return result


def dsl2(I):
    # ── OBJECT‑EXTRACTION ──  bounding box → relative centres → pattern
    objs    = objects(I, False, False, True)
    merged  = merge(objs)

    top  = uppermost(merged)
    left = leftmost(merged)
    origin = astuple(top, left)                            # top‑left of whole slice

    slice_ = subgrid(merged, I)                           # tight box
    hb, wb = height(slice_), width(slice_)

    base = canvas(0, (hb, wb))

    rows_pat = merge(apply(
        lambda o: fill(base,
                       color(o),
                       {(r, c) for (r, c) in asindices(base)
                        if r == subtract(center(o), origin)[0]}),
        objs))

    cols_pat = merge(apply(
        lambda o: fill(base,
                       color(o),
                       {(r, c) for (r, c) in asindices(base)
                        if c == subtract(center(o), origin)[1]}),
        objs))

    pat = branch(greater(hb, wb), rows_pat, cols_pat)      # tall → rows, short → cols
    tmp = vupscale(pat, divide(height(I), hb))
    result = hupscale(tmp, divide(width(I), wb))
    return result


def dsl3(I):
    # ── COLOR‑FILTERING ── treat each colour separately
    objs   = objects(I, False, False, True)
    merged = merge(objs)
    top  = uppermost(merged)
    left = leftmost(merged)
    origin = astuple(top, left)

    slice_ = subgrid(merged, I)
    hb, wb = height(slice_), width(slice_)

    base = canvas(0, (hb, wb))

    rows_pat = merge(apply(
        lambda col:
            let_o = first(colorfilter(objs, col));
            fill(base,
                 col,
                 {(r, c) for (r, c) in asindices(base)
                  if r == subtract(center(let_o), origin)[0]}),
        palette(I)))

    cols_pat = merge(apply(
        lambda col:
            let_o = first(colorfilter(objs, col));
            fill(base,
                 col,
                 {(r, c) for (r, c) in asindices(base)
                  if c == subtract(center(let_o), origin)[1]}),
        palette(I)))

    pat = branch(greater(hb, wb), rows_pat, cols_pat)
    tmp = vupscale(pat, divide(height(I), hb))
    result = hupscale(tmp, divide(width(I), wb))
    return result


def dsl4(I):

def dsl5(I):
    # ── RELATIONAL REPEAT ── use actual row/col spacing as repeat unit
    objs = objects(I, False, False, True)
    merged = merge(objs)

    top  = uppermost(merged)
    left = leftmost(merged)
    origin = astuple(top, left)

    slice_ = subgrid(merged, I)
    hb, wb = height(slice_), width(slice_)

    # sets of distinct row / column indices of object centres
    rows_set = {center(o)[0] for o in objs}
    cols_set = {center(o)[1] for o in objs}

    # smallest non‑zero distance (fallback to slice size)
    drow = branch(sizefilter(rows_set, 2),
                  valmin({abs(r2 - r1) for r1 in rows_set for r2 in rows_set
                         if r1 != r2 and abs(r2 - r1) > 0},
                         identity),
                  hb)

    dcol = branch(sizefilter(cols_set, 2),
                  valmin({abs(c2 - c1) for c1 in cols_set for c2 in cols_set
                         if c1 != c2 and abs(c2 - c1) > 0},
                         identity),
                  wb)

    base = canvas(0, (hb, wb))

    rows_pat = merge(apply(
        lambda o: fill(base,
                       color(o),
                       {(r, c) for (r, c) in asindices(base)
                        if r == subtract(center(o), origin)[0]}),
        objs))

    cols_pat = merge(apply(
        lambda o: fill(base,
                       color(o),
                       {(r, c) for (r, c) in asindices(base)
                        if c == subtract(center(o), origin)[1]}),
        objs))

    pat = branch(greater(hb, wb), rows_pat, cols_pat)

    # repeat according to the measured spacing, then tile the other axis
    tmp = vupscale(pat,
                   branch(greater(hb, wb), divide(height(I), drow), divide(height(I), hb)))
    result = hupscale(tmp,
                      branch(greater(hb, wb), divide(width(I), wb), divide(width(I), dcol)))
    return result
