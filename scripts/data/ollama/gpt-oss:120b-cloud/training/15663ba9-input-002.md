# Program 1
## DSL
```python
def dsl1(I):
    # Object‑based endpoint recoloring
    #  – for every non‑background object recolor the cells that lie on its
    #    leftmost column with 4,
    #    the cells that lie on its rightmost column with 2.
    #  – left‑side (4) is painted after right‑side (2) so that width‑1
    #    objects keep colour 4.
    objs      = fgpartition(I)                                      # all objects
    non_bg   = mfilter(objs, lambda o: color(o) > 0)                # ignore bg
    # left‑most cells of every object
    lefts = apply(
        lambda o: recolor(
            4,
            { (r, c) for (col, (r, c)) in o if c == leftmost(o) }
        ),
        non_bg
    )
    # right‑most cells of every object
    rights = apply(
        lambda o: recolor(
            2,
            { (r, c) for (col, (r, c)) in o if c == rightmost(o) }
        ),
        non_bg
    )
    # paint right side first, then left side (4 overrides 2 on width‑1 objects)
    O = I
    for p in rights:   O = paint(O, p)
    for p in lefts:    O = paint(O, p)
    return O
```

def dsl3(I):
    # Geometric mirroring with colour priority
    #  – mirror the whole picture vertically,
    #    recolour every non‑background cell of the mirrored picture with 4,
    #    recolour every original non‑background cell with 2.
    #  – because 4 is painted after 2 it overwrites any overlap.
    M = vmirror(I)

    # cells of the mirrored picture (all non‑zero colours)
    objs_mir   = fgpartition(M)
    non_bg_mir = mfilter(objs_mir, lambda o: color(o) > 0)
    mir_patch  = apply(lambda o: { (r, c) for (col, (r, c)) in o }, non_bg_mir)
    mir_cells  = combine(*mir_patch)          # union of all mirrored cells

def dsl4(I):
    # Relational – distance from the outer border
    #  – cells at Manhattan distance 1 from the outer border become 4,
    #    cells at distance 2 become 2, everything else unchanged.
    h, w      = shape(I)
    border    = ({ (0, c) for c in range(w) } |
                { (h-1, c) for c in range(w) } |
                { (r, 0) for r in range(h) } |
                { (r, w-1) for r in range(h) })
    d1        = { p for b in border for p in neighbors(b)
                 if 0 <= p[0] < h and 0 <= p[1] < w }
    d2        = { p for d in d1 for p in neighbors(d)
                 if 0 <= p[0] < h and 0 <= p[1] < w and p not in border and p not in d1 }
    O = recolor(4, d1)
    O = recolor(2, d2)
    return O

def dsl5(I):
    # Periodicity – colour the first and last column of each horizontal period
    #  – horizontal period of the non‑zero pattern is obtained with hperiod().
    #    Cells that lie in the leftmost column of each period are recoloured 4,
    #    cells that lie in the rightmost column of each period are recoloured 2.
    #  – 4 is painted after 2 so it has priority where periods have width 1.
    objs      = fgpartition(I)
    non_bg    = mfilter(objs, lambda o: color(o) > 0)
    # union of all cells belonging to any non‑background object
    all_cells = combine(*apply(lambda o: { (r, c) for (col, (r, c)) in o }, non_bg))

    # horizontal period of the whole pattern (same for every row because of regularity)
    p         = hperiod(ofcolor(I, color(first(non_bg))))   # any non‑zero colour works
    # start columns of each period
    starts    = interval(0, w := shape(I)[1], p)
    # left‑most column indices of the periods
    left_cols = { c for c in starts }
    # right‑most column indices of the periods
    right_cols = { c-1 for c in starts if c-1 >= 0 }

    left_patch  = { (r, c) for (r, c) in all_cells if c in left_cols }
    right_patch = { (r, c) for (r, c) in all_cells if c in right_cols }

    O = I
    O = paint(O, recolor(2, right_patch))   # 2 first
    O = paint(O, recolor(4, left_patch))    # 4 on top
    return O
