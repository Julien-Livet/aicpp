def dsl1(I):
    # ---------- Global transformation ----------
    # Remove the outermost one‑pixel border, then put the original border back.
    trimmed = trim(I)                                           # interior without border
    top    = crop(I, (0, 0),                (1, width(I)))       # original top row
    bottom = crop(I, (height(I)-1, 0),      (1, width(I)))       # original bottom row
    left   = crop(I, (0, 0),                (height(I), 1))     # original left column
    right  = crop(I, (0, width(I)-1),       (height(I), 1))     # original right column
    middle = hconcat(vconcat(left, trimmed), right)            # rebuild interior with side borders
    O      = vconcat(top, vconcat(middle, bottom))             # re‑attach top & bottom
    return O


def dsl2(I):
    # ---------- Object extraction ----------
    # Keep the largest object (by cell count) and paint it on a canvas filled with the
    # colour that appears least often in the whole picture (the background colour).
    bg      = leastcolor(I)                                    # background colour
    objects = fgpartition(I)                                    # all foreground objects
    biggest = argmax(objects, lambda o: size(o))               # the biggest one
    canvas  = canvas(bg, shape(I))                              # empty canvas with bg
    O       = paint(canvas, biggest)                            # paint the object
    return O


def dsl3(I):
    # ---------- Color‑filtering ----------
    # Any cell that has the dominant colour and at least three orthogonal neighbours
    # of a different colour is recoloured to the most common neighbour colour.
    dom      = mostcolor(I)                                     # dominant colour
    others   = difference(palette(I), initset(dom))             # the set of other colours
    O = I
    O = mapply(
            lambda c:
                recolor(
                    c,
                    intersection(
                        ofcolor(O, dom),                         # cells of dominant colour
                        delta(ofcolor(O, c))                     # cells adjacent to colour c
                    )
                ),
            others
        )
    return O


def dsl4(I):
    # ---------- Geometric reasoning ----------
    # Rotate the picture 180° and keep only the cells that are identical in the
    # original and the rotated version.
    rot      = rot180(I)                                        # 180° rotation
    O        = cellwise(I, rot, -1)                             # keep matching cells, else -1 (background)
    return O


def dsl5(I):
    # ---------- Relational / structural ----------
    # Every non‑background object is shifted one step toward the nearest edge of the grid.
    bg       = leastcolor(I)                                   # background colour
    objs     = fgpartition(I)                                  # all objects
    moved    = apply(
                lambda o: move(
                              I,
                              o,
                              gravitate(o, ofcolor(I, bg))
                           ),
                objs
              )
    O = underpaint(I, combine(*moved))                          # paint the moved objects over the original background
    return O
