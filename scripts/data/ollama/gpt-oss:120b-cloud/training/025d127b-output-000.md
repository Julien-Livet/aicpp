def dsl1(I):
    # Global transformation: trim empty border, prepend a zero column (shifting everything right),
    # then crop back to the original size.
    T = trim(I)                                 # remove empty outer rows/cols
    h, w = shape(T)                             # height and width of trimmed grid
    Z = canvas(ZERO, (h, 1))                    # column of zeros
    shifted = hconcat(Z, T)                     # prepend zero column (right‑shift)
    O = crop(shifted, (0, 0), shape(I))        # restore original dimensions
    return O


def dsl2(I):

    # non‑background object with its own color.
