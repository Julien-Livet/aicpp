def dsl1(I):
    # GLOBAL – draw a segment from the first empty cell after the last 1 to the bottom‑right corner
    ones      = ofcolor(I, 1)
    last_one  = astuple(lowermost(ones), rightmost(ones))
    start     = increment(last_one)                                 # first empty cell after the diagonal of 1s
    h, w      = shape(I)
    end       = subtract(astuple(h, w), astuple(1, 1))               # bottom‑right corner
    diag      = connect(start, end)                                   # straight line (1,1) to the edge
    O         = fill(I, 2, diag)                                      # paint it with colour 2
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION – keep the part of the main diagonal that lies beyond the last 1
    ones       = ofcolor(I, 1)
    h, w      = shape(I)
    end        = subtract(astuple(h, w), astuple(1, 1))
    full_diag  = connect(astuple(0, 0), end)                          # whole main diagonal
    last_one   = astuple(lowermost(ones), rightmost(ones))
    part_up_to = connect(astuple(0, 0), last_one)                    # diagonal up to (and including) the last 1
    mask       = difference(full_diag, part_up_to)                    # cells after the last 1
    O          = fill(I, 2, mask)                                     # colour those cells with 2
    return O


def dsl3(I):
    # COLOR‑FILTERING – under‑paint the missing diagonal cells after the last 1
    ones       = ofcolor(I, 1)
    h, w      = shape(I)
    end        = subtract(astuple(h, w), astuple(1, 1))
    full_diag  = connect(astuple(0, 0), end)
    last_one   = astuple(lowermost(ones), rightmost(ones))
    part_up_to = connect(astuple(0, 0), last_one)
    mask       = difference(full_diag, part_up_to)
    O          = underpaint(I, mask)                                   # paint only on background cells
    return O


def dsl4(I):
    # GEOMETRIC – extend the diagonal from the cell after the final 1 using a ray
    ones      = ofcolor(I, 1)
    last_one  = astuple(lowermost(ones), rightmost(ones))
    start     = increment(last_one)                                   # first empty cell after the 1‑chain
    line      = shoot(start, astuple(1, 1))                           # ray in direction (1,1) to the border
    O         = fill(I, 2, line)                                       # colour the ray with 2
    return O


def dsl5(I):
    # RELATIONAL / STRUCTURAL – derive the unit step from the span of the 1‑pixels
    ones       = ofcolor(I, 1)
    first_one  = ulcorner(ones)                                        # upper‑left 1
    last_one   = astuple(lowermost(ones), rightmost(ones))
    delta      = subtract(last_one, first_one)                         # overall displacement of the diagonal
    unit_step  = divide(delta, delta)                                 # reduces to (1,1)
    start      = increment(last_one)
    line       = shoot(start, unit_step)                               # continue the diagonal
    O          = fill(I, 2, line)                                      # paint with colour 2
    return O
