def dsl1(I):
    # ----- ROW‑TEMPLATE (most frequent row without colour 8) -----
    # rows that contain at least one non‑zero cell and no 8
    good_rows = apply(
        lambda r: r if any(c != 0 for c in r) and not any(c == 8 for c in r) else None,
        I,
    )
    kept      = sfilter(good_rows, lambda x: x is not None)          # discard the Nones
    tmpl      = mostcommon(kept)                                   # most common “good’’ row
    # replace every “good’’ row by the template, leave everything else untouched
    O = apply(
        lambda r: tmpl if any(c != 0 for c in r) and not any(c == 8 for c in r) else r,
        I,
    )
    return O


def dsl2(I):
    # ----- OBJECT‑EXTRACTION (preserve colour 8 objects) + ROW‑TEMPLATE -----
    # keep all colour‑8 objects as they are
    objs_8   = colorfilter(objects(I, True, False, True), 8)

    # rows that are non‑zero and contain no 8
    good_rows = apply(
        lambda r: r if any(c != 0 for c in r) and not any(c == 8 for c in r) else None,
        I,
    )
    kept      = sfilter(good_rows, lambda x: x is not None)
    tmpl_row  = mostcommon(kept)                                    # template row

    # turn the template row into a 1‑row object (all its coloured cells stay coloured)
    tmpl_obj  = toobject({(0, j) for j in interval(0, width(I), 1)}, (tmpl_row,))

    # indices of rows that satisfied the “good’’ condition
    good_idx  = ofcolor(I, 0)                                        # all background cells
    good_idx  = sfilter(
        good_idx,
        lambda p: any(I[p[0]][c] != 0 for c in range(width(I))) and
                  not any(I[p[0]][c] == 8 for c in range(width(I))),
    )
    # move the 1‑row object to each good row
    moved_obj = mapply(lambda p: move(I, tmpl_obj, (p[0], 0)), good_idx)

    # paint the moved objects, then restore the colour‑8 objects
    O = paint(I, merge(moved_obj))
    O = paint(O, merge(objs_8))
    return O


def dsl3(I):
    # ----- KEEP COLOURS THAT OCCUPY ≥ 5 % OF THE GRID -----
    total   = size(asindices(I))                                 # total number of cells
    # colours whose frequency is > 5 %
    major   = sfilter(
        palette(I),
        lambda c: colorcount(I, c) * 20 > total,                # *20  →  > 5 %
    )
    # union of all cells of those colours
    mask    = merge(mapply(lambda c: ofcolor(I, c), major))
    # paint them on a blank canvas
    O = paint(canvas(0, shape(I)), mask)
    return O


def dsl4(I):
    # ----- HORIZONTAL MIRRORING -----
    # mirror the full picture, keep only the left half (original width)
    mirrored   = hmirror(I)                # mirror along the vertical axis
    concatened  = hconcat(I, mirrored)       # side‑by‑side
    O = crop(concatened, (0, 0), shape(I))
    return O


def dsl5(I):
    # ----- KEEP CELLS THAT HAVE AT LEAST TWO NON‑ZERO NEIGHBOURS -----
    # all non‑zero cells of the grid
    nz_cells = merge(mapply(lambda c: ofcolor(I, c), sfilter(palette(I), lambda c: c != 0))

    # keep a cell if at least two of its (direct) neighbours are also non‑zero
    keep = sfilter(
        nz_cells,
        lambda p: size(intersection(neighbors(p), nz_cells)) >= 2,
    )
    # paint the kept cells on a blank canvas, preserving colour‑8 objects
    O = paint(canvas(0, shape(I)), keep)
    O = paint(O, merge(colorfilter(objects(I, True, False, True), 8)))
    return O
