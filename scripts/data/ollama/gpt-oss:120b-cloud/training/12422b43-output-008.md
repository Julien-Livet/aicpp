def dsl1(I):
    # OBJECT‑EXTRACTION – rows that contain a 5 are cleared and tiled vertically
    anchor      = ofcolor(I, 5)                                            # cells with colour 5
    rows_set    = { r for r,_ in anchor }                                   # rows that hold a 5
    rows_cells  = product(rows_set, interval(ZERO, width(I), ONE))          # full‑width cells of those rows
    cleared     = recolor(0, anchor)                                        # erase the 5s
    I0          = paint(I, cleared)                                         # grid with the block cleared
    pattern_g   = subgrid(rows_cells, I0)                                   # cleared rows as a grid
    pattern_o   = asobject(pattern_g)                                       # object representation

    nz          = difference(asindices(I), ofcolor(I, 0))                    # all non‑zero cells in the original
    max_nz_row = valmax(nz, lambda p: p[0])                                # lowest occupied row
    empty_rows  = subtract(height(I), add(max_nz_row, ONE))                 # rows still empty at the bottom
    repeat_cnt  = divide(empty_rows, size(rows_set))                       # how many full copies fit

    offsets = mapply(lambda i: astuple(multiply(add(i, ONE), size(rows_set)),
                                      ZERO),
                    interval(ZERO, repeat_cnt, ONE))
    shifted = mapply(lambda off: shift(pattern_o, off), offsets)           # shifted copies
    tiled   = merge(shifted)                                                # union of all copies

    base   = subgrid(backdrop(nz), I)                                       # keep original picture
    O      = trim(paint(base, tiled))                                       # overlay the tiled block
    return O
