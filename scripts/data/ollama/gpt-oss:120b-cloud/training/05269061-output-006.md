def dsl1(I):
    # OBJECT‑CENTRIC + TILING: infer the (row+col) mod‑3 → colour map
    # from the three coloured objects, then tile the whole grid with it.
    objs   = objects(I, univalued=True, diagonal=False, without_bg=True)

    sum_of = lambda o: add(centerofmass(o)[0], centerofmass(o)[1])
    o0     = argmin(objs, sum_of)
    o1     = argmin(difference(objs, {o0}), sum_of)
    o2     = argmin(difference(objs, {o0, o1}), sum_of)

    c0 = color(o0)          # colour for residue 0
    c1 = color(o1)          # colour for residue 1
    c2 = color(o2)          # colour for residue 2

    # build a column object of a given colour
    col_obj = lambda col, colc: recolor(colc,
                        toobject(apply(lambda r: (r, col), interval(0, height(I), 1)), I))

    # three families of columns according to column‑mod‑3
    cols0 = interval(0, width(I), 3)
    cols1 = interval(1, width(I), 3)
    cols2 = interval(2, width(I), 3)

    obj0  = combine(apply(lambda c: col_obj(c, c0), cols0))
    obj1  = combine(apply(lambda c: col_obj(c, c1), cols1))
    obj2  = combine(apply(lambda c: col_obj(c, c2), cols2))

    empty = canvas(0, shape(I))
    G0    = paint(empty, obj0)
    G1    = paint(G0,    obj1)
    O     = paint(G1,    obj2)
    return O


def dsl2(I):
    # TILING‑FROM‑BASE‑BLOCK: construct a 3×3 prototype using the colour map
    # and then repeat it over the whole grid via offsets.
    objs   = objects(I, univalued=True, diagonal=False, without_bg=True)
    sum_of = lambda o: add(centerofmass(o)[0], centerofmass(o)[1])
    o0 = argmin(objs, sum_of); o1 = argmin(difference(objs,{o0}), sum_of)
    o2 = argmin(difference(objs,{o0,o1}), sum_of)
    c0 = color(o0); c1 = color(o1); c2 = color(o2)

    # colour for a given residue (row+col)%3 → colour
    col_of = lambda r,c: branch(equality(modulo(add(r,c),3),0),c0,
                    branch(equality(modulo(add(r,c),3),1),c1,c2)

    # build the 3×3 prototype as an object
    rows = interval(0,3,1)
    cols = interval(0,3,1)
    cells = apply(lambda r: apply(lambda c: (r,c), cols), rows)          # nested containers
    cells_flat = combine(cells)                                       # set of (r,c)
    proto_obj = recolor(0, toobject(cells_flat, I))                   # dummy colour
    proto_obj = replace(proto_obj, 0,
                combine(apply(lambda rc:
                           initset((col_of(rc[0],rc[1]), rc)), cells_flat))

    # tile the prototype over the whole grid
    roffs = interval(0, height(I), 3)
    coffs = interval(0, width(I), 3)
    anchors = combine(apply(lambda r: apply(lambda c: (r,c), coffs), roffs))
    empty   = canvas(0, shape(I))
    tiled   = apply(lambda off: paint(empty, shift(proto_obj, off)), anchors)
    O       = first(tiled)                     # any order gives the full tiling
    return O


def dsl3(I):
    # COLOR‑FILTER + MOD‑3 MASKS: create three masks for the three residues
    # and colour each mask with the appropriate colour.
    objs   = objects(I, univalued=True, diagonal=False, without_bg=True)
    sum_of = lambda o: add(centerofmass(o)[0], centerofmass(o)[1])
    o0 = argmin(objs, sum_of); o1 = argmin(difference(objs,{o0}), sum_of)
    o2 = argmin(difference(objs,{o0,o1}), sum_of)
    c0 = color(o0); c1 = color(o1); c2 = color(o2)

    # mask for a given residue: all cells whose (row+col)%3 == r
    mask = lambda r: apply(
            lambda rc: (rc[0], rc[1]),
            apply(lambda i: apply(lambda j: (i,j), interval(0,width(I),1)),
                  interval(0,height(I),1))
        )
    # keep only cells with wanted residue
    keep = lambda r: apply(
            lambda rc: branch(equality(modulo(add(rc[0],rc[1]),3), r, None),
            mask(r))
    # build coloured objects from the masks
    obj0 = recolor(c0, toobject(apply(lambda rc: rc, keep(0)), I))
    obj1 = recolor(c1, toobject(apply(lambda rc: rc, keep(1)), I))
    obj2 = recolor(c2, toobject(apply(lambda rc: rc, keep(2)), I))

    empty = canvas(0, shape(I))
    O = paint(paint(paint(empty, obj0), obj1), obj2)
    return O


def dsl4(I):
    # GEOMETRIC + PATTERN REPROJECTION: rotate the input so that the
    # diagonal becomes horizontal, then copy that line across the grid
    # and finally undo the rotation.
    rot   = rot90(I)                      # diagonal → horizontal
    line  = argmax(objects(rot, univalued=True, diagonal=False, without_bg=True),
                  lambda o: size(o))      # longest horizontal line
    h, w  = shape(subgrid(line, rot))
    anchors = asindices(canvas(0, ((height(I)-h+1, width(I)-w+1))))
    empty   = canvas(0, shape(rot))
    tiled   = apply(lambda off: paint(empty, shift(line, off)), anchors)
    Orot    = first(tiled)
    O       = rot270(Orot)                # back to original orientation
    return O


def dsl5(I):
    # RELATIONAL NORMALISATION + RE‑APPLICATION:
    # normalise each colour patch to the origin, then repeatedly
    # paste the normalised patch at every possible offset.
    cols   = palette(I) - {0}
    empty  = canvas(0, shape(I))

    norm = lambda c: let(
            patch   = ofcolor(I, c),
            obj     = toobject(patch, I),
            shifted = shift(obj, (-uppermost(patch), -leftmost(patch))),
            recolor(c, shifted))

    normed = apply(norm, cols)
    allobj = combine(normed)

    # all possible top‑left anchors where a full patch fits
    h, w   = shape(subgrid(first(normed), I))
    anchors = asindices(canvas(0, ((height(I)-h+1, width(I)-w+1))))
    grids   = apply(lambda off: paint(empty, shift(allobj, off)), anchors)
    O       = first(grids)               # overlay everything
    return O
