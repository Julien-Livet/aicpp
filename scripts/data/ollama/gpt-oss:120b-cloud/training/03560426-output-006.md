def dsl1(I):
    # GLOBAL – keep all non‑zero colours, trim outer empty border,
    #         then collapse every empty row / column (including those that
    #         appear after the first collapse).  Paint the compacted pattern
    #         on a fresh canvas of the original size.
    colours = palette(I) - {0}
    mask    = ofcolor(I, colours)                     # all coloured cells
    filtered = underpaint(I, mask)                     # erase everything else
    compact  = compress(compress(trim(filtered)))      # external + internal gaps
    empty    = canvas(0, shape(I))
    return paint(empty, asobject(compact))


def dsl2(I):
    # OBJECT‑LEVEL – for each coloured object normalise its position
    #                (move its bounding box to the origin) while preserving
    #                its colour, then redraw all objects together.
    objs   = fgpartition(I)                           # set of coloured objects
    def norm(o):
        col   = color(o)                              # object's colour
        coords = normalize(toindices(o))              # shift to (0,0) origin
        return recolor(col, coords)                   # coloured patch again
    normed = apply(norm, objs)                         # set of normalised objects
    united = merge(normed)                             # one object containing all
    empty  = canvas(0, shape(I))
    return paint(empty, united)


def dsl3(I):
    # COLOUR‑FILTER – keep only the three most frequent non‑zero colours,
    #                erase everything else and compact the remaining pattern.
    colours   = palette(I) - {0}
    ordered   = order(colours, lambda c: -colorcount(I, c))
    top3      = {first(apply(lambda _: ordered, interval(0, 3, 1)))}
    mask      = ofcolor(I, top3)                       # cells of the three colours
    filtered  = underpaint(I, mask)                    # delete all other colours
    compact   = compress(compress(trim(filtered)))     # collapse gaps
    empty     = canvas(0, shape(I))
    return paint(empty, asobject(compact))


def dsl4(I):
    # GEOMETRIC – mirror across the main diagonal, then remove every empty
    #            row and column (external and internal), finally restore the
    #            result to the original canvas size.
    mirrored = dmirror(I)                             # diagonal mirror
    compact  = compress(compress(trim(mirrored)))     # trim + double‑compress
    empty    = canvas(0, shape(I))
    return paint(empty, asobject(compact))


def dsl5(I):
    # RELATIONAL – for each coloured object move it as far up as possible,
    #              then as far left as possible, keeping its colour.
    objs = fgpartition(I)                             # coloured objects

    def up_left(o):
        col   = color(o)                              # object's colour
        pts   = toindices(o)                          # set of coordinates
        # move up until just above the top row
        up    = gravitate(pts,
                          {(0, c) for c in interval(0, width(I), 1)})
        pts_up = shift(pts, up)
        # then move left until just left of the leftmost column
        left  = gravitate(pts_up,
                          {(r, 0) for r in interval(0, height(I), 1)})
        new_pts = shift(pts_up, left)
        return recolor(col, new_pts)

    shifted = apply(up_left, objs)                    # set of shifted objects
    united  = merge(shifted)                           # one object with all colours
    empty   = canvas(0, shape(I))
    return paint(empty, united)
