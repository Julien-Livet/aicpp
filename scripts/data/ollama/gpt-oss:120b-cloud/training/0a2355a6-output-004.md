    # BORDER‑FOCUSED: interior → 2, touching frame → 1, isolated singletons → 3
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    interior = mfilter(
        objs,
        lambda o: size(
            intersection(
                o,
                intersection(shift(o, UP),
                    intersection(shift(o, DOWN),
                        intersection(shift(o, LEFT), shift(o, RIGHT))
                    )
                )
            ) > 0)

    border = mfilter(objs, lambda o: bordering(o, I))

    def recol(o):
        col = branch(contained(interior, o), 2,
               branch(contained(border, o), 1,
                     branch(size(o) == 1, 3, 1)))
        return recolor(col, o)

    return paint(canvas(0, shape(I)), merge(apply(recol, objs)))


def dsl4(I):
    # COLOR‑FILTERING: start with all 8 → 2, then overlay border →1 and singletons →3
    O = replace(I, 8, 2)

    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    border_objs = mfilter(objs, lambda o: bordering(o, I))
    O = paint(O, merge(apply(lambda o: recolor(1, o), border_objs)))

    singleton_objs = mfilter(objs, lambda o: size(o) == 1)
    O = paint(O, merge(apply(lambda o: recolor(3, o), singleton_objs)))

    return O


def dsl5(I):
    # GEOMETRIC‑RELATIONAL: interior cells via intersection of four shifted copies
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    interior = mfilter(
        objs,
        lambda o: size(
            intersection(
                shift(o, UP),
                shift(o, DOWN),
                shift(o, LEFT),
                shift(o, RIGHT)
            )
        ) > 0)

    non = difference(objs, interior)

    max_sz = valmax(apply(lambda o: size(o), non), lambda x: x)
    min_sz = valmin(apply(lambda o: size(o), non), lambda x: x)

    def recol(o):
        col = branch(contained(interior, o), 2,
               branch(size(o) == max_sz, 1,
                     branch(size(o) == min_sz, 3, 1)))
        return recolor(col, o)

    return paint(canvas(0, shape(I)), merge(apply(recol, objs)))
