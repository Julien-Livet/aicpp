def dsl1(I):
    # GLOBAL – build three horizontal‑scaled bands and stack them vertically.

def dsl2(I):

    # and place three copies (orig‑mirr‑orig) with explicit vertical shifts.
    orig_objs = objects(I, True, False, False)                 # one‑cell objects
    mirr_grid = hmirror(I)
    mirr_objs = objects(mirr_grid, True, False, False)

    # scale each cell horizontally (factor 3) – keeps the original vertical row.
    orig_up = apply(lambda o: hupscale(toobject(o, I), 3), orig_objs)
    mirr_up = apply(lambda o: hupscale(toobject(o, mirr_grid), 3), mirr_objs)

    # shift the three bands to rows 0‑1, 2‑3 and 4‑5 respectively.
    top    = orig_up                                          # no shift

    bottom = apply(lambda o: shift(o, (4, 0)), orig_up)       # down 4 rows

    O = paint(canvas(0, (6, 6)), combine(combine(top, middle), bottom))
    return O
