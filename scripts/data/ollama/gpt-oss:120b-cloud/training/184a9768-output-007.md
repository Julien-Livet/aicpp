def dsl1(I):
    # GLOBAL‑PIXELATION + MARKER REMOVAL
    G   = downscale(I, 2)                       # coarse version
    U   = upscale(G, 2)                         # back to original size
    MARK = combine(combine(ofcolor(I, 3), ofcolor(I, 5)), ofcolor(I, 8))
    O   = fill(U, 0, MARK)                     # erase all marker cells
    return O


def dsl2(I):
    # LARGEST‑OBJECT RE‑COLOURING
    OBJ   = objects(I, univalued=False, diagonal=False, without_bg=True)
    BIG   = argmax(OBJ, size)                    # biggest component
    MC    = mostcolor(I)                        # dominant colour in the whole picture
    O     = canvas(0, shape(I))
    O     = paint(O, recolor(MC, BIG))          # paint the biggest component with that colour
    return O


def dsl3(I):
    # COLOUR‑FILTERING (fixed syntax)
    MARKERS = combine(combine(ofcolor(I, 3), ofcolor(I, 5)), ofcolor(I, 8))
    OBJ     = objects(I, univalued=False, diagonal=False, without_bg=True)
    SINGLE  = mfilter(OBJ, lambda o: equality(size(o), 1))
    BAD     = combine(MARKERS, SINGLE)          # cells to be cleared
    O       = fill(I, 0, BAD)
    return O


def dsl4(I):
    # GEOMETRIC‑OVERLAY WITH 180° ROTATION
    H   = hmirror(I)                            # horizontal mirror
    V   = vmirror(I)                            # vertical mirror
    R   = rot180(I)                             # 180° rotation
    M1  = cellwise(H, V, fallback=0)           # union of the two mirrors
    M2  = cellwise(M1, R, fallback=0)           # add the rotated picture
    O   = cellwise(I, M2, fallback=0)           # original colour dominates
    return O
