def dsl1(I):
    # GLOBAL – cut the empty border, double size, then crop back to original shape.
    C = compress(I)                                   # remove empty outer border
    H = hupscale(C, 2)                                # double width

    O = crop(S, (0, 0), (height(I), width(I)))        # keep original dimensions
    return O


def dsl2(I):
