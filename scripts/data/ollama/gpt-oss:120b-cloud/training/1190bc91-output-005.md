def dsl2(I):
    # Object‑level transformation: rotate every foreground object 90° clockwise
    objs   = fgpartition(I)                                   # set of foreground objects
    rotobj = apply(lambda o: rot90(paint(canvas(0, shape(I)), o)), objs)
    O      = paint(canvas(0, shape(I)), combine(*rotobj))    # paint all rotated objects on a blank canvas
    return O

    # Relational/structural transformation: isolate the first foreground object,

    O     = hupscale(sub, 2)                      # enlarge the subgrid vertically (horizontal size follows)
