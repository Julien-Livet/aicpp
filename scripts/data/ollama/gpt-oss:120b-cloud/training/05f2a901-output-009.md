def dsl1(I):
    # ---------- H-MATCH / V-MATCH ----------
    # If the objects share any row, shift horizontally; otherwise shift vertically.
    obj2 = ofcolor(I, TWO)
    obj8 = ofcolor(I, EIGHT)

    rows_overlap = hmatching(obj2, obj8)          # share a row → horizontal move
    cols_overlap = vmatching(obj2, obj8)          # share a column → vertical move

    # ----- horizontal quantities -----
    left2  = leftmost(obj2);   right2 = rightmost(obj2)
    left8  = leftmost(obj8);   right8 = rightmost(obj8)

    h_gap   = branch(greater(left8, right2),
                     subtract(left8, add(right2, ONE)),
                     subtract(left2, add(right8, ONE)))
    h_sign  = branch(greater(left8, right2), ONE, NEG_ONE)
    dc_h    = multiply(h_sign, h_gap)

    # ----- vertical quantities -----
    up2   = uppermost(obj2);   low2 = lowermost(obj2)
    up8   = uppermost(obj8);   low8 = lowermost(obj8)

    v_gap  = branch(greater(up8, low2),
                     subtract(up8, add(low2, ONE)),
                     subtract(up2, add(low8, ONE)))
    v_sign = branch(greater(up8, low2), ONE, NEG_ONE)
    dr_v   = multiply(v_sign, v_gap)

    # ----- pick the proper direction -----
    dr = branch(rows_overlap, ZERO, dr_v)
    dc = branch(rows_overlap, dc_h, ZERO)

    moved   = recolor(TWO, shift(obj2, (dr, dc)))
    cleared = cover(I, obj2)
    O = paint(cleared, moved)
    return O


def dsl2(I):
    # ---------- CENTER OF MASS + DOMINANT AXIS ----------
    # Move the 2‑object along the axis where the two centers are farther apart.
    obj2 = ofcolor(I, TWO)
    obj8 = ofcolor(I, EIGHT)

    c2 = centerofmass(obj2)
    c8 = centerofmass(obj8)

    drc = subtract(c8[0], c2[0])          # row difference
    dcc = subtract(c8[1], c2[1])          # col difference

    # absolute differences (no builtin, use branch)
    abs_dr = branch(greater(drc, ZERO), drc, multiply(NEG_ONE, drc))
    abs_dc = branch(greater(dcc, ZERO), dcc, multiply(NEG_ONE, dcc))

    vertical = greater(abs_dr, abs_dc)      # True → move vertically

    # ----- vertical gap -----
    up2   = uppermost(obj2);   low2 = lowermost(obj2)
    up8   = uppermost(obj8);   low8 = lowermost(obj8)

    v_gap  = branch(greater(up8, low2),
                    subtract(up8, add(low2, ONE)),
                    subtract(up2, add(low8, ONE)))
    v_sign = branch(greater(up8, low2), ONE, NEG_ONE)
    dr_v   = multiply(v_sign, v_gap)

    # ----- horizontal gap -----
    left2  = leftmost(obj2);   right2 = rightmost(obj2)
    left8  = leftmost(obj8);   right8 = rightmost(obj8)

    h_gap  = branch(greater(left8, right2),
                    subtract(left8, add(right2, ONE)),
                    subtract(left2, add(right8, ONE)))
    h_sign = branch(greater(left8, right2), ONE, NEG_ONE)
    dc_h   = multiply(h_sign, h_gap)

    dr = branch(vertical, dr_v, ZERO)

def dsl3(I):
    # ---------- ROTATE‑AND‑SHIFT ----------
    # Turn a vertical problem into a horizontal one, solve it, then rotate back.
    obj2 = ofcolor(I, TWO)
    obj8 = ofcolor(I, EIGHT)

    rows_overlap = hmatching(obj2, obj8)          # share a row → horizontal

    def horiz_shift(grid):
        o2 = ofcolor(grid, TWO)
        o8 = ofcolor(grid, EIGHT)

        left2  = leftmost(o2);   right2 = rightmost(o2)
        left8  = leftmost(o8);   right8 = rightmost(o8)

        gap  = branch(greater(left8, right2),
                     subtract(left8, add(right2, ONE)),
                     subtract(left2, add(right8, ONE)))
        sign = branch(greater(left8, right2), ONE, NEG_ONE)
        dc   = multiply(sign, gap)

        moved   = recolor(TWO, shift(o2, (ZERO, dc)))
        cleared = cover(grid, o2)
        return paint(cleared, moved)

    O = branch(rows_overlap,
               horiz_shift(I),
               rot270(horiz_shift(rot90(I))))   # rotate, shift horizontally, rotate back
    return O


def dsl4(I):
    # ---------- RECONSTRUCTION WITH VERTICAL CONCAT ----------
    # Compute the needed shift, paste the moved object, then rebuild the picture
    # by concatenating the unchanged top part, the rows that now contain the
    # moved object, and the unchanged bottom part.
    obj2 = ofcolor(I, TWO)
    obj8 = ofcolor(I, EIGHT)

    left2  = leftmost(obj2);   right2 = rightmost(obj2)
    left8  = leftmost(obj8);   right8 = rightmost(obj8)
    up2    = uppermost(obj2);   low2  = lowermost(obj2)
    up8    = uppermost(obj8);   low8  = lowermost(obj8)

    rows_overlap = hmatching(obj2, obj8)          # share a row → horizontal move

    # ----- horizontal gap -----
    h_gap  = branch(greater(left8, right2),
                    subtract(left8, add(right2, ONE)),
                    subtract(left2, add(right8, ONE)))
    h_sign = branch(greater(left8, right2), ONE, NEG_ONE)
    dc_h   = multiply(h_sign, h_gap)

    # ----- vertical gap -----
    v_gap  = branch(greater(up8, low2),
                    subtract(up8, add(low2, ONE)),
                    subtract(up2, add(low8, ONE)))
    v_sign = branch(greater(up8, low2), ONE, NEG_ONE)
    dr_v   = multiply(v_sign, v_gap)

    dr = branch(rows_overlap, ZERO, dr_v)
    dc = branch(rows_overlap, dc_h, ZERO)

    moved   = recolor(TWO, shift(obj2, (dr, dc)))
    cleared = cover(I, obj2)

    # three vertical slices
    top    = crop(cleared, (ZERO, ZERO), (up2, width(cleared)))
    middle = crop(paint(cleared, moved), (add(up2, dr), ZERO),

    bottom = crop(cleared,
                  (add(low2, dr, ONE), ZERO),
                  (subtract(height(cleared), add(low2, dr, ONE)), width(cleared)))

    O = vconcat(vconcat(top, middle), bottom)
    return O
