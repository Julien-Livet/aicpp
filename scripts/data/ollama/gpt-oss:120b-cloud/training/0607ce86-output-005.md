def dsl1(I):
    # ------- ROW‑TEMPLATE (most frequent non‑8 row) -------
    # rows that contain any non‑zero cell and no colour 8
    good = apply(lambda r: any(c != 0 for c in r) and not any(c == 8 for c in r), I)

    # keep only those rows, discard the others (None)
    cand = apply(lambda r, g: r if g else None, I, good)
    # filter out the Nones
    kept = sfilter(cand, lambda x: x is not None)

    # the template row is the most common among the kept rows
    tmpl = mostcommon(kept)

    # build the output: every “good’’ row becomes tmpl, everything else stays unchanged
    O = apply(lambda r, g: tmpl if g else r, I, good)
    return O


def dsl2(I):
    # ------- OBJECT‑EXTRACTION (colour 8) + TEMPLATE‑OVERLAY -------
    # colour 8 objects stay untouched
    objs_8   = colorfilter(objects(I, True, False, True), 8)

    # rows that are non‑zero and do not contain 8
    good = apply(lambda r: any(c != 0 for c in r) and not any(c == 8 for c in r), I)
    cand = apply(lambda r, g: r if g else None, I, good)
    kept = sfilter(cand, lambda x: x is not None)

    # most common such row – will be turned into a 1‑row object
    tmpl_row = mostcommon(kept)

    # turn the template row into an object (its cells keep their colours)
    tmpl_obj = toobject({(0, j) for j in range(width(I))}, tuple([tmpl_row]))

    # shift the template object to every row that satisfied the “good’’ condition
    rows = ofcolor(I, 0)                     # all background cells
    good_rows = ofcolor(I, 0)                # placeholder, will be replaced below
    # build a patch containing the indices of all rows that are “good’’
    good_patch = sfilter(ofcolor(I, 0),          # background cells
                         lambda p: any(I[p[0]][c] != 0 for c in range(width(I))) and
                                   not any(I[p[0]][c] == 8 for c in range(width(I))))
    # move the 1‑row object to each of those rows
    over = mapply(lambda r: move(I, tmpl_obj, (r[0], 0)), good_patch)

    # paint the moved objects on top of the original grid, then restore the 8‑objects
    O = paint(I, merge(over))
    O = paint(O, merge(objs_8))
    return O


def dsl3(I):
    # ------- COLOR‑FREQUENCY (dominant colours) -------
    # the colours that occupy at least 5 % of the whole grid
    total   = size(asindices(I))
    freqcol = sfilter(palette(I),
                     lambda c: colorcount(I, c) * 20 > total)   # > 5 %
    # union of all cells of those colours
    mask    = merge(apply(lambda c: ofcolor(I, c), freqcol))
    # keep those cells, everything else becomes 0
    O = paint(canvas(0, shape(I)), mask)
    return O


def dsl4(I):
    # ------- GEOMETRIC‑MIRRORING (horizontal) -------
    # mirror the whole picture and keep only the left half
    mirrored   = hmirror(I)                # mirror along the vertical axis
    concatened = hconcat(I, mirrored)      # side‑by‑side
    O = crop(concatened, (0, 0), shape(I))   # cut back to original size
    return O


def dsl5(I):
    # ------- RELATIONAL‑ADJACENCY (touch ≥ 2 other rows) -------
    # treat each row as a “patch’’ and count how many other rows it touches
    rows = apply(lambda r: {(i, j) for j, v in enumerate(r) if v != 0},
                 I)                               # set of occupied cells per row

    adj_cnt = lambda p: size(
                sfilter(rows,
                        lambda q: q != p and adjacent(p, q)))

    # keep rows that touch at least two other rows (i.e. belong to a “band’’)
    keep = sfilter(rows, lambda p: adj_cnt(p) >= 2)

    # build a mask from the kept rows
    mask = merge(keep)

    # paint the mask on a blank canvas, preserving colour 8 objects
    O = paint(canvas(0, shape(I)), mask)
    O = paint(O, merge(colorfilter(objects(I, True, False, True), 8)))
    return O
