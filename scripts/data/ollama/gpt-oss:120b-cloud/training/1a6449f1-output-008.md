def dsl1(I):
    # 1️⃣  Global periodic‑tile extraction (non‑background)
    bg        = mostcolor(I)                                          # most frequent colour → background
    all_idx   = asindices(I)                                          # every cell
    nonbg_idx = difference(all_idx, ofcolor(I, bg))                   # cells that are not background
    obj       = toobject(nonbg_idx, I)                                 # coloured object of those cells
    vp        = vperiod(obj)                                          # vertical period of the pattern
    hp        = hperiod(obj)                                          # horizontal period of the pattern
    O         = crop(I, ORIGIN, (vp, hp))                             # first repeating tile, anchored at (0,0)
    return O


def dsl2(I):
    # 2️⃣  Largest connected non‑background component
    bg          = mostcolor(I)                                        # background colour
    no_bg_grid  = replace(I, bg, -1)                                   # mark background with a sentinel
    comps       = objects(no_bg_grid, univalued=True,
                         diagonal=False, without_bg=True)            # all monochrome objects (including sentinel)
    non_sentinel = difference(comps, colorfilter(comps, -1))            # discard the huge sentinel component
    largest     = argmax(non_sentinel, lambda o: size(o))              # component with most cells
    O           = subgrid(toindices(largest), I)                       # its minimal bounding rectangle
    return O


def dsl3(I):
    # 3️⃣  Double‑mirror invariance limited to non‑background cells
    bg          = mostcolor(I)                                        # background colour
    nonbg_idx   = difference(asindices(I), ofcolor(I, bg))            # indices of foreground
    obj         = toobject(nonbg_idx, I)                               # foreground object
    hm_obj      = asobject(hmirror(I))                                 # mirror of whole grid as object
    vm_obj      = asobject(vmirror(I))                                 # vertical mirror of whole grid as object
    invariant   = intersection(obj,
                               intersection(hm_obj, vm_obj))         # cells that survive both mirrors
    O           = subgrid(toindices(invariant), I)                    # bounding box of the invariant region
    return O


def dsl4(I):
    # 4️⃣  Smallest connected non‑background component
    bg          = mostcolor(I)
    no_bg_grid  = replace(I, bg, -1)
    comps       = objects(no_bg_grid, univalued=True,
                         diagonal=False, without_bg=True)
    non_sentinel = difference(comps, colorfilter(comps, -1))
    smallest    = argmin(non_sentinel, lambda o: size(o))             # component with fewest cells
    O           = subgrid(toindices(smallest), I)
    return O


def dsl5(I):
    # 5️⃣  Trimmed grid → periodic tile of the remaining pattern
    trimmed     = trim(I)                                             # cut uniform border
    bg          = mostcolor(trimmed)
    nonbg_idx   = difference(asindices(trimmed), ofcolor(trimmed, bg))
    obj         = toobject(nonbg_idx, trimmed)
    vp          = vperiod(obj)
    hp          = hperiod(obj)
    O           = crop(trimmed, ORIGIN, (vp, hp))                     # first tile after trimming
    return O
