def dsl1(I):
    # Global – extract the smallest repeating tile of the non‑background pattern
    bg        = mostcolor(I)                                            # background colour
    non_bg    = difference(asindices(I), ofcolor(I, bg))                # indices of non‑background cells
    vp        = vperiod(non_bg)                                          # vertical period of that set
    hp        = hperiod(non_bg)                                          # horizontal period of that set
    O         = crop(I, ORIGIN, (vp, hp))                                 # first tile of the pattern
    return O


def dsl2(I):
    # Object extraction – tight box around the colour that appears most often
    # (ignoring the background colour)
    bg        = mostcolor(I)                                            # background colour
    fg        = mostcolor(replace(I, bg, -1))                            # most frequent non‑background colour
    O         = subgrid(ofcolor(I, fg), I)                                # bounding box of that colour
    return O


def dsl3(I):
    # Colour filtering – delete the background colour and trim empty margins
    bg        = mostcolor(I)                                            # background colour
    O_no_bg   = replace(I, bg, 0)                                       # turn background cells to 0
    O         = compress(O_no_bg)                                       # remove outer rows / columns that are all 0
    return O


def dsl4(I):
    # Geometric reasoning – keep the part of the grid that stays unchanged
    # after a horizontal mirror (vertical symmetry axis)
    mirrored  = hmirror(I)                                               # horizontally mirrored grid
    invariant = intersection(asobject(I), asobject(mirrored))           # cells unchanged by the mirror
    O         = subgrid(toindices(invariant), I)                         # minimal rectangle enclosing them
    return O


def dsl5(I):
    # Relational / structural – region of the rarest colour together with its 4‑neighbourhood
    rare      = leastcolor(I)                                            # colour that occurs the fewest times
    rc        = ofcolor(I, rare)                                         # cells of that colour
    neigh     = combine(rc,
                         combine(shift(rc, DOWN),
                                 combine(shift(rc, UP),
                                         combine(shift(rc, LEFT),
                                                 shift(rc, RIGHT)))))
    O         = subgrid(neigh, I)                                        # tight box around the whole neighbourhood
    return O
