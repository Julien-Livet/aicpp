    ones   = ofcolor(I, 1)
    fours  = ofcolor(I, 4)

    # colours that may take part in a swap (everything except 0,1,4)

    # 1 → c  : a 1‑cell that sees a c‑cell below it, with no 4 in‑between
    def up_to_c(c):
        below = lambda cell: {(r, cell[1]) for r in range(cell[0] + 1, len(I))}
        return {cell for cell in ones

                and not intersection(below(cell), fours)}
