```python
def dsl1(I):
    # GLOBAL – just remove empty border and shrink by a factor of 2.
    O = downscale(trim(I), 2)
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION – compress the picture (remove all‑zero frontiers)
    # then shrink by 2.  compress behaves like trim but also discards
    # interior uniform rows/columns, giving a slightly different route.
    O = downscale(compress(I), 2)
    return O


def dsl3(I):
    # GEOMETRIC – cut out the exact bounding box that contains the 2‑cells
    # and then down‑scale it.
    patch   = ofcolor(I, 2)                       # all 2‑cells
    top     = uppermost(patch)
    left    = leftmost(patch)
    h       = height(patch)
    w       = width(patch)
    cropped = crop(I, (top, left), (h, w))
    O       = downscale(cropped, 2)
    return O


def dsl4(I):
    # COLOR‑FILTERING – turn every 0 into 1 (so that down‑scale has a
    # non‑empty colour in each block), compute the horizontal period of the
    # 2‑pattern and down‑scale by that period (which is 2 for all three
    # training examples).
    I1      = replace(I, 0, 1)
    period  = hperiod(ofcolor(I, 2))
    O       = downscale(I1, period)
    return O


def dsl5(I):
    # RELATIONAL – centre the picture on the centre of mass of the 2‑cells,
    # take the 7×7 window that starts at the origin and finally rotate
    # it 180° to match the required orientation.
    patch   = ofcolor(I, 2)                       # indices of colour 2
    cm      = centerofmass(patch)                 # (row, col) centre of mass
    obj     = toobject(patch, I)                  # coloured object
    shifted = move(I, obj, (-cm[0], -cm[1]))       # shift whole grid
    window  = crop(shifted, (0, 0), (7, 7))        # 7×7 window centred on origin
    O       = rot180(window)                     # final orientation
    return O
```
