def dsl1(I):
    # GLOBAL‑COMBINE: union of all colour masks, then colour the extreme columns.
    colours   = {c for c in palette(I) if c > 0}
    masks     = apply(lambda c: ofcolor(I, c), colours)          # patches of each colour
    union     = combine(masks)                                # all coloured cells together
    leftcol   = leftmost(union)                               # global leftmost column
    rightcol  = rightmost(union)                              # global rightmost column
    lefts     = recolor(4, sfilter(union, lambda rc: rc[1] == leftcol))
    rights    = recolor(2, sfilter(union, lambda rc: rc[1] == rightcol))
    O = I
    O = paint(O, rights)
    O = paint(O, lefts)
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION: work on each connected component, then take the global extremes.
    objs      = fgpartition(I)                                 # all objects (including background)
    coloured  = mfilter(objs, lambda o: color(o) > 0)           # drop background
    # column extremes of every object
    lefts_obj = apply(lambda o: leftmost(toindices(o)), coloured)
    rights_obj= apply(lambda o: rightmost(toindices(o)), coloured)
    leftcol   = minimum(lefts_obj)                              # global leftmost
    rightcol  = maximum(rights_obj)                             # global rightmost
    # all coloured cells again
    all_cells = combine(apply(lambda o: toindices(o), coloured))
    lefts     = recolor(4, sfilter(all_cells, lambda rc: rc[1] == leftcol))
    rights    = recolor(2, sfilter(all_cells, lambda rc: rc[1] == rightcol))
    O = I
    O = paint(O, rights)
    O = paint(O, lefts)
    return O


def dsl3(I):

def dsl4(I):
    # GEOMETRIC‑REASONING: draw the bounding rectangle of all coloured cells,
    # keep only the rectangle sides that actually intersect the original colours.
    colours   = {c for c in palette(I) if c > 0}
    masks     = apply(lambda c: ofcolor(I, c), colours)
    union     = combine(masks)                                 # all coloured cells
    rect      = box(toindices(union))                          # outline of the bounding box
    leftcol   = leftmost(rect)
    rightcol  = rightmost(rect)
    # vertical sides of the rectangle
    left_side  = sfilter(rect, lambda rc: rc[1] == leftcol)
    right_side = sfilter(rect, lambda rc: rc[1] == rightcol)
    # intersect with original coloured cells so we only colour existing material
    lefts  = recolor(4, intersection(left_side, union))
    rights = recolor(2, intersection(right_side, union))
    O = I
    O = paint(O, rights)
    O = paint(O, lefts)
    return O
