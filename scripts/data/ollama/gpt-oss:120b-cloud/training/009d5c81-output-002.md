def dsl1(I):
    # ------------------------------------------------------------------
    # 1️⃣  Object‑extraction + vertical periodicity
    #    – extract the 8‑object,
    #    – compute its vertical period,
    #    – map the period to the new colour (7/3/2),
    #    – erase the 1‑shape.
    # ------------------------------------------------------------------
    patch8 = ofcolor(I, 8)                          # the 8‑object
    per    = vperiod(patch8)                         # vertical period
    col    = branch(greater(per, 2), 7,
                    branch(greater(per, 1), 3, 2))   # 7 if >2, 3 if =2, else 2
    O      = replace(I, 8, col)                     # recolour the 8‑object
    O      = replace(O, 1, 0)                       # delete the 1‑shape
    return O


def dsl2(I):
    # ------------------------------------------------------------------
    # 2️⃣  Relational – number of “holes” inside the 8‑object
    #    – area = height × width of the bounding box,
    #    – cnt  = number of 8 cells,
    #    – holes = area – cnt,
    #    – colour decided from the amount of holes.
    # ------------------------------------------------------------------
    patch8 = ofcolor(I, 8)
    area   = multiply(height(patch8), width(patch8))
    cnt    = colorcount(patch8, 8)
    holes  = subtract(area, cnt)
    col    = branch(greater(holes, 30), 2,          # many holes → 2
                branch(greater(holes, 10), 3, 7)) # medium → 3, else 7
    O      = replace(I, 8, col)
    O      = replace(O, 1, 0)
    return O
