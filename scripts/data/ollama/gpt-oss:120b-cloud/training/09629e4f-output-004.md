def dsl1(I):
    # ---------- global‑style solution ----------
    # 1) locate the full rows/columns of 5s as horizontal/vertical lines
    h5      = apply(lambda p: p[0],
                    mfilter(ofcolor(I, 5),
                            lambda p: hline({p})))          # rows that belong to a horizontal line of 5
    v5      = apply(lambda p: p[1],
                    mfilter(ofcolor(I, 5),
                            lambda p: vline({p})))          # columns that belong to a vertical line of 5
    # 2) compute the limits of each block using the separators
    rcuts   = sort(combine({0}, combine(h5, {shape(I)[0]})))   # 0, each separator row, height
    ccuts   = sort(combine({0}, combine(v5, {shape(I)[1]})))   # 0, each separator col, width
    # 3) fill every block with its own most common colour
    O = I
    for rs in interval(0, size(rcuts) - 1, 1):
        for cs in interval(0, size(ccuts) - 1, 1):
            r0 = list(rcuts)[rs]
            r1 = list(rcuts)[rs+1]
            c0 = list(ccuts)[cs]
            c1 = list(ccuts)[cs+1]
            blk = crop(I, (r0, c0), (r1 - r0 - 1, c1 - c0 - 1))
            col = mostcolor(blk)
            idx = mfilter(asindices(I),
                         lambda p: p[0] > r0 and p[0] < r1 and p[1] > c0 and p[1] < c1)
            O   = paint(O, recolor(col, idx))
    return O

def dsl3(I):
    # ---------- colour‑filtering solution ----------
    # swap 5 ↔ 0 so that separator lines become background (0),
    # then replace every non‑background cell by the dominant colour of its 3×3 neighbourhood.
    S       = switch(I, 5, 0)                       # separators become 0
    G       = smear(S, 3)                           # custom primitive that spreads colour in 3×3 neighbourhood
    O       = switch(G, 0, 5)                       # put the separators back
    return O
