# ----------------------------------------------------------------------
# 1️⃣  Global‑propagation hypothesis
#    – Propagate the colour 6 upward until it meets a different non‑4 colour.
#    – All cells that are “1” and have a 6 somewhere below in the same column
#      become 6.  Everything else is left unchanged.
# ----------------------------------------------------------------------
def dsl1(I):
    # set of all cells coloured 6
    sixes   = ofcolor(I, 6)
    # set of all cells coloured 1
    ones    = ofcolor(I, 1)

    # for each 1, look straight down; if a 6 is encountered before a 4,
    # mark the cell for conversion.
    def should_be_six(cell):
        r, c = cell
        below = {(rr, c) for rr in range(r + 1, len(I))}
        return bool(intersection(below, sixes)) and not bool(intersection(below, ofcolor(I, 4)))

    conv = {cell for cell in ones if should_be_six(cell)}
    O = recolor(6, conv)
    O = paint(O, recolor(1, ones - conv))       # restore the remaining 1‑cells
    O = paint(O, recolor(4, ofcolor(I, 4)))    # keep all 4‑cells as they were
    return O


# ----------------------------------------------------------------------
# 2️⃣  Object‑copy hypothesis
#    – Identify the lowest non‑background coloured object (the block that
#      contains the colour 6) and copy its shape to the top‑most block of
#      the same size, overwriting any 1‑coloured cells there.
# ----------------------------------------------------------------------
def dsl2(I):
    # all objects (background colour is 0)
    objs      = fgpartition(I)
    # object that contains a 6 (the “source”)
    src_obj   = extract(lambda o: 6 in palette(I), objs)
    # bounding box of the source object
    src_box   = backdrop(src_obj)
    # height and width of the source
    h, w      = shape(src_box)

    # find the highest object that does NOT contain a 6 and has the same size
    candidate = {o for o in objs if 6 not in palette(I) and shape(backdrop(o)) == (h, w)}
    dst_obj   = extract(lambda o: True, candidate)          # the first such object
    dst_box   = backdrop(dst_obj)

    # move the source object into the destination location
    shift_vec = (min(p[0] for p in dst_box) - min(p[0] for p in src_box),
                 min(p[1] for p in dst_box) - min(p[1] for p in src_box))
    moved_src = shift(src_obj, shift_vec)

    # paint the moved source over the original grid, preserving all other colours
    O = paint(I, moved_src)
    return O


# ----------------------------------------------------------------------
# 3️⃣  Colour‑swap hypothesis
#    – Within every horizontal strip that is bounded by two rows of 4’s,
#      swap the colours 1 ↔ 6.  (Rows of all‑4 act as separators.)
# ----------------------------------------------------------------------
def dsl3(I):
    # rows that are completely 4
    full4_rows = {r for r in range(len(I))
                 if all(index(I, (r, c)) == 4 for c in range(len(I[0])))}
    # split the grid vertically at those rows
    parts = hsplit(I, sorted(full4_rows))

    def swap_part(part):

        sixes = ofcolor(part, 6)
        part = recolor(1, sixes)(part)
        part = recolor(6, ones)(part)
        return part

    O = hconcat(*(apply(swap_part, [p]) for p in parts))
    return O

# 4️⃣  Mirroring hypothesis
#    – Take the lower‑right quadrant (the region that contains the colour 6)
#      mirror it vertically and paste it onto the upper‑left quadrant,
#      replacing any 1‑coloured cells there.
# ----------------------------------------------------------------------
def dsl4(I):
    h, w = shape(I)
    # define the lower‑right quadrant
    br = subgrid({(r, c) for r in range(h//2, h) for c in range(w//2, w)}, I)
    # mirror that quadrant vertically (top‑bottom flip)
    mir = vmirror(br)

    # position of the upper‑left quadrant

    # overwrite 1‑cells with the moved block

#    – In every row, if a cell coloured 1 is immediately left of a 6,
