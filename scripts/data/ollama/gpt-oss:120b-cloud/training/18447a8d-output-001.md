def dsl1(I):
    # --------- GLOBAL SWAP 7↔other colours (horizontal adjacency) ----------
    # 1. keep a copy of the original grid for reference
    O0 = I
    # 2. all non‑7, non‑8 colours that appear
    other = palette(I) - {7, 8}
    # 3. for each such colour c, turn the 7‑cells that have c on their right
    #    into that colour, and turn the colour cells themselves into 7.
    #    (set of 7‑cells that will become c)
    to_c = mapply(lambda c:
                   {p for p in ofcolor(O0, 7)
                     if add(p, RIGHT) in ofcolor(O0, c)},
                   other)
    #    (set of colour‑c cells that will become 7)
    to_7 = mapply(lambda c: ofcolor(O0, c), other)
    # 4. start from the original grid and apply the replacements
    O = O0
    #    turn each target‑c patch into colour c
    O = mapply(lambda pair: fill(O, pair[0], pair[1]),
               zip(other, to_c))
    #    turn each colour‑c patch into 7
    O = mapply(lambda c: fill(O, 7, to_7[c]), other)
    return O
