def dsl1(I):
    # Global transformation:
    #   1. Remove the outer zero border.
    #   2. Turn interior background (0) into colour 1.
    #   3. Swap colours 2 and 3 (colour 4 stays unchanged).
    G = compress(I)
    B = replace(G, 0, 1)
    O = switch(B, 2, 3)
    return O


def dsl2(I):
    # Object‑extraction route:
    #   1. Crop the border.
    #   2. Replace interior 0 → 1 (background).
    #   3. Extract all coloured objects.
    #   4. Recolour objects of colour 2 → 3 and colour 3 → 2,
    #      keep colour 4 as‑is.
    #   5. Paint the recoloured objects on the new background.
    G   = compress(I)
    BG  = replace(G, 0, 1)
    OBJ = fgpartition(G)                     # all non‑zero objects
    SET2 = colorfilter(OBJ, 2)                # objects coloured 2
    SET3 = colorfilter(OBJ, 3)                # objects coloured 3
    SET4 = colorfilter(OBJ, 4)                # objects coloured 4
    SW2 = apply(lambda o: recolor(3, o), SET2)   # 2 → 3
    SW3 = apply(lambda o: recolor(2, o), SET3)   # 3 → 2
    ALL = combine(combine(SW2, SW3), SET4)     # union of all recoloured objects
    O   = paint(BG, ALL)
    return O


def dsl3(I):
    # Colour‑filtering with explicit patches:
    #   1. Crop the border.
    #   2. Convert interior 0 → 1.
    #   3. Build patches for colours 2, 3, 4,
    #      turn them into coloured objects,
    #      swap the colours of the first two patches,
    #      then paint them over the background.
    G   = compress(I)
    BG  = replace(G, 0, 1)
    P2  = ofcolor(G, 2)                     # indices of colour 2
    P3  = ofcolor(G, 3)                     # indices of colour 3
    P4  = ofcolor(G, 4)                     # indices of colour 4
    O2  = toobject(P2, G)                    # (value, index) pairs for colour 2
    O3  = toobject(P3, G)                    # (value, index) pairs for colour 3
    O4  = toobject(P4, G)                    # (value, index) pairs for colour 4
    SW2 = recolor(3, O2)                     # 2 → 3
    SW3 = recolor(2, O3)                     # 3 → 2
    ALL = merge({SW2, SW3, O4})
    O   = paint(BG, ALL)
    return O


def dsl4(I):
    # Geometric reasoning via rotation:
    #   1. Rotate the picture clockwise.
    #   2. Crop the zero border.
    #   3. Convert remaining background (0) to colour 1.
    #   4. Swap colours 2 and 3.
    #   5. Rotate back to original orientation.
    R   = rot90(I)                # clockwise rotation
    C   = compress(R)              # remove outer zeros
    B   = replace(C, 0, 1)        # background → 1
    S   = switch(B, 2, 3)         # colour swap
    O   = rot270(S)               # rotate back (90° counter‑clockwise)
    return O
