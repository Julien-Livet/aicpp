def dsl1(I):
    # Global – compress the whole picture by a factor 3 and restore its size.
    O = hupscale(vupscale(downscale(I, 3), 3), 3)
    return O


def dsl2(I):
    # Object‑extraction – locate the vertical line of colour 5,
    # split the grid into three column zones (left, middle, right)
    # and recolour each zone with a constant (0, 3 and 6 respectively).
    five      = ofcolor(I, 5)                                 # indices of colour‑5 cells
    mincol    = valmin(apply(lambda p: p[1], five), lambda x: x)
    maxcol    = valmax(apply(lambda p: p[1], five), lambda x: x)

    all_idx   = asindices(I)
    left      = mfilter(all_idx, lambda p: p[1] < mincol)
    middle    = mfilter(all_idx, lambda p: mincol < p[1] < maxcol)
    right     = mfilter(all_idx, lambda p: p[1] > maxcol)

    base      = canvas(0, shape(I))                           # start with a blank canvas (all 0)
    O1        = recolor(3, middle)                           # colour 3 in the middle zone
    O2        = recolor(6, right)                            # colour 6 in the right zone
    O         = paint(paint(base, left), O1)                 # add the two coloured zones
    O         = paint(O, O2)                                 # final picture
    return O


def dsl3(I):

    O = switch(switch(I, 4, 6), 6, 4)
    return O


def dsl4(I):
    # Geometric – rotate the whole picture 90° clockwise.
    O = rot90(I)
    return O


def dsl5(I):
    # Relational – move every 5‑coloured cell one step toward the nearest 0‑coloured cell.
    five_cells   = ofcolor(I, 5)
    zero_cells   = ofcolor(I, 0)
    shift_dir    = gravitate(five_cells, zero_cells)          # direction (dx,dy)
    moved_idx    = shift(five_cells, shift_dir)               # shifted positions (indices)
    moved_obj    = toobject(moved_idx, I)                     # give the shifted cells their original colour (5)
    cleared      = fill(I, 0, five_cells)                     # erase original 5s
    O            = paint(cleared, moved_obj)                 # paint them at the new positions
    return O
