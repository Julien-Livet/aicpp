def dsl1(I):

    # 1) compress : cut away any solid colour border.
    # 2) hupscale  : double horizontally.
    # 3) vupscale  : double vertically.

def dsl2(I):
    # Object‑extraction – keep every object of the most frequent colour
    # and paint them on a canvas filled with the least frequent colour.
    bg        = leastcolor(I)                                   # background colour
    objects   = fgpartition(I)                                   # all foreground objects
    dom_col   = mostcolor(I)                                     # dominant colour
    dom_objs  = colorfilter(objects, dom_col)                    # only those objects
    canvas    = canvas(bg, shape(I))                             # empty canvas
    # combine all dominant objects (combine works on two arguments, we fold)
    combined  = fold(lambda a, b: combine(a, b), dom_objs)       # union of all objects
    O = paint(canvas, combined)                                 # draw them
    return O


def dsl3(I):
    # Colour‑filtering – any cell whose colour is not the dominant colour
    # but touches a dominant‑coloured cell is recoloured to the dominant colour.
    dom          = mostcolor(I)                                 # dominant colour
    other_cols   = difference(palette(I), initset(dom))         # all other colours
    O = I
    O = mapply(
            lambda c:
                recolor(
                    dom,
                    intersection(
                        ofcolor(O, c),                # cells of colour c
                        delta(ofcolor(O, dom))        # those adjacent to a dominant cell
                    )
                ),
            other_cols
        )
    return O


def dsl4(I):
    # Geometric – intersect the picture with its vertical mirror;
    # keep the common cells, fill the rest with the background colour.
    bg      = leastcolor(I)                                     # background colour

                ofcolor(I, mostcolor(I)),                        # (any colour, we will mask later)
                ofcolor(mir, mostcolor(I))

    O = underfill(I, bg, delta(common))                        # write bg everywhere else,

    return O


def dsl5(I):

    bg       = leastcolor(I)                                    # background colour
    objs     = fgpartition(I)                                    # all objects
    mid_row  = halve(height(I))
    mid_col  = halve(width(I))
    centre   = initset(astuple(mid_row, mid_col))                # centre cell as a set
    moved    = apply(
