def dsl1(I):
    # Global: crop to the tight non‑zero bounding box,
    # then fill every remaining background cell with the most common colour.
    T = trim(I)                                    # interior with any non‑zero cell
    O = canvas(0, shape(T))                       # empty canvas of the trimmed size
    O = paint(O, asobject(T))                     # copy the trimmed pattern
    C = mostcolor(T)                              # dominant colour inside the box
    O = fill(O, C, ofcolor(O, 0))                 # recolour all zero cells to that colour
    return O


def dsl2(I):
    # Object extraction: keep only the single largest connected component
    # (regardless of its colour).
    OBJ = objects(I, univalued=False, diagonal=False, without_bg=True)
    BIG = argmax(OBJ, lambda o: size(o))          # the component with maximal size
    O = canvas(0, shape(I))
    O = paint(O, BIG)                             # render the chosen component
    return O


def dsl3(I):

def dsl4(I):
    # Geometric reasoning: create horizontal and vertical mirrors,
    # overlay them (original cells dominate) and finally combine with the original.
    H = hmirror(I)                                # horizontal flip (same dimensions)
    V = vmirror(I)                                # vertical flip (same dimensions)
    M = cellwise(H, V, fallback=0)               # union of the two mirrors
    O = cellwise(I, M, fallback=0)                # original non‑zero cells keep precedence
    return O


def dsl5(I):
    # Relational/structural: draw the outline of every coloured region
    # with its own colour.
    OBJ = objects(I, univalued=False, diagonal=False, without_bg=True)
    O = canvas(0, shape(I))
    # for each object: recolour its border (box) with the object's colour
    O = apply(lambda o: paint(O, recolor(color(o), box(o))), OBJ)
    return O
