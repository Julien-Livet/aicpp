def dsl1(I):
    # 1️⃣ Relational – Manhattan distance ≤2 from any 4 (shift‑based dilation)
    fours   = ofcolor(I, 4)

    d1 = combine(
            combine(combine(combine(shift(fours, UP),
                                   shift(fours, DOWN)),
                           shift(fours, LEFT)),
                    shift(fours, RIGHT))

    d2 = combine(
            combine(combine(combine(shift(d1, UP),
                                   shift(d1, DOWN)),
                           shift(d1, LEFT)),
                    shift(d1, RIGHT))

    reach   = combine(fours, combine(d1, d2))               # all cells ≤2 from a 4
    target  = intersection(ofcolor(I, 5), reach)            # 5‑cells close to a 4
    O = fill(I, 2, toobject(target, I))
    return O


def dsl2(I):

    reach   = combine(fours, combine(n1, n2))

def dsl3(I):
    # 3️⃣ Object‑extraction – rectangular hull of every 4 expanded by two cells
    fours   = ofcolor(I, 4)

    # minimal rectangle that contains the 4‑cells
    rect    = box(fours)

    # expand the rectangle by one cell in the four directions
    exp1 = combine(
            combine(combine(combine(shift(rect, UP),

def dsl4(I):
