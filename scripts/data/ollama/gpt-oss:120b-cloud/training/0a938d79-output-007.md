def dsl1(I):
    # OBJECT‑EXTRACTION + ORIENTATION BY BOUNDING‑BOX
    objs = objects(I, False, False, True)
    box  = subgrid(merge(objs), I)                     # tight slice of all colours
    hb   = height(box);  wb = width(box)

    # empty canvas of the slice size
    base = canvas(0, (hb, wb))

    rows_pat = merge(apply(
        lambda o: fill(base,
                        color(o),
                        {(r, c) for (r, c) in asindices(base)
                         if r == center(o)[0]}),
        objs))

    cols_pat = merge(apply(
        lambda o: fill(base,
                        color(o),
                        {(r, c) for (r, c) in asindices(base)
                         if c == center(o)[1]}),
        objs))

    pat = branch(greater(wb, hb), rows_pat, cols_pat)   # rows if slice wider, else columns

    result = branch(greater(wb, hb),
                    vupscale(pat, divide(height(I), hb)),
                    hupscale(pat, divide(width(I),  wb)))
    return result


def dsl2(I):

def dsl3(I):
    # LINE‑SHOOTING FROM CENTRES, ORIENTED BY BOUNDING‑BOX
    objs = objects(I, False, False, True)
    box  = subgrid(merge(objs), I)
    hb   = height(box); wb = width(box)

    rows = merge(apply(
        lambda o: fill(I,
                      color(o),

        objs))

    result = branch(greater(wb, hb),

                    hupscale(cols, divide(width(I),  wb)))
    return result


def dsl4(I):
    # MIRROR‑PLUS‑TILE (mirroring across the diagonal of the shorter side)
    objs   = objects(I, False, False, True)
    slice_ = subgrid(merge(objs), I)

    vp = height(slice_)
    hp = width (slice_)

    mirrored = branch(greater(vp, hp),
                      cmirror(slice_),   # mirror across main diagonal
                      dmirror(slice_))   # mirror across counter‑diagonal

    result = branch(greater(vp, hp),
                    hupscale(mirrored, divide(width(I),  hp)),
                    vupscale(mirrored, divide(height(I), vp)))
    return result


def dsl5(I):
    # RELATIONAL REPEAT: use the vertical distance between distinct rows
    objs = objects(I, False, False, True)
    box  = subgrid(merge(objs), I)
    hb   = height(box); wb = width(box)

    # set of distinct row indices inside the slice
    rows_set = {center(o)[0] for o in objs}
    # smallest positive distance between any two rows (spacing)
    drow = branch(sizefilter(rows_set, 2),
                  valmin({abs(r2 - r1) for r1 in rows_set for r2 in rows_set
                          if r1 != r2 and abs(r2 - r1) > 0},
                         identity),
                  height(I))                     # fallback: whole height

                       {(0, c) for c in range(wb)

        objs))

    # tile the single‑row pattern vertically using the computed spacing
    result = vupscale(pat, divide(height(I), drow))
    return result
