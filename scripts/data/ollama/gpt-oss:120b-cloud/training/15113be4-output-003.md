def dsl1(I):
    # 1️⃣  Column‑wise swap 1 ↔ any other colour (≠ 0,4) when they see each other
    ones   = ofcolor(I, 1)
    fours  = ofcolor(I, 4)

    # colours that may take part in a swap (everything except 0,1,4)
    others = sfilter(palette(I), lambda c: c not in {0, 1, 4})

    # a 1‑cell that sees a colour c below it, with no 4 in‑between
    up_to_c = lambda c: {
        cell for cell in ones
        if intersection(
                {(r, cell[1]) for r in range(cell[0] + 1, len(I))},
                ofcolor(I, c)
           )
        and not intersection(
                {(r, cell[1]) for r in range(cell[0] + 1, len(I))},
                fours)
    }

    # a colour‑c cell that sees a 1 above it, with no 4 in‑between
    down_to_one = lambda c: {
        cell for cell in ofcolor(I, c)
        if intersection(
                {(r, cell[1]) for r in range(0, cell[0])},
                ones
           )
        and not intersection(
                {(r, cell[1]) for r in range(0, cell[0])},
                fours)
    }

    recol_up   = apply(lambda c: recolor(c, up_to_c(c)),   others)   # 1 → c
    recol_down = apply(lambda c: recolor(1, down_to_one(c)), others) # c → 1

    O = paint(I, merge(recol_up))
    O = paint(O, merge(recol_down))
    return O


def dsl2(I):
    # 2️⃣  Global replacement: turn every 1 into the most frequent non‑0/1/4 colour
    other_colors = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    # colour that appears the most (ignoring 0,1,4)
    main = argmax(other_colors, lambda c: colorcount(I, c))
    O = paint(I, recolor(main, ofcolor(I, 1)))
    return O


def dsl3(I):

    objs = fgpartition(I)                                   # all coloured objects

    src = argmax(coloured, lowermost)                       # lowest block (the propagating colour)

    tgt_box = backdrop(tgt_patch)

    moved = shift(src, shift_vec)                           # copy of the source object
    O = paint(I, moved)                                     # overlay it onto the 1‑region
    return O


def dsl4(I):
    # 4️⃣  Mirror the lower‑right quadrant onto the upper‑left, then replace 1’s
    h, w = shape(I)

    # lower‑right quadrant (contains the propagating colour)
    br_coords = {(r, c) for r in range(h // 2, h) for c in range(w // 2, w)}
    br = subgrid(br_coords, I)

    # mirror vertically then horizontally
    mir = hmirror(vmirror(br))

    # move the mirrored quadrant to the upper‑left corner
    tl_coords = {(r, c) for r in range(h // 2) for c in range(w // 2)}
    shift_vec = (min(r for r, _ in tl_coords) - min(r for r, _ in backdrop(mir)),
                min(c for _, c in tl_coords) - min(c for _, c in backdrop(mir)))
    moved = shift(mir, shift_vec)

    # replace every 1 with the colour coming from the mirrored content
    O = paint(I, recolor(1, moved))
    return O


def dsl5(I):

        return (paint(part, recolor(tgt, ofcolor(part, 1)))
                .paint(part, recolor(1, ofcolor(part, tgt))) if tgt is not None else part

    O = hconcat(*apply(swap, strips))
    return O
