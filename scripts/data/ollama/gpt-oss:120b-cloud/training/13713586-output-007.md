def dsl1(I):
    # OBJECT‑RECTANGLE: extend each non‑border object to the border line by
    # taking the backdrop of the object together with the border and erasing the border.
    BG   = mostcommon(I)
    CAN  = canvas(BG, shape(I))
    CLS  = difference(palette(I), initset(BG))
    BC   = argmax(CLS, lambda c: size(ofcolor(I, c)))          # colour of the solid border
    B    = ofcolor(I, BC)                                      # border cells (indices)
    OBJ  = objects(I, False, False, True)                     # all coloured objects
    BOBJ = colorfilter(OBJ, BC)                               # border object(s)
    OTH  = difference(OBJ, BOBJ)                              # objects that are not the border
    RCT  = apply(lambda o: recolor(
                difference(
                    backdrop(combine(toindices(o), B)),      # rectangle up to but not over border
                    B),
                color(o)),
                OTH)                                            # coloured rectangles
    O    = paint(CAN, merge(RCT))
    return O


def dsl2(I):
    # UNDERPAINT‑RECTANGLE: same rectangle construction as dsl1 but
    # applied with underpaint so the original border colour stays untouched.
    BG   = mostcommon(I)
    CAN  = canvas(BG, shape(I))
    CLS  = difference(palette(I), initset(BG))
    BC   = argmax(CLS, lambda c: size(ofcolor(I, c)))
    B    = ofcolor(I, BC)
    OBJ  = objects(I, False, False, True)
    BOBJ = colorfilter(OBJ, BC)
    OTH  = difference(OBJ, BOBJ)
    RCT  = apply(lambda o: recolor(
                difference(

                    B),
                color(o)),
                OTH)
    O    = underpaint(CAN, merge(RCT))
    return O


def dsl3(I):
    # ERASE‑BORDER + BACKDROP: first erase the border, then each remaining object’s
    # backdrop already spans the whole inner area; recolour that backdrop.
    BG   = mostcommon(I)
    CAN  = canvas(BG, shape(I))
    CLS  = difference(palette(I), initset(BG))
    BC   = argmax(CLS, lambda c: size(ofcolor(I, c)))

    # horizontally and vertically, then removing the border cells.

def dsl5(I):
    # CONNECT‑CORNER: draw a rectangle by connecting the object’s top‑left corner
    # with the farthest corner of the border, then clamp it to the border region.
    BG   = mostcommon(I)
    CAN  = canvas(BG, shape(I))
    CLS  = difference(palette(I), initset(BG))
    BC   = argmax(CLS, lambda c: size(ofcolor(I, c)))
    B    = ofcolor(I, BC)
