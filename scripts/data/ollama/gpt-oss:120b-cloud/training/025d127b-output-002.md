def dsl1(I):
    # Global right‑shift: treat every coloured cell (including all colours) as one patch,
    # shift it one step to the right and paint the result on an empty canvas.
    empty   = canvas(0, shape(I))                     # all‑zero background of same size
    shifted = shift(asobject(I), RIGHT)                # shift whole coloured patch right
    O       = paint(empty, shifted)                    # paint shifted cells onto background
    return O


def dsl2(I):
    # Object‑level interior‑fill: for each foreground object fill the cells that lie
    # inside its bounding box but are not part of the object, using the object’s colour.
    objs = fgpartition(I)
    fill_one = lambda o: fill(I, color(o), delta(o))
    O = apply(fill_one, objs)                          # apply the fill to every object
    return O


def dsl3(I):
    # Most‑common‑colour right‑shift: isolate the dominant non‑zero colour,
    # shift its cells one step right and render the shifted patch on a blank canvas.
    main   = mostcolor(I)                               # dominant foreground colour
    patch  = ofcolor(I, main)                           # all cells of that colour
    shifted = shift(patch, RIGHT)                       # move those cells right by one
    empty   = canvas(0, shape(I))                      # empty background
    O       = paint(empty, shifted)                     # paint shifted patch onto background
    return O


def dsl4(I):
    # Left‑right mirroring into the original background:
    # mirror the whole grid, keep only the cells that were background in the input,
    # and paint them over the original grid.
    mirrored = vmirror(I)                               # horizontal mirror of the grid
    bg_mask  = ofcolor(I, ZERO)                         # indices where input is background
    # turn the mirrored grid into an object and intersect with the background mask
    mirrored_obj = toobject(bg_mask, mirrored)
    O = paint(I, mirrored_obj)                          # paint the intersected cells
    return O


def dsl5(I):
    # Relational gravity: move every foreground object one step toward the
    # centre of the background (the cell that makes the object adjacent to the
    # background’s centre).  The direction is given by gravitate.
    objs = fgpartition(I)
    bg   = ofcolor(I, ZERO)                             # background cells
    move_one = lambda o: move(I, o, gravitate(o, bg))   # shift object by one step
    O = apply(move_one, objs)                           # apply to all objects
    return O
