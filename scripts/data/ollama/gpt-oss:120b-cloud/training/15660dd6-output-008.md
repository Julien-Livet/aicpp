def dsl1(I):
    # GLOBAL – trim, recolour, find the topmost pattern row, compute its
    # vertical period and crop exactly one period (full width).
    c   = trim(I)

    start = uppermost(pat)                 # first row that contains the pattern
    per = vperiod(pat)                     # height of one repeating block
    w   = width(G)
    O   = crop(G, (start, 0), (per, w))
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION – trim & recolour, isolate the whole pattern,
    # restrict to its bounding box, split that box vertically by the period
    # and keep the first block.
    c   = trim(I)
    c1  = replace(c, 8, 5)
    c2  = replace(c1, 1, 9)
    c3  = replace(c2, 3, 8)
    G   = replace(c3, 2, 7)
    pat = asobject(G)                     # full coloured object
    box = subgrid(pat, G)                 # minimal sub‑grid that encloses all repeats
    per = vperiod(pat)                    # vertical period of the pattern
    pieces = vsplit(box, per)              # vertical blocks of height = period
    O = first(pieces)                      # keep the first block
    return O

def dsl4(I):
