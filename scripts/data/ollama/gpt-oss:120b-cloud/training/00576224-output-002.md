def dsl1(I):
    # GLOBAL TILING: repeat the whole input three times vertically,
    # using the original orientation on the first and third blocks,
    # and its horizontal mirror on the middle block.
    A = hupscale(I, 3)                 # 2×6, original orientation
    B = hupscale(hmirror(I), 3)        # 2×6, mirrored orientation
    O = vconcat(A, vconcat(B, A))      # 6×6 final pattern
    return O


def dsl2(I):
    # OBJECT‑LEVEL: expand every 1×1 cell to a 3×3 block.
    # The blocks coming from the original grid occupy the
    # first and last vertical thirds; the blocks coming from
    # the horizontally‑mirrored grid are shifted down one third.
    # All blocks are painted on a blank 6×6 canvas.
    # – create 3×3 blocks for the original cells
    orig_objs   = objects(I, True, False, False)
    orig_blocks = apply(lambda o: upscale(toobject(o, I), 3), orig_objs)

    # – create 3×3 blocks for the mirrored cells and shift them down
    mirr_grid   = hmirror(I)
    mirr_objs   = objects(mirr_grid, True, False, False)
    mirr_blocks = apply(
        lambda o: shift(upscale(toobject(o, mirr_grid), 3), (3, 0)),
        mirr_objs)

    # – combine all blocks and paint them
    all_blocks  = combine(combine(orig_blocks, mirr_blocks))
    blank       = canvas(0, (6, 6))
    O = paint(blank, all_blocks)
    return O


def dsl3(I):
    # COLOR FILTERING: build the correct 3×3 tiling (as in dsl1),
    # then replace every background (0) with the most frequent colour
    # of the tiled picture.
    A = hupscale(I, 3)
    B = hupscale(hmirror(I), 3)
    tiled   = vconcat(A, vconcat(B, A))          # 6×6 correct layout
    bgcol   = mostcolor(tiled)                   # dominant colour in tiled grid
    O = underfill(tiled, bgcol, ofcolor(tiled, 0))
    return O


def dsl4(I):
    # GEOMETRIC REASONING: rotate the input 180°, apply the same
    # three‑block tiling as in dsl1, then rotate the result back.
    R = rot180(I)                                 # 2×2 rotated version
    A = hupscale(R, 3)
    B = hupscale(hmirror(R), 3)
    tiled = vconcat(A, vconcat(B, A))
    O = rot180(tiled)                             # restore original orientation
    return O


def dsl5(I):
    # RELATIONAL / STRUCTURAL: start from a blank canvas, paint the
    # original‑orientation block at the top, the mirrored block exactly
    # three rows lower, and finally the original block again three rows
    # below the mirror.  The relative vertical displacement (0, 3, 6)
    # encodes the required relationship between the three parts.
    blank = canvas(0, (6, 6))

    A = hupscale(I, 3)                 # top block (rows 0‑1)
    B = hupscale(hmirror(I), 3)        # middle block (rows 2‑3)

    top   = paint(blank, asobject(A))                     # place A at (0,0)
    middle = paint(top,   asobject(shift(B, (2, 0))))     # shift B down 2 rows
    O      = paint(middle, asobject(shift(A, (4, 0))))   # shift A down another 2 rows
    return O
