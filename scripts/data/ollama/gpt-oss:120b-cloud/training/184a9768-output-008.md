def dsl1(I):
    # GLOBAL‑PIXELATION + MARKER ERASURE
    G   = downscale(I, 2)                                   # coarse version
    U   = upscale(G, 2)                                     # back to original size
    MARK = combine(combine(ofcolor(I, 3), ofcolor(I, 5)), ofcolor(I, 8))
    O   = fill(U, 0, MARK)                                   # erase all marker cells
    return O


def dsl2(I):
    # OBJECT‑BASED INTERIOR‑FILLING
    OBJ   = objects(I, univalued=False, diagonal=False, without_bg=True)
    BIG   = argmax(OBJ, size)                               # largest connected component
    INNER = delta(BIG)                                       # interior cells of that component
    O     = fill(I, 0, INNER)                               # turn interior to background
    return O


def dsl3(I):
    # GEOMETRIC OVERLAY OF MIRRORS AND ROTATION
    H   = hmirror(I)                                         # horizontal mirror
    V   = vmirror(I)                                         # vertical mirror
    R   = rot180(I)                                          # 180° rotation
    M1  = cellwise(H, V, fallback=0)                        # union of H and V
    M2  = cellwise(M1, R, fallback=0)                       # add rotated picture
    O   = cellwise(I, M2, fallback=0)                        # original colours dominate
    return O


def dsl4(I):
    # RE‑COLOUR THE LARGEST OBJECT WITH THE GLOBAL DOMINANT COLOUR
    OBJ = objects(I, univalued=False, diagonal=False, without_bg=True)
    BIG = argmax(OBJ, size)                                 # biggest component
    MC  = mostcolor(I)                                      # most frequent colour overall
    O   = canvas(0, shape(I))
    O   = paint(O, recolor(MC, BIG))                        # paint that object with MC
    return O


def dsl5(I):
    # SIMPLE SCALING FILTER (down‑scale then up‑scale)
    G = downscale(I, 3)                                      # coarse representation
    O = upscale(G, 3)                                        # restore size
    return O
