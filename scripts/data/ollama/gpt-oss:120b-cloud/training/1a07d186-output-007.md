```python
def dsl1(I):
    # GLOBAL – fill the whole canvas with the most common colour,
    # then paint all original objects on top of it.
    BG      = mostcolor(I)                                           # background colour
    CANVAS  = canvas(BG, shape(I))                                   # whole grid filled with BG
    OBJ     = asobject(I)                                            # original coloured objects
    O       = underpaint(CANVAS, OBJ)                                 # overlay objects on the background
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION – keep only objects that are a straight line
    # (horizontal or vertical) and discard everything else.
    OBJS        = fgpartition(I)                                    # all coloured objects
    LINE_OBJS   = sfilter(OBJS, lambda o: hline(o) or vline(o))      # objects that form a line
    COMBINED    = merge(LINE_OBJS)                                   # union of the retained objects
    O           = underpaint(canvas(0, shape(I)), COMBINED)           # paint them on a blank canvas
    return O


def dsl3(I):
    # COLOR‑FILTERING – retain colours whose cells occupy a square patch
    # (i.e. horizontal and vertical periods are equal).
    COLORS   = palette(I)                                            # all colours present
    GOOD     = mfilter(COLORS,
                       lambda c: equality(hperiod(ofcolor(I, c)),
                                         vperiod(ofcolor(I, c))))   # colour forms a square
    PATCHES  = apply(lambda c: recolor(c, ofcolor(I, c)), GOOD)       # recolour each retained patch
    COMBINED = merge(PATCHES)                                        # union of those patches
    O        = underpaint(canvas(0, shape(I)), COMBINED)            # render on blank canvas
    return O


def dsl4(I):
    # GEOMETRIC – overlay the original grid with its vertical mirror.
    ORIG   = asobject(I)                                             # original object
    VMIR   = asobject(vmirror(I))                                    # vertically mirrored object
    COMBINED = merge({ORIG, VMIR})                                   # union of both objects
    O        = underpaint(canvas(0, shape(I)), COMBINED)            # paint union on blank canvas
    return O


def dsl5(I):
    # RELATIONAL – align every object horizontally with the leftmost column
    # of the most frequent colour object (the “reference” object).
    OBJS       = fgpartition(I)                                      # all objects
    REF_COLOR  = mostcolor(I)                                        # colour that appears most
    REF_OBJ    = argmax(OBJS, lambda o: color(o) == REF_COLOR)       # one of the objects of that colour
    REF_LEFT   = leftmost(toindices(REF_OBJ))                       # its leftmost column index
    SHIFTED    = apply(
                    lambda o:
                        shift(
                            toindices(o),
                            astuple(0,
                                    subtract(REF_LEFT,
                                             leftmost(toindices(o))))
                        ),
                    OBJS)                                            # shift each object
    COMBINED   = merge(SHIFTED)                                      # reunite shifted objects
    O          = underpaint(canvas(0, shape(I)), COMBINED)           # render on blank canvas
    return O
```
