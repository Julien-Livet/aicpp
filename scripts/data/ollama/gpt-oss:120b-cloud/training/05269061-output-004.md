def dsl1(I):
    # GLOBAL: three‑color periodic tiling inferred from the input.
    # 1. Get the three non‑zero colours.
    cols = palette(I) - {0}
    # 2. Build three vertical strips (one per colour) and tile them every 3 columns.
    strip0 = canvas(first(cols), (height(I), 1))
    strip1 = canvas(valmin(cols - {first(cols)}, lambda c: colorcount(I, c)),
                   (height(I), 1))
    strip2 = canvas(valmax(cols - {first(cols), valmin(cols - {first(cols)},
                                    lambda c: colorcount(I, c))},
                   (height(I), 1))
    offs0 = apply(lambda j: (0, j), interval(0, width(I), 3))
    offs1 = apply(lambda j: (0, j), interval(1, width(I), 3))
    offs2 = apply(lambda j: (0, j), interval(2, width(I), 3))
    V0   = apply(lambda o: paint(canvas(0, shape(I)), shift(strip0, o)), offs0)
    V1   = apply(lambda o: paint(canvas(0, shape(I)), shift(strip1, o)), offs1)
    V2   = apply(lambda o: paint(canvas(0, shape(I)), shift(strip2, o)), offs2)
    vertical = first(V0)          # any of the three already contains all vertical strips

    # 3. Build three horizontal strips (one per colour) and tile them every 3 rows.
    hstrip0 = canvas(first(cols), (1, width(I)))
    hstrip1 = canvas(valmin(cols - {first(cols)}, lambda c: colorcount(I, c)),
                    (1, width(I)))

                                    lambda c: colorcount(I, c))}),
                    (1, width(I)))
    roffs0 = apply(lambda i: (i, 0), interval(0, height(I), 3))
    roffs1 = apply(lambda i: (i, 0), interval(1, height(I), 3))

    H1 = apply(lambda o: paint(canvas(0, shape(I)), shift(hstrip1, o)), roffs1)

def dsl2(I):

    objs   = objects(I, univalued=True, diagonal=False, without_bg=True)
    pat    = argmin(objs, lambda o: size(o))                 # smallest object
    h, w   = shape(subgrid(pat, I))
    H, W   = shape(I)

    anchors = asindices(canvas(0, ((H - h + 1, W - w + 1))))

def dsl3(I):

    # and vice‑versa.

    most   = valmax(cols, lambda c: colorcount(I, c))
    least  = valmin(cols, lambda c: colorcount(I, c))
    O = switch(I, most, least)
    return O

def dsl5(I):

    # to the origin, then overlay all such normalised patches.
    cols   = palette(I) - {0}
