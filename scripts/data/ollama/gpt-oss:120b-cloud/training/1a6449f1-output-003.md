def dsl1(I):
    # Global – extract the first repetition tile of the non‑background pattern
    bg        = mostcolor(I)                                          # background colour (most common)
    pat_cells = difference(asindices(I), ofcolor(I, bg))              # cells that are not background
    vp        = vperiod(pat_cells)                                   # vertical period of that pattern
    hp        = hperiod(pat_cells)                                   # horizontal period of that pattern
    O         = crop(I, ORIGIN, (vp, hp))                             # first tile (full grid if period = size)
    return O


def dsl2(I):
    # Object extraction – bounding box of the most frequent colour that is not the background
    bg          = mostcolor(I)                                        # background colour
    I_tmp       = replace(I, bg, -1)                                    # hide background with a sentinel
    dom         = mostcolor(I_tmp)                                      # most frequent colour among the rest
    O           = subgrid(ofcolor(I, dom), I)                           # tight box around every cell of that colour
    return O


def dsl3(I):
    # Colour filtering – delete the background colour and then trim the result
    bg          = mostcolor(I)                                        # background colour
    O_no_bg     = replace(I, bg, 0)                                    # turn background to 0
    O           = compress(O_no_bg)                                   # remove outer rows / columns that are all 0
    return O


def dsl4(I):
    # Geometric reasoning – keep only the part of the grid that is invariant under a horizontal mirror
    Hmirrored   = hmirror(I)                                           # mirror horizontally
    invariant   = intersect(asobject(I), asobject(Hmirrored))           # cells that are unchanged by the mirror
    O           = subgrid(toindices(invariant), I)                    # minimal rectangle enclosing that region
    return O


def dsl5(I):
    # Relational / structural – region consisting of the rarest colour together with its 4‑neighbourhood
    rare        = leastcolor(I)                                        # colour that occurs the fewest times
    rc_cells    = ofcolor(I, rare)                                     # cells of that colour
    neigh       = combine(rc_cells,
                         combine(shift(rc_cells, DOWN),
                                 combine(shift(rc_cells, UP),
                                         combine(shift(rc_cells, LEFT),
                                                 shift(rc_cells, RIGHT)))))
    O           = subgrid(neigh, I)                                    # tight box around the whole neighbourhood
    return O
