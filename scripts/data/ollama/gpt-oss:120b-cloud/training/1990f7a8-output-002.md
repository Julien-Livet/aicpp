def dsl1(I):

def dsl3(I):
    # crop the exact 2‑patch, rotate, then down‑scale
    patch   = ofcolor(I, 2)
    top     = uppermost(patch)
    left    = leftmost(patch)
    h       = height(patch)
    w       = width(patch)
    cropped = crop(I, (top, left), (h, w))
    O       = downscale(rot90(cropped), 2)
    return O


def dsl4(I):

    norm    = normalize(patch)                 # shift patch to (0,0)
    bbox    = subgrid(norm, I)                 # minimal sub‑grid containing the shifted patch
    O       = downscale(bbox, 2)
    return O
