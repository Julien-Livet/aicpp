def dsl1(I):
    # ---------- Global mirroring with rotation ----------
    # TL = original, TR = left‑right mirror, BL = up‑down mirror obtained via
    # a 180° rotation of the left‑right mirror, BR = 180° rotation of the original.
    TL = I
    TR = vmirror(I)
    BL = rot180(vmirror(I))          # produces the up‑down mirror without using hmirror
    BR = rot180(I)
    top    = hconcat(TL, TR)
    bottom = hconcat(BL, BR)
    O = vconcat(top, bottom)
    return O


def dsl2(I):
    # ---------- Object‑wise colour tiling ----------
    # For each colour build a solid colour layer, tile it with the four
    # symmetries (LR‑mirror, UD‑mirror, 180° rotation) and finally merge
    # all coloured layers.
    colors = palette(I)

    def tiled_layer(col):
        mask   = ofcolor(I, col)                     # cells of this colour
        solid  = canvas(col, shape(I))               # uniform canvas of the colour
        layer  = paint(solid, toobject(mask, I))     # colour only on the mask
        # tile the layer
        tr = vmirror(layer)                          # left‑right mirror
        bl = rot180(vmirror(layer))                  # up‑down mirror (via 180° of LR‑mirror)
        br = rot180(layer)                          # 180° rotation
        top    = hconcat(layer, tr)
        bottom = hconcat(bl, br)
        return vconcat(top, bottom)

    tiled = apply(tiled_layer, colors)               # tuple of tiled colour‑layers
    O = merge(tiled)                                 # union of all layers
    return O


def dsl3(I):
    # ---------- Geometric rotation cascade ----------
    # Rotate the picture by 0°, 90°, 180°, 270° and arrange them
    # in a 2×2 square after mirroring the appropriate quadrants.
    r0   = I
    r90  = rot90(I)
    r180 = rot180(I)
    r270 = rot270(I)

    # mirror the right‑hand quadrants so that the final picture is
    # symmetric with respect to both axes
    top    = hconcat(r0,      vmirror(r90))
    bottom = hconcat(hmirror(r270), r180)
    O = vconcat(top, bottom)
    return O


def dsl4(I):
    # ---------- Relational shifting of mirrored copies ----------
    # Create the four symmetric copies, shift each to its quadrant
    # and overlay them on a blank canvas twice the size of the input.
    h, w = shape(I)
    canvas_big = canvas(0, (h * 2, w * 2))

    tl = toobject(ofcolor(I, leastcolor(I)), I)               # any colour works; we keep the whole picture
    tr = toobject(ofcolor(I, leastcolor(I)), I)               # will be mirrored later
    bl = toobject(ofcolor(I, leastcolor(I)), I)
    br = toobject(ofcolor(I, leastcolor(I)), I)

    # apply the required symmetries
    tr = toobject(ofcolor(vmirror(I), leastcolor(I)), vmirror(I))
    bl = toobject(ofcolor(hmirror(I), leastcolor(I)), hmirror(I))
    br = toobject(ofcolor(rot180(I), leastcolor(I)), rot180(I))

    # shift each quadrant to its place
    tr = shift(tr, (0, w))          # right
    bl = shift(bl, (h, 0))          # down
    br = shift(br, (h, w))          # down‑right

    # overlay everything
    O = merge({tl, tr, bl, br}, canvas_big)
    return O


def dsl5(I):
    # ---------- Colour‑periodic replication ----------
    # Replicate the whole picture periodically with period equal to the
    # input dimensions, then keep only the 2×2 block that starts at the origin.
    # The replication is performed by taking the Cartesian product of the
    # grid with itself and placing each copy at the appropriate offset.
    h, w = shape(I)

    # build a 2×2 repetition by shifting copies
    copy00 = toobject(ofcolor(I, leastcolor(I)), I)                     # original
    copy01 = shift(toobject(ofcolor(vmirror(I), leastcolor(I)), vmirror(I)),
                   (0, w))                                            # right‑mirrored copy
    copy10 = shift(toobject(ofcolor(hmirror(I), leastcolor(I)), hmirror(I)),
                   (h, 0))                                            # down‑mirrored copy
    copy11 = shift(toobject(ofcolor(rot180(I), leastcolor(I)), rot180(I)),
                   (h, w))                                            # 180° rotated copy

    O = merge({copy00, copy01, copy10, copy11})
    return O
