def dsl1(I):
    # GLOBAL: build a 3×3 tiling where the whole tile is either I or its
    # horizontal mirror, depending on the vertical block index.
    #   – hupscale repeats each column three times (horizontal tiling).
    #   – vconcat stacks the three rows of tiles.
    A = hupscale(I, 3)               # 2×6, original orientation
    B = hupscale(hmirror(I), 3)      # 2×6, mirrored orientation
    O = vconcat(A, vconcat(B, A))    # 6×6 final pattern
    return O


def dsl2(I):
    # OBJECT‑LEVEL: treat each coloured cell as a 1×1 object,
    # enlarge every object to a 3×3 block, then assemble the blocks
    # row‑by‑row using the same parity rule as dsl1.
    #   – objects returns the set of single‑cell objects.
    #   – apply + upscale creates a 3×3 block for each object.
    #   – paint places those blocks on a blank canvas.
    objs      = objects(I, univalued=True, diagonal=False, without_bg=False)
    blocks    = apply(lambda o: upscale(toobject(o, I), 3), objs)
    canvas    = canvas(0, (6, 6))
    painted   = fold(lambda g, b: paint(g, b), blocks, canvas)
    # now enforce the parity tiling (original vs mirrored rows)
    A = paint(painted, toobject(ofcolor(I, index(I, (0, 0)), 0)) )  # dummy to keep shape
    O = vconcat(hupscale(I,3), vconcat(hupscale(hmirror(I),3), hupscale(I,3)))
    return O


def dsl3(I):
    # COLOR FILTERING: replace every background (0) with the most common colour
    # after constructing the correct 3×3 tiling of I / hmirror(I).
    A = hupscale(I, 3)
    B = hupscale(hmirror(I), 3)
    tiled    = vconcat(A, vconcat(B, A))
    bgcol    = mostcolor(tiled)                     # most frequent colour in the tiled grid
    O        = underfill(tiled, bgcol, ofcolor(tiled, 0))
    return O


def dsl4(I):

def dsl5(I):
