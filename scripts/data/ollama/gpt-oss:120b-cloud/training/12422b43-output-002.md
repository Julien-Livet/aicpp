def dsl1(I):
    # -------------------------------------------------
    # Strategy 1 – Row‑wise replication (clear‑anchor version)
    # -------------------------------------------------
    # 1.  Anchor rows = rows that contain colour 5 in column 0.
    anchor       = ofcolor(I, 5)                                 # {(r,0)}
    rows_patch   = mapply(lambda d: shift(anchor, (0, d)),
                         interval(ZERO, width(I), ONE))          # whole rows
    # 2.  Clear the leading 5 in those rows.
    cleared      = recolor(0, anchor)
    I0           = paint(I, cleared)                             # colour 5 → 0
    pattern      = subgrid(rows_patch, I0)                       # rows without leading 5
    # 3.  How many empty rows are below the lowest non‑zero cell ?
    nz           = difference(asindices(I), ofcolor(I, 0))        # all non‑zero cells
    max_nz_row   = valmax(nz, lambda p: p[0])                   # bottom row index
    empty_rows   = subtract(height(I), add(max_nz_row, ONE))
    # 4.  Number of rows that belong to the pattern.
    pattern_rows = size({ r for r,_ in anchor })
    repeat_cnt   = divide(empty_rows, pattern_rows)             # whole blocks to add
    # 5.  Build the repeated block (repeat_cnt copies of *pattern*).
    repeat_one   = lambda g: vconcat(g, pattern)
    repeat_block = branch(greater(repeat_cnt, ONE),
                         power(repeat_one, subtract(repeat_cnt, ONE))(pattern),
                         pattern)
    # 6.  Keep the original non‑empty part (crop to the bbox of nz).
    base         = subgrid(backdrop(nz), I)
    # 7.  Append the repeated block and trim trailing zero rows.
    O = trim(vconcat(base, repeat_block))
    return O


def dsl2(I):
    # -------------------------------------------------
    # Strategy 2 – Crop → vertical‑mirror only the anchor rows
    # -------------------------------------------------
    # 1.  Crop to the tight bounding box of all non‑zero cells.
    nz        = difference(asindices(I), ofcolor(I, 0))
    core      = subgrid(backdrop(nz), I)
    # 2.  Extract the rows that contain a 5 (anchor rows) from *core*.
    anchor    = ofcolor(core, 5)
    rows_pat  = mapply(lambda d: shift(anchor, (0, d)),
                      interval(ZERO, width(core), ONE))
    cleared   = recolor(0, anchor)
    core0     = paint(core, cleared)             # same as *core* but leading 5 → 0
    pat_rows  = subgrid(rows_pat, core0)        # pattern rows
    # 3.  Mirror the pattern vertically (which simply reverses the order).
    mirror    = vmirror(pat_rows)
    # 4.  Stack the original core with the mirrored pattern and trim.
    O = trim(vconcat(core, mirror))
    return O


def dsl3(I):
    # -------------------------------------------------
    # Strategy 3 – Colour‑filtering then full duplication
    # -------------------------------------------------
    # 1.  Remove colour 5 (set to 0) everywhere.
    cleared   = recolor(0, ofcolor(I, 5))
    I0        = paint(I, cleared)
    # 2.  Duplicate the whole picture vertically until the grid is filled.
    #    The number of needed copies is the ratio of total height to the height
    #    of the non‑zero part (computed from the bbox of nz).
    nz        = difference(asindices(I0), ofcolor(I0, 0))
    core_h    = add(valmax(nz, lambda p: p[0]), ONE)               # height of content
    repeats   = divide(height(I0), core_h)                         # how many blocks
    dup_one   = lambda g: vconcat(g, I0)
    dup_block = branch(greater(repeats, ONE),
                      power(dup_one, subtract(repeats, ONE))(I0),
                      I0)
    O = trim(dup_block)
    return O

def dsl5(I):

    # -------------------------------------------------
