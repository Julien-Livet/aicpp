def dsl1(I):
    # Global transformation:
    #   1. Remove the outer zero border.
    #   2. Turn the remaining background (0) into colour 1.
    #   3. Swap colours 2 and 3 (colour 4 stays unchanged).
    O = switch(replace(compress(I), 0, 1), 2, 3)
    return O


def dsl2(I):
    # Object‑extraction & re‑painting:
    #   1. Crop the zero border.
    #   2. Replace the interior background (0) with colour 1.
    #   3. Extract all non‑zero objects and paint them back unchanged.
    G   = compress(I)
    BG  = replace(G, 0, 1)                     # background → 1
    OBJ = fgpartition(G)                      # set of all coloured objects
    ALL = merge(OBJ)                           # unite the objects into one patch
    O   = paint(BG, ALL)                       # paint the objects onto the new background
    return O


def dsl3(I):
    # Colour‑filtering chain (alternative ordering):
    #   1. Crop the border.
    #   2. Turn interior zeros into colour 1.
    #   3. Swap colours 2 and 3.
    G = compress(I)
    G1 = replace(G, 0, 1)
    O  = switch(G1, 2, 3)
    return O


def dsl4(I):
    # Geometric reasoning via mirroring:
    #   1. Mirror the whole picture horizontally.
    #   2. Remove the all‑zero border.
    #   3. Convert the remaining background (0) to colour 1.
    O = replace(compress(hmirror(I)), 0, 1)
    return O


def dsl5(I):
    # Relational / structural:
    #   1. Crop the zero border.
    #   2. Shift every coloured object leftwards until its
    #      leftmost cell touches column 0 (normalisation).
    #   3. Fill the emptied background with colour 1.
    G       = compress(I)
    OBJSET  = fgpartition(G)                                 # objects of colours ≠0
    SHIFTED = apply(lambda o: shift(o, (-leftmost(o), 0)), OBJSET)
    ALL     = merge(SHIFTED)                                 # unite shifted objects
    BG      = replace(G, 0, 1)                               # interior background → 1
    O       = paint(BG, ALL)                                 # paint normalised objects onto BG
    return O
