    O = canvas(0, shape(I))                       # blank canvas of original size
    O = cover(O, core)                            # paste the core at the top‑left corner
    return O


def dsl2(I):
    # OBJECT – keep the largest object, move every other object one step
    #           towards the centre of the largest object.
    objs   = fgpartition(I)                       # all colour‑based objects
    big    = argmax(objs, lambda o: size(o))      # biggest object
    ctr_big = center(big)                         # its centre
    others = difference(objs, initset(big))       # all the remaining objects

    def move_to_big(o):
        d = subtract(ctr_big, center(o))          # direction to align centres
        return move(I, o, d)                      # shift the object by d

    moved  = apply(move_to_big, others)           # shifted smaller objects
    moved_union = merge(moved)                    # single patch containing all of them
    O = canvas(0, shape(I))
    O = cover(O, big)                              # paste the biggest object
    O = cover(O, moved_union)                      # paste the moved ones
    return O


def dsl3(I):
    # COLOR‑FILTER – swap colours 1↔4 and 2↔7, then erase any non‑zero cell
    #                that touches a zero‑cell.
    S1 = replace(I, 1, 10)        # temporary colour for 1
    S2 = replace(S1, 4, 1)
    S3 = replace(S2,10,4)          # 1 ↔ 4 completed

    T1 = replace(S3, 2, 11)        # temporary colour for 2
    T2 = replace(T1, 7, 2)
    O0 = replace(T2,11,7)          # 2 ↔ 7 completed

    zero   = ofcolor(O0, 0)                                 # indices of all zeroes
    adj    = apply(lambda p: shoot(p, UP) |
                         shoot(p, DOWN) |
                         shoot(p, LEFT) |
                         shoot(p, RIGHT), zero)          # orthogonal neighbours of zeros
    to_erase = intersection(adj,
                difference(asindices(O0), zero))              # coloured cells touching a zero
    O = cover(O0, difference(asindices(O0), to_erase))       # erase them
    return O


def dsl4(I):
    # GEOMETRIC – mirror about the main diagonal and pull every colour‑3
    #            cell one step toward the nearest colour‑5 cell.
    M   = dmirror(I)                                 # diagonal mirror
    O0  = combine(I, M)                              # super‑impose original and its mirror

    three = ofcolor(O0, 3)                           # indices of all 3‑cells
    five  = ofcolor(O0, 5)                           # indices of all 5‑cells

    def pull_one(p):
        src = toobject({p}, O0)                     # single‑cell object
        d   = gravitate(src, five)                  # direction to the closest 5
        return move(O0, src, d)                     # move the 3‑cell one step

    moved_three = apply(pull_one, three)              # each 3‑cell after one pull
    moved_union = merge(moved_three)                  # single patch with all moved 3‑cells
    O = cover(canvas(0, shape(I)),
              difference(asindices(O0), three))       # erase the original 3‑cells
    O = cover(O, moved_union)                         # paste the moved ones
    return O


def dsl5(I):
    # RELATIONAL – recolour each non‑zero cell by (original colour +
    #               Manhattan distance to the nearest colour‑5 cell),
    #               capping the added distance at 9.
    five    = ofcolor(I, 5)                                   # all 5‑indices
    nonzero = difference(asindices(I), ofcolor(I, 0))        # all coloured cells

    def recolour(idx):
        d   = manhattan({idx}, five)                         # distance to nearest 5
        cap = branch(greater(d, 9), 9, d)                    # cap at 9
        new = add(index(I, idx), cap)                        # original + capped distance
        return recolor(new, {idx})                           # recolour this cell

    patches = apply(recolour, nonzero)                        # recoloured single‑cell patches
    patches_union = merge(patches)                            # single patch containing them all
    O = canvas(0, shape(I))
    O = cover(O, patches_union)                               # paint all recoloured cells
    return O
