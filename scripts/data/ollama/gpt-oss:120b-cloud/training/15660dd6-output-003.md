def dsl1(I):
    # Global transformation – remove the uniform outer 8‑border with compress,
    # then recolour the remaining cells.
    c      = compress(I)                     # strip outer 8 frame
    c1     = replace(c, 5, 9)               # original 5 → background 9
    c2     = replace(c1, 1, 9)              # 1 → 9
    c3     = replace(c2, 8, 5)               # 8 → 5
    c4     = replace(c3, 3, 8)               # 3 → 8
    O      = replace(c4, 2, 7)               # 2 → 7

def dsl2(I):
    # Object‑extraction with underfill – turn everything that is not part of the
    # four pattern colours into 9, then keep only the minimal rectangle that
    # contains the pattern.
    pat    = ofcolor(I, 1) | ofcolor(I, 2) | ofcolor(I, 3) | ofcolor(I, 8)
    base   = underfill(I, 9, pat)           # background → 9
    bbox   = backdrop(pat)                  # minimal rectangle of the pattern
    sub    = subgrid(bbox, base)            # extract that rectangle
    sub1   = replace(sub, 1, 9)
    sub2   = replace(sub1, 8, 5)
    sub3   = replace(sub2, 3, 8)
    O      = replace(sub3, 2, 7)
    return O


def dsl3(I):
    # Colour‑filtering with explicit fill – compute the rectangle,
    # fill its background cells with 9, then recolour the pattern.
    pat    = ofcolor(I, 1) | ofcolor(I, 2) | ofcolor(I, 3) | ofcolor(I, 8)
    bbox   = backdrop(pat)
    sub    = subgrid(bbox, I)                       # rectangle containing the pattern
    bg     = difference(asindices(sub), pat)        # cells inside the rectangle that are not pattern
    filled = fill(sub, 9, bg)                       # paint those cells 9
    f1     = replace(filled, 1, 9)
    f2     = replace(f1, 8, 5)
    f3     = replace(f2, 3, 8)
    O      = replace(f3, 2, 7)
    return O


def dsl4(I):
    # Relational cropping – obtain min/max row/column of the pattern with
    # argmin/argmax, crop to that box, then recolour.
    pat    = ofcolor(I, 1) | ofcolor(I, 2) | ofcolor(I, 3) | ofcolor(I, 8)
    rows   = {r for (r, _) in pat}
    cols   = {c for (_, c) in pat}
    top    = argmin(rows,   lambda x: x)
    bottom = argmax(rows,   lambda x: x)
    left   = argmin(cols,   lambda x: x)
    right  = argmax(cols,   lambda x: x)
    height = add(subtract(bottom, top), 1)
    width  = add(subtract(right, left), 1)
    sub    = crop(I, (top, left), (height, width))
    s1     = replace(sub, 1, 9)
    s2     = replace(s1, 8, 5)
    s3     = replace(s2, 3, 8)
    O      = replace(s3, 2, 7)
    return O
