def dsl1(I):
    # GLOBAL‑SHAPE + OPPOSITE‑OFFSET
    # decide orientation from the compressed shape, then place the 3× tiling
    # on the side opposite to the (first) full‑row / full‑column object
    COMP   = compress(I)
    HCOND  = greater(width(COMP), height(COMP))

    OBJ    = objects(I, univalued=True, diagonal=False, without_bg=False)
    ROWS   = sfilter(OBJ, lambda o: height(o) == ONE)   # full rows
    COLS   = sfilter(OBJ, lambda o: width(o)  == ONE)   # full columns

    # opposite side offset: (2 - index) * 3   (grid of 3 rows/cols → 9 after tiling)
    YOFF   = branch(positive(size(ROWS)),
                    multiply(subtract(TWO, uppermost(first(ROWS))), THREE), ZERO)
    XOFF   = branch(positive(size(COLS)),
                    multiply(subtract(TWO, leftmost(first(COLS))), THREE), ZERO)

    HROW   = hconcat(hconcat(I, I), I)      # 3 × 9
    VCOL   = vconcat(vconcat(I, I), I)      # 9 × 3
    CANV   = canvas(ZERO, (9, 9))

    HOUT   = move(CANV, asobject(HROW), (YOFF, ZERO))
    VOUT   = move(CANV, asobject(VCOL), (ZERO, XOFF))

    O = branch(HCOND, HOUT, VOUT)
    return O

    HCOND  = equality(height(LONG), ONE)            # row → horizontal tiling

    YBASE  = uppermost(LONG)
    XBASE  = leftmost(LONG)

    XBASE  = leftmost(BB)

    VCOL   = vconcat(vconcat(I, I), I)
