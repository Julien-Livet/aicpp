def dsl1(I):
    # Global transformation – rotate the whole picture 180°.
    O = rot180(I)
    return O


def dsl2(I):
    # Object‑centric tiling – take the first non‑background object and
    # paint it at every location where it occurs in the input.
    objs   = objects(I, univalued=True, diagonal=False, without_bg=True)
    pat    = first(objs)                                   # the pattern object
    places = occurrences(I, pat)                            # all its anchor positions
    base   = canvas(0, shape(I))                            # empty canvas
    O = apply(lambda p: paint(base, shift(pat, p)), places) # paint the pattern everywhere
    return O


def dsl3(I):
    # Color‑filtering – swap the two most frequent non‑zero colors.
    colors = palette(I) - {0}
    col1  = valmax(colors, lambda c: colorcount(I, c))               # most common
    col2  = valmax(colors - {col1}, lambda c: colorcount(I, c))      # second most common
    O = switch(I, col1, col2)
    return O


def dsl4(I):
    # Geometric reasoning – reflect the picture vertically and then horizontally
    # (equivalent to a 180° rotation around the centre of the grid).
    O = vmirror(hmirror(I))
    return O


def dsl5(I):
    # Relational / structural reasoning – for every non‑zero colour move its whole
    # patch to the top‑left corner (normalise each colour patch) and render them
    # together on a blank canvas.
    cols   = palette(I) - {0}
    empty  = canvas(0, shape(I))
    O = apply(
            lambda c: paint(
                    empty,
                    shift(
                        ofcolor(I, c),
                        (-uppermost(ofcolor(I, c)), -leftmost(ofcolor(I, c)))
                ),
            cols)
    return O
