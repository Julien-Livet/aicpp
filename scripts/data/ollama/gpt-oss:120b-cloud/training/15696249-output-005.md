def dsl1(I):
    # OBJECT‑EXTRACTION : detect a full monochrome row → horizontal tiling,
    # otherwise a full monochrome column → vertical tiling.
    OBJ      = objects(I, univalued=True, diagonal=False, without_bg=False)
    FULLROW = sfilter(OBJ,
                      lambda o: both(equality(height(o), ONE),
                                    equality(width(o), width(I))))
    FULLCOL = sfilter(OBJ,
                      lambda o: both(equality(width(o), ONE),
                                    equality(height(o), height(I))))
    HCOND   = positive(size(FULLROW))          # true → horizontal
    HROW    = hconcat(hconcat(I, I), I)         # three copies side‑by‑side
    VCOL    = vconcat(vconcat(I, I), I)         # three copies stacked
    CANV    = canvas(ZERO, (9, 9))
    HOUT    = move(CANV, asobject(HROW), (ZERO, ZERO))
    VOUT    = move(CANV, asobject(VCOL), (ZERO, ZERO))
    O       = branch(HCOND, HOUT, VOUT)
    return O
