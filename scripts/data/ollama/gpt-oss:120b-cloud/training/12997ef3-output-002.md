def dsl1(I):
    # 1️⃣ GLOBAL – just crop away any all‑zero border rows/columns.
    O = trim(I)
    return O


def dsl2(I):
    # 2️⃣ OBJECT‑TEMPLATE – use the shape of colour 1 as a stencil.
    #   For every other colour c, paste that stencil coloured c
    #   at the position of the first pixel of colour c, then trim.
    objs      = fgpartition(I)                                 # all objects
    tmpl_obj  = first(colorfilter(objs, 1))                     # the 1‑object
    tmpl      = normalize(tmpl_obj)                            # stencil at (0,0)

    other     = difference(objs, initset(tmpl_obj))            # everything but the 1‑object

    # recolour the stencil for each remaining object and shift it to the
    # object’s upper‑left corner
    pasted    = apply(
                lambda o: recolor(
                               color(o),
                               shift(tmpl, ulcorner(o))
                           ),
                other)

    # union of all coloured stencil cells (flatten the set of patches)
    all_cells = apply(lambda o: o, pasted)                     # each element already a patch
    all_cells = combine(all_cells, difference(all_cells, all_cells))  # makes a single flat patch

    blank     = canvas(0, shape(I))
    painted   = paint(blank, all_cells)
    O         = trim(painted)
    return O


def dsl3(I):
    # 3️⃣ COLOR‑FILTER – erase colour 1, keep everything else,
    #   then crop to the remaining content.
    no_one = replace(I, 1, 0)
    O      = trim(no_one)
    return O


def dsl4(I):
    # 4️⃣ GEOMETRIC – mirror the whole picture about the vertical axis
    #   and then remove any all‑zero border rows/columns.
    mirrored = vmirror(I)
    O        = trim(mirrored)
    return O


def dsl5(I):
    # 5️⃣ RELATIONAL – move each coloured object so that its centre of mass
    #   coincides with the top‑left corner (0,0); afterwards crop.
    objs    = fgpartition(I)
    shifted = apply(
                lambda o: shift(
                             o,
                             subtract((0, 0), centerofmass(o))   # –center → move to origin
                         ),
                objs)

    blank   = canvas(0, shape(I))
    painted = apply(lambda o: paint(blank, o), shifted)   # paint each shifted object
    O       = first(painted)                              # the fully painted grid
    O       = trim(O)
    return O
