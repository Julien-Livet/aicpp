def dsl1(I):
    # colour‑filtering (mapply variant)
    # 1‑cells that lie inside the bounding box of any colour >1 are recoloured
    # with that colour.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)
    patches = mapply(
                lambda c, b: recolor(c,
                                     intersection(b, ofcolor(I, 1))),
                seeds, boxes)
    O = paint(I, merge(patches))
    return O


def dsl2(I):
    # geometric (extreme‑corner rectangle)
    # a rectangle is built from the upper‑left and lower‑right corners of each
    # colour >1; the part of that rectangle that originally contained a 1 is
    # recoloured with the colour.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                box(connect(
                                        ulcorner(ofcolor(I, c)),
                                        lrcorner(ofcolor(I, c))),
                                ofcolor(I, 1))),
                seeds)
    O = paint(I, merge(patches))
    return O


def dsl3(I):
    # object‑extraction (objects primitive)
    # all connected components of the same colour are merged, their joint box
    # is taken and 1‑cells inside that box are recoloured.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    objs    = objects(I, False, False, True)                     # all objects
    unions  = apply(lambda c: merge(colorfilter(objs, c)), seeds)  # per colour union
    boxes   = apply(lambda o: box(o), unions)
    patches = prapply(
                lambda c, b: recolor(c,
                                     intersection(b, ofcolor(I, 1))),
                seeds, boxes)
    O = paint(I, merge(patches))
    return O


def dsl4(I):
    # relational / structural (erase 1‑cells then paint)
    # the original 1‑cells are removed; for each colour >1 the part of its box
    # that previously held a 1 is painted with that colour.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)
    masks   = apply(lambda b: intersection(b, ofcolor(I, 1)), boxes)
    patches = prapply(lambda c, m: recolor(c, m), seeds, masks)
    base    = cover(I, ofcolor(I, 1))               # grid without any 1‑cell
    O = paint(base, merge(patches))
    return O


def dsl5(I):
    # relational / structural (underpaint + colour‑swap)
    # instead of painting over the whole grid, we first erase the 1‑cells,
    # then we swap the colour >1 with 1 only inside the colour’s box.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)
    masks   = apply(lambda b: intersection(b, ofcolor(I, 1)), boxes)
    swaps   = prapply(lambda c, m: switch(I, 1, c), seeds, masks)   # flip 1↔c inside mask
    base    = cover(I, ofcolor(I, 1))                               # clear original 1‑cells
    O = underpaint(base, merge(swaps))                             # only fill cleared cells
    return O
