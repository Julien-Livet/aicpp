def dsl1(I):
    # OBJECT‑EXTRACTION – build the missing diagonal with a ray (shoot) instead of connect
    ones       = ofcolor(I, 1)
    last_one   = astuple(lowermost(ones), rightmost(ones))
    # whole main diagonal from (0,0) to bottom‑right
    full_diag  = shoot(astuple(0, 0), astuple(1, 1))
    # diagonal up to and including the last 1
    part_up_to = shoot(astuple(0, 0), last_one)
    # cells that have to become 2
    mask       = difference(full_diag, part_up_to)
    O          = fill(I, 2, mask)
    return O


def dsl2(I):
    # COLOR‑FILTERING – recolour the mask to the background colour and under‑paint it
    ones       = ofcolor(I, 1)
    last_one   = astuple(lowermost(ones), rightmost(ones))
    full_diag  = connect(astuple(0, 0), subtract(shape(I), astuple(1, 1)))
    part_up_to = connect(astuple(0, 0), last_one)
    mask_idx   = difference(full_diag, part_up_to)                 # set of (i,j)
    # turn the index set into an object coloured 0 (the background colour)
    mask_obj   = recolor(0, mask_idx)
    O          = underpaint(I, mask_obj)                           # paint only where background is 0
    return O


def dsl3(I):
    # GEOMETRIC – fire a ray starting just after the last 1 and paint it
    ones   = ofcolor(I, 1)
    last_one = astuple(lowermost(ones), rightmost(ones))
    start = increment(last_one)                                    # first empty cell after the chain
    line  = shoot(start, astuple(1, 1))                           # ray in direction (1,1) up to the border
    O     = fill(I, 2, line)
    return O


def dsl4(I):
    # RELATIONAL / STRUCTURAL – derive the unit step from the span of the 1‑pixels
    ones       = ofcolor(I, 1)
    first_one  = ulcorner(ones)                                    # upper‑left 1
    last_one   = astuple(lowermost(ones), rightmost(ones))
    delta      = subtract(last_one, first_one)                     # overall displacement
    unit_step  = divide(delta, delta)                              # normalises to (1,1)
    start      = increment(last_one)
    line       = shoot(start, unit_step)                           # continue the diagonal with the unit step
    O          = fill(I, 2, line)
    return O


def dsl5(I):
    # GLOBAL – rotate, draw the missing column, then rotate back
    # after a 90° clockwise rotation the main diagonal becomes a vertical line
    R          = rot90(I)
    ones_R     = ofcolor(R, 1)                                     # 1‑pixels are already on the same vertical line
    last_one_R = astuple(lowermost(ones_R), rightmost(ones_R))
    start_R    = increment(last_one_R)                             # first empty cell below the last 1 in the rotated view
    col_R      = shoot(start_R, astuple(0, 1))                     # vertical ray downwards
    R_filled   = fill(R, 2, col_R)                                 # paint the ray in the rotated grid
    O          = rot270(R_filled)                                  # rotate back to original orientation
