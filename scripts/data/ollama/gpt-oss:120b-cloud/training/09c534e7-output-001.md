def dsl1(I):
    # ---- colour‑filtering (seed expansion) -----------------
    # seeds are all colours > 1
    seeds   = mfilter(palette(I), lambda c: c > 1)

    # for each seed colour build the patch of 1‑cells that lie inside its
    # orthogonal out‑box and recolour that patch with the seed colour
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                outbox(ofcolor(I, c)),
                                ofcolor(I, 1))
                         ),
                seeds)

    # paint all recoloured patches on the original grid
    O = paint(I, combine(patches))
    return O


def dsl2(I):
    # ---- global‑mirroring + seed expansion -----------------
    seeds = mfilter(palette(I), lambda c: c > 1)

    # mirror the whole picture vertically and work on the doubled grid
    G = vconcat(I, vmirror(I))

    # expand each seed inside the doubled grid (same rule as dsl1)
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                outbox(ofcolor(G, c)),
                                ofcolor(G, 1))
                         ),
                seeds)

    G2 = paint(G, combine(patches))

    # keep only the left half, i.e. the size of the original picture
    O = lefthalf(G2)
    return O


def dsl3(I):
    # ---- geometric reasoning – draw a 5×5 square centred on each seed ------------
    seeds = mfilter(palette(I), lambda c: c > 1)

    # for every seed, compute its centre of mass and build a square of side 5
    patches = apply(
                lambda c:
                    recolor(
                        c,
                        box(
                            connect(
                                add(centerofmass(ofcolor(I, c)), (-2, -2)),
                                add(centerofmass(ofcolor(I, c)), ( 2,  2))))
                ,
                seeds)

    O = paint(I, combine(patches))
    return O


def dsl4(I):
    # ---- pixel‑wise colour‑filtering (strongest neighbour) -----------------
    cells = asindices(I)

    # for each cell that is a 1 and touches at least one seed (>1),
    # replace it by the maximal neighbouring seed colour
    patches = apply(
                lambda p:
                    branch(
                        both(

                            greater(

                                        apply(lambda n: index(I, n), neighbors(p)),

                                    lambda v: v > 1),

                            toobject(initset(p), I)),
                        initset()),
                cells)

    O = paint(I, combine(patches))
    return O


def dsl5(I):
    # ---- relational / object‑centric dilation -----------------
    # each non‑background object is processed separately
    objs = fgpartition(I, False, False, True)

    patches = apply(
                lambda obj:
                    let(
                        b   = backdrop(obj),                 # bounding box of the object
                        col = color(obj),                  # its unique colour
                        # 1‑cells that lie in the orthogonal out‑box of the object
                        cand = intersection(
                                    outbox(ofcolor(I, col)),
                                    ofcolor(I, 1)),
                        recolor(col, cand)                 # recolour them with the seed colour
                    ),
                objs)

    O = paint(I, combine(patches))
    return O

1. **Colour‑filtering** – expand seeds into adjacent 1‑cells (object‑centric).
2. **Global transformation** – vertical mirroring, seed expansion, then crop.
