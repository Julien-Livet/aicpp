def dsl2(I):
    # set of all 8‑objects (4‑connected, background excluded)
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    # helper that maps an object to a colour based on its cardinality
    def obj_to_colour(obj):
        sz = size(obj)                     # number of cells in the object
        col = branch(sz == 9, 2,
               branch(sz == 5, 3, 1))      # fallback colour 1
        return fill(I, col, obj)           # recolour this object

    # recolour every object and merge the results
