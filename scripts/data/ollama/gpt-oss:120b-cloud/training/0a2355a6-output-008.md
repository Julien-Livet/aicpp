def dsl1(I):
    # OBJECT‑LEVEL: per component interior →2, edge →1, singletons →3
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    def recol(o):
        interior = intersection(o, shift(o, UP))
        interior = intersection(interior, shift(o, DOWN))
        interior = intersection(interior, shift(o, LEFT))
        interior = intersection(interior, shift(o, RIGHT))
        border = difference(o, interior)
        return branch(
            size(o) == 1,
            recolor(3, o),
            combine(recolor(2, interior), recolor(1, border))
        )

    return paint(canvas(0, shape(I)), merge(apply(recol, objs)))


def dsl2(I):
    # GLOBAL‑RELATIONAL: interior / border of the whole 8‑mask,
    # singles = 8‑cells with no orthogonal 8 neighbour →3
    mask = ofcolor(I, 8)

    interior = intersection(mask, shift(mask, UP))
    interior = intersection(interior, shift(mask, DOWN))
    interior = intersection(interior, shift(mask, LEFT))
    interior = intersection(interior, shift(mask, RIGHT))

    border = difference(mask, interior)

    # cells that have at least one orthogonal 8 neighbour
    neigh = combine(shift(mask, UP),
            combine(shift(mask, DOWN),
                combine(shift(mask, LEFT), shift(mask, RIGHT))))
    singles = difference(mask, neigh)          # isolated 8‑cells

    return paint(
        canvas(0, shape(I)),
        merge([recolor(2, interior), recolor(1, border), recolor(3, singles)])
    )


def dsl3(I):
    # COLOR‑FILTERING + OVERLAY:
    #   8 →1 (base), interior →2, isolated cells →3
    base = replace(I, 8, 1)                     # turn every 8 into 1

    mask = ofcolor(I, 8)

    interior = intersection(mask, shift(mask, UP))
    interior = intersection(interior, shift(mask, DOWN))
    interior = intersection(interior, shift(mask, LEFT))
    interior = intersection(interior, shift(mask, RIGHT))

    # isolated 8‑cells : those not having an orthogonal 8 neighbour
    neigh = combine(shift(mask, UP),
            combine(shift(mask, DOWN),
                combine(shift(mask, LEFT), shift(mask, RIGHT))))
    singles = difference(mask, neigh)

    O1 = paint(base, recolor(2, interior))
    return paint(O1, recolor(3, singles))


def dsl4(I):
    # GEOMETRIC: outline (box) →1, interior →2, singletons →3
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    def recol(o):
        outline = intersection(o, box(o))          # cells on the perimeter
        interior = difference(o, outline)
        return branch(
            size(o) == 1,
            recolor(3, o),
            combine(recolor(2, interior), recolor(1, outline))
        )

    return paint(canvas(0, shape(I)), merge(apply(recol, objs)))


def dsl5(I):
    # RELATIONAL: border = 8‑cells adjacent to a 0‑cell,
    # interior = remaining 8‑cells, isolated 8‑cells →3
    mask = ofcolor(I, 8)          # all 8‑indices
    bg   = ofcolor(I, 0)          # background (0)

    b_up    = intersection(mask, shift(bg, UP))
    b_down  = intersection(mask, shift(bg, DOWN))
    b_left  = intersection(mask, shift(bg, LEFT))
    b_right = intersection(mask, shift(bg, RIGHT))

    border = combine(b_up,
              combine(b_down,
                combine(b_left, b_right)))

    interior = difference(mask, border)

    # isolated cells = border cells that have no orthogonal 8 neighbour
    neigh = combine(shift(mask, UP),
            combine(shift(mask, DOWN),
                combine(shift(mask, LEFT), shift(mask, RIGHT))))
    singles = difference(border, intersection(border, neigh))

    return paint(
        canvas(0, shape(I)),
        merge([recolor(2, interior), recolor(1, border), recolor(3, singles)])
    )
