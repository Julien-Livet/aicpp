def dsl1(I):
    # -------- 1️⃣  Vertical propagation (generic colour) ----------
    # colour 1 cells that have a non‑zero / non‑4 colour below them
    ones   = ofcolor(I, 1)
    fours  = ofcolor(I, 4)

    # colours that are allowed to propagate (everything except 0,1,4)
    prop_cols = mfilter(palette(I), lambda c: c not in {0, 1, 4})

    # for a given colour c, cells of colour 1 that see c somewhere below
    def conv(c):
        below = lambda cell: {(r, cell[1]) for r in range(cell[0] + 1, len(I))}
        return {cell for cell in ones
                if intersection(below(cell), ofcolor(I, c))
                and not intersection(below(cell), fours)}

    # recolour the identified cells with their corresponding colour
    recoloured = apply(lambda c: recolor(c, conv(c)), prop_cols)
    changed    = merge(apply(conv, prop_cols))

    O = paint(I, merge(recoloured))                     # paint the new colours
    O = paint(O, recolor(1, ones - changed))           # keep the remaining 1‑cells
    O = paint(O, recolor(4, fours))                   # keep all 4‑cells
    return O


def dsl2(I):
    # -------- 2️⃣  Object‑copy (lowest coloured block → topmost 1‑block) ----------
    objs   = fgpartition(I)                     # all coloured objects
    # lowest object that contains a colour other than 0,1,4
    src    = argmax(objs, lowermost)

    # shape of the source block
    src_box = backdrop(src)
    h, w   = shape(src_box)

    # highest object that is *only* colour 1 and has the same size
    dst_candidates = {o for o in objs
                     if palette(I) & {1} == {1} and shape(backdrop(o)) == (h, w)}
    dst    = argmin(dst_candidates, uppermost)

    dst_box = backdrop(dst)

    # vector that moves the source block onto the destination position
    shift_vec = (min(r for r, _ in dst_box) - min(r for r, _ in src_box),
                min(c for _, c in dst_box) - min(c for _, c in src_box)

    moved_src = shift(src, shift_vec)           # source object moved to destination

    O = paint(I, moved_src)                    # overlay the moved block
    return O


def dsl3(I):
    # -------- 3️⃣  Strip‑wise colour swap ----------
    # rows that are completely filled with colour 4 act as separators
    full4 = {r for r in range(len(I))
            if all(index(I, (r, c)) == 4 for c in range(len(I[0])))}
    parts = hsplit(I, sorted(full4))

    def swap(part):
        # colour(s) that are not 0,1,4 inside the part
        other = mfilter(palette(part), lambda c: c not in {0, 1, 4})
        # if there is such a colour, take any (they are all the same in the data)
        tgt = next(iter(other), None)
        if tgt is None:
            return part
        ones   = ofcolor(part, 1)
        others = ofcolor(part, tgt)
        part   = recolor(1, others)(part)          # 1 ↔ tgt
        part   = recolor(tgt, ones)(part)
        return part

    O = hconcat(*(apply(swap, [p]) for p in parts))
    return O


def dsl4(I):
    # -------- 4️⃣  Quadrant mirroring ----------
    h, w = shape(I)
    # lower‑right quadrant (contains the propagating colour)
    br = subgrid({(r, c) for r in range(h//2, h) for c in range(w//2, w)}, I)
    # mirror it vertically (top‑bottom flip)
    mir = vmirror(br)

    # shift the mirrored block to the upper‑left corner
    tl_coords = {(r, c) for r in range(h//2) for c in range(w//2)}
    shift_vec = (min(r for r, _ in tl_coords) - min(r for r, _ in backdrop(mir)),
                 min(c for _, c in tl_coords) - min(c for _, c in backdrop(mir)))
    moved = shift(mir, shift_vec)

    # overwrite the 1‑cells with the moved block (keeps the propagated colour)
    O = paint(I, recolor(1, moved))
    return O


def dsl5(I):
    # -------- 5️⃣  Column‑wise adjacency (1 directly above propagating colour) ----------
    ones   = ofcolor(I, 1)
    fours  = ofcolor(I, 4)

    # colours that may propagate (everything except 0,1,4)
    prop_cols = mfilter(palette(I), lambda c: c not in {0, 1, 4})

    # for each propagating colour c, collect the cells that are 1 and sit
    # immediately above a c‑cell without a 4 between them
    def up_conv(c):
        above = lambda cell: {(r, cell[1]) for r in range(0, cell[0])}
        return {cell for cell in ones
                if intersection({(cell[0]-1, cell[1])}, ofcolor(I, c))
                and not intersection(above(cell), fours)}

    recoloured = apply(lambda c: recolor(c, up_conv(c)), prop_cols)
    changed    = merge(apply(up_conv, prop_cols))

    O = paint(I, merge(recoloured))                 # paint the new colours
    O = paint(O, recolor(1, ones - changed))       # keep the remaining 1‑cells
    O = paint(O, recolor(4, fours))                # keep all 4‑cells
