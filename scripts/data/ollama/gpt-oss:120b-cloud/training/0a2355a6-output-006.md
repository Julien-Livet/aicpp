def dsl1(I):
    # OBJECT‑LEVEL: per component interior →2, edge →1, singletons →3
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    def recol(o):
        interior = intersection(o, shift(o, UP))
        interior = intersection(interior, shift(o, DOWN))
        interior = intersection(interior, shift(o, LEFT))
        interior = intersection(interior, shift(o, RIGHT))
        border   = difference(o, interior)
        return branch(
            size(o) == 1,
            recolor(3, o),
            merge({recolor(2, interior), recolor(1, border)})
        )

    return paint(canvas(0, shape(I)), merge(apply(recol, objs)))


def dsl2(I):
    # GLOBAL‑MORPHOLOGY: compute all interior 8‑cells at once,
    # then colour borders 1 and isolated cells 3
    mask      = ofcolor(I, 8)                     # all 8‑indices
    mask_obj  = toobject(mask, I)                # object with colour 8

    interior = intersection(mask_obj, shift(mask_obj, UP))
    interior = intersection(interior, shift(mask_obj, DOWN))
    interior = intersection(interior, shift(mask_obj, LEFT))
    interior = intersection(interior, shift(mask_obj, RIGHT))

    border = difference(mask_obj, interior)

    singletons = mfilter(
        objects(I, univalued=True, diagonal=False, without_bg=True),
        lambda o: size(o) == 1
    )

    return paint(
        canvas(0, shape(I)),
        merge({
            recolor(2, interior),
            recolor(1, border),
            recolor(3, union(singletons))
        })
    )


def dsl3(I):
    # COLOR‑FILTERING + OVERLAY:
