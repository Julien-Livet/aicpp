def dsl5(I):
    # RELATIONAL / STRUCTURAL – extract every coloured object, shift each object
    #   to the upper‑left corner (normalisation) and repaint them on a blank canvas.
    objs      = fgpartition(I)                 # set of all objects (including their colour)
    normed    = apply(normalize, objs)          # move every object to origin
    combined  = merge(normed)                   # union of all normalised patches
    blank     = canvas(0, shape(I))              # empty grid of original size
    O = underpaint(blank, combined)            # paint the normalised objects onto the blank grid
