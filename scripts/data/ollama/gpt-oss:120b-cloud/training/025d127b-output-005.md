def dsl1(I):
    # GLOBAL‑RELATIONAL: move every left‑most cell one step right,
    # except the bottom cells of that column.
    all_cells  = asobject(I)
    left_adj   = intersection(all_cells, shift(all_cells, RIGHT))
    leftmost   = difference(all_cells, left_adj)                     # no neighbour on the left
    has_below  = intersection(leftmost, shift(leftmost, UP))        # cells that have a neighbour below
    bottom     = difference(leftmost, has_below)                     # bottom cells of the column
    movable    = difference(leftmost, bottom)                        # cells to be shifted
    shifted    = shift(movable, RIGHT)
    static     = difference(all_cells, movable)                      # everything else
    empty      = canvas(0, shape(I))
    return paint(paint(empty, shifted), static)


def dsl2(I):
    # OBJECT‑LEVEL: apply the same rule to each connected component.
    objs = fgpartition(I)

    def proc(o):
        col   = color(o)
        inds  = toindices(o)                     # just the coordinates
        patch = initset(inds)

        left_adj  = intersection(patch, shift(patch, RIGHT))
        leftmost  = difference(patch, left_adj)
        has_below = intersection(leftmost, shift(leftmost, UP))
        bottom    = difference(leftmost, has_below)
        movable   = difference(leftmost, bottom)
        shifted   = shift(movable, RIGHT)
        keep      = difference(patch, movable)

        return combine(recolor(col, shifted), recolor(col, keep))

    transformed = apply(proc, objs)               # set of coloured patches
    return underpaint(canvas(0, shape(I)), merge(transformed))


def dsl3(I):
    # COLOR‑FILTERED: treat every non‑zero colour independently.
    colours = difference(palette(I), {ZERO})
    empty   = canvas(0, shape(I))

    def proc(c):
        pat       = ofcolor(I, c)                # indices of colour c
        left_adj  = intersection(pat, shift(pat, RIGHT))
        leftmost  = difference(pat, left_adj)
        has_below = intersection(leftmost, shift(leftmost, UP))
        bottom    = difference(leftmost, has_below)
        movable   = difference(leftmost, bottom)
        shifted   = shift(movable, RIGHT)
        keep      = difference(pat, movable)
        return combine(recolor(c, shifted), recolor(c, keep))

    percol = apply(proc, colours)                # one coloured patch per colour
    return underpaint(empty, merge(percol))


def dsl4(I):
    # GEOMETRIC (OBJECT‑LEVEL): use the leftmost x‑coordinate of each object
    # and a vertical line (shoot) to isolate the column.
    objs = fgpartition(I)

    def left_column(o):
        col   = leftmost(o)                                          # integer column index
        line  = shoot((0, col), DOWN)                               # whole vertical line
        inds  = intersection(toindices(o), initset(line))           # indices of o on that line
        return initset(inds)

    def proc(o):
        col   = color(o)
        left  = left_column(o)
        has_below = intersection(left, shift(left, UP))
        bottom    = difference(left, has_below)
        movable   = difference(left, bottom)
        shifted   = shift(movable, RIGHT)
        keep      = difference(toindices(o), movable)
        return combine(recolor(col, shifted), recolor(col, keep))

    transformed = apply(proc, objs)
    return underpaint(canvas(0, shape(I)), merge(transformed))
