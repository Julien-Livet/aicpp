def dsl1(I):
    # ---------- GLOBAL –  trim + compress + rotate‑90 ----------
    # 1. Cut off any completely empty border rows/cols.
    T = trim(I)
    # 2. Remove any completely empty interior rows/cols.
    C = compress(T)
    # 3. Turn the remaining pattern clockwise.
    O = rot90(C)
    return O


def dsl2(I):
    # ---------- OBJECT – keep the largest colour‑5 object ----------
    # 1. Split the grid into colour‑based objects.
    objs = fgpartition(I)
    # 2. Keep only the objects that have colour 5.
    f5   = colorfilter(objs, 5)
    # 3. From those, select the biggest (most cells).
    big  = argmax(f5, lambda o: size(o))
    # 4. Paint it on a blank canvas of the original size.
    O    = canvas(0, shape(I))
    O    = paint(O, big)
    return O


def dsl3(I):
    # ---------- COLOR‑FILTER + GRAVITY ----------
    # 1. Swap 2↔9 and 3↔8.
    O = replace(I, 2, 9)
    O = replace(O, 9, 2)
    O = replace(O, 3, 8)
    O = replace(O, 8, 3)
    # 2. Move every non‑5 cell one step toward the nearest 5‑cell.
    five   = ofcolor(O, 5)
    all_ix = asindices(O)
    non0   = difference(all_ix, ofcolor(O, 0))          # all coloured cells
    non5   = difference(non0, five)                     # coloured but not 5
    def pull(idx):
        src   = toobject({idx}, O)                     # single‑cell object
        dvec  = gravitate(src, five)                   # direction to nearest 5
        return move(O, src, dvec)                     # shifted object
    moved  = apply(pull, non5)                         # list of shifted objects
    # 3. Overlay the moved objects on a background that contains only the 5‑patch.
    background = cover(O, five)
    O = background
    for g in moved:
        O = cover(O, g)
    return O


def dsl4(I):
    # ---------- GEOMETRIC – pure “gravity” of every non‑5 cell ----------
    five   = ofcolor(I, 5)                                 # patch of 5’s
    # all cells that are coloured and not 5
    non5   = difference(difference(asindices(I), ofcolor(I, 0)), five)
    def pull(idx):
        src  = toobject({idx}, I)
        dvec = gravitate(src, five)
        return move(I, src, dvec)
    moved  = apply(pull, non5)                             # shifted single‑cell objects
    O = cover(I, five)                                     # start from 5‑background
    for g in moved:
        O = cover(O, g)                                    # overlay each moved cell
    return O


def dsl5(I):
    # ---------- RELATIONAL – Manhattan distance to the nearest 5 ----------
    five = ofcolor(I, 5)
    # auxiliary: distance (capped at 9) from an index to the 5‑patch
    def capped_dist(idx):
        d = manhattan({idx}, five)
        return min(d, 9)

    # start from a blank canvas
    O = canvas(0, shape(I))

    # fill every non‑zero cell with its distance; keep zeros as zero
    def recolor_idx(idx):
        return recolor(capped_dist(idx), {idx})

    non0 = difference(asindices(I), ofcolor(I, 0))          # all non‑zero indices
    updates = apply(recolor_idx, non0)                     # list of 1‑cell patches
    for g in updates:
        O = cover(O, g)                                    # overlay distances
    return O
