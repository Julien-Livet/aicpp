def dsl3(I):

    objs8 = colorfilter(objects(I, True, False, True), 8)

def dsl4(I):
    # RE‑COLOR EVERY NON‑ZERO, NON‑8 CELL WITH THE MOST COMMON SUCH COLOR
    non8 = sfilter(palette(I), lambda c: c != 0 and c != 8)
    # most frequent non‑zero, non‑8 colour in the whole grid
    most_common = argmax(non8, lambda c: colorcount(I, c))
    # all cells that have a colour from ‘non8’
    mask = merge(mapply(lambda c: ofcolor(I, c), non8))
    # recolor the whole mask with the dominant colour
    recolored = recolor(most_common, mask)
    O = paint(canvas(0, shape(I)), recolored)
    # finally restore the original 8‑objects
    O = paint(O, merge(colorfilter(objects(I, True, False, True), 8)))
    return O


def dsl5(I):
    # ROWS THAT INTERSECT THE BOUNDING‑BOX OF ANY 8‑OBJECT ARE LEFT INTACT,
    # ALL OTHER ROWS ARE ZEROED.
    zero_grid = canvas(0, shape(I))
    objs8 = colorfilter(objects(I, True, False, True), 8)
    # union of the out‑boxes of all 8‑objects
    outb = merge(mapply(outbox, objs8))
    # one representative cell per row that appears in the union
    rows = { (r, 0) for (r, _) in outb }
    # full horizontal lines for those rows
    lines = mapply(lambda p: shoot(p, (0, 1)), rows)
    O = paint(zero_grid, merge(lines))          # rows that intersect 8‑boxes become blank (zero)
    O = paint(O, merge(objs8))                 # restore the original 8‑objects
