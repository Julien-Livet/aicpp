def dsl1(I):
    # Global crop: keep only the interior that contains any non‑zero cell.
    T = trim(I)                                 # remove outer zero borders
    O = canvas(0, shape(T))                    # new empty canvas of the cropped size
    O = paint(O, asobject(T))                  # paste the trimmed pattern back
    return O


def dsl2(I):
    # Object extraction: keep the biggest component of each non‑zero colour.
    OBJ = objects(I, univalued=False, diagonal=False, without_bg=True)
    BIG = {argmax(colorfilter(OBJ, c), lambda o: size(o))
           for c in palette(I) if c != 0 and size(colorfilter(OBJ, c)) > 0}
    O = canvas(0, shape(I))
    O = apply(lambda o: paint(O, o), BIG)       # paint every selected object
    return O


def dsl3(I):
    # Colour filtering: erase colours 5 and 8.
    BAD = combine(ofcolor(I, 5), ofcolor(I, 8)) # cells to be cleared
    O = fill(I, 0, BAD)                         # replace them by background
    return O


def dsl4(I):
    # Geometric reasoning: mirror by a 90° clockwise rotation, keeping original cells.
    R = rot90(I)                                 # rotated copy
    O = cellwise(I, R, fallback=0)               # original non‑zero cells dominate
    return O


def dsl5(I):
    # Relational: recolour the outline of every colour‑2 region with colour 3.
    BORDER = box(ofcolor(I, 2))                 # cells that touch a 2‑pixel
    O = recolor(3, BORDER)                      # make them colour 3
    O = paint(I, O)                              # overlay on the original grid
    return O
