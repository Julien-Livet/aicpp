def dsl1(I):
    # GLOBAL – choose orientation by which side (rows or columns) is less numerous
    objs   = objects(I, False, False, True)

    rows   = apply(lambda o: center(o)[0], objs)
    cols   = apply(lambda o: center(o)[1], objs)

    top    = minimum(rows)
    bottom = maximum(rows)
    left   = minimum(cols)
    right  = maximum(cols)

    slice_h = add(subtract(bottom, top), ONE)        # height of the tight row‑slice
    slice_w = add(subtract(right, left), ONE)        # width  of the tight col‑slice

    # ---- row‑mode (fill each occupied row, then repeat vertically) ----
    row_slice   = crop(I, (top, ZERO), (slice_h, width(I)))
    row_objs    = objects(row_slice, False, False, True)
    rows_filled = apply(
        lambda o: fill(row_slice,
                       color(o),
                       { (r, c) for (r, c) in asindices(row_slice)
                         if r == center(o)[0] }),
        row_objs)
    row_bg      = { (r, c) for (r, c) in asindices(row_slice)
                   if index(row_slice, (r, c)) == ZERO }
    row_result  = vupscale(combine(rows_filled, row_bg),
                           divide(height(I), slice_h))

    # ---- column‑mode (fill each occupied column, then repeat horizontally) ----
    col_slice   = crop(I, (ZERO, left), (height(I), slice_w))
    col_objs    = objects(col_slice, False, False, True)
    cols_filled = apply(
        lambda o: fill(col_slice,
                       color(o),
                       { (r, c) for (r, c) in asindices(col_slice)
                         if c == center(o)[1] }),
        col_objs)
    col_bg      = { (r, c) for (r, c) in asindices(col_slice)
                   if index(col_slice, (r, c)) == ZERO }
    col_result  = hupscale(combine(cols_filled, col_bg),
                           divide(width(I), slice_w))

    # choose the orientation whose side is smaller
    O = branch(greater(size(rows), size(cols)),         # more distinct rows → row‑mode
               row_result,
               col_result)
    return O
