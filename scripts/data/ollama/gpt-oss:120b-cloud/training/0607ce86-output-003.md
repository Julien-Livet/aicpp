def dsl1(I):
    # Global‑row‑template: find the most common non‑zero, non‑8 row and
    # replace every row that contains any non‑zero and no 8 with that template;
    # all other rows become zero.
    non8_rows = [r for r in I
                 if any(c != 0 for c in r) and not any(c == 8 for c in r)]
    tmpl = mostcommon(non8_rows) if non8_rows else tuple([0] * width(I))
    zero_r = tuple([0] * width(I))
    O = tuple(apply(
            lambda r: tmpl if any(c != 0 for c in r) and not any(c == 8 for c in r)
                     else zero_r,
            I))
    return O


def dsl2(I):
    # Object‑extraction: keep only objects whose colour is 8.
    all_objs = objects(I, True, False, True)
    keep     = colorfilter(all_objs, 8)                      # objects of colour 8
    O = paint(canvas(0, shape(I)), merge(keep))
    return O


def dsl3(I):
    # Colour‑frequency filter: keep colours that occupy > 5 % of the grid.
    total      = size(asindices(I))
    frequent   = sfilter(palette(I),
                         lambda c: colorcount(I, c) * 20 > total)   # > 5 %
    masks      = apply(lambda c: ofcolor(I, c), frequent)          # set of colour‑patches
    mask_union = merge(masks)
    O = paint(canvas(0, shape(I)), mask_union)
    return O


def dsl4(I):
    # Geometric‑mirroring: concatenate the grid with its horizontal mirror
    # and then crop back to the original size.
    mirrored   = hmirror(I)
    concatened  = hconcat(I, mirrored)
    O = crop(concatened, (0, 0), shape(I))
    return O


def dsl5(I):
    # Relational‑adjacency: keep objects that touch at least two other objects.
    objs = objects(I, True, False, True)
