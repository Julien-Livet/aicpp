def dsl1(I):

    O       = downscale(bbox, 2)                # halve each dimension

    O       = downscale(trimmed, 2)              # halve each dimension
    return O

def dsl5(I):

    rotated = rot180(cropped)                                        # turn it upside‑down
    O       = downscale(rotated, 2)                                   # halve each dimension
