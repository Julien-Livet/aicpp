def dsl1(I):
    # OBJECT‑EXTRACTION + BOUNDING‑BOX :
    # find the unique object that fills an entire row or an entire column
    all_idx   = asindices(I)                                   # all grid cells
    objs      = objects(I, True, False, False)                  # each colour region (univalued)
    full_row  = lambda o: both(                                      # same row start‑to‑end
                    equality(uppermost(o), lowermost(o)),
                    both(equality(leftmost(o), leftmost(all_idx)),
                         equality(rightmost(o), rightmost(all_idx))))
    full_col  = lambda o: both(                                      # same column top‑to‑bottom
                    equality(leftmost(o), rightmost(o)),
                    both(equality(uppermost(o), uppermost(all_idx)),
                         equality(lowermost(o), lowermost(all_idx))))
    target    = first(sfilter(objs, lambda o: either(full_row(o), full_col(o))))
    O = canvas(color(target), (1, 1))
    return O


def dsl2(I):
    # FG‑PARTITION + ROW / COLUMN CONSISTENCY :
    # each colour’s patches are examined; the colour whose patch occupies a whole
    # row or a whole column is selected.
    bg        = mostcolor(I)
    parts     = fgpartition(I)                                 # colour → set of objects
    non_bg    = sfilter(parts,
                lambda p: flip(equality(color(p), bg)))          # discard background colour objects
    full_line = lambda o: either(                               # full horizontal line
                     both(equality(uppermost(o), lowermost(o)),
                          equality(size(o), subtract(rightmost(asindices(I)),
                                                leftmost(asindices(I))) ),
                     both(equality(leftmost(o), rightmost(o)),
                          equality(size(o), subtract(lowermost(asindices(I)),
                                                uppermost(asindices(I)))))
    candidate = first(sfilter(non_bg, lambda o: full_line(o)))
    O = canvas(color(candidate), (1, 1))
    return O


def dsl3(I):
    # COLOR‑FILTERING + SIZE MATCHING :
    # build a binary mask for each colour, keep the mask whose size equals the grid width
    # (horizontal full line) or the grid height (vertical full line).
    bg        = mostcolor(I)
    cols      = palette(I)
    w_h       = subtract(rightmost(asindices(I)), leftmost(asindices(I)))   # (0, width‑1)
    h_h       = subtract(lowermost(asindices(I)), uppermost(asindices(I)))  # (height‑1,0)
    width     = add(w_h, (0, 1))[1]      # width
    height    = add(h_h, (1, 0))[0]      # height
    masks     = mapply(lambda c:
                combine({recolor(1, o) for o in objects(I, True, False, False)
                        if equality(color(o), c)}, {}), cols)
    sizes     = {c: size(o) for c, o in zip(cols, masks)}
    target_c  = first(sfilter(cols,
                lambda c: either(equality(sizes[c], width),
                              equality(sizes[c], height))))
    O = canvas(target_c, (1, 1))
    return O
