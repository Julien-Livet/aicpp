def dsl1(I):
    # Global geometric transformation: rotate the whole grid 180°.
    O = rot180(I)
    return O


def dsl2(I):
    # Object‑level transformation: fill the bounding box of every
    # cell that has colour 4 with colour 4 (reinforces the rectangle).
    O = fill(I, 4, backdrop(ofcolor(I, 4)))
    return O


def dsl3(I):
    # Colour‑filtering transformation: replace every background (0) cell
    # with the most common non‑zero colour in the grid.
    most = mostcommon(I)
    O = replace(I, 0, most)
    return O


def dsl4(I):
    # Geometric reasoning: take the first foreground object,
    # normalise its position (move its upper‑left corner to (0,0))
    # and paint it on a blank canvas of the original size.
    obj   = first(fgpartition(I))
    norm  = normalize(obj)
    O     = paint(canvas(0, shape(I)), norm)
    return O


def dsl5(I):
    # Relational/structural reasoning: move all cells that are NOT the
    # most common colour towards the cluster of the most common colour.
    main_col   = mostcommon(I)
    main_patch = ofcolor(I, main_col)
    other_patch = difference(ofcolor(I, main_col), main_patch)
    direction   = gravitate(other_patch, main_patch)
    moved       = move(I, other_patch, direction)
    # paint the main colour patch first, then the moved other cells
    O = paint(paint(canvas(0, shape(I)), main_patch), moved)
    return O
