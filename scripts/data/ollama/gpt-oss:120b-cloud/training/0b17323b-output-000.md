def dsl1(I):
    # 1️⃣  Find the lowest/right‑most 1 (they are on a \ diagonal)
    ones      = ofcolor(I, 1)                       # set of (r,c) where colour 1 occurs
    last_row  = lowermost(ones)                     # row of the lowest 1
    last_col  = rightmost(ones)                     # column of the right‑most 1
    start     = increment(astuple(last_row, last_col))   # first empty cell after the diagonal
    direction = add(DOWN, RIGHT)                    # step (1,1)
    line      = shoot(start, direction)              # all cells from start to the border
    O = fill(I, 2, line)                            # paint the line with colour 2
    return O


def dsl2(I):
    # 2️⃣  Use the bounding‑box of the 1‑pixels and extend its bottom‑right corner
    ones      = ofcolor(I, 1)
    box       = backdrop(ones)                      # full rectangular patch containing the 1‑pixels
    br_row    = lowermost(box)                     # bottom row of the box
    br_col    = rightmost(box)                     # rightmost column of the box
    start     = increment(astuple(br_row, br_col))
    direction = add(DOWN, RIGHT)
    line      = shoot(start, direction)
    O = fill(I, 2, line)
    return O


def dsl3(I):
    # 3️⃣  Build the diagonal by connecting the last 1 to the far corner of the grid
    ones      = ofcolor(I, 1)
    last_row  = lowermost(ones)
    last_col  = rightmost(ones)
    h, w      = shape(I)                           # (height, width)
    corner    = astuple(h - 1, w - 1)               # bottom‑right corner of the grid
    start     = increment(astuple(last_row, last_col))
    # line from the cell after the last 1 to the bottom‑right corner
    line_full = connect(start, corner)
    # keep only the fresh part that is still background (0)
    fresh     = difference(line_full, ofcolor(I, 1))
    O = fill(I, 2, fresh)
    return O


def dsl4(I):
    # 4️⃣  Geometric reasoning: the three 1‑pixels define a line; replicate that line with colour 2
    ones       = ofcolor(I, 1)
    # vector between any two successive 1‑pixels (they are equally spaced on the diagonal)
    pts        = tuple(sorted(ones))                # deterministic order
    v          = subtract(pts[1], pts[0])           # (Δr, Δc) = (step, step)
    last_pt    = pts[-1]
    start      = add(last_pt, v)                   # first cell after the existing line
    line       = shoot(start, v)                  # continue in the same direction to the border
    O = fill(I, 2, line)
    return O


def dsl5(I):
    # 5️⃣  Relational: place a 2 in every cell that is farther from the origin than all 1‑pixels
