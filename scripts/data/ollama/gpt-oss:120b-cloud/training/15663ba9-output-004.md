def dsl1(I):
    # OBJECT‑LEVEL: colour the extreme columns of every non‑background object.
    objs      = fgpartition(I)                                     # all colour‑connected objects
    non_bg    = mfilter(objs, lambda o: color(o) > 0)               # drop the background

    lefts  = apply(lambda o: recolor(4,
                     {(r, c) for (r, c) in toindices(o)
                      if c == leftmost(toindices(o))}),
                    non_bg)

    rights = apply(lambda o: recolor(2,
                     {(r, c) for (r, c) in toindices(o)
                      if c == rightmost(toindices(o))}),
                    non_bg)

    O = I
    for p in rights: O = paint(O, p)      # paint right side first (value 2)
    for p in lefts:  O = paint(O, p)      # then left side (value 4)
    return O


def dsl2(I):
    # BOUNDING‑BOX: use the rectangle that encloses each object and colour its vertical sides.
    objs      = fgpartition(I)
    non_bg    = mfilter(objs, lambda o: color(o) > 0)

    boxes    = apply(lambda o: box(toindices(o)), non_bg)           # outline of each object

    lefts  = apply(lambda b: recolor(4,
                     {(r, c) for (r, c) in b if c == leftmost(b)}),
                    boxes)

    rights = apply(lambda b: recolor(2,
                     {(r, c) for (r, c) in b if c == rightmost(b)}),
                    boxes)

    O = I
    for p in rights: O = paint(O, p)
    for p in lefts:  O = paint(O, p)
    return O


def dsl3(I):
    # SHIFT‑DIFFERENCE: cells that disappear when shifted left/right are the outer columns.
    objs      = fgpartition(I)
    non_bg    = mfilter(objs, lambda o: color(o) > 0)

    lefts  = apply(lambda o: recolor(4,
                     difference(toindices(o), shift(toindices(o), LEFT))),
                    non_bg)

    rights = apply(lambda o: recolor(2,
                     difference(toindices(o), shift(toindices(o), RIGHT))),
                    non_bg)

    O = I
    for p in rights: O = paint(O, p)
    for p in lefts:  O = paint(O, p)
    return O


def dsl4(I):
    # PER‑COLOUR BOX: treat each non‑zero colour as a separate “object”, box it,
    # and colour the vertical sides of that box.
    colours   = {c for c in palette(I) if c > 0}

    masks   = apply(lambda c: ofcolor(I, c), colours)                # cells of a given colour
    boxes   = apply(lambda m: box(m), masks)

    lefts  = apply(lambda b: recolor(4,
                     {(r, c) for (r, c) in b if c == leftmost(b)}),
                    boxes)

    rights = apply(lambda b: recolor(2,
                     {(r, c) for (r, c) in b if c == rightmost(b)}),
                    boxes)

    O = I
    for p in rights: O = paint(O, p)
    for p in lefts:  O = paint(O, p)
    return O


def dsl5(I):
    # GLOBAL ROW‑WISE EXTREMA: work on the union of all coloured cells.
    # Cells that have no neighbour on the left are left‑most,
    # cells that have no neighbour on the right are right‑most.
    colours   = {c for c in palette(I) if c > 0}
    masks     = apply(lambda c: ofcolor(I, c), colours)
    all_cells = apply(lambda m: m, masks)        # identity, just to keep a container

    # union of every colour‑mask
    union = reduce(lambda a, b: combine(a, b), all_cells)   # fold using combine

    left_mask  = difference(union, shift(union, LEFT))
    right_mask = difference(union, shift(union, RIGHT))

    O = paint(paint(I, recolor(2, right_mask)), recolor(4, left_mask))
    return O
