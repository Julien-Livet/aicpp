def dsl1(I):
    # colour‑filtering: expand each colour >1 into the 1‑cells that lie
    # inside its orthogonal bounding box.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                box(ofcolor(I, c)),      # full bounding box of colour c
                                ofcolor(I, 1))           # the 1‑cells
                         ),
                seeds)
    O = paint(I, combine(patches))
    return O


def dsl2(I):
    # global mirroring + same seed‑expansion as dsl1.
    seeds = sfilter(palette(I), lambda c: c > 1)

    # double the picture by mirroring it left‑right
    G = vconcat(I, vmirror(I))

    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                box(ofcolor(G, c)),
                                ofcolor(G, 1))
                         ),
                seeds)

    G2 = paint(G, combine(patches))

    # keep only the part that corresponds to the original size
    O = lefthalf(G2)
    return O


def dsl3(I):
    # geometric reasoning – draw a 5×5 square centred on each seed (>1)
    seeds = sfilter(palette(I), lambda c: c > 1)

    patches = apply(
                lambda c: recolor(
                            c,
                            box(
                                connect(
                                    add(centerofmass(ofcolor(I, c)), (-2, -2)),
                                    add(centerofmass(ofcolor(I, c)), ( 2,  2))))
                ,
                seeds)

    O = paint(I, combine(patches))
    return O


def dsl4(I):
    # pixel‑wise strongest‑neighbour replacement.
    # keep only those cells that are 1 and touch at least one colour >1,
    # then recolour them with the maximal neighbouring seed colour.
    cand_cells = sfilter(
                    asindices(I),
                    lambda p: both(
                                 equality(index(I, p), 1),
                                 greater(
                                      size(
                                          sfilter(
                                              apply(lambda n: index(I, n), neighbors(p)),
                                              lambda v: v > 1)),
                                      0)))

    patches = apply(
                lambda p: recolor(
                            valmax(
                                sfilter(
                                    apply(lambda n: index(I, n), neighbors(p)),
                                    lambda v: v > 1),
                                lambda v: v),
                            toobject(initset(p), I)),
                cand_cells)

    O = paint(I, combine(patches))
    return O


def dsl5(I):
    # relational / object‑centric dilation.
    # for every non‑background object, recolour the 1‑cells that lie
    # inside its orthogonal out‑box with the object’s colour.
    objs = fgpartition(I, False, False, True)

    patches = apply(
                lambda obj: recolor(
                            color(obj),
                            intersection(
                                outbox(obj),          # orthogonal bounding box of the object
                                ofcolor(I, 1))),      # the 1‑cells of the whole picture
                objs)

    O = paint(I, combine(patches))
    return O
