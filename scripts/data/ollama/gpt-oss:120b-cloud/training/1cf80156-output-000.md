# ----------------------------------------------------------------------
# 1️⃣  GLOBAL TRANSFORMATION – trim empty border and compress
# ----------------------------------------------------------------------
def dsl1(I):
    # remove any fully‑zero rows/columns that surround the picture
    O = trim(I)                 # strip outer empty rows/cols
    O = compress(O)              # collapse any remaining empty fringe
    return O


# ----------------------------------------------------------------------
# 2️⃣  OBJECT‑LEVEL EXTRACTION – crop to the bounding box of the
#      most frequent non‑zero colour
# ----------------------------------------------------------------------
def dsl2(I):
    C = mostcolor(I)                          # colour that appears most often (ignores 0)
    OBJ = ofcolor(I, C)                        # all cells of that colour
    TOP    = uppermost(OBJ)                     # top row of the object
    LEFT   = leftmost(OBJ)                     # left column of the object
    BBOX   = backdrop(OBJ)                     # rectangle that encloses the object
    H      = height(BBOX)                      # height of the rectangle
    W      = width(BBOX)                       # width of the rectangle
    O      = crop(I, (TOP, LEFT), (H, W))      # slice out the minimal rectangle
    return O


# ----------------------------------------------------------------------
# 3️⃣  COLOR FILTERING – keep only the most common non‑zero colour,
#      replace everything else with 0
# ----------------------------------------------------------------------
def dsl3(I):
    C   = mostcolor(I)                         # dominant non‑background colour
    OBJ = ofcolor(I, C)                        # cells of that colour
    O   = fill(canvas(0, shape(I)), 0, OBJ)    # start from a zero‑filled canvas,
                                            # then paint the kept cells on top
    return O


# ----------------------------------------------------------------------
# 4️⃣  GEOMETRIC REASONING – mirror the picture horizontally and
#      overlay the mirror onto the original
# ----------------------------------------------------------------------
def dsl4(I):
    MIR = hmirror(I)                           # reflect across the vertical axis
    O   = combine(I, MIR)                      # union of the two grids (pixelwise OR)
    return O


# ----------------------------------------------------------------------
# 5️⃣  RELATIONAL / STRUCTURAL – move the dominant object so that its
#      upper‑left corner sits at the origin (0,0)
# ----------------------------------------------------------------------
def dsl5(I):
    C      = mostcolor(I)                      # dominant colour
    OBJ    = ofcolor(I, C)                     # its cells
    TOP    = uppermost(OBJ)
    LEFT   = leftmost(OBJ)
    SHIFT  = (-TOP, -LEFT)                     # vector that brings the object to (0,0)
    MOVED  = shift(OBJ, SHIFT)                 # translate the object
    BG     = canvas(0, shape(I))               # empty background of the same size
    O      = paint(BG, MOVED)                  # place the shifted object onto the background
    return O
