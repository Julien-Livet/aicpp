```python
def dsl1(I):
    # ------------ Global mirroring tiling ------------
    # TL = original, TR = left‑right mirror, BL = up‑down mirror,
    # BR = 180° rotation (both mirrors).  Assemble a 2×2 block.
    TL = I
    TR = vmirror(I)
    BL = hmirror(I)
    BR = rot180(I)
    top    = hconcat(TL, TR)
    bottom = hconcat(BL, BR)
    O = vconcat(top, bottom)
    return O
```

```python
def dsl2(I):
    # ------------ Object‑wise reconstruction ------------
    # Split the grid into its four quadrants, transform each
    # quadrant separately, then stitch them back together.
    rows, cols = shape(I)
    half_r = halve(rows)
    half_c = halve(cols)

    UL = crop(I, (0, 0), (half_r, half_c))
    UR = crop(I, (0, half_c), (half_r, half_c))
    LL = crop(I, (half_r, 0), (half_r, half_c))
    LR = crop(I, (half_r, half_c), (half_r, half_c))

    # Transform each quadrant with a different symmetry
    UL_t = UL                     # keep as is
    UR_t = vmirror(UR)           # left‑right mirror
    LL_t = hmirror(LL)           # up‑down mirror
    LR_t = rot180(LR)            # rotate 180°

    top    = hconcat(UL_t, UR_t)
    bottom = hconcat(LL_t, LR_t)
    O = vconcat(top, bottom)
    return O
```

```python
def dsl3(I):
    # ------------ Color‑filter tiling ------------
    # Create a separate mask for each colour, mirror the mask,
    # then recombine the colour layers.
    colors = palette(I)

    def layer(col):
        mask = ofcolor(I, col)                     # positions of this colour
        filled = canvas(col, shape(I))             # solid colour board
        layer_grid = paint(filled, toobject(mask, I))  # colour only on the mask
        # tile the layer with mirroring
        mir = vmirror(layer_grid)
        top = hconcat(layer_grid, mir)
        bottom = hconcat(hmirror(layer_grid), rot180(layer_grid))
        return vconcat(top, bottom)

    # build all layers and merge them
    layered = apply(layer, colors)                # tuple of tiled colour‑layers
    O = merge(layered)                            # union of all layers
    return O
```
