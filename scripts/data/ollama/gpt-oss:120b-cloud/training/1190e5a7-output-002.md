def dsl1(I):
    # ── Global periodic tiling ──
    #   The dominant colour repeats with a vertical and a horizontal period.
    #   Compute those periods and crop the top‑left tile.
    dom   = mostcolor(I)                                 # dominant colour
    mask  = ofcolor(I, dom)                              # its cells (indices)
    obj   = toobject(mask, I)                           # coloured object
    vsize = vperiod(obj)                                 # vertical period
    hsize = hperiod(obj)                                 # horizontal period
    O     = crop(I, (0, 0), (vsize, hsize))              # extract the tile
    return O


def dsl2(I):
    # ── Object‑extraction based tiling ──
    #   Get the connected components of the dominant colour, keep one of them,
    #   read its periods and crop the original grid accordingly.
    dom   = mostcolor(I)
    comps = objects(I, True, False, True)                # all uni‑coloured objects
    dom_c = colorfilter(comps, dom)                      # only dominant‑colour objects
    comp  = first(dom_c)                                 # any component (they are all alike)
    vsize = vperiod(comp)                                # vertical period of the component
    hsize = hperiod(comp)                                # horizontal period of the component
    O     = crop(I, (0, 0), (vsize, hsize))
    return O


def dsl3(I):
    # ── Colour‑filter → periodic tile ──
    #   Erase everything except the dominant colour, then use its periods.
    dom   = mostcolor(I)
    bg    = difference(asindices(I), ofcolor(I, dom))   # all non‑dominant cells
    F     = fill(I, 0, bg)                               # keep only the dominant colour
    mask  = ofcolor(F, dom)                              # indices of the kept colour
    obj   = toobject(mask, F)
    vsize = vperiod(obj)
    hsize = hperiod(obj)
    O     = crop(F, (0, 0), (vsize, hsize))
    return O


def dsl4(I):
    # ── Geometric core extraction ──
    #   Keep only cells that are unchanged by both horizontal and vertical mirrors,
    #   then tile‑extract the remaining dominant pattern.
    H     = hmirror(I)                                   # horizontal mirror
    V     = vmirror(I)                                   # vertical mirror
    core  = cellwise(I, H, 0)                            # equal to horizontal mirror
    core  = cellwise(core, V, 0)                        # also equal to vertical mirror
    dom   = mostcolor(core)                             # dominant colour inside the core
    mask  = ofcolor(core, dom)
    obj   = toobject(mask, core)
    vsize = vperiod(obj)
    hsize = hperiod(obj)
    O     = crop(core, (0, 0), (vsize, hsize))
    return O


def dsl5(I):
    # ── Relational / structural tiling ──
    #   From the set of dominant‑colour cells obtain the coloured object,
    #   read its vertical and horizontal periods and crop the original grid.

    obj   = toobject(mask, I)                           # coloured object
    vsize = vperiod(obj)                                 # vertical repetition length
    hsize = hperiod(obj)                                 # horizontal repetition length
    O     = crop(I, (0, 0), (vsize, hsize))              # top‑left tile
