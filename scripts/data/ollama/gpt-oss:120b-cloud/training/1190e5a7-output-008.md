def dsl1(I):
    # Global: periodicity of the non‑dominant colour mask
    dom      = mostcolor(I)
    non_mask = difference(asindices(I), ofcolor(I, dom))
    obj      = toobject(non_mask, I)
    vsize    = vperiod(obj)

def dsl2(I):
    # Object‑extraction: bounding box of the largest non‑dominant object
    dom   = mostcolor(I)
    objs  = objects(I, False, False, True)                         # all objects (no background)
    ndobs = mfilter(objs, lambda o: color(o) != dom)               # keep only non‑dominant ones
    big   = argmax(ndobs, size)                                    # largest by size
    bbox  = box(toindices(big))                                     # (top‑left, bottom‑right)
    vsize = add(subtract(bbox[1], bbox[0])[0], ONE)
    hsize = add(subtract(bbox[1], bbox[0])[1], ONE)
    O     = canvas(dom, (vsize, hsize))
    return O


def dsl3(I):

    non_mask = difference(asindices(I), ofcolor(I, dom))
