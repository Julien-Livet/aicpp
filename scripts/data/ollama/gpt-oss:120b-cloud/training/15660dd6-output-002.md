def dsl1(I):
    # ── Object‑extraction + explicit background fill.
    #   Take the rectangle that contains every cell of the four “pattern” colours
    #   (1,2,3,8).  All cells of that rectangle which are *not* part of the pattern
    #   become 9.  Finally recolour the four pattern colours as required.
    pat   = ofcolor(I, 1) | ofcolor(I, 2) | ofcolor(I, 3) | ofcolor(I, 8)
    bbox  = backdrop(pat)                                 # minimal rectangle covering the pattern
    sub   = subgrid(bbox, I)                              # extract that rectangle
    bg    = difference(asindices(sub), pat)               # cells of the rectangle that are not pattern
    filled = fill(sub, 9, bg)                            # paint them 9
    step1 = replace(filled, 1, 9)
    step2 = replace(step1,   8, 5)
    step3 = replace(step2,   3, 8)
    O     = replace(step3,   2, 7)
    return O
