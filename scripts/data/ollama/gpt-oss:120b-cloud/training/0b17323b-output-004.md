def dsl1(I):
    # GLOBAL‑TRANSFORMATION:
    #   Crop the lower‑right region that starts right after the last 1,
    #   draw its main diagonal, shift it back and paint with 2.
    ones      = ofcolor(I, 1)
    last_one  = astuple(lowermost(ones), rightmost(ones))
    start     = increment(last_one)                                    # first empty cell after the line of 1's
    h, w      = shape(I)
    end       = subtract(astuple(h, w), astuple(1, 1))                 # bottom‑right corner of the whole grid
    dims      = add(subtract(end, start), astuple(1, 1))              # size of the sub‑grid to crop
    subgrid   = crop(I, start, dims)                                   # lower‑right part of the picture
    diag      = connect(astuple(0, 0), subtract(dims, astuple(1, 1)))   # diagonal inside the cropped part
    O         = fill(I, 2, shift(diag, start))                         # place the diagonal back into the original grid
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION:
    #   Take the whole main diagonal of the grid, remove the already‑filled prefix
    #   (up to the last 1) and colour the remaining cells with 2.
    ones       = ofcolor(I, 1)
    last_one   = astuple(lowermost(ones), rightmost(ones))
    prefix     = connect(astuple(0, 0), last_one)                       # diagonal cells already occupied by 1's
    h, w       = shape(I)
    full_diag  = connect(astuple(0, 0), subtract(astuple(h, w), astuple(1, 1)))  # complete main diagonal
    remaining  = difference(full_diag, prefix)                         # cells of the diagonal that are still empty
    O          = fill(I, 2, remaining)
    return O


def dsl3(I):
    # COLOUR‑FILTERING:
    #   Build the set of diagonal cells after the last 1 and use underfill
    #   (which colours only background cells) to paint them with 2.
    ones       = ofcolor(I, 1)
    last_one   = astuple(lowermost(ones), rightmost(ones))
    prefix     = connect(astuple(0, 0), last_one)                       # occupied prefix of the diagonal
    h, w       = shape(I)
    full_diag  = connect(astuple(0, 0), subtract(astuple(h, w), astuple(1, 1)))
    remaining  = difference(full_diag, prefix)                         # diagonal cells still empty
    O          = underfill(I, 2, remaining)                            # colour only the background cells among “remaining”
    return O


def dsl4(I):
    # RELATIONAL / STRUCTURAL (periodicity):
    #   The horizontal and vertical periods of the 1‑pixels give the step (Δr,Δc)
    #   of the diagonal; continue the line from the cell after the last 1.

    start    = increment(last_one)
    line     = shoot(start, step)                                        # continuation of the diagonal
    O        = fill(I, 2, line)
    return O
