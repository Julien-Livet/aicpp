def dsl1(I):
    # ---------- GLOBAL TRANSFORMATION ----------
    # 1. Remove the original colour (8) completely.
    # 2. Down‑scale the picture by a factor of 2 (integer division of coordinates).
    # 3. Replace every non‑zero cell of the down‑scaled picture by colour 2.
    # This yields a simple size‑reduction that keeps the overall layout.
    G = replace(I, 8, 0)                # erase the 8‑foreground
    S = downscale(G, 2)                  # integer down‑scale (keeps shape)
    O = replace(S, 0, 2)                # colour the remaining pattern
    return O


def dsl2(I):
    # ---------- OBJECT‑LEVEL BORDER / INTERIOR / ISOLATED ----------
    # 1. Detect every 8‑object (4‑connected, background excluded).
    # 2. For each object decide:
    #        – touches the outer border          → colour 1
    #        – has at least one interior cell    → colour 2
    #        – otherwise (no interior cell)      → colour 3
    # 3. Paint the coloured objects on a clean canvas.
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    def colour_obj(patch):
        # does the patch touch the outer grid border ?
        is_border = bordering(patch, I)

        # four shifted versions of the patch
        up    = shift(patch, UP)
        down  = shift(patch, DOWN)
        left  = shift(patch, LEFT)
        right = shift(patch, RIGHT)

        # cells that have all four orthogonal neighbours inside the patch
        interior = intersection(patch,
                               intersection(up,
                                           intersection(down,
                                                       intersection(left, right))))
        has_interior = size(interior) > 0

        col = branch(is_border, 1,
               branch(has_interior, 2, 3))

        # recolour the patch to the selected colour
        return recolor(col, patch)          # returns an object (col, loc) set

    coloured_objs = apply(colour_obj, objs)   # set of coloured objects
    canvas = replace(I, 8, 0)                # background without the original colour
    O = paint(canvas, combine(coloured_objs))
    return O

        return recolor(col, comp)

    recoloured = apply(recolour, comps)
    O = paint(canvas(0, shape(I)), combine(recoloured))
    return O


def dsl4(I):
    # ---------- GEOMETRIC SYMMETRY REUSE ----------
    # 1. Rotate the whole picture 90° clockwise.
    # 2. Overlay the rotated non‑zero cells on top of the original picture
    #    (rotated cells have priority).
    # 3. Replace any remaining 8 with colour 1.
    R = rot90(I)                                   # rotated grid
    rot_objs = objects(R, univalued=True, diagonal=False, without_bg=True)
    O = paint(I, combine(rot_objs))               # overlay rotated objects
    O = replace(O, 8, 1)                           # leftover 8 → 1
    return O


def dsl5(I):
    # ---------- RELATIONAL / STRUCTURAL REASONING ----------
    # 1. Extract every 8‑object.
    # 2. For each object compute its centre of mass.
    # 3. Two objects are “touching” if the Manhattan distance
    #    between their centres of mass equals 1.
    # 4. Objects that touch at least one other become colour 3,
    #    otherwise colour 1.
    # 5. Paint the coloured objects onto a zero background.
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    # map each object to its centre of mass (a point)
    centres = apply(lambda o: centerofmass(o), objs)

    # build a set of objects that have at least one neighbour at distance 1
    touching = initset(frozenset())
    # we cannot write explicit loops; instead we build all pairs,
    # filter those at distance 1 and collect the first component of each pair.
    pairs = product(objs, objs)                     # all ordered pairs
    close  = mfilter(pairs,
                     lambda p: p[0] != p[1] and
                               manhattan(centerofmass(p[0]),
                                         centerofmass(p[1])) == 1)
    touching = apply(lambda p: p[0], close)        # objects that appear in a close pair
    touching_set = combine(touching)               # unified set of objects

    def recolour(o):
        col = branch(contains(touching_set, o), 3, 1)
        return recolor(col, o)

    recoloured = apply(recolour, objs)
    O = paint(canvas(0, shape(I)), combine(recoloured))
    return O
