def dsl1(I):
    # GLOBAL – rotate so the diagonal becomes a column,
    # extend the column one step to the right and fill downward.
    R         = rot90(I)                                 # diagonal → column (plus offset)
    ones_R    = ofcolor(R, 1)                             # column of 1‑pixels
    last_R    = astuple(lowermost(ones_R), rightmost(ones_R))
    start_R   = increment(last_R)                         # first empty cell after the last 1
    col_R     = shoot(start_R, astuple(0, 1))              # vertical ray in the same column
    R2        = fill(R, 2, col_R)                         # paint the ray with colour 2
    O         = rot270(R2)                                # restore original orientation
    return O


def dsl2(I):

    # and continue the diagonal with that step, painting colour 2.

    step_vec   = subtract(second_one, first_one)           # (dx,dy) = (gap,gap)
    last_one   = astuple(lowermost(ones), rightmost(ones))

    line       = shoot(start, step_vec)                    # continue the diagonal with the same step
    O          = fill(I, 2, line)                         # paint the new cells with 2
    return O


def dsl3(I):
