def dsl1(I):
    # GLOBAL – translate all non‑zero cells one step to the right (clipped at border)
    all_cells = asobject(I)
    moved     = shift(all_cells, RIGHT)
    empty     = canvas(0, shape(I))
    return paint(empty, moved)


def dsl2(I):
    # OBJECT – for each connected component shift its left‑most column cells right
    comps = fgpartition(I)

    def proc(comp):
        left_adj   = intersection(comp, shift(comp, LEFT))
        leftmost_c = difference(comp, left_adj)          # cells with no neighbour on the left (column‑wise)
        moved      = shift(leftmost_c, RIGHT)
        keep       = difference(comp, leftmost_c)
        return combine(moved, keep)

    transformed = apply(proc, comps)
    return underpaint(canvas(0, shape(I)), merge(transformed))


def dsl3(I):
    # COLOR – for each colour move cells that have an empty neighbour on the right
    colours = difference(palette(I), {ZERO})

    def proc(col):
        cells   = ofcolor(I, col)
        blocked = intersection(cells, shift(cells, RIGHT))      # cells that already have a same‑col neighbour to the right
        movable = difference(cells, blocked)                   # those whose right neighbour is empty
        moved   = shift(movable, RIGHT)
        stay    = difference(cells, movable)
        return combine(recolor(col, moved), recolor(col, stay))

    percol = apply(proc, colours)
    return underpaint(canvas(0, shape(I)), merge(percol))


def dsl4(I):
    # GEOMETRIC – cells that disappear after a horizontal mirror are the left edge;
    # shift those cells right
    mirrored = hmirror(I)
    diff_obj = difference(asobject(I), asobject(mirrored))     # cells present only in the original (left‑most side)
    left_adj = intersection(diff_obj, shift(diff_obj, LEFT))
    leftmost = difference(diff_obj, left_adj)                # true left‑most cells
    moved    = shift(leftmost, RIGHT)
    stay     = difference(asobject(I), leftmost)
    empty    = canvas(0, shape(I))
    return paint(paint(empty, moved), stay)


def dsl5(I):

    cells        = asobject(I)
    left_adj_row = intersection(cells, shift(cells, LEFT))

    stay         = difference(cells, movable)
