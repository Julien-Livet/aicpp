```python
def dsl1(I):
    # GLOBAL – crop to the tight box, double its size, rotate 90° clockwise,
    # then paste the result in the top‑left corner of a fresh canvas.
    sub   = compress(I)               # minimal rectangle containing all non‑zero cells
    big   = upscale(sub, 2)           # 2× enlargement
    rot   = rot90(big)                # quarter‑clockwise rotation
    base  = canvas(0, shape(I))      # empty canvas of the original dimensions
    O = paint(base, asobject(rot))   # embed at (0,0)
    return O
```

```python
def dsl2(I):
    # OBJECT‑LEVEL – for every colour, extract its exact bounding box,
    # enlarge it 2× and move it to (2·ulcorner) before painting all together.
    cols   = palette(I)                                   # all present colours (excluding 0)
    objs   = apply(
        lambda c: shift(
            asobject(upscale(subgrid(backdrop(ofcolor(I, c)), I), 2)),
            multiply(ulcorner(ofcolor(I, c)), (TWO, TWO))
```
