def dsl1(I):
    # GLOBAL – remove uniform outer margin, detect period on the remaining object,
    # crop the fundamental tile and recolour its background with the tile's dominant colour.
    bg      = mostcolor(I)
    O0      = replace(I, bg, 0)                     # background → 0
    O1      = compress(O0)                          # strip outer 0‑rows/cols
    nb      = difference(asindices(O1), ofcolor(O1, 0))
    obj     = toobject(nb, O1)                      # coloured object for period detection
    hp      = hperiod(obj)                          # horizontal period
    vp      = vperiod(obj)                          # vertical period
    anchor  = first(nb)                              # top‑left non‑bg cell after trim
    tile    = crop(O1, anchor, astuple(vp, hp))     # fundamental tile
    fillcol = mostcolor(tile)                       # dominant colour inside the tile
    O       = replace(tile, 0, fillcol)             # restore background with that colour
    return O


def dsl2(I):
    # OBJECT EXTRACTION – pick the first non‑background object, compute its period,
    # extract the tile anchored at the object's top‑left cell and fill its background.
    bg      = mostcolor(I)
    objs    = objects(I, False, False, True)        # all coloured objects (bg excluded)
    obj     = first(objs)                           # the first such object
    nb      = toindices(obj)                        # its cell indices
    hp      = hperiod(obj)                          # period from the coloured object
    vp      = vperiod(obj)
    anchor  = leftmost(nb), uppermost(nb)           # top‑left corner of the object
    tile    = crop(I, anchor, astuple(vp, hp))
    fillcol = mostcolor(tile)
    O       = replace(tile, bg, fillcol)
    return O


def dsl3(I):
    # COLOR FILTERING – erase the global background, keep only the repeating pattern,
    # then retrieve the tile via its detected period and recolour its former background.
    bg      = mostcolor(I)
    O0      = replace(I, bg, 0)                     # turn background to 0
    nb      = difference(asindices(O0), ofcolor(O0, 0))
    obj     = toobject(nb, O0)
    hp      = hperiod(obj)
    vp      = vperiod(obj)
    anchor  = first(nb)
    tile    = crop(O0, anchor, astuple(vp, hp))
    fillcol = mostcolor(tile)
    O       = replace(tile, 0, fillcol)             # 0 → dominant colour
    return O


def dsl4(I):
    # GEOMETRIC – use the centre of mass of all non‑background cells as a reference,
    # offset it by half the period to obtain the top‑left corner of the tile.
    bg      = mostcolor(I)
    nb      = difference(asindices(I), ofcolor(I, bg))
    obj     = toobject(nb, I)
    hp      = hperiod(obj)
    vp      = vperiod(obj)
    centre  = centerofmass(nb)                      # (row, col) of mass
    tl      = subtract(centre,
                       astuple(divide(vp, TWO), divide(hp, TWO)))  # top‑left corner
    tile    = crop(I, tl, astuple(vp, hp))
    fillcol = mostcolor(tile)
    O       = replace(tile, bg, fillcol)
    return O


def dsl5(I):
    # RELATIONAL – analyse the border of the repeating pattern,
    # use the most common border colour as the fill colour for the tile background.
    bg      = mostcolor(I)
    nb      = difference(asindices(I), ofcolor(I, bg))
    obj     = toobject(nb, I)
    hp      = hperiod(obj)
    vp      = vperiod(obj)
    anchor  = first(nb)
    tile    = crop(I, anchor, astuple(vp, hp))
    borderc = mostcolor(border(obj))                 # dominant colour on the object's border
    O       = replace(tile, bg, borderc)
    return O
