def dsl1(I):
    # Global – smallest repeating tile of the whole picture
    # (periods are computed on the full coloured index set)
    vp = vperiod(asobject(I))                 # vertical period of the coloured cells
    hp = hperiod(asobject(I))                 # horizontal period of the coloured cells
    O  = crop(I, ORIGIN, (vp, hp))           # first tile of that period
    return O


def dsl2(I):
    # Object extraction – keep the tight box around cells that stay unchanged
    # after a horizontal mirror (horizontal symmetry axis)
    mirrored   = hmirror(I)                                   # horizontally mirrored picture
    invariant  = intersection(asobject(I), asobject(mirrored)) # cells identical in both
    O          = subgrid(toindices(invariant), I)              # minimal rectangle enclosing them
    return O


def dsl3(I):

    bg        = mostcolor(I)                                   # most frequent colour (background)

    O         = crop(no_bg, ORIGIN, (vp, hp))                  # first tile of that pattern
    return O
