def dsl1(I):
    # RELATIONAL MOVE (pure mapply) : any non‑7/8 cell that touches a 7 and an 8
    # is moved one step towards the nearest 7 (by gravitate) and the source cell
    # is recoloured 7.
    other   = palette(I) - {7, 8}
    # candidate cells of each colour that have a diagonal 7 neighbour and a diagonal 8 neighbour
    cand    = mapply(
                lambda c: {p for p in ofcolor(I, c)
                           if dneighbors(p) & ofcolor(I, 7)
                           and dneighbors(p) & ofcolor(I, 8)},
                other)

    # compute the direction from each candidate to the (any) adjacent 7
    dirs    = mapply(
                lambda p: gravitate(
                            initset({p}),
                            initset({first(dneighbors(p) & ofcolor(I, 7))})),
                cand)

    # move every candidate simultaneously, then recolour all original positions 7
    O = mapply(lambda p, d: move(I, initset({p}), d), zip(cand, dirs))
    O = recolor(7, cand)
    return O


def dsl2(I):
    # OBJECT‑LEVEL EXCHANGE : for every object whose colour is not 7 or 8,
    # swap its colour with every 7 that lies inside its bounding box.
    objs    = {o for o in objects(I, univalued=False, diagonal=False, without_bg=True)
                if color(o) not in {7, 8}}

    # 7‑cells that belong to the bounding box of each object
    inside7 = mapply(lambda o: intersection(ofcolor(I, 7), box(o)), objs)

    # turn the 7‑cells into the object's original colour
    O = mapply(lambda p, o: fill(I, color(o), toobject(p, I)), zip(inside7, objs))

    # turn the original object cells into 7
    O = mapply(lambda o: fill(O, 7, o), objs)
    return O


def dsl3(I):
    # GLOBAL HORIZONTAL SWAP : a 7 becomes the colour that appears immediately
    # to its right (if that right‑hand neighbour is a non‑7/8 colour);
    # the right‑hand colour becomes 7.
    other   = palette(I) - {7, 8}

    # for each colour c, the set of 7‑cells whose right neighbour is colour c
    to_c    = mapply(
                lambda c: {p for p in ofcolor(I, 7)
                             if add(p, RIGHT) in ofcolor(I, c)},
                other)

    # the original cells of colour c (they will become 7)
    orig_c   = mapply(lambda c: ofcolor(I, c), other)

    O = I
    # turn the selected 7‑cells into the colour they point to
    O = mapply(lambda pair:
                fill(O, pair[0], toobject(pair[1], I)),
                zip(other, to_c))

    # turn the original colour‑c cells into 7
    O = mapply(lambda c: fill(O, 7, toobject(orig_c[c], I)), other)
    return O


def dsl4(I):
    # MIRROR‑SHIFT‑COMBINE + RECOLOUR “sandwiched” 7
    # 1) mirror vertically, shift the mirror one column right and overlay.
    # 2) any 7 that has an 8 on both its left and right (in the original grid)
    #    is recoloured with the least frequent non‑7/8 colour.
    mirrored = vmirror(I)
    shifted  = move(mirrored, asobject(mirrored), (0, 1))
    O = combine(I, shifted)

    filler   = leastcommon(palette(I) - {7, 8})          # new colour
    sand     = {p for p in ofcolor(O, 7)
                  if add(p, LEFT)  in ofcolor(I, 8)
                  and add(p, RIGHT) in ofcolor(I, 8)}
    O = recolor(filler, sand)
    return O


def dsl5(I):
    # VERTICAL PROJECTION SWAP : for every column, the leftmost non‑7/8
    # colour that shares the column with a 7 becomes 7, and the 7 becomes
    # that colour.  This is achieved by pairing each 7 with the nearest
    # non‑7/8 cell in the same column (using gravitate) and swapping.
    other   = palette(I) - {7, 8}
    # cells of colours other than 7/8
    others  = mapply(lambda c: ofcolor(I, c), other)

    # for each such cell, the nearest 7 in the same column (if any)
    pairs   = mapply(
                lambda p: (p,
                           first({q for q in ofcolor(I, 7)
                                   if q[1] == p[1]}))
                if {q for q in ofcolor(I, 7) if q[1] == p[1]} else None,
                others)

    # keep only pairs where a 7 was found
    valid   = {pair for pair in pairs if pair is not None}

    # swap colours by moving the non‑7/8 cell onto the 7 position
    O = I
    O = mapply(lambda pr: move(O, initset({pr[0]}), subtract(pr[1], pr[0])),
               valid)
    # recolour the original 7 positions (now empty) to the colour of the moved cell
    O = mapply(lambda pr: recolor(color(toobject({pr[0]}, I)), {pr[1]}), valid)
    return O
