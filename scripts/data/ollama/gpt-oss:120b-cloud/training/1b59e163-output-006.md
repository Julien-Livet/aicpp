def dsl1(I):
    # Global – rotate 180° and keep the rotated non‑background cells.
    R = rot180(I)                                 # 180° rotation
    O = cellwise(I, R, 4)                         # original cells win, background is 4
    return O


def dsl2(I):
    # Object extraction – move every non‑background pixel to the right half,
    # erasing the left half completely.
    bg      = ofcolor(I, 4)                                   # background cells
    w       = add(1, subtract(rightmost(bg), leftmost(bg)))  # full width
    half    = divide(w, 2)                                    # centre column
    all_idx = asindices(I)                                    # every index
    left    = sfilter(all_idx, lambda p: p[1] < half)          # cells left of centre
    left_o  = toobject(left, I)                               # left‑side coloured object
    O       = fill(I, 4, left)                                # erase left half
    O       = move(O, left_o, (0, half))                       # paste it shifted right
    return O


def dsl3(I):
    # Colour‑statistics – swap the background colour (4) with the most common
    # non‑background colour.
    bg   = 4
    nb   = palette(I) - {bg}                                  # colours ≠ background
    tgt  = argmax(nb, lambda c: colorcount(I, c))             # dominant non‑bg colour
    O    = recolor(bg,  ofcolor(I, tgt))                      # background → dominant
    O    = recolor(tgt, ofcolor(I, bg))                       # dominant  → background
    return O


def dsl4(I):
    # Geometric – vertical mirror overlay, keeping the original pixel whenever it
    # differs from its mirror (fallback colour = background = 4).
    V = vmirror(I)                                            # vertical flip
    O = cellwise(I, V, 4)                                      # original wins, else background
    return O


def dsl5(I):

    # vertically downwards using the intrinsic vertical period.
    objs   = objects(I, False, False, True)                    # all coloured objects
    patch  = {c for obj in objs for c in obj}                 # union of their cells
    p      = vperiod(patch)                                   # basic vertical period
    shifted= shift(patch, (p, 0))                             # copy one period down
    O = fill(I, 4, patch)                                    # erase original pattern
    O = fill(O, mostcolor(I), shifted)                        # paste the replicated copy
    return O
