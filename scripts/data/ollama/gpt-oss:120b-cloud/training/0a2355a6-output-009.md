def dsl1(I):
    # GLOBAL‑RELATIONAL: classify each 8‑cell by its orthogonal neighbourhood count.
    mask      = ofcolor(I, 8)

    up        = shift(mask, UP)
    down      = shift(mask, DOWN)
    left      = shift(mask, LEFT)
    right     = shift(mask, RIGHT)

    # cells that have all four orthogonal neighbours → interior (2)
    interior  = intersection(mask, up)
    interior  = intersection(interior, down)
    interior  = intersection(interior, left)
    interior  = intersection(interior, right)

    # cells with no orthogonal neighbour → isolated (3)
    neighbours = combine(up,
                    combine(down,
                        combine(left, right)))
    singles   = difference(mask, neighbours)

    # remaining 8‑cells are border (1)
    border    = difference(mask, combine(interior, singles))

    return paint(
        canvas(0, shape(I)),
        merge([recolor(2, interior), recolor(1, border), recolor(3, singles)])
    )


def dsl2(I):
    # OBJECT‑LEVEL (BOX‑OUTLINE): outline of each component → 1,
    # interior (inside the outline) → 2, isolated cells → 3.
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    def recol(o):
        # outline = cells of the object that lie on its minimal bounding box
        outline   = intersection(o, box(o))
        interior  = difference(o, outline)

        # isolated cells: those with no orthogonal neighbour inside the same object
        neigh = combine(shift(o, UP),
                combine(shift(o, DOWN),
                    combine(shift(o, LEFT), shift(o, RIGHT))))
        singles = difference(o, neigh)

        # border = outline minus isolated cells
        border   = difference(outline, singles)

        return merge([
            recolor(2, interior),
            recolor(1, border),
            recolor(3, singles)
        ])

    return paint(canvas(0, shape(I)), merge(apply(recol, objs)))

def dsl5(I):
    # RELATIONAL‑CENTER: for each connected component,
    # cells that are not on the object's outermost rows/columns are interior (2);
    # outermost cells are border (1); isolated cells (3).
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    def recol(o):
        # rows and columns occupied by the object
        top    = uppermost(o)
        bottom = lowermost(o)
        leftc  = leftmost(o)
        rightc = rightmost(o)

        # interior = cells whose row is strictly between top/bottom
        #            and whose column is strictly between left/right
        rows   = interval(top + 1, bottom, 1)   # rows not on edge
        cols   = interval(leftc + 1, rightc, 1)  # cols not on edge
        interior = ofcolor(I, 8)                # start from all 8‑cells
        interior = intersect(interior,
                     initset((r, c)) for r in rows for c in cols)  # keep only interior rows/cols

        # isolated cells: no orthogonal neighbour inside the same object
        neigh = combine(shift(o, UP),
                 combine(shift(o, DOWN),
                     combine(shift(o, LEFT), shift(o, RIGHT))))
        singles = difference(o, neigh)

        border = difference(o, combine(interior, singles))

        return merge([
            recolor(2, interior),
            recolor(1, border),
            recolor(3, singles)
        ])

    return paint(canvas(0, shape(I)), merge(apply(recol, objs)))
