    #   Remove uniform outer borders and then shrink the inner pattern to its
    #   minimal representation.
    O = compress(trim(I))
    return O

def dsl2(I):
    # ── Object extraction ──
    #   Take the largest connected component of the most frequent colour and
    #   return exactly the sub‑grid that contains it.
    main_c   = mostcolor(I)
    comps    = objects(I, True, False, True)               # all uni‑coloured objects
    target   = argmax(comps, lambda o: both(color(o) == main_c,
                                            size(o)))  # keep only those of main colour
    O        = subgrid(target, I)                          # crop to its bounding box
    return O
```

def dsl3(I):
    # ── Colour filtering ──
    #   Keep only the dominant colour, replace everything else with the
    #   background (0) and finally trim the empty margin.
    dom      = mostcolor(I)
    keep     = ofcolor(I, dom)                              # indices of dominant colour
    background = difference(asindices(I), keep)              # all other cells
    O        = fill(I, 0, background)                      # recolour the background to 0
    O        = trim(O)                                       # cut off the uniform border
    return O
```
