def dsl1(I):
    # dominant‑colour periodicity → solid tile
    dom   = mostcolor(I)                         # most frequent colour
    mask  = ofcolor(I, dom)                     # indices of that colour
    obj   = toobject(mask, I)                   # treat the mask as an object
    vsize = vperiod(obj)                        # vertical period of the mask
    hsize = hperiod(obj)                        # horizontal period of the mask
    O     = canvas(dom, (vsize, hsize))         # solid rectangle of the dominant colour
    return O


def dsl2(I):
    # dominant‑colour periodicity → crop original tile
    dom   = mostcolor(I)
    mask  = ofcolor(I, dom)
    obj   = toobject(mask, I)
    vsize = vperiod(obj)
    hsize = hperiod(obj)
    O     = crop(I, (0, 0), (vsize, hsize))     # top‑left elementary tile from the input
    return O


def dsl3(I):
    # colour‑filter → compress → tile size from periods
    dom      = mostcolor(I)
    mask_dom = ofcolor(I, dom)                       # keep only dominant colour
    non_dom  = difference(asindices(I), mask_dom)   # everything else
    cleared  = cover(I, non_dom)                     # turn non‑dominant cells to 0
    comp     = compress(cleared)                      # collapse identical rows / columns
    obj      = toobject(ofcolor(comp, dom), comp)    # object of remaining dominant cells
    vsize    = vperiod(obj)
    hsize    = hperiod(obj)
    O        = canvas(dom, (vsize, hsize))
    return O


def dsl4(I):
    # dominant‑colour mask → bounding box of its first occurrence → solid tile
    dom       = mostcolor(I)
    mask      = ofcolor(I, dom)
    obj       = toobject(mask, I)
    occs      = occurrences(I, obj)                     # all origins of the mask
    first_pos = first(occs)                            # top‑left corner of the first block
    # distance to the next occurrence (period) – if there is only one occurrence we fall back to the whole size
    rest      = difference(occs, initset(first_pos))
    period    = branch(not equal(rest, initset()), 
                      lambda: add(first(rest), (-first_pos[0], -first_pos[1])),
                      lambda: shape(I))
    vsize     = period[0]                               # vertical size of the tile
    hsize     = period[1]                               # horizontal size of the tile
    O         = canvas(dom, (vsize, hsize))
    return O


def dsl5(I):
    # most common object (any colour) → its bounding box → solid tile of its colour
    objs   = objects(I, False, False, True)            # all colour‑agnostic objects
    tile_o = argmax(objs, lambda o: size(o))            # pick the largest (most frequent) object
    col    = mostcolor(paint(canvas(0, shape(I)), tile_o))  # colour of that object
    bbox   = box(toindices(tile_o))                    # bounding box of the object
    O      = canvas(col, (bbox[1][0] - bbox[0][0] + 1,
                         bbox[1][1] - bbox[0][1] + 1))
    return O
