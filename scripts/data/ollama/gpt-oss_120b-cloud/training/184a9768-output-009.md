def dsl1(I):
    # OBJECT‑LEVEL FILTERING: erase all tiny components (size ≤ 2)
    OBJ   = objects(I, univalued=False, diagonal=False, without_bg=True)
    SMALL = sizefilter(OBJ, 2)            # set of very small objects

    O     = fill(I, 0, MASK)               # turn those cells to background
    return O

    MC = mostcolor(I)                                 # dominant colour in the whole grid
    O  = paint(canvas(0, shape(I)), ofcolor(I, MC))    # paint only cells of that colour on a zero canvas

def dsl3(I):

def dsl4(I):

    INNER = delta(BIG)                        # interior cells of the biggest component
    O     = fill(I, color(SMALL), INNER)     # recolour those interior cells
    return O
