def dsl1(I):
    # GLOBAL‑LEVEL: shrink the picture, keep the coarse shape as colour 2,
    # then restore the original size – this reproduces the pattern of 8‑blocks
    # as a solid colour 2 (all 8 → 2) and leaves the background 0.
    G = replace(I, 8, 0)                 # 8 → background
    S = downscale(G, 2)                  # coarse version
    O = replace(S, 0, 2)                 # background of coarse picture → 2
    return O


def dsl2(I):
    # OBJECT‑LEVEL:  three‑way classification
    #   – objects that touch the outer frame become colour 1
    #   – objects that contain at least one interior cell become colour 2
    #   – all other objects become colour 3
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    def recolour(patch):
        is_border = bordering(patch, I)                     # touches outer frame ?
        interior = intersection(
            patch,
            intersection(shift(patch, UP),
                intersection(shift(patch, DOWN),
                    intersection(shift(patch, LEFT),
                                 shift(patch, RIGHT)))))
        has_interior = size(interior) > 0

        col = branch(is_border, 1,
               branch(has_interior, 2, 3))
        return recolor(col, patch)

    coloured = apply(recolour, objs)
    O = paint(canvas(0, shape(I)), merge(coloured))
    return O


def dsl3(I):
    # COLOR‑FILTERING: keep the three size‑groups of the 8‑components
    #   – the largest group (always the interior blocks) → 2
    #   – the medium‑sized group (border blocks) → 1
    #   – the smallest group (isolated single‑pixel blocks) → 3
    comps = objects(I, univalued=True, diagonal=False, without_bg=True)

    sizes = apply(lambda o: size(o), comps)                # size of each component
    max_sz = valmax(sizes, lambda x: x)                    # biggest size
    min_sz = valmin(sizes, lambda x: x)                    # smallest size
    med_sz = valmin(setdifference(sizes, {max_sz, min_sz}),
                    lambda x: x)                           # middle size (if any)

    def recol(o):
        s = size(o)
        col = branch(s == max_sz, 2,
               branch(s == med_sz, 1, 3))
        return recolor(col, o)

    recoloured = apply(recol, comps)
    O = paint(canvas(0, shape(I)), merge(recoloured))
    return O
