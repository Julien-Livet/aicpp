def dsl1(I):
    # GLOBAL ROW‑ZEROING
    # create a blank row of the proper width
    zero_row = repeat(0, width(I))
    # replace every row that contains a non‑zero cell but no 8 by the blank row
    O = apply(

        if any(c != 0 for c in r) and not any(c == 8 for c in r)
        else r,
        I,
    )
    return O


def dsl2(I):
    # OBJECT‑LEVEL LINE PAINTING
    # rows that must be cleared: non‑zero and no 8
    rows_to_blank = sfilter(
        ofcolor(I, 0),
        lambda p: any(I[p[0]][c] != 0 for c in range(width(I)))
    rows_to_blank = sfilter(
        rows_to_blank,
        lambda p: not any(I[p[0]][c] == 8 for c in range(width(I))),
    )
    # for each such row create a full horizontal line (shoot from left border)
    lines = mapply(lambda r: shoot((r[0], 0), (0, 1)), rows_to_blank)
    # paint the lines on a blank canvas, then restore colour‑8 objects

    O = paint(O, merge(colorfilter(objects(I, True, False, True), 8)))
    return O

    O = paint(canvas(0, shape(I)), merge(objs8))
