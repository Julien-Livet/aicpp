def dsl1(I):
    # ---------- Object‑extraction ----------
    # 1. background = most frequent colour
    bg   = mostcolor(I)
    # 2. indices of every cell
    allI = asindices(I)
    # 3. non‑background indices
    nb   = difference(allI, ofcolor(I, bg))
    # 4. make a binary grid: 0 = background, 1 = foreground
    empty = canvas(0, shape(I))
    binG  = paint(empty, recolor(1, nb))
    # 5. connected components of the foreground (colour 1)
    comps = objects(binG, False, False, True)
    # 6. output = number of those components
    O = canvas(size(comps), (1, 1))
    return O


def dsl2(I):
    # ---------- Colour‑filtering ----------
    # background colour
    bg   = mostcolor(I)
    # foreground objects: each colour ≠ background as a separate object
    fgal = colorfilter(objects(I, True, False, False), bg)   # drop background objects
    # unify all foreground objects to a single colour (2) → new grid
    empty = canvas(0, shape(I))
    # paint every foreground object with colour 2
    fore  = mapply(lambda o: recolor(2, o), fgal)           # set of recoloured objects
    binG  = paint(empty, combine(fore, {}))
    # connected components of colour 2
    comps = objects(binG, False, False, True)
    O = canvas(size(comps), (1, 1))
    return O


def dsl3(I):
    # ---------- Geometric reasoning ----------
    # treat the whole picture as a binary mask (foreground = not background)
    bg      = mostcolor(I)
    mask    = difference(asindices(I), ofcolor(I, bg))          # foreground indices
    # compute the convex‑hull‑like bounding box of the mask, then cut it out
    bb      = backdrop(mask)                                   # bounding box of mask
    sub     = crop(I, (ulcorner(bb)), shape(bb))                # sub‑grid containing the mask
    # recolour everything inside the sub‑grid that is not background to colour 1
    sub_bg  = mostcolor(sub)
    sub_all = asindices(sub)
    sub_nb  = difference(sub_all, ofcolor(sub, sub_bg))
    empty   = canvas(0, shape(sub))
    binSub  = paint(empty, recolor(1, sub_nb))
    # count connected components inside the sub‑grid
    comps   = objects(binSub, False, False, True)
    O = canvas(size(comps), (1, 1))
    return O


def dsl4(I):
    # ---------- Relational / structural ----------
    # obtain every non‑background object (colour‑wise)
    bg      = mostcolor(I)
    objs    = objects(I, True, False, False)          # all colour‑wise objects, background included
    # keep only those that are not background
    fgs     = difference(objs, colorfilter(objs, bg))
    # build the adjacency graph of those objects
    adj = mfilter(product(fgs, fgs),
                  lambda p: both(adjacent(p[0], p[1]), flip(equality(p[0], p[1]))))
    # each connected component of the graph corresponds to one foreground region
    # we obtain the components by repeatedly merging adjacent objects
    merged = mapply(lambda o: union(o, adj), fgs)      # union with all neighbours
    # the distinct merged sets are the foreground components
    comps = sizefilter(merged, 1)                    # keep only non‑empty sets
    O = canvas(size(comps), (1, 1))
    return O


def dsl5(I):
    # ---------- Global transformation ----------
    # replace every background colour (most common) by 0,
    # replace every other colour by 1 – a pure binary grid
    bg   = mostcolor(I)
    # first turn background into 0 (already 0 after the next step)
    step = replace(I, bg, 0)
    # turn everything else into 1
    step2 = switch(step, bg, 1)      # swaps bg (0) with 1 → all non‑bg become 1
    # now count orthogonal connected components of colour 1
    comps = objects(step2, False, False, True)
    O = canvas(size(comps), (1, 1))
    return O
