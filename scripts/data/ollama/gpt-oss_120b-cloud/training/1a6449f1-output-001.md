def dsl1(I):
    # GLOBAL: erase the dominant colour then compress away all empty rows/columns
    mc = mostcolor(I)                         # most frequent colour (background)
    O  = compress(replace(I, mc, 0))           # set it to 0 and remove every empty line/column
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION: take the tight sub‑grid that contains every cell
    #                  that is *not* the dominant colour
    mc   = mostcolor(I)
    non  = difference(asindices(I), ofcolor(I, mc))   # all indices except the background
    O    = subgrid(non, I)                           # rectangle around those cells
    return O


def dsl3(I):
    # COLOUR‑FILTERING: erase the rarest colour and trim the surrounding zeros
    lc = leastcolor(I)                       # colour that appears the fewest times
    O  = trim(cover(I, ofcolor(I, lc)))      # delete it, then remove the outer zero border
    return O


def dsl4(I):
    # GEOMETRIC‑REASONING: keep only one repetition tile of the pattern
    #                      (using the vertical and horizontal period of the
    #                      non‑background cells)
    mc   = mostcolor(I)
    pat  = difference(asindices(I), ofcolor(I, mc))   # cells that are not background
    vp   = vperiod(pat)                               # vertical period
    hp   = hperiod(pat)                               # horizontal period
    O    = crop(I, ORIGIN, astuple(vp, hp))            # extract the first tile
    return O


def dsl5(I):
    # RELATIONAL / STRUCTURAL: extract the region consisting of the least‑frequent
    #                          colour together with all orthogonal neighbours
    lc_patch = ofcolor(I, leastcolor(I))                # cells of the rare colour
    neigh   = combine(
                lc_patch,
                shift(lc_patch, DOWN),
                shift(lc_patch, UP),
                shift(lc_patch, LEFT),
                shift(lc_patch, RIGHT)
              )
    O = subgrid(neigh, I)                               # minimal rectangle around them
    return O
