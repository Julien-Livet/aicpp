def dsl1(I):
    # Global – replace the background colour (least frequent) everywhere
    # with the dominant colour (most frequent).
    bg  = leastcolor(I)               # background colour
    dom = mostcolor(I)                # dominant colour
    O   = replace(I, bg, dom)         # swap colours globally
    return O


def dsl2(I):
    # Object‑extraction – keep every object whose colour equals the
    # dominant colour and draw them on a canvas coloured with the least
    # frequent colour.
    bg   = leastcolor(I)                                        # background colour
    dom  = mostcolor(I)                                         # dominant colour
    objs = colorfilter(fgpartition(I), dom)                     # objects of dominant colour
    canvas_grid = canvas(bg, shape(I))                          # empty background canvas
    O = apply(lambda o: paint(canvas_grid, o), objs)           # paint each object
    return last(O)                                             # final grid


def dsl3(I):
    # Colour‑filtering – any cell that is not the dominant colour but
    # touches a dominant‑coloured cell becomes the dominant colour.
    dom = mostcolor(I)                                         # dominant colour
    other = difference(palette(I), initset(dom))                # all non‑dominant colours
    O = I
    O = mapply(
            lambda c:
                fill(O, dom,
                     intersection(ofcolor(O, c),               # cells of colour c
                                  delta(ofcolor(O, dom))       # touching a dom cell
                                 )
                ),
            other
        )
    return O


def dsl4(I):
    # Geometric – intersect the picture with its vertical mirror; keep the
    # common dominant‑coloured cells, everything else becomes background.
    bg   = leastcolor(I)                                           # background colour
    mirr = vmirror(I)                                             # vertical mirror
    common = intersection(
                ofcolor(I, mostcolor(I)),                         # dominant cells in I
                ofcolor(mirr, mostcolor(I))                       # dominant cells in mirror
             )
    O = underfill(I, bg, delta(common))                           # background elsewhere
    O = paint(O, common)                                           # restore intersecting cells
    return O


def dsl5(I):
    # Relational – shift every foreground object one step toward the
    # centre of the grid and redraw them on a clean canvas.
    bg      = leastcolor(I)                                            # background colour
    objs    = fgpartition(I)                                            # all foreground objects
    centre  = initset(astuple(halve(height(I)), halve(width(I))))       # centre cell (as a set)
    moved   = apply(lambda o: move(I, o, gravitate(o, centre)), objs)   # one‑step move toward centre
    canvas_grid = canvas(bg, shape(I))                                 # empty canvas
    O = paint(canvas_grid, combine(*moved))                           # draw all moved objects
    return O
