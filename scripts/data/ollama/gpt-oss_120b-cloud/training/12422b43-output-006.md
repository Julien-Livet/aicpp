def dsl1(I):
    # GLOBAL – crop the rows that contain a 5, clear them and tile them downwards
    anchor      = ofcolor(I, 5)                                          # cells with colour 5
    rows_set    = { r for r,_ in anchor }                                # rows that hold a 5
    rows_patch  = mapply(lambda d: shift(anchor, (0, d)),
                        interval(ZERO, width(I), ONE))                  # whole rows of the block
    cleared     = recolor(0, anchor)                                      # turn the 5s into 0
    I0          = paint(I, cleared)                                      # grid with the block cleared
    pattern_g   = subgrid(rows_patch, I0)                                 # the cleared block as a grid

    # how many full copies fit underneath the existing non‑zero cells ?
    nz          = difference(asindices(I), ofcolor(I, 0))
    max_nz_row  = valmax(nz, lambda p: p[0])
    empty_rows  = subtract(height(I), add(max_nz_row, ONE))
    repeat_cnt  = divide(empty_rows, size(rows_set))

    # build the tiled block as a set of shifted objects
    pat_obj     = asobject(pattern_g)                                     # {(value,(r,c))}
    shifted     = mapply(lambda i: shift(pat_obj,
                                         (multiply(add(i, ONE), size(rows_set)), ZERO)),
                         interval(ZERO, repeat_cnt, ONE))
    tiled_obj   = merge(shifted)

    # keep the original picture (including the leading 5) and overlay the copies
    base        = subgrid(backdrop(nz), I)
    O           = trim(paint(base, tiled_obj))
    return O
