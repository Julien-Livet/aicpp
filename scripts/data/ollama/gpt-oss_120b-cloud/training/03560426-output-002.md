def dsl1(I):
    # GLOBAL – remove every all‑zero row and column (both exterior and interior)
    trimmed   = trim(I)                      # delete outer empty rows / cols
    compact   = compress(trimmed)            # delete any remaining all‑zero rows / cols
    empty     = canvas(0, shape(I))          # full‑size blank canvas
    O = paint(empty, asobject(compact))      # paste the compact pattern at (0,0)
    return O


def dsl2(I):

    #   top‑left corner of its own bounding box, then concatenate the
    #   resulting sub‑grids horizontally (preserving left‑to‑right order)
    objs      = fgpartition(I)                                      # coloured objects
    ordered   = order(objs, lambda o: leftmost(o))                    # left‑most first
    subgs     = apply(lambda o: subgrid(o, I), ordered)            # each object's minimal sub‑grid
    # fold the list of sub‑grids left‑to‑right using hconcat

    empty     = canvas(0, shape(I))
    O = paint(empty, asobject(combined))
    return O
