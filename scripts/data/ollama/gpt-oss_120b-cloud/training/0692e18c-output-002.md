def dsl1(I):
    # Object‑extraction + shift + paint (basic version)
    fg   = argmax(difference(palette(I), initset(0)), lbind(colorcount, I))
    inv  = replace(replace(I, fg, 0), 0, fg)                # inverse pattern
    N    = height(I)
    Pobj = asobject(inv)                                   # (color,idx) set
    cells = ofcolor(I, fg)                                  # foreground cells
    shifted = apply(lambda p: shift(Pobj, (p[0]*N, p[1]*N)), cells)
    O = paint(canvas(0, (N*N, N*N)), merge(shifted))
    return O
