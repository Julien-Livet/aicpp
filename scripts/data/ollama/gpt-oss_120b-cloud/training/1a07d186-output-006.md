    O   = underpaint(canvas(0, shape(I)), OBJ)        # paint it back on a blank canvas
    return O


def dsl2(I):
    # OBJECT EXTRACTION – keep only objects that occupy more than one cell
    OBJS      = fgpartition(I)                        # all coloured objects
    BIG       = sizefilter(OBJS, lambda o: size(o) > 1) # discard single‑pixel objects
    COMBINED  = merge(BIG)                            # union of the remaining objects
    O         = underpaint(canvas(0, shape(I)), COMBINED)
    return O


def dsl3(I):
    # COLOR FILTERING – retain colours that form a line (horizontal or vertical)
    COLORS    = palette(I)                                            # set of colours present
    GOOD      = mfilter(COLORS,
                       lambda c:
                           hmatching(ofcolor(I, c), ofcolor(I, c)) or
                           vmatching(ofcolor(I, c), ofcolor(I, c)))   # colour has a line
    PATCHES   = apply(lambda c: recolor(c, ofcolor(I, c)), GOOD)     # recoloured patches
    COMBINED  = merge(PATCHES)                                        # union of the selected patches
    O         = underpaint(canvas(0, shape(I)), COMBINED)
    return O


def dsl4(I):
    # GEOMETRIC – overlay the original with a 90° clockwise rotation
    ORIG      = asobject(I)                         # original object
    ROT90     = asobject(rot90(I))                   # 90° rotation of the grid
    COMBINED  = merge({ORIG, ROT90})                # union of both objects
    O         = underpaint(canvas(0, shape(I)), COMBINED)
    return O


def dsl5(I):
    # RELATIONAL – align every object horizontally with the leftmost column
    #               of the largest object (by cell count)
    OBJS      = fgpartition(I)                                         # all objects
    LARGEST   = argmax(OBJS, lambda o: size(o))                        # biggest object
    REF_LEFT  = leftmost(toindices(LARGEST))                           # its leftmost column
    SHIFTED   = apply(
                   lambda o:
                       shift(
                           toindices(o),
                           astuple(0, subtract(REF_LEFT,
                                               leftmost(toindices(o)))
                                 )
                       ),
                   OBJS)                                             # move each object
    COMBINED  = merge(SHIFTED)                                          # reunite shifted objects
    O         = underpaint(canvas(0, shape(I)), COMBINED)
    return O
