def dsl1(I):
    # OBJECT‑EXTRACTION – build rectangle objects, merge them and paint.
    BG   = mostcommon(I)                                           # background colour
    CAN  = canvas(BG, shape(I))                                    # empty canvas
    OBJ  = objects(I, False, False, True)                           # coloured objects
    RCT  = apply(lambda o: recolor(backdrop(o), color(o)), OBJ)    # rectangle as coloured object
    ALL  = merge(RCT)                                               # union of all rectangles
    O    = paint(CAN, ALL)                                          # paint onto canvas
    return O


def dsl2(I):
    # COLOR‑FILTERING – work directly with colours, ignore the background.
    BG   = mostcommon(I)                                            # background colour
    CAN  = canvas(BG, shape(I))                                     # empty canvas
    CLS  = difference(palette(I), initset(BG))                       # set of non‑background colours
    RCT  = apply(lambda c: recolor(backdrop(ofcolor(I, c)), CLS)    # rectangle for each colour
    ALL  = merge(RCT)                                               # union of rectangles
    O    = paint(CAN, ALL)                                          # paint onto canvas
    return O


def dsl3(I):
    # UNDERPAINT – use underpaint to colour rectangles only on background cells.
    BG   = mostcommon(I)                                            # background colour
    CAN  = canvas(BG, shape(I))                                     # empty canvas
    OBJ  = objects(I, False, False, True)                           # coloured objects
    RCT  = apply(lambda o: recolor(backdrop(o), color(o)), OBJ)    # rectangle as coloured object
    O    = apply(lambda r: underpaint(CAN, r), RCT)                # underpaint each rectangle
    # `apply` returns a container of grids; the final grid is the last element.
    # Since rectangles never overlap, every element is identical.
    O    = last(O)
    return O
