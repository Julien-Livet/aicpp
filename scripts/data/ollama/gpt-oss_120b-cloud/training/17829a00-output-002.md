def dsl1(I):
    # Global transformation – pixelate the picture.
    # 1) Remove any uniform border (compress).
    # 2) Upscale the result twice in both dimensions (hupscale + vupscale).
    compressed = compress(I)
    upscaled    = vupscale(hupscale(compressed), 2)
    return upscaled


def dsl2(I):
    # Object extraction – keep every object whose colour is the most common,
    # and draw them on a canvas filled with the least‑common colour.
    bg          = leastcolor(I)                                   # background colour
    fg_objs     = fgpartition(I)                                 # all foreground objects
    dom_color   = mostcolor(I)                                    # dominant colour
    dom_objs    = colorfilter(fg_objs, dom_color)                # objects of that colour
    canvas_grid = canvas(bg, shape(I))                            # empty canvas
    O           = paint(canvas_grid, combine(*dom_objs))        # paint the kept objects
    return O


def dsl3(I):

    # dominant‑coloured cell is recoloured to the dominant colour.
    dom        = mostcolor(I)                                    # dominant colour
    others     = difference(palette(I), initset(dom))           # all other colours

    O = mapply(
            lambda c:
                recolor(

                    )
                ),
            others
        )
    return O


def dsl4(I):

def dsl5(I):
