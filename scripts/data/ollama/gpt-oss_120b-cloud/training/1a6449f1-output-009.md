def dsl1(I):
    # Global periodic tile of the whole grid (including background)
    obj = asobject(I)                     # treat every cell as one object
    vp  = vperiod(obj)                    # vertical period of the pattern
    hp  = hperiod(obj)                    # horizontal period of the pattern
    O   = crop(I, ORIGIN, (vp, hp))       # first repeating tile
    return O


def dsl2(I):
    # Periodic tile after removing the uniform outer border
    trimmed = trim(I)                      # cut away the uniform background border
    obj = asobject(trimmed)               # object of the trimmed grid
    vp  = vperiod(obj)                    # vertical period
    hp  = hperiod(obj)                    # horizontal period
    O   = crop(trimmed, ORIGIN, (vp, hp)) # tile of the trimmed picture
    return O


def dsl3(I):
    # Bounding box of the most frequently occurring monochrome component
    comps      = objects(I, univalued=True, diagonal=False, without_bg=False)
    # pick component whose set of occurrences is largest
    target     = argmax(comps, lambda o: size(occurrences(I, o)))
    # take the first occurrence of that component as the representative tile
    occ_first  = first(occurrences(I, target))
    O          = subgrid(toindices(occ_first), I)
    return O


def dsl4(I):
    # Region that is invariant under both horizontal and vertical mirroring
    hm_obj = asobject(hmirror(I))         # horizontal mirror as an object
    vm_obj = asobject(vmirror(I))         # vertical mirror as an object
    invariant = intersection(hm_obj, vm_obj)
    O = subgrid(toindices(invariant), I)  # minimal bounding box of the invariant part
    return O


def dsl5(I):
    # Simple compression: strip away any uniform outer rows / columns
    # (the remaining grid often corresponds to the repeating tile)
    O = compress(I)
    return O
