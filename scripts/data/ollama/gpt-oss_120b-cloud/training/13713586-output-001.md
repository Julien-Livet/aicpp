def dsl1(I):
    # Global – compress away empty border, then upscale uniformly by 2×,
    # finally keep the brighter colour at each cell.
    C = compress(I)                     # remove outer zero‑border
    H = hupscale(C, 2)                  # horizontal upscale 2×
    S = vupscale(H, 2)                  # vertical upscale 2×  →  overall 2× scaling
    O = cellwise(I, S, max)             # keep larger colour value cell‑wise
    return O

def dsl3(I):
