def dsl1(I):
    # ----------  GLOBAL – choose the shorter bounding side ----------
    # all non‑zero pixels as separate objects
    objs   = objects(I, False, False, True)

    # rows / columns occupied by the objects
    rows   = apply(lambda o: center(o)[0], objs)
    cols   = apply(lambda o: center(o)[1], objs)

    top    = minimum(rows)
    bottom = maximum(rows)
    left   = minimum(cols)
    right  = maximum(cols)

    # size of the bounding box
    slice_h = subtract(add(bottom, ONE), top)   # bottom‑top+1
    slice_w = subtract(add(right, ONE), left)   # right‑left+1

    # which side is smaller ?
    # if height < width → work with rows, else work with columns
    O = branch(greater(slice_w, slice_h),
        # ----------  ROW‑MODE ----------
        # crop the minimal vertical slice (all columns)
        let_slice   = crop(I, (top, ZERO), (slice_h, width(I))),
        # fill every row that contains a colour with that colour
        rows_filled = apply(
            lambda o: fill(let_slice,
                           color(o),
                           { (r, c) for (r, c) in asindices(let_slice) if r == center(o)[0] }),
            objects(let_slice, False, False, True)),
        # repeat the slice vertically to the original height

                  divide(height(I), slice_h)),
