    objs       = colorfilter(fgpartition(I), top_colour)    # objects of that colour
    canvas_g   = canvas(dom, shape(I))                       # empty background
    painted    = apply(lambda o: paint(canvas_g, o), objs)   # paint each object
    O = last(painted)                                       # final grid
    return O


def dsl3(I):
    # COLOUR‑FILTERING – turn every non‑dominant cell that touches a dominant
    # cell into the dominant colour, but keep the original outer frame.
    dom = mostcolor(I)
    other = difference(palette(I), initset(dom))
    # cells of a non‑dominant colour that are orthogonal neighbours of a dominant cell
    masks = apply(lambda c: intersection(ofcolor(I, c),
                                        delta(ofcolor(I, dom))),
                 other)
    grow = combine(*masks)                                 # union of all those cells
    h, w = shape(I)
    top    = asindices(crop(I, (0, 0), (1, w)))
    bottom = asindices(crop(I, (h-1, 0), (1, w)))
    left   = asindices(crop(I, (0, 0), (h, 1)))
    right  = asindices(crop(I, (0, w-1), (h, 1)))
    border = combine(combine(top, bottom), combine(left, right))
    interior = difference(grow, border)
    recoloured = fill(I, dom, interior)                     # recolour the interior part
    O = paint(recoloured, toobject(border, I))             # restore border
    return O


def dsl4(I):
    # GEOMETRIC – keep only the dominant‑coloured cells that survive a
    # vertical mirror, everything else becomes the least frequent colour.
    bg   = leastcolor(I)            # background colour (least common)
    mirr = vmirror(I)               # vertical mirror of the picture
    dom  = mostcolor(I)             # dominant colour
    common = intersection(ofcolor(I, dom), ofcolor(mirr, dom))
    O = underfill(I, bg, delta(common))            # turn everything else into bg
    O = paint(O, toobject(common, I))               # restore the surviving dominant cells
    return O


def dsl5(I):
    # RELATIONAL – shift every object of the top‑row colour upwards until it
    # touches the top border, and redraw them on a canvas of the dominant colour.
    top_colour = mostcolor(crop(I, (0, 0), (1, width(I))))   # colour of first row
    dom        = mostcolor(I)                               # background colour
    objs       = fgpartition(I)                               # all foreground objects
    top_set    = ofcolor(I, top_colour)                     # all cells of the top‑row colour
    moved      = apply(lambda o: move(I, o, gravitate(o, top_set)), objs)
    canvas_g   = canvas(dom, shape(I))                      # clean canvas
    O = paint(canvas_g, combine(*moved))                    # draw shifted objects
    return O
