def dsl1(I):
    # 1️⃣ Relational – Manhattan distance ≤2 from any 4
    fours   = ofcolor(I, 4)

    # distance‑1 from the 4‑cells
    d1 = combine(combine(combine(combine(shift(fours, UP),
                                         shift(fours, DOWN)),
                                 shift(fours, LEFT)),
                        shift(fours, RIGHT))

    # distance‑2 from the 4‑cells (neighbors of the distance‑1 set)
    d2 = combine(combine(combine(combine(shift(d1, UP),
                                         shift(d1, DOWN)),
                                 shift(d1, LEFT)),
                        shift(d1, RIGHT))

    reach   = combine(fours, combine(d1, d2))               # all cells ≤2 from a 4
    target  = intersection(ofcolor(I, 5), reach)            # 5‑cells close to a 4
    O = fill(I, 2, toobject(target, I))
    return O


def dsl2(I):
    # 2️⃣ Colour‑filter – 5 that is horizontally sandwiched between two 4’s
    fours   = ofcolor(I, 4)

    # cells that have a 4 on the left side
    left_of_fours  = shift(fours, RIGHT)

    # cells that have a 4 on the right side
    right_of_fours = shift(fours, LEFT)

    sandwiched = intersection(intersection(left_of_fours,
                                           right_of_fours),
                              ofcolor(I, 5))
    O = fill(I, 2, toobject(sandwiched, I))
    return O


def dsl3(I):

    lower_pair = intersection(fives, below_five)           # the lower members of vertical pairs
    O = fill(I, 2, toobject(lower_pair, I))

def dsl4(I):
    # 4️⃣ Object‑extraction – interior of the bounding rectangle of all 4’s
    fours   = ofcolor(I, 4)
    outline = box(fours)                                 # border of the minimal rectangle containing the 4’s

    target = intersection(ofcolor(I, 5), interior_mask)
    O = fill(I, 2, toobject(target, I))
