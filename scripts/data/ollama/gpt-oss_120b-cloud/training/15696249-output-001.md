def dsl1(I):
    # ---------- OBJECT‑EXTRACTION ----------
    # all monochrome objects (rows or columns)
    OBJ   = objects(I, univalued=True, diagonal=False, without_bg=False)

    # ── horizontal case ──   (exactly‑one‑row objects exist)
    ROWS  = sfilter(OBJ, lambda o: height(o) == ONE)
    HCOND = positive(size(ROWS))
    ROBJ  = first(ROWS)                         # a uniform row object
    ROWIX = uppermost(ROBJ)                      # its row index 0,1 or 2
    YOFF  = multiply(ROWIX, THREE)              # 0,3 or 6

    # tile the full 3×3 input three times horizontally
    HROW  = hconcat(hconcat(I, I), I)            # 3 × 9
    CANV  = canvas(ZERO, (NINE, NINE))
    HOUT  = move(CANV, asobject(HROW, I), (YOFF, ZERO))

    # ── vertical case ──   (exactly‑one‑column objects exist)
    COLS  = sfilter(OBJ, lambda o: width(o) == ONE)
    COBJ  = first(COLS)                         # a uniform column object
    COLIX = leftmost(COBJ)                       # its column index 0,1 or 2
    XOFF  = multiply(COLIX, THREE)               # 0,3 or 6

    # tile the full 3×3 input three times vertically
    VCOL  = vconcat(vconcat(I, I), I)            # 9 × 3
    VOUT  = move(CANV, asobject(VCOL, I), (ZERO, XOFF))

    # choose the proper orientation
    O = branch(HCOND, HOUT, VOUT)
    return O


def dsl2(I):
    # ---------- COLOR‑FILTERING ----------
    # colour that appears most often
    MC    = mostcolor(I)
    IDX   = ofcolor(I, MC)                       # set of its cells

    # does this colour have horizontal period 1 ?  (i.e. occupies a whole row)
    HCOND = equality(hperiod(IDX), ONE)

    # offset where the uniform line sits
    P0    = first(IDX)                           # any cell of that colour
    YOFF  = multiply(uppermost(P0), THREE)      # for horizontal case
    XOFF  = multiply(leftmost(P0),  THREE)       # for vertical   case

    # two possible tilings
    HROW  = hconcat(hconcat(I, I), I)              # 3 × 9
    VCOL  = vconcat(vconcat(I, I), I)            # 9 × 3
    CANV  = canvas(ZERO, (NINE, NINE))

    HOUT  = move(CANV, asobject(HROW, I), (YOFF, ZERO))
    VOUT  = move(CANV, asobject(VCOL, I), (ZERO, XOFF))

    O = branch(HCOND, HOUT, VOUT)
    return O


def dsl3(I):

def dsl4(I):
    # ---------- RELATIONAL / STRUCTURAL ----------
    # compare total width versus total height of all monochrome objects
    OBJ   = objects(I, univalued=True, diagonal=False, without_bg=False)

    WLIST = apply(width,  OBJ)                   # list of widths
    HLIST = apply(height, OBJ)                   # list of heights

    WSUM  = sum(WLIST)
    HSUM  = sum(HLIST)

    HCOND = greater(WSUM, HSUM)                 # more “horizontal” mass → horizontal tiling

    # two possible expansions
    HR    = hconcat(hconcat(I, I), I)            # 3 × 9
    VT    = vconcat(vconcat(I, I), I)            # 9 × 3
    CANV  = canvas(ZERO, (NINE, NINE))

    # locate the uniform line (if any) to obtain the correct offset
    ROWS  = sfilter(OBJ, lambda o: height(o) == ONE)
    COLS  = sfilter(OBJ, lambda o: width(o)  == ONE)

    YOFF  = multiply(uppermost(first(ROWS)), THREE) if positive(size(ROWS)) else ZERO
    XOFF  = multiply(leftmost(first(COLS)),   THREE) if positive(size(COLS)) else ZERO

    HOUT  = move(CANV, asobject(HR, I), (YOFF, ZERO))
    VOUT  = move(CANV, asobject(VT, I), (ZERO, XOFF))

    O = branch(HCOND, HOUT, VOUT)
    return O
