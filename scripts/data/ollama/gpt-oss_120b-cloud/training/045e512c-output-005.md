def dsl1(I):
    # GLOBAL – crop to the tight box, double its size, rotate 90° clockwise,
    # then paste the result in the top‑left corner of a fresh canvas.
    sub   = compress(I)               # minimal rectangle containing all non‑zero cells
    big   = upscale(sub, 2)           # 2× enlargement
    rot   = rot90(big)                # quarter‑clockwise rotation
    base  = canvas(0, shape(I))      # empty canvas of the original dimensions
    O = paint(base, asobject(rot))   # embed at (0,0)
    return O


def dsl2(I):
    # OBJECT‑LEVEL – for every colour, extract its exact bounding box,
    # enlarge it 2× and move it to (2·ulcorner) before painting all together.
    cols   = palette(I)                                   # all present colours (excluding 0)
    objs   = apply(
        lambda c: shift(
            asobject(upscale(subgrid(backdrop(ofcolor(I, c)), I), 2)),
            multiply(ulcorner(ofcolor(I, c)), (TWO, TWO))
        ,
        cols)
    base   = canvas(0, shape(I))
    O = paint(base, merge(objs))                         # paint the union of shifted objects
    return O


def dsl3(I):
    # GLOBAL – remove the outer zero border, enlarge 2×,
    # then centre the enlarged pattern on the original canvas.
    trimmed = trim(I)                       # drop empty outer rows / cols
    big     = upscale(trimmed, 2)           # 2‑fold enlargement
    # compute the shift needed to centre the block:
    dh      = divide(subtract(shape(big), shape(I)), TWO)   # (height diff // 2, width diff // 2)
    shifted = shift(asobject(big), negate(dh))              # move by the negative offset
    base    = canvas(0, shape(I))
    O = paint(base, shifted)
    return O


def dsl4(I):

    return O


def dsl5(I):

    # replace each pixel by a 2×2 checker‑board of that colour and zero,
    # then paint the resulting pattern on a clean canvas.

        lambda patch:
            cellwise(
                paint(canvas(0, shape(I)),
                      recolor(c, patch)),      # recolor the patch with its colour
                canvas(0, shape(I)),         # background of zeros

        nonzero)
