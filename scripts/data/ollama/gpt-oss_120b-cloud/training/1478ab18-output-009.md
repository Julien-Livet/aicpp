def dsl1(I):
    # GLOBAL – build the rectangle with interval/merge instead of backdrop
    five        = ofcolor(I, 5)                                 # cells = {(r,c)...}
    rows        = {r for r, _ in five}
    cols        = {c for _, c in five}
    rmin        = valmin(rows, lambda x: x)                      # topmost row
    rmax        = valmax(rows, lambda x: x)                      # bottommost row
    cmin        = valmin(cols, lambda x: x)                      # leftmost column
    cmax        = valmax(cols, lambda x: x)                      # rightmost column
    all_rows    = interval(rmin, rmax + ONE, ONE)                # all rows inside bbox
    rect_cells  = merge(apply(lambda r: shoot((r, cmin), RIGHT), all_rows)  # full rectangle
    O1          = fill(I, 8, rect_cells)                        # colour rectangle 8
    O           = paint(O1, toobject(five, O1))                  # restore the 5s
    return O


def dsl2(I):
    # OBJECT EXTRACTION – union of the individual bounding boxes of each 5
    five_objs   = objects(I, univalued=True, diagonal=False, without_bg=False)  # each 5 as object
    five_boxes  = merge(apply(lambda obj: backdrop(obj), five_objs))            # bbox of every 5
    O1          = fill(I, 8, five_boxes)                     # colour every box 8
    O           = paint(O1, toobject(ofcolor(I, 5), O1))      # keep the original 5s
    return O


def dsl3(I):
    # GEOMETRIC – extend a ray in all four cardinal directions from each 5
    five       = ofcolor(I, 5)
    left_ray   = merge(apply(lambda p: shoot(p, LEFT),  five))
    right_ray  = merge(apply(lambda p: shoot(p, RIGHT), five))
    up_ray     = merge(apply(lambda p: shoot(p, UP),    five))
    down_ray   = merge(apply(lambda p: shoot(p, DOWN),  five))
    mask       = combine(combine(left_ray, right_ray), combine(up_ray, down_ray))
    O1         = fill(I, 8, mask)                           # colour the cross‑shaped mask
    O          = paint(O1, toobject(five, O1))                # restore 5s
    return O


def dsl4(I):
    # RELATIONAL / STRUCTURAL – take the Cartesian product of rows and columns
    # that contain at least one 5 (same idea as dsl5 but built with interval)

    row_cells  = merge(apply(lambda r: shoot((r, 0), RIGHT), rows_set)        # full rows
    col_cells  = merge(apply(lambda c: shoot((0, c), DOWN),  cols_set)        # full columns
    mask       = intersection(row_cells, col_cells)                           # rectangle
    O1         = fill(I, 8, mask)
    O          = paint(O1, toobject(five, O1))
