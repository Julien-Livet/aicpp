def dsl1(I):
    # Global transformation:
    #   1. Crop the outer zero border.
    #   2. Turn the interior background (0) into colour 1.
    #   3. Swap colours 2 and 3 (colour 4 stays unchanged).
    G = compress(I)
    B = replace(G, 0, 1)
    O = switch(B, 2, 3)
    return O


def dsl2(I):
    # Object‑extraction & recolouring:
    #   1. Crop the border.
    #   2. Replace interior 0 → 1.
    #   3. Extract all non‑zero objects.
    #   4. Recolour colour 2 objects to 3 and colour 3 objects to 2,
    #      keep colour 4 unchanged.
    #   5. Paint the recoloured objects back onto the new background.
    G   = compress(I)
    BG  = replace(G, 0, 1)
    OBJ = fgpartition(G)                                 # all coloured objects
    SET2 = colorfilter(OBJ, 2)                          # objects of colour 2
    SET3 = colorfilter(OBJ, 3)                          # objects of colour 3
    SET4 = colorfilter(OBJ, 4)                          # objects of colour 4
    SW2 = recolor(3, SET2)                               # 2 → 3
    SW3 = recolor(2, SET3)                               # 3 → 2
    ALL = merge({SW2, SW3, SET4})
    O   = paint(BG, ALL)
    return O


def dsl3(I):
    # Colour‑filtering via explicit patches:
    #   1. Crop the border.
    #   2. Convert interior 0 → 1.
    #   3. Build separate patches for colours 2, 3, 4,
    #      swap the colours of the first two patches,
    #      then paint them over the background.
    G   = compress(I)
    BG  = replace(G, 0, 1)
    P2  = ofcolor(G, 2)                                 # all cells of colour 2
    P3  = ofcolor(G, 3)                                 # all cells of colour 3
    P4  = ofcolor(G, 4)                                 # all cells of colour 4
    SW2 = recolor(3, P2)                                 # 2 → 3
    SW3 = recolor(2, P3)                                 # 3 → 2
    ALL = merge({SW2, SW3, P4})
    O   = paint(BG, ALL)
    return O


def dsl4(I):
    # Geometric reasoning via mirroring:
    #   1. Mirror the whole picture horizontally.
    #   2. Crop the zero border.
    #   3. Convert remaining background (0) to colour 1.
    #   4. Swap colours 2 and 3.
    #   5. Mirror back to restore the original orientation.
    M   = hmirror(I)                                     # geometric step
    C   = compress(M)                                    # remove outer zeros
    B   = replace(C, 0, 1)                               # background → 1
    S   = switch(B, 2, 3)                                 # colour swap
    O   = hmirror(S)                                     # undo the mirroring
    return O


def dsl5(I):

    T3  = replace(T2, 9, 3)                               # temp 9 → 3
    O   = replace(T3, 0, 1)                               # background 0 → 1
