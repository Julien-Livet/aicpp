def dsl1(I):

    # but keep any border row or column that contains more than one 5 unchanged.

    rows_multi = apply(lambda p: {p[0]}, five)        # set of row indices of each 5

    # columns of the box that host ≥2 fives
    cols_multi = apply(lambda p: {p[1]}, five)        # set of column indices of each 5

    # turn the duplicated rows / columns into patches on the grid
    row_patches = {shoot((r, 0), RIGHT) for r in dup_rows}
