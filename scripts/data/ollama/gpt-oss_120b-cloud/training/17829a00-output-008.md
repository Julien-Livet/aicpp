def dsl1(I):
    # GLOBAL – keep the outer frame unchanged, recolour every interior cell
    # with the most common colour of the whole picture.
    h, w        = shape(I)
    top_row     = crop(I, (0, 0), (1, w))
    bot_row     = crop(I, (h-1, 0), (1, w))
    left_col    = crop(I, (0, 0), (h, 1))
    right_col   = crop(I, (0, w-1), (h, 1))
    border_mask = union(
                    union(asindices(top_row),   asindices(bot_row)),
                    union(asindices(left_col), asindices(right_col))
                  )
    interior    = difference(asindices(I), border_mask)
    dom         = mostcolor(I)                     # dominant colour
    recoloured  = fill(I, dom, interior)           # paint interior
    # put the original border back (same colours as in I)
    O = paint(recoloured, toobject(border_mask, I))
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION – draw only the objects whose colour is the
    # colour of the top row on a canvas filled with the dominant colour.
    top_colour = mostcolor(crop(I, (0, 0), (1, width(I))))      # colour of the first row
    dom        = mostcolor(I)                                 # dominant colour (background)
    objs       = colorfilter(fgpartition(I), top_colour)      # objects of the top‑row colour
    canvas_g   = canvas(dom, shape(I))                         # empty background
    painted    = apply(lambda o: paint(canvas_g, o), objs)      # paint each object
    O = last(painted)                                         # final grid
    return O


def dsl3(I):
    # COLOUR‑FILTERING – any non‑dominant cell that touches a dominant cell
    # becomes dominant; everything else stays as it is.
    dom   = mostcolor(I)                                 # dominant colour
    other = difference(palette(I), initset(dom))        # all non‑dominant colours
    # masks of cells of a non‑dominant colour that are orthogonal neighbours of a dominant cell
    masks = apply(lambda c: intersection(ofcolor(I, c),
                                 delta(ofcolor(I, dom))),
                 other)
    grow  = combine(*masks)                              # union of all those masks
    O = fill(I, dom, grow)                               # recolour them
    return O


def dsl4(I):
    # GEOMETRIC – keep only the dominant‑coloured cells that survive a
    # vertical mirror, everything else becomes the least frequent colour.
    bg    = leastcolor(I)                                # background colour
    mirr  = vmirror(I)                                   # vertical mirror of the picture
    dom   = mostcolor(I)                                 # dominant colour
    # cells of the dominant colour that are present on both sides
    common = intersection(ofcolor(I, dom), ofcolor(mirr, dom))
    # turn everything else into the background colour
    O = underfill(I, bg, delta(common))
    # restore the surviving dominant cells with their original colour
    O = paint(O, toobject(common, I))
    return O


def dsl5(I):

    canvas_g   = canvas(dom, shape(I))                     # clean canvas
