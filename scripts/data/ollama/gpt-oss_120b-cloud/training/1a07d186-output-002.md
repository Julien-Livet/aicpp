def dsl1(I):
    # GLOBAL – compress away empty border and place the compressed pattern
    # back at the top‑left corner of a blank canvas of the original size.
    C = compress(I)                          # remove outer empty rows/cols
    O = underpaint(canvas(0, shape(I)), C)    # paint the compacted pattern
    return O


def dsl2(I):
    # OBJECT EXTRACTION – keep only those objects that form a full horizontal line.
    objs   = fgpartition(I)                                   # all coloured objects
    hlines = mfilter(objs, lambda o: hline(toindices(o)))      # keep the horizontal ones
    combined = merge(hlines)                                   # union of the kept patches
    O = underpaint(canvas(0, shape(I)), combined)             # render on a blank grid
    return O


def dsl3(I):

def dsl5(I):

    # down by an amount equal to its leftmost column index.
    objs     = fgpartition(I)                                            # all objects
    shifted  = apply(lambda o: shift(o, (leftmost(o), 0)), objs)          # vertical shift

    O = underpaint(canvas(0, shape(I)), combined)                        # paint on blank canvas
