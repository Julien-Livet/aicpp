def dsl1(I):

    mask = asindices(I)                     # set of every coordinate in the grid
    O = underfill(I, 0, mask)               # nothing is removed → output equals input

    # whole grid (indices + original colors) and paint it back.
    whole_obj = toobject(asindices(I), I)    # object that contains every cell with its colour
    O = paint(I, whole_obj)                 # painting the whole object restores the grid
    return O

    O = paint(I, shifted)                   # paint back → original grid
