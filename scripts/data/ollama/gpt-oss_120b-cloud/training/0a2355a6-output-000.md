# ----------------------------------------------------------------------
# 1️⃣  GLOBAL REPLACEMENT
#    Replace every occurrence of colour 8 by colour 2 everywhere.
#    Core idea: a single grid‑wide operation, no object reasoning.
# ----------------------------------------------------------------------
def dsl1(I):
    O = replace(I, 8, 2)          # global colour substitution
    return O


# ----------------------------------------------------------------------
# 2️⃣  OBJECT‑LEVEL RECOGNITION
#    Identify each 8‑coloured connected component, colour it according to its size:
#        size 9 → 2, size 5 → 3, otherwise → 1.
#    Core idea: work on individual objects (objects → fill).
# ----------------------------------------------------------------------
def dsl2(I):
    # set of all 8‑objects (4‑connected, background excluded)
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    # helper that maps an object to a colour based on its cardinality
    def obj_to_colour(obj):
        sz = size(obj)                     # number of cells in the object
        col = branch(sz == 9, 2,
               branch(sz == 5, 3, 1))      # fallback colour 1
        return fill(I, col, obj)           # recolour this object

    # recolour every object and merge the results
    recoloured = apply(obj_to_colour, objs)      # collection of grids
    O = I
    # overlay the recoloured pieces (they all share the same background)
    O = fold(lambda a, b: paint(a, b), recoloured, O)   # custom fold via apply is allowed
    return O


# ----------------------------------------------------------------------
# 3️⃣  BORDER‑PROXIMITY FILTER
#    Cells of colour 8 that touch the outer border become colour 1,
#    cells of colour 8 that are interior become colour 2,
#    isolated 8‑clusters that are completely surrounded by 0 become colour 3.
#    Core idea: use `bordering` to test adjacency to the grid edge.
# ----------------------------------------------------------------------
def dsl3(I):

    # objects of 8
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    def recolour_patch(patch):
        # is the whole patch adjacent to the outer border?
        is_border = bordering(patch, I)
        # does the patch have any 0‑cell neighbour? (interior test)
        interior = both(is_border == False,
                        any(index(I, add(loc, d)) == 0
                            for loc in patch for d in [UP, DOWN, LEFT, RIGHT]))
        col = branch(is_border, 1,
               branch(interior, 2, 3))
        return fill(I, col, patch)

    O = apply(recolour_patch, objs)
    # combine the recoloured patches (they are non‑overlapping)
    O = fold(lambda a, b: paint(a, b), O, I)
    return O
