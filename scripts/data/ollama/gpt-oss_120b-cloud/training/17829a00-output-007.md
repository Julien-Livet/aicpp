def dsl1(I):
    # GLOBAL – keep the outermost frame of the picture untouched,
    # recolour every interior cell with the most frequent colour.
    h, w      = shape(I)
    top       = crop(I, (0, 0), (1, w))                # first row
    bottom    = crop(I, (h-1, 0), (1, w))              # last row
    left      = crop(I, (0, 0), (h, 1))                # first column
    right     = crop(I, (0, w-1), (h, 1))              # last column
    frame     = vconcat(hconcat(top, bottom), hconcat(left, right))
    dom       = mostcolor(I)                           # dominant colour
    inner     = replace(I, leastcolor(I), dom)          # colour‑swap everywhere
    # restore the unchanged border
    O = paint(inner, frame)
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION – draw only the objects whose colour is the
    # dominant one on a canvas coloured with the least frequent colour.
    bg    = leastcolor(I)                                # background colour
    dom   = mostcolor(I)                                 # dominant colour
    objs  = colorfilter(fgpartition(I), dom)            # all dominant‑colour objects
    canvas_grid = canvas(bg, shape(I))                   # empty background canvas
    O = apply(lambda o: paint(canvas_grid, o), objs)      # paint each object
    return last(O)                                       # final grid


def dsl3(I):
    # COLOUR‑FILTERING – any non‑dominant cell that touches a dominant cell
    # becomes dominant; everything else stays as it is.
    dom   = mostcolor(I)                                 # dominant colour
    other = difference(palette(I), initset(dom))          # all non‑dominant colours
    O = foldl(
            lambda acc, c:
                fill(acc, dom,
                     intersection(ofcolor(acc, c),                # cells of colour c
                                  delta(ofcolor(acc, dom))      # orthogonal neighbours of dom
                                 )
                ),
            I,
            other
        )
    return O


def dsl4(I):
    # GEOMETRIC – intersect the picture with its vertical mirror;
    # keep only the dominant‑coloured cells that survive the intersection,
    # turn everything else into the background colour.
    bg    = leastcolor(I)                                       # background colour
    mirr  = vmirror(I)                                          # vertical mirror
    common = intersection(ofcolor(I, mostcolor(I)),
                         ofcolor(mirr, mostcolor(I)))          # dominant cells common to both halves
    O = underfill(I, bg, delta(common))                         # background elsewhere
    O = paint(O, common)                                        # restore intersecting cells
    return O


def dsl5(I):

    # of the grid and redraw them on a clean canvas.
    bg      = leastcolor(I)                                         # background colour
    objs    = fgpartition(I)                                         # all foreground objects
    centre  = initset(astuple(halve(height(I)), halve(width(I))))   # centre cell (as a set)
    moved   = apply(lambda o: move(I, o, gravitate(o, centre)), objs) # one‑step move toward centre
    canvas_grid = canvas(bg, shape(I))                               # empty canvas
    O = paint(canvas_grid, combine(*moved))                         # draw all moved objects
