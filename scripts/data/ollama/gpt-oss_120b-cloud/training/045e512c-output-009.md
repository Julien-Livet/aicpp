def dsl1(I):
    # COLOR‑BY‑COLOR replication.
    #   – For each colour present, upscale the exact shape of that colour.
    #   – Paste a copy of the enlarged shape at every original pixel of that colour.
    #   – Keep the original pixels (they are reproduced by the copies).
    cols   = palette(I)
    # enlarged copies for each colour
    ups_per_col = apply(
        lambda c:
            apply(
                lambda p: shift(
                    asobject(upscale(subgrid(backdrop(ofcolor(I, c)), I), TWO)),
                p),
                ofcolor(I, c)),
        cols)
    ups    = merge(apply(lambda s: merge(s), ups_per_col))        # all up‑scaled copies
    olds   = merge(apply(lambda c: asobject(ofcolor(I, c)), cols))  # original pixels
    O      = paint(canvas(0, shape(I)), combine(ups, olds))
    return O


def dsl2(I):
    # WHOLE‑PATTERN replication.
    #   – Enlarge the whole coloured pattern 2×.
    #   – Paste a copy of the enlarged pattern at every original non‑zero cell.
    whole   = merge(objects(I, F, F, T))
    sub     = subgrid(backdrop(whole), I)
    big     = upscale(sub, TWO)                      # 2× whole pattern
    # set of all non‑zero locations
    anchors = merge(apply(lambda c: ofcolor(I, c), palette(I)))
    copies  = apply(lambda p: shift(asobject(big), p), anchors)
    O       = paint(canvas(0, shape(I)), merge(copies))
    return O


def dsl3(I):
    # OBJECT‑LEVEL replication.
    #   – Treat every connected component (any colour) as an object.
    #   – Enlarge each object 2× and paste a copy at each of its own pixels.
    objs    = objects(I, T, F, T)                    # all connected components
    ups_per_obj = apply(
        lambda o:
            apply(
                lambda p: shift(
                    asobject(upscale(subgrid(backdrop(o), I), TWO)),
                p),
                o),
        objs)
    O = paint(canvas(0, shape(I)), merge(apply(lambda s: merge(s), ups_per_obj)))
    return O


def dsl4(I):
    # CENTER‑ALIGN replication (per colour).
    #   – For each colour, enlarge its shape.
    #   – Align the centre of mass of the enlarged shape with every original pixel of that colour.
    cols = palette(I)
    ups_per_col = apply(
        lambda c:
            let_big = asobject(upscale(subgrid(backdrop(ofcolor(I, c)), I), TWO) in
            let_cbig = centerofmass(toindices(let_big)) in
            apply(
                lambda p: shift(let_big, subtract(p, let_cbig)),
                ofcolor(I, c)),
        cols)
    O = paint(canvas(0, shape(I)), merge(apply(lambda s: merge(s), ups_per_col)))
    return O


def dsl5(I):
    # LOWER‑LEFT‑CORNER anchoring replication of the whole pattern.
    #   – Upscale the whole coloured pattern 2×.
    #   – Use the lower‑left corner of the enlarged pattern as the anchor.
    #   – Paste a copy anchored at every original non‑zero cell.
    whole   = merge(objects(I, F, F, T))
    sub     = subgrid(backdrop(whole), I)
    big     = upscale(sub, TWO)
    anchor  = llcorner(asobject(big))
    anchors = merge(apply(lambda c: ofcolor(I, c), palette(I)))   # all coloured cells
    copies  = apply(lambda p: shift(asobject(big), subtract(p, anchor)), anchors)
    O = paint(canvas(0, shape(I)), merge(copies))
    return O
