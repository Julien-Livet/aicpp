def dsl1(I):
    # ── GLOBAL‑TRANSFORMATION ──
    #   Down‑scale the whole grid by a factor of 2, count the non‑zero cells
    #   (a proxy for overall density of colour 8).  The denser the pattern,
    #   the higher the colour that replaces the original 8’s.
    tiny   = downscale(I, 2)                                 # global view, 2× smaller
    dens   = colorcount(tiny, 8)                             # how many 8‑pixels survive
    col    = branch(greater(dens, 20), 7,
               branch(greater(dens, 8), 3, 2))               # 7 / 3 / 2  (high‑mid‑low)
    O      = replace(I, 8, col)                              # recolour the 8‑object
    O      = replace(O, 1, 0)                                # erase the 1‑shape
    return O


def dsl2(I):
    # ── OBJECT‑EXTRACTION ──
    #   Work on the 8‑object alone.  Compare its height and width; a tall object
    #   gets colour 7, a wide one gets colour 3, otherwise colour 2.
    p8       = ofcolor(I, 8)                                 # the set of 8 cells
    h        = height(p8)                                    # vertical size
    w        = width(p8)                                     # horizontal size
    col      = branch(greater(h, w), 7,
               branch(greater(w, h), 3, 2))                    # 7 / 3 / 2
    O        = replace(I, 8, col)                            # recolour
    O        = replace(O, 1, 0)                               # erase the 1‑shape
    return O


def dsl3(I):
    # ── COLOR‑FILTERING ──
    #   Keep only those 8‑pixels that are adjacent to a background 0‑pixel
    #   (i.e. on the interior boundary).  Recolour the resulting mask.
    p8            = ofcolor(I, 8)                             # all 8 cells
    interior0      = intersection(ofcolor(I, 0), backdrop(p8))  # 0‑cells inside the bbox
    boundary8      = intersection(p8, box(interior0))          # 8‑cells touching those 0‑cells
    cnt           = size(boundary8)                           # how many boundary cells
    col           = branch(greater(cnt, 30), 7,
                  branch(greater(cnt, 10), 3, 2))
    O             = replace(I, 8, col)                         # recolour whole 8‑object
    O             = replace(O, 1, 0)                          # erase the 1‑shape
    return O


def dsl4(I):
    # ── GEOMETRIC‑REASONING ──
    #   Use the centre of mass of the 8‑object.  If it lies in the upper‑half
    #   of its bounding box we colour it 7, else if it lies in the lower‑half we
    #   colour it 3, otherwise colour 2.
    p8          = ofcolor(I, 8)
    c_mass      = centerofmass(p8)                            # (row, col) of COM
    c_box       = center(p8)                                  # centre of the bbox
    above       = greater(c_mass[0], c_box[0])                 # COM below centre ?
    col         = branch(above, 3,
                 branch(greater(c_box[0], c_mass[0]), 7, 2))
    O           = replace(I, 8, col)
    O           = replace(O, 1, 0)
    return O


def dsl5(I):
    # ── RELATIONAL / STRUCTURAL ──
    #   Measure the Manhattan distance between the 8‑object and the separate 1‑object.
    #   If they touch (distance = 0) we erase the 8‑object (colour 0);
    #   otherwise we recolour it according to how far apart they are:
    #   distance ≤ 3 → 7, distance ≤ 6 → 3, else 2.
    p8          = ofcolor(I, 8)
    p1          = ofcolor(I, 1)
    d           = manhattan(p8, p1)                           # minimal distance
    col         = branch(equality(d, 0), 0,
                 branch(greater(d, 6), 2,
                 branch(greater(d, 3), 3, 7)))
    O           = replace(I, 8, col)
    O           = replace(O, 1, 0)
    return O
