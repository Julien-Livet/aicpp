def dsl1(I):
    # Object‑based endpoint recoloring (explicit left/right column via leftmost/rightmost)
    objs       = fgpartition(I)                                 # all objects (color,coord)
    non_bg    = mfilter(objs, lambda o: color(o) > 0)           # drop background
    # turn objects into plain coordinate patches
    patches   = apply(lambda o: { (r, c) for (col, (r, c)) in o }, non_bg)

    # recolor leftmost column of every patch with 4
    lefts     = apply(lambda p: recolor(4,
                        { (r, c) for (r, c) in p if c == leftmost(p) }), patches)
    # recolor rightmost column of every patch with 2
    rights    = apply(lambda p: recolor(2,
                        { (r, c) for (r, c) in p if c == rightmost(p) }), patches)

    O = I
    for p in rights:  O = paint(O, p)      # 2 first
    for p in lefts:   O = paint(O, p)      # 4 overwrites when width‑1
    return O

def dsl3(I):

    objs      = fgpartition(I)
    non_bg    = mfilter(objs, lambda o: color(o) > 0)

def dsl4(I):
    # Shift‑difference based endpoint recoloring
    objs      = fgpartition(I)
    non_bg    = mfilter(objs, lambda o: color(o) > 0)
    patches   = apply(lambda o: { (r, c) for (col, (r, c)) in o }, non_bg)

    # cells that disappear when shifted left are the leftmost column
    lefts  = apply(lambda p: recolor(4,
                difference(p, shift(p, LEFT))), patches)
    # cells that disappear when shifted right are the rightmost column

    O = I
    for p in rights: O = paint(O, p)
    for p in lefts:  O = paint(O, p)
    return O
