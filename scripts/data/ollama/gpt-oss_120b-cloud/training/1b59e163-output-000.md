def dsl2(I):
    # 2️⃣ Color‑filtering: replace every non‑background (≠4) cell with the background color 4
    #   build a patch that contains all cells of colors 1,2,3,5,6,8 and fill them with 4
    patch = combine(
                combine(ofcolor(I, 1), ofcolor(I, 2)),

                                combine(ofcolor(I, 6), ofcolor(I, 8)))))
    O = fill(I, 4, patch)
    return O
