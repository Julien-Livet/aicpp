def dsl1(I):
    # colour‑filtering: for every colour > 1 take the orthogonal
    # bounding box of its cells, intersect it with the 1‑cells and
    # recolour the intersection with the seed colour.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                box(ofcolor(I, c)),   # full bounding box of colour c
                                ofcolor(I, 1))        # the 1‑cells
                         ),
                seeds)
    O = paint(I, merge(patches))
    return O


def dsl2(I):
    # object‑extraction: obtain every non‑background object,
    # recolour the 1‑cells that lie inside its orthogonal out‑box
    # with the object colour.
    objs    = sfilter(fgpartition(I), lambda o: color(o) > 1)
    patches = apply(
                lambda o: recolor(
                            color(o),
                            intersection(
                                outbox(o),           # orthogonal bounding box of the object
                                ofcolor(I, 1))       # all the 1‑cells
                         ),
                objs)
    O = paint(I, merge(patches))
    return O


def dsl3(I):
    # global‑mask construction: build a single mask that contains every
    # 1‑cell that belongs to at least one colour‑>1 bounding box,
    # then recolour those cells with the colour of the (unique) box
    # that contains them (boxes never overlap in the training data).
    seeds   = sfilter(palette(I), lambda c: c > 1)

    # for each seed colour make a coloured patch (its box ∩ 1‑cells)
    colored_patches = apply(
                        lambda c: recolor(
                                    c,
                                    intersection(
                                        box(ofcolor(I, c)),
                                        ofcolor(I, 1))
                                 ),
                        seeds)

    # union of all coloured patches gives the final coloured grid
    O = paint(I, merge(colored_patches))
    return O


def dsl4(I):
    # geometric‑reasoning using explicit corners:
    # for each colour >1 compute the upper‑left and lower‑right
    # corners of its bounding box, connect the two corners (diagonal)
    # and turn the whole rectangle into a patch with `box`.
    seeds = sfilter(palette(I), lambda c: c > 1)

    patches = apply(
                lambda c:
                    recolor(
                        c,
                        intersection(
                            box(
                                connect(
                                    ulcorner(ofcolor(I, c)),   # upper‑left corner of colour‑c cells
                                    lrcorner(ofcolor(I, c))    # lower‑right corner of colour‑c cells
                                )
                            ),
                            ofcolor(I, 1))
                    ),
                seeds)

    O = paint(I, merge(patches))
    return O


def dsl5(I):

    # cell that shares the same bounding‑box row or column.
    # Because each colour‑>1 occupies a single rectangular region,

    O = paint(I, merge(patches))
