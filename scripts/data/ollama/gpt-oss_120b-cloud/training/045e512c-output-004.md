def dsl1(I):
    # GLOBAL – crop to the tight bounding box, enlarge 2× and paste at the
    # top‑left corner of a canvas that has the original size.
    sub   = compress(I)                     # minimal rectangle containing all colours
    big   = upscale(sub, 2)                  # 2‑fold enlargement
    base  = canvas(0, shape(I))              # empty canvas of the original dimensions
    O     = paint(base, asobject(big))       # place the enlarged block at (0,0)
    return O


def dsl2(I):
    # OBJECT‑LEVEL – scale each colour separately, shift the enlarged block by
    # twice its original upper‑left corner and paint all blocks on a fresh canvas.
    cols   = palette(I)                                      # colours that appear
    canvas0 = canvas(0, shape(I))                             # empty canvas
    # for every colour produce a shifted, enlarged object
    shifted_objs = apply(
        lambda c: shift(
            asobject(upscale(subgrid(backdrop(ofcolor(I, c)), I), 2)),
            multiply(ulcorner(ofcolor(I, c)), (TWO, TWO))
        ),
        cols
    )
    all_objs = merge(shifted_objs)                            # union of the objects
    O = paint(canvas0, all_objs)                              # paint them together
    return O


def dsl3(I):
    # GLOBAL – remove the outer zero border (trim), enlarge 2× and embed at (0,0).
    trimmed = trim(I)                                          # drop empty outer rows/cols
    big     = upscale(trimmed, 2)                              # 2‑fold enlargement
    base    = canvas(0, shape(I))                             # empty canvas of original size
    O       = paint(base, asobject(big))                      # paste at the top‑left
    return O


def dsl4(I):
    # OBJECT‑LEVEL – obtain the global bounding box via objects, crop, enlarge,
    # and place the result on a canvas of the original dimensions.
    objs    = objects(I, True, False, True)                   # all coloured objects
    all_ix  = merge(apply(toindices, objs))                    # union of their indices
    sub     = subgrid(backdrop(all_ix), I)                     # minimal sub‑grid containing everything
    big     = upscale(sub, 2)                                 # 2‑fold enlargement
    base    = canvas(0, shape(I))                             # empty canvas
    O       = paint(base, asobject(big))                      # embed at (0,0)
    return O


def dsl5(I):
    # OBJECT‑LEVEL – compute exact corner coordinates, crop, enlarge,
    # and paint onto a zero canvas of the original size.
    objs    = objects(I, True, False, True)
    all_ix  = merge(apply(toindices, objs))
    ul      = ulcorner(all_ix)                               # upper‑left of the occupied region
    lr      = lrcorner(all_ix)                               # lower‑right of the occupied region
    dims    = add(subtract(lr, ul), (ONE, ONE))               # width × height of the box
    sub     = crop(I, ul, dims)                               # exact rectangle
    big     = upscale(sub, 2)                                 # 2‑fold enlargement
    base    = canvas(0, shape(I))                             # empty canvas
    O       = paint(base, asobject(big))                      # paste at (0,0)
