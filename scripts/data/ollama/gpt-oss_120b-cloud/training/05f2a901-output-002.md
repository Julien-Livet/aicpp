def dsl1(I):
    obj2 = ofcolor(I, TWO)
    obj8 = ofcolor(I, EIGHT)

    up2 = uppermost(obj2); low2 = lowermost(obj2)
    left2 = leftmost(obj2); right2 = rightmost(obj2)
    up8 = uppermost(obj8); low8 = lowermost(obj8)
    left8 = leftmost(obj8); right8 = rightmost(obj8)

    rows_overlap = both(flip(greater(up8, low2)), flip(greater(up2, low8)))

    h_shift = branch(greater(left8, right2),
                    subtract(left8, add(right2, ONE)),
                    subtract(left2, add(right8, ONE)))
    v_shift = branch(greater(up8, low2),
                    subtract(up8, add(low2, ONE)),
                    subtract(up2, add(low8, ONE)))

    dr = branch(rows_overlap, ZERO, v_shift)
    dc = branch(rows_overlap, h_shift, ZERO)

    O = move(I, obj2, astuple(dr, dc))
    return O
