def dsl1(I):
    # OBJECT‑EXTRACTION : pick object whose height equals grid height
    #                     or whose width  equals grid width
    objs = objects(I, True, False, False)
    h   = shape(I)[0]
    w   = shape(I)[1]
    full_obj = lambda o: either(
        both(equality(height(o), h),     True),
        both(equality(width(o),  w),     True))
    target = first(sfilter(objs, full_obj))
    return canvas(color(target), (1, 1))


def dsl2(I):
    # COLOR‑FILTERING : a colour that appears in every row
    #                  (or in every column) of the grid
    colors = palette(I)
    bg     = mostcolor(I)
    h      = shape(I)[0]
    w      = shape(I)[1]

    rows_covered = lambda c: both(
        equality(size(apply(lambda pt: pt[0], ofcolor(I, c))), h),
        equality(width(ofcolor(I, c)), w))

    cols_covered = lambda c: both(
        equality(size(apply(lambda pt: pt[1], ofcolor(I, c))), w),
        equality(height(ofcolor(I, c)), h))

    good_c = lambda c: either(rows_covered(c), cols_covered(c))

    target_c = first(sfilter(colors,
                lambda c: both(not equality(c, bg), good_c(c))))
    return canvas(target_c, (1, 1))


def dsl3(I):
    # GEOMETRIC‑REASONING : object touching both opposite borders
    objs   = fgpartition(I)                     # each coloured object
    bg     = mostcolor(I)
    idx    = asindices(I)
    top    = uppermost(idx);   bottom = lowermost(idx)
    left   = leftmost(idx);    right  = rightmost(idx)

    full_obj = lambda o: either(
        both(equality(uppermost(o), top),    equality(lowermost(o), bottom)),
        both(equality(leftmost(o),  left),    equality(rightmost(o), right)))

    nonbg = sfilter(objs, lambda o: flip(equality(color(o), bg)))
    target = first(sfilter(nonbg, full_obj))
    return canvas(color(target), (1, 1))


def dsl4(I):
    # LINE‑DETECTION : object that is a vertical line spanning the grid
    #                  or a horizontal line spanning the grid
    objs = objects(I, True, False, False)
    h    = shape(I)[0]
    w    = shape(I)[1]

    full_obj = lambda o: either(
        both(vline(o), equality(height(o), h)),
        both(hline(o), equality(width(o),  w)))

    target = first(sfilter(objs, full_obj))
    return canvas(color(target), (1, 1))


def dsl5(I):
    # RELATIONAL / OCCURRENCE : compare each coloured object with
    #                          a full‑grid column or row generated by shoot
    h   = shape(I)[0]
    w   = shape(I)[1]
    bg  = mostcolor(I)

    # all full columns (top → bottom)
    cols = apply(lambda c: shoot((0, c), DOWN), interval(0, w, 1))
    # all full rows (left → right)
    rows = apply(lambda r: shoot((r, 0), RIGHT), interval(0, h, 1))

    objs = objects(I, True, False, False)

    col_match = lambda o: greater(
        size(mfilter(cols,
            lambda l: both(vmatching(o, l), equality(size(o), h)))), 0)

    row_match = lambda o: greater(
        size(mfilter(rows,
            lambda l: both(hmatching(o, l), equality(size(o), w)))), 0)

    full_obj = lambda o: either(col_match(o), row_match(o))

    target = first(sfilter(objs,
                lambda o: both(not equality(color(o), bg), full_obj(o))))
    return canvas(color(target), (1, 1))
