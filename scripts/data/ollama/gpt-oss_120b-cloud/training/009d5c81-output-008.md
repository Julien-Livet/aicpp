def dsl1(I):
    # RELATIONAL‑DENSITY – interior 8‑cells (have 4‑neighbour 8‑support)
    p8      = ofcolor(I, 8)
    up      = shift(p8, UP)
    down    = shift(p8, DOWN)
    left    = shift(p8, LEFT)
    right   = shift(p8, RIGHT)
    interior = intersection(p8, up)
    interior = intersection(interior, down)
    interior = intersection(interior, left)
    interior = intersection(interior, right)
    cnt     = size(interior)                         # number of interior cells
    col = branch(greater(cnt, 30), 7,
           branch(greater(cnt, 10), 3, 2))
    O = replace(I, 8, col)
    O = replace(O, 1, 0)
    return O
