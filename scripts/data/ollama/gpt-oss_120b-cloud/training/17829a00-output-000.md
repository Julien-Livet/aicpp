def dsl1(I):
    # ---------- Global transformation ----------
    # Remove the outermost border, then restore it unchanged.
    # (trim removes one-cell border; we put the trimmed part back
    #  together with the original border using vconcat/hconcat.)
    trimmed = trim(I)                               # global trim
    top    = crop(I, (0, 0), (1, width(I)))       # original top row
    bottom = crop(I, (height(I)-1, 0), (1, width(I)))   # original bottom row
    left   = crop(I, (0, 0), (height(I), 1))       # original left column
    right  = crop(I, (0, width(I)-1), (height(I), 1))   # original right column
    # rebuild the grid (border stays the same, interior is trimmed)
    middle = hconcat(vconcat(left, trimmed), right)
    O      = vconcat(top, vconcat(middle, bottom))
    return O


def dsl2(I):
    # ---------- Object extraction ----------
    # Keep only the largest non‑background object (by cell count)
    # and paint it on a canvas of the original size filled with the
    # background color (the least common color).
    bg        = leastcolor(I)                                    # background
    objs      = fgpartition(I)                                   # all objects
    largest   = argmax(objs, lambda o: size(o))                   # biggest object
    canvas_bg = canvas(bg, shape(I))                             # blank canvas
    O         = paint(canvas_bg, largest)                         # paint largest object
    return O


def dsl3(I):
    # ---------- Color‑filtering ----------
    # Replace every cell of the most common color that has at least two
    # orthogonal neighbours of a different color with that neighbour colour.
    main_col   = mostcolor(I)                                    # dominant colour
    other_cols = difference(palette(I), initset(main_col))        # set of other colours
    # start from the original grid
    O = I
    # for each secondary colour c:
    O = mapply(lambda c:
                # cells of main colour adjacent (4‑dir) to at least two c‑cells
                recolor(c,
                        intersection(
                            ofcolor(O, main_col),
                            # neighbours of colour c taken twice (self‑intersection)
                            delta(ofcolor(O, c))
                        )
                ),
                other_cols)
    return O
