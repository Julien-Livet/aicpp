## DSL

def dsl1(I):
    # GLOBAL – repeatedly eliminate every empty row and column,
    #           first outer border then any interior gaps that appear.
    #   trim   – remove the outer empty border.
    #   compress – collapse any completely empty rows/cols;
    #   a second compress catches new gaps created by the first one.
    T   = trim(I)
    C1  = compress(T)
    C2  = compress(C1)
    O   = canvas(0, shape(I))
    return paint(O, asobject(C2))
```

def dsl2(I):
    # OBJECT‑LEVEL – extract each coloured object, shift it to the
    #   top‑left corner of its own bounding box, then redraw all
    #   objects on a fresh canvas (order of colours is irrelevant).
    objs    = fgpartition(I)                       # all coloured objects
    normed  = apply(normalize, objs)                # each to origin
    # fold the set of objects with the binary ‘combine’
    united  = apply(lambda o: combine(o, initset(())), normed)
    empty   = canvas(0, shape(I))
    return paint(empty, united)
```

def dsl3(I):
    # COLOUR‑FILTER – keep only the three most frequent non‑zero colours
    #   (ignoring the background), erase everything else and
    #   compact the remaining pattern.
    colours   = palette(I) - {0}
    ordered   = order(colours, lambda c: -colorcount(I, c))
    top3      = {first(apply(lambda _: ordered, interval(0, 3, 1)))}
    mask      = ofcolor(I, top3)                    # cells of the three colours
    filtered  = underpaint(I, mask)                 # delete all other colours
    compact   = compress(trim(filtered))            # remove outer & interior gaps
