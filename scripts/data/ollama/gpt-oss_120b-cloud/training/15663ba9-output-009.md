def dsl1(I):
    # GLOBAL EXTREME COLUMNS – colour the leftmost non‑zero column with 4
    # and the rightmost non‑zero column with 2 (whole picture at once).
    nz      = sfilter(asindices(I), lambda rc: index(I, rc) != 0)
    left    = leftmost(nz)
    right   = rightmost(nz)
    left_m  = recolor(4, sfilter(nz, lambda rc: rc[1] == left))
    right_m = recolor(2, sfilter(nz, lambda rc: rc[1] == right))
    O = paint(I, left_m)
    O = paint(O, right_m)
    return O


def dsl2(I):
    # OBJECT‑LEVEL EXTREME COLUMNS – for every connected component colour its
    # own leftmost column with 4 and its own rightmost column with 2.
    objs      = fgpartition(I)
    coloured  = mfilter(objs, lambda o: color(o) > 0)
    lefts = merge(apply(
        lambda o: recolor(
            4,
            sfilter(toindices(o),
                    lambda rc: rc[1] == leftmost(toindices(o)))) , coloured))
    rights = merge(apply(
        lambda o: recolor(
            2,
            sfilter(toindices(o),
                    lambda rc: rc[1] == rightmost(toindices(o)))) , coloured))
    O = paint(I, lefts)
    O = paint(O, rights)
    return O


def dsl3(I):
    # RECTANGLE‑SIDES PER OBJECT – keep only those border cells of an object's
    # bounding box that already belong to the object; colour left side 4,
    # right side 2.
    objs      = fgpartition(I)
    coloured  = mfilter(objs, lambda o: color(o) > 0)
    lefts = merge(apply(
        lambda o: recolor(
            4,
            intersection(
                sfilter(box(toindices(o)),
                        lambda rc: rc[1] == leftmost(toindices(o))),
                toindices(o)))
        , coloured))
    rights = merge(apply(
        lambda o: recolor(
            2,
            intersection(
                sfilter(box(toindices(o)),
                        lambda rc: rc[1] == rightmost(toindices(o))),
                toindices(o)))
        , coloured))
    O = paint(I, lefts)
    O = paint(O, rights)
    return O


def dsl4(I):
    # VERTICAL‑LINE INTERSECTION PER OBJECT – draw a full‑height line at each
    # object's leftmost and rightmost columns, then keep only the cells that
    # already belong to the object; colour the left line 4 and the right line 2.
    objs      = fgpartition(I)
    coloured = mfilter(objs, lambda o: color(o) > 0)
    lefts = merge(apply(
        lambda o: recolor(
            4,
            intersection(
                shoot((0, leftmost(toindices(o))), DOWN),
                toindices(o)))
        , coloured))
    rights = merge(apply(
        lambda o: recolor(
            2,
            intersection(
                shoot((0, rightmost(toindices(o))), DOWN),
                toindices(o)))
        , coloured))
    O = paint(I, lefts)
    O = paint(O, rights)
    return O


def dsl5(I):
    # COLOUR‑GROUP BOUNDING‑BOX – for each non‑zero colour compute its bounding box;
    # colour the cells of that colour that lie on the box's left side with 4
    # and on its right side with 2.
    colours = {c for c in palette(I) if c > 0}
    lefts = merge(apply(
        lambda c: recolor(
            4,
            sfilter(
                ofcolor(I, c),
                lambda rc: rc[1] == leftmost(ofcolor(I, c)))),
        colours))
    rights = merge(apply(
        lambda c: recolor(
            2,
            sfilter(
                ofcolor(I, c),
                lambda rc: rc[1] == rightmost(ofcolor(I, c)))),
        colours))
    O = paint(I, lefts)
    O = paint(O, rights)
    return O
