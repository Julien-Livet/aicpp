def dsl1(I):
    # 1️⃣  Propagation by north‑west shifting of all coloured cells.
    #    Objects are moved by (‑k,‑k) for every k and over‑laid.
    maxd   = maximum({height(I), width(I)})
    offs   = interval(0, maxd + 1)                     # 0 … maxd
    # shift the whole set of coloured objects by each offset
    shifted = apply(lambda k: move(I, fgpartition(I), (-k, -k)), offs)
    # combine from farthest to nearest so the nearest colour overwrites
    O      = fold(combine, shifted)
    return O


def dsl2(I):
    # 2️⃣  Build a north‑west rectangle for each non‑zero cell.
    #    The rectangle is the bounding box of the cell together with (0,0).
    nonzero = difference(palette(I), initset(0))
    # one patch per coloured cell
    cells   = apply(lambda c: ofcolor(I, c), nonzero)
    # rectangle from origin to the cell (inclusive)
    rects   = apply(lambda p: recolor(color(p), backdrop(combine(p, initset((0, 0))))), cells)
    # farthest rectangles first → nearest colour wins
    O       = fold(combine, rects)
    return O
