def dsl1(I):
    # Global – cut uniform outer fringe, then double every pixel.
    C = compress(I)                 # remove solid‑colour border
    H = hupscale(C, 2)              # double horizontally
    O = vupscale(H, 2)              # double vertically

def dsl2(I):
    # Object‑extraction – keep every object of the most frequent colour,
    # draw them on a canvas coloured with the least frequent colour.
    bg      = leastcolor(I)                                 # background colour
    dom_col = mostcolor(I)                                   # dominant colour
    objs    = colorfilter(fgpartition(I), dom_col)           # objects of dominant colour
    canvas_grid = canvas(bg, shape(I))                        # empty background canvas
    # paint each object on the canvas (paint returns a grid, we keep the last one)
    O = apply(lambda o: paint(canvas_grid, o), objs)
    # the above produces a container of grids; the final grid is the last element
    return last(O)


def dsl3(I):
    # Colour‑filtering – any cell that is not the dominant colour but touches a
    # dominant‑coloured cell becomes the dominant colour.
    dom   = mostcolor(I)                                         # dominant colour
    other = difference(palette(I), initset(dom))                  # all non‑dominant colours
    O = I
    O = mapply(
            lambda c:
                fill(O,                                   # recolour directly on the grid
                     dom,
                     intersection(ofcolor(O, c),            # cells of colour c
                                  delta(ofcolor(O, dom))  # touching a dominant cell
                                 )
                ),
            other
        )
    return O


def dsl4(I):
    # Geometric – intersect the picture with its horizontal mirror; keep the
    # common cells, everything else becomes the background colour.
    bg   = leastcolor(I)                                          # background colour
    mirr = hmirror(I)                                             # horizontal mirror
    common = intersection(
                ofcolor(I, mostcolor(I)),                         # dominant cells in I
                ofcolor(mirr, mostcolor(I))                       # dominant cells in mirror
             )
    O = underfill(I, bg, delta(common))                           # background elsewhere
    O = paint(O, common)                                          # restore the intersecting cells
    return O


def dsl5(I):
    # Relational – shift every non‑background object one step toward the centre.
    bg      = leastcolor(I)                                      # background colour
    objs    = fgpartition(I)                                      # all foreground objects
    centre  = initset(astuple(halve(height(I)), halve(width(I)))) # centre cell
    moved   = apply(
                lambda o: move(I, o, gravitate(o, centre)),      # one‑step move toward centre
                objs
            )
    canvas_grid = canvas(bg, shape(I))                            # clean background canvas
    O = paint(canvas_grid, combine(*moved))                      # draw all moved objects
