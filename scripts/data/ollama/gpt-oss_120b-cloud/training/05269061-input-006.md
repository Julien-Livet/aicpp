def dsl1(I):
    # GLOBAL PATTERN: 3‑color (row+col) mod 3 tiling.
    # 1) collect the three non‑zero colours as objects
    objs   = objects(I, univalued=True, diagonal=False, without_bg=True)
    # 2) order them by the sum of their centre‑of‑mass coordinates
    sum_of = lambda o: add(centerofmass(o)[0], centerofmass(o)[1])
    o0     = argmin(objs, sum_of)
    o1     = argmin(difference(objs, {o0}), sum_of)
    o2     = argmin(difference(objs, {o0, o1}), sum_of)
    c0     = color(o0)          # smallest (row+col) residue
    c1     = color(o1)          # middle residue
    c2     = color(o2)          # largest residue

    # 3) build three vertical strips (one colour per column‑mod‑3)
    strip0 = canvas(c0, (height(I), 1))
    strip1 = canvas(c1, (height(I), 1))
    strip2 = canvas(c2, (height(I), 1))
    offs0  = apply(lambda j: (0, j), interval(0, width(I), 3))
    offs1  = apply(lambda j: (0, j), interval(1, width(I), 3))
    offs2  = apply(lambda j: (0, j), interval(2, width(I), 3))
    V0     = apply(lambda o: paint(canvas(0, shape(I)), shift(strip0, o)), offs0)
    V1     = apply(lambda o: paint(canvas(0, shape(I)), shift(strip1, o)), offs1)
    V2     = apply(lambda o: paint(canvas(0, shape(I)), shift(strip2, o)), offs2)
    vertical = first(V0)      # any already contains the full vertical pattern

    # 4) build three horizontal strips (one colour per row‑mod‑3)
    hstrip0 = canvas(c0, (1, width(I)))
    hstrip1 = canvas(c1, (1, width(I)))
    hstrip2 = canvas(c2, (1, width(I)))
    roffs0   = apply(lambda i: (i, 0), interval(0, height(I), 3))
    roffs1   = apply(lambda i: (i, 0), interval(1, height(I), 3))
    roffs2   = apply(lambda i: (i, 0), interval(2, height(I), 3))
    H0       = apply(lambda o: paint(canvas(0, shape(I)), shift(hstrip0, o)), roffs0)
    H1       = apply(lambda o: paint(canvas(0, shape(I)), shift(hstrip1, o)), roffs1)
    H2       = apply(lambda o: paint(canvas(0, shape(I)), shift(hstrip2, o)), roffs2)
    horizontal = first(H0)

    # 5) combine: keep vertical where it is non‑zero, otherwise take horizontal.
    O = cellwise(vertical, horizontal, 0)
    return O
```
