def dsl1(I):
    # IDENTITY – return the input unchanged.
    O = I

def dsl2(I):
    # MOST‑COLOR – keep only the cells of the most frequent colour,
    # everything else becomes background (0).
    MC   = mostcolor(I)                                 # most common colour
    PATCH = ofcolor(I, MC)                              # its cells
    O     = paint(canvas(0, shape(I)), PATCH)           # recolour on blank canvas
    return O


def dsl3(I):
    # VERTICAL‑MIRROR – overlay the grid with its vertical mirror.
    MIR   = vmirror(I)                                   # mirror left‑right
    OBJ   = merge({asobject(I), asobject(MIR)})           # union of original and mirror
    O     = underpaint(canvas(0, shape(I)), OBJ)        # draw on blank background
    return O


def dsl4(I):
    # HORIZONTAL‑LINES – retain only objects that form a horizontal line.
    OBJS      = fgpartition(I)                                   # all coloured objects
    LINE_OBJS = sfilter(OBJS, lambda o: hline(o))                 # keep horizontal lines
    COMBINED  = merge(LINE_OBJS)                                 # union of those objects
    O         = underpaint(canvas(0, shape(I)), COMBINED)        # paint on blank canvas
    return O


def dsl5(I):
    # COMPRESS‑ZERO‑ROWS – remove rows that are completely background (0)
    # and shift the remaining rows upward, keeping the original width.
    ROWS      = apply(lambda r: r if any(c != 0 for c in r) else None,
                    I)                                      # keep non‑empty rows
    COMPRESSED = tuple(r for r in ROWS if r is not None)          # filter out Nones
    # pad with empty rows at the bottom to preserve original height
    PAD       = canvas(0, (height(I) - len(COMPRESSED), width(I)))
    O         = vconcat(COMPRESSED, PAD)
    return O
