def dsl1(I):
    # GLOBAL‑RELATIONAL: operate on the whole grid at once, using UP to detect the bottom cells.
    all_cells = asobject(I)
    left_adj  = intersection(all_cells, shift(all_cells, RIGHT))
    leftmost  = difference(all_cells, left_adj)                     # cells with no neighbour on the left
    has_below = intersection(leftmost, shift(leftmost, UP))         # cells that have something above them
    bottom    = difference(leftmost, has_below)                     # bottom cells of each column
    movable  = difference(leftmost, bottom)                         # cells that will move right
    shifted   = shift(movable, RIGHT)
    static    = difference(all_cells, movable)
    empty     = canvas(0, shape(I))
    return paint(paint(empty, shifted), static)
