    # GEOMETRIC‑MIRROR: extract the sub‑grid that contains the 5’s,
    # mirror it vertically, recolour the mirrored 5‑pixels with 8,
    # and overlay the result on the original grid.
    five_pts = ofcolor(I, 5)                     # all 5‑cells
    box      = backdrop(five_pts)                # bounding box of the 5’s
    sub      = subgrid(box, I)                  # the rectangle that encloses the 5’s
    mir      = vmirror(sub)                     # vertical mirror of that rectangle
    mir5     = ofcolor(mir, 5)                   # where the mirrored 5‑pixels are
    O        = fill(I, 8, mir5)                 # colour those positions 8
    O        = paint(O, toobject(five_pts, O))  # restore the original 5‑pixels
    return O


def dsl5(I):
    # RELATIONAL‑DISTANCE: take the 5‑bounding box, dilate the set of 5’s by
    # one orthogonal step (i.e. all cells at Manhattan distance ≤ 1 from a 5),
    # intersect that dilation with the bounding box, and recolour the result 8
    # (keeping the original 5‑pixels).
    five      = ofcolor(I, 5)                                 # 5‑cells
    box      = backdrop(five)                                # bounding rectangle of all 5’s
    dilated  = combine(five,
                       combine(hupscale(five, 1),
                               vupscale(five, 1)))          # 5‑cells + their orthogonal neighbours
    region   = intersection(dilated, box)                    # keep only the part inside the box
    O        = fill(I, 8, region)                           # colour those cells 8
    O        = paint(O, toobject(five, O))                    # restore the original 5‑pixels
    return O
