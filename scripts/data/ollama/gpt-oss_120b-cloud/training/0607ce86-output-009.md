def dsl1(I):
    # RELATIONAL – replace each non‑zero, non‑8 row by the most frequent such row
    zero_row = repeat(0, width(I))
    rows         = apply(identity, I)                                   # all rows as tuples
    cand_rows    = mfilter(rows, lambda r: any(c != 0 and c != 8 for c in r))
    most_common  = mostcommon(cand_rows)                                 # the prototype row
    O = apply(
        lambda r: r if any(c == 8 for c in r) else                      # keep rows that contain 8
                 (most_common if any(c != 0 for c in r) else zero_row), # replace others
        I)
    return O


def dsl2(I):
    # OBJECT‑LEVEL – keep the 8‑objects, repaint every other cell with the
    # shape of the most frequent non‑8 object (which in these tasks coincides with a whole row)
    objs8        = colorfilter(objects(I, True, False, True), 8)         # 8‑objects
    non8_colors  = sfilter(palette(I), lambda c: c != 0 and c != 8)
    non8_objs    = merge(mapply(lambda c: objects(I, False, False, True) &
                                   colorfilter(objects(I, True, False, True), c),
                         non8_colors))                                 # all non‑8 objects
    most_obj     = mostcommon(non8_objs)                                 # most frequent non‑8 object
    # paint a full‑grid with that object (its rows are reproduced everywhere it appears)
    mask_rows    = { (r, 0) for (r, _) in outbox(most_obj) }             # rows occupied by the object
    rows_mask    = merge(mapply(lambda p: shoot(p, (0, 1)), mask_rows))  # horizontal lines on those rows
    O0           = paint(canvas(0, shape(I)), rows_mask)               # rows become the prototype
    O            = paint(O0, merge(objs8))                               # restore 8‑objects
    return O
