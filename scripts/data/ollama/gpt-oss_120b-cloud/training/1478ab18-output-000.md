def dsl1(I):
    # --- Global bounding‑box fill ---------------------------------
    # 1. Locate every cell of colour 5.
    pts = ofcolor(I, 5)
    # 2. Bounding box that contains all 5‑cells.
    bb = backdrop(pts)
    # 3. All cells strictly inside that box (border excluded).
    interior = delta(bb)
    # 4. Paint the interior with colour 8, leaving the original 5‑cells intact.
    O = fill(I, 8, interior)
    return O


def dsl2(I):
    # --- Adjacent‑to‑5 fill ---------------------------------------
    # 1. All 5‑cells.
    pts = ofcolor(I, 5)
    # 2. Orthogonal neighbours of every 5‑cell (union of the neighbour sets).
    neigh = mapply(neighbors, pts)
    # 3. Keep only those neighbour cells that are not themselves 5.
    target = difference(neigh, pts)
    # 4. Recolour the target cells with 8.
    O = fill(I, 8, target)
    return O


def dsl3(I):
    # --- Global colour switch --------------------------------------
    # Replace every background 7 with 8 everywhere.
    O = switch(I, 7, 8)
    return O


def dsl4(I):
    # --- Geometric mirroring ---------------------------------------
    # Produce the horizontal mirror of the whole grid.
    O = hmirror(I)
    return O


def dsl5(I):
    # --- Local block around the furthest 5 -------------------------
    # 1. Set of all 5‑cells.
    pts = ofcolor(I, 5)
    # 2. Choose the 5‑cell with the greatest (row+col) sum (the
    #    “most distant” towards the bottom‑right).
    max_pt = valmax(pts, lambda p: add(p[0], p[1]))
    # 3. Build a 3×3 square centred on that cell.
    #    (use a tiny 3×3 object centred on max_pt).
    square = backdrop(initset(max_pt))          # 1×1 box
    square = expand(square, (1, 1))            # grow it to 3×3
    # 4. Paint that square with colour 8.
    O = fill(I, 8, square)
    return O
