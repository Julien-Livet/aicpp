def dsl1(I):
    # GLOBAL‑CROP → SHIFT INSIDE SUB‑GRID → RE‑EMBED
    obj2 = ofcolor(I, TWO)
    obj8 = ofcolor(I, EIGHT)

    # bounding box that contains both objects
    bb = combine(backdrop(obj2), backdrop(obj8))
    top, left   = ulcorner(bb)
    bottom, right = lrcorner(bb)

    h = add(add(bottom, -top), ONE)
    w = add(add(right, -left), ONE)

    # work only inside that region
    sub = crop(I, (top, left), (h, w))
    o2 = ofcolor(sub, TWO)
    o8 = ofcolor(sub, EIGHT)

    # extreme coordinates inside the sub‑grid
    up2   = uppermost(o2);   low2  = lowermost(o2)
    left2  = leftmost(o2);   right2 = rightmost(o2)
    up8   = uppermost(o8);   low8  = lowermost(o8)
    left8  = leftmost(o8);   right8 = rightmost(o8)

    rows_overlap = both(flip(greater(up8, low2)),
                         flip(greater(up2, low8)))

    h_gap = branch(greater(left8, right2),
                 subtract(left8, add(right2, ONE)),
                 subtract(left2, add(right8, ONE)))
    v_gap = branch(greater(up8, low2),
                 subtract(up8, add(low2, ONE)),
                 subtract(up2, add(low8, ONE)))

    dr_sign = branch(greater(up2, low8), NEG_ONE,
                 branch(greater(low2, up8), ONE, ZERO))
    dc_sign = branch(greater(left2, right8), NEG_ONE,
                 branch(greater(right2, left8), ONE, ZERO))

    dr = branch(rows_overlap, ZERO, multiply(dr_sign, v_gap))
    dc = branch(rows_overlap, multiply(dc_sign, h_gap), ZERO)

    # shift the 2‑object inside the cropped window
    moved_o2 = recolor(TWO, shift(o2, (dr, dc)))
    moved_global = shift(moved_o2, (top, left))

    cleared = cover(I, obj2)
    O = paint(cleared, moved_global)
    return O
