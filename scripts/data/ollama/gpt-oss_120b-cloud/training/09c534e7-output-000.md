# 1️⃣  Global dilation of every non‑1 colour into orthogonal 1‑cells
# --------------------------------------------------------------
def dsl1(I):
    # colours that may act as seeds (all colours > 1)
    seeds   = mfilter(palette(I), lambda c: c > 1)

    # start from the original grid
    O = I

    # for each seed colour c:
    #   – take its orthogonal out‑box,
    #   – intersect it with the cells that are colour 1,
    #   – recolour that intersection to c,
    #   – paint the recoloured object back onto the grid.
    O = apply(
            lambda c:
                paint(
                    O,
                    recolor(
                        c,
                        intersection(
                            outbox(ofcolor(I, c)),
                            ofcolor(I, 1))
                    )
                ),
            seeds)

    # `apply` returns a container; the last element of the container
    # is the fully updated grid.
    return last(O)

def dsl3(I):
