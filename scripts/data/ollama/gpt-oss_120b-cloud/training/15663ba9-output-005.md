def dsl1(I):
    # GLOBAL‑TRANSFORM: find the leftmost / rightmost column that contains any
    # non‑zero colour in the whole grid and colour those columns (4 = left, 2 = right).
    colours   = {c for c in palette(I) if c > 0}
    masks     = apply(lambda c: ofcolor(I, c), colours)

    leftmost_col  = minimum(apply(lambda m: leftmost(m), masks))
    rightmost_col = maximum(apply(lambda m: rightmost(m), masks))

    lefts  = apply(lambda m: recolor(4,
                {(r, c) for (r, c) in m if c == leftmost_col}), masks)
    rights = apply(lambda m: recolor(2,
                {(r, c) for (r, c) in m if c == rightmost_col}), masks)

    O = I
    for p in rights: O = paint(O, p)
    for p in lefts : O = paint(O, p)
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION: treat each *connected* component as an object,
    # take its bounding box and colour the vertical sides of that box.
    objs   = fgpartition(I)                                   # all objects
    nonbg  = mfilter(objs, lambda o: color(o) > 0)            # drop background

    boxes  = apply(lambda o: box(toindices(o)), nonbg)      # outline of each component

    lefts  = apply(lambda b: recolor(4,
                {(r, c) for (r, c) in b if c == leftmost(b)}), boxes)
    rights = apply(lambda b: recolor(2,
                {(r, c) for (r, c) in b if c == rightmost(b)}), boxes)

    O = I
    for p in rights: O = paint(O, p)
    for p in lefts : O = paint(O, p)
    return O
