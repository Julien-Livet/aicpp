def dsl1(I):
    # GLOBAL‑REPLACE: replace every 1 with the most frequent non‑{0,1,4} colour.
    other = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main  = argmax(other, lambda c: colorcount(I, c))
    O = replace(I, 1, main)
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION: recolour each connected component made of 1‑cells
    # to the dominant non‑{0,1,4} colour and paint them back.
    other = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main  = argmax(other, lambda c: colorcount(I, c))

    # binary mask that keeps only the 1‑cells
    mask = paint(canvas(0, shape(I)), recolor(1, ofcolor(I, 1)))

    # connected components of the mask (background = 0)
    comps = objects(mask, False, False, True)

    recol = lambda comp: recolor(main, comp)          # turn the whole component into “main”
    recoloured = apply(recol, comps)                  # recolour every component
    merged = merge(recoloured)                        # union of all recoloured patches

    O = paint(I, merged)                               # paint them over the original grid
    return O


def dsl3(I):
    # PALETTE‑MAP: build a recolour‑patch for each colour, keep everything
    # unchanged except colour 1, which is remapped to the dominant non‑{0,1,4} colour.
    other = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main  = argmax(other, lambda c: colorcount(I, c))

    patches = apply(
        lambda c: branch(
            c == 1,
            recolor(main, ofcolor(I, 1)),      # 1 → main
            recolor(c,   ofcolor(I, c))        # other colours stay the same
        ),
        palette(I)
    )
    O = paint(I, merge(patches))
    return O


def dsl4(I):
    # DIRECT‑PAINT: create a patch that maps every 1‑cell to the dominant colour
    # and paint it on the original grid.
    other   = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main    = argmax(other, lambda c: colorcount(I, c))

    map_one = recolor(main, ofcolor(I, 1))
    O = paint(I, map_one)
    return O


def dsl5(I):
    # PER‑STRIP (relational): rows consisting entirely of 4 are separators.
    # 1) erase those rows (turn 4 → 0);
    # 2) recolour every component that contains a 1 with the dominant colour;
    # 3) restore the original 4‑rows.
    other = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main  = argmax(other, lambda c: colorcount(I, c))

    # 1) erase the separator rows
    no4 = paint(I, recolor(0, ofcolor(I, 4)))

    # 2) recolour each component that touches a 1‑cell
    comps = objects(no4, False, False, True)
    recol = lambda comp: branch(
        greater(size(intersection(comp, ofcolor(I, 1))),  # component contains a 1?
        recolor(main, comp),                           # recolour it
        comp                                            # otherwise keep unchanged
    )
    newc   = apply(recol, comps)
    merged = merge(newc)

    # 3) restore the separator rows
    O = paint(merged, recolor(4, ofcolor(I, 4)))
    return O
