def dsl1(I):
    # ---------- object‑extraction ----------
    # 1) turn the separator colour (5) into background (0) so that the
    #    remaining coloured regions become separate objects.
    S = switch(I, 5, 0)

    # 2) extract every object (connected component) from the modified grid.
    objs = objects(S, False, False, False)

    # 3) keep only the objects that do NOT contain the separator colour.
    #    (they are the original blocks)
    blocks = colorfilter(objs, 5)

    # 4) for each block compute its dominant colour in the ORIGINAL grid
    #    and recolour the whole block with that colour.
    recoloured = apply(
        lambda o: recolor(mostcolor(subgrid(o, I)), o),
        blocks
    )

    # 5) paint all recoloured blocks on top of the original input.
    O = paint(I, combine(recoloured))
    return O


def dsl2(I):
    # ---------- colour‑filtering (local majority) ----------
    # 1) indices of all cells that are NOT a separator (5).
    non5 = mfilter(asindices(I), lambda p: index(I, p) != 5)

    # 2) for every such cell take its 3×3 neighbourhood (clipped at the border)
    #    and recolour the centre with the most frequent colour inside that neighbourhood
    #    (the neighbourhood still contains the original 5‑cells, but they are ignored
    #     because they are never the most frequent colour inside a block).
    patches = apply(
        lambda p: recolor(
            mostcolor(
                crop(
                    I,
                    (max(p[0] - 1, 0), max(p[1] - 1, 0)),
                    (min(3, shape(I)[0] - max(p[0] - 1, 0)),
                    (min(3, shape(I)[1] - max(p[1] - 1, 0)))
                )
            ),
            {p}
        ),
        non5
    )

    # 3) paint the obtained recolourings over the input.
    O = paint(I, combine(patches))
    return O


def dsl3(I):
    # ---------- geometric (rotation‑based) ----------
    # 1) rotate so that rows of 5 become columns.
    R = rot90(I)

    # 2) positions of the separator columns (still colour 5 after rotation).
    v5   = apply(lambda p: p[1], ofcolor(R, 5))
    cuts = sort(combine({0}, combine(v5, {shape(R)[1]})))   # 0, each separator, width

    # 3) all indices of the rotated grid.
    idx = asindices(R)

    # 4) fill each vertical strip (between two consecutive separators)
    #    with the most common colour that appears in that strip in the ORIGINAL grid.
    strips = apply(
        lambda pair: recolor(
            mostcolor(subgrid(
                mfilter(idx, lambda q: q[1] > pair[0] and q[1] < pair[1]),
                R)),
            mfilter(idx, lambda q: q[1] > pair[0] and q[1] < pair[1])),
        zip([0] + list(rest(cuts)), list(rest(cuts)) + [shape(R)[1]])
    )
    Orot = paint(R, combine(strips))

    # 5) rotate back to the original orientation.
    O = rot270(Orot)
    return O


def dsl4(I):
    # ---------- relational / structural ----------
    # 1) the set of all separator cells (colour 5).
    sep = ofcolor(I, 5)

    # 2) for each non‑separator cell locate the nearest separator in the
    #    four cardinal directions (up, down, left, right).
    non5 = mfilter(asindices(I), lambda p: index(I, p) != 5)

    # 3) build a patch consisting of all cells that share the same enclosing
    #    rectangle defined by those four nearest separators.
    #    The rectangle is described by the four limits:
    #    row_min = p[0] + gravitate({p}, sep)   (up direction)
    #    row_max = p[0] + gravitate({p}, sep)   (down direction)
    #    col_min = p[1] + gravitate({p}, sep)   (left direction)
    #    col_max = p[1] + gravitate({p}, sep)   (right direction)
    #    Because gravitate always returns the shortest vector that makes the source
    #    adjacent to the destination, the same call works for both opposite
    #    directions (the sign of the returned vector tells the direction).
    rects = apply(
        lambda p: recolor(
            mostcolor(
                subgrid(
                    mfilter(asindices(I),
                             lambda q: q[0] > p[0] + gravitate({p}, sep)[0] and
                                      q[0] < p[0] + gravitate({p}, sep)[0] and
                                      q[1] > p[1] + gravitate({p}, sep)[1] and
                                      q[1] < p[1] + gravitate({p}, sep)[1]),
                    I)),
            {p}),
        non5
    )
    O = paint(I, combine(rects))
    return O


def dsl5(I):
    # ---------- global (periodicity‑based) ----------
    # 1) rows and columns that are completely filled with the separator colour (5).
    full_rows = apply(lambda p: p[0],
                     mfilter(ofcolor(I, 5), lambda p: hline({p})))
    full_cols = apply(lambda p: p[1],
                     mfilter(ofcolor(I, 5), lambda p: vline({p})))

    # 2) the vertical and horizontal periods of those separator lines.
    #    (the distance between two consecutive separators).
    vperiod = sub(minimum(full_cols), maximum(full_cols))   # not used directly, only to force a dependency
    hperiod = sub(minimum(full_rows), maximum(full_rows))

    # 3) cut positions (including the borders of the grid).
    rcuts = sort(combine({0}, combine(full_rows, {shape(I)[0]})))
    ccuts = sort(combine({0}, combine(full_cols, {shape(I)[1]})))

    # 4) generate every block by pairing consecutive cut lines.
    blocks = apply(
        lambda r: apply(
            lambda c: (list(rcuts)[r], list(ccuts)[c]),
            interval(0, size(ccuts) - 1, 1)
        ),
        interval(0, size(rcuts) - 1, 1)
    )

    # 5) for each block compute its dominant colour and recolour the whole block.
    recolours = apply(
        lambda corner: recolor(
            mostcolor(
                crop(
                    I,
                    corner,
                    (list(rcuts)[index(rcuts, corner[0]) + 1] - corner[0],
                     list(ccuts)[index(ccuts, corner[1]) + 1] - corner[1]))
            ),
            shoot(corner,
                  (list(rcuts)[index(rcuts, corner[0]) + 1] - corner[0],
                  (list(ccuts)[index(ccuts, corner[1]) + 1] - corner[1]))
        ),
        flatten(blocks)
    )
    O = paint(I, combine(recolours))
    return O
