    # patches that turn the 1‑cells into the corresponding “other” colour
    from_one = apply(lambda c: recolor(1, ofcolor(I, c)), others)

    O = paint(I, merge(to_one))
    O = paint(O, merge(from_one))
    return O


def dsl4(I):
    # mirror the lower‑right quadrant onto the upper‑left and replace 1’s
    h, w = shape(I)

    # coordinates of the lower‑right quadrant
    br = {(r, c) for r in range(h // 2, h) for c in range(w // 2, w)}
    quad = subgrid(br, I)

    # mirror vertically then horizontally (counter‑diagonal symmetry)
    mir = hmirror(vmirror(quad))

    # place the mirrored quadrant in the upper‑left corner
    moved = shift(mir, (-h // 2, -w // 2))

    # replace every 1 in the original grid by the colour found at that cell in the mirrored patch
    O = paint(I, recolor(1, moved))
    return O


def dsl5(I):
    # column‑wise “visibility” – a 1 exchanges colour with the first different colour it sees
    ones   = ofcolor(I, 1)
    fours  = ofcolor(I, 4)
    others = sfilter(palette(I), lambda c: c not in {0, 1, 4})

    sees_up = lambda c: {
        cell for cell in ones
        if intersection({(r, cell[1]) for r in range(cell[0] + 1, len(I))},
                        ofcolor(I, c))
           and not intersection({(r, cell[1]) for r in range(cell[0] + 1, len(I))},
                               fours)}

    sees_down = lambda c: {
        cell for cell in ofcolor(I, c)
        if intersection({(r, cell[1]) for r in range(0, cell[0])},
                        ones)
           and not intersection({(r, cell[1]) for r in range(0, cell[0])},
                               fours)}

    recol_up   = apply(lambda c: recolor(c, sees_up(c)),   others)
    recol_down = apply(lambda c: recolor(1, sees_down(c)), others)

    O = paint(I, merge(recol_up))
    O = paint(O, merge(recol_down))
    return O
