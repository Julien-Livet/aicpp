def dsl1(I):
    # recolor every 1 that lies inside the bounding box of the 8‑pixels
    patch8      = ofcolor(I, 8)                     # all 8 cells
    bbox8       = backdrop(patch8)                  # their bounding box
    target      = intersection(ofcolor(I, 1), bbox8) # 1‑cells inside that box
    O = fill(I, 3, target)                         # paint them 3
    return O


def dsl2(I):
    # like dsl1 but first remove the 8‑pixels themselves, i.e. use the interior
    patch8      = ofcolor(I, 8)
    bbox8       = backdrop(patch8)
    interior8   = difference(bbox8, patch8)          # box without the 8‑border
    target      = intersection(ofcolor(I, 1), interior8)
    O = fill(I, 3, target)
    return O


def dsl3(I):
    # recolor every 1 that is orthogonally adjacent to an 8
    neigh8      = mapply(neighbors, ofcolor(I, 8))   # all orthogonal neighbours of 8’s
    target      = intersection(ofcolor(I, 1), neigh8)
    O = fill(I, 3, target)
    return O


def dsl4(I):
    # recolor 1‑cells that are inside the convex hull of the 8‑pixels.
    # The convex hull is approximated by the set of cells that are
    # reachable from an 8 by repeatedly moving orthogonally
    # (i.e. the whole connected component of 0/1/8 that contains an 8).
    patch8      = ofcolor(I, 8)
    # start from the 8‑cells and grow outward while staying on non‑background (≠0)
    def grow(p):
        return union(p, mapply(neighbors, p))
    # a fixed‑point expansion (here three iterations are enough for the given sizes)
    expanded    = grow(grow(grow(patch8)))
    target      = intersection(ofcolor(I, 1), expanded)
    O = fill(I, 3, target)
    return O


def dsl5(I):
    # recolor 1‑cells whose Manhattan distance to the nearest 8 is exactly 2.
    # First build the set of cells at distance 1 from any 8,
    # then expand it one more step.
    neigh1      = mapply(neighbors, ofcolor(I, 8))
    neigh2      = mapply(neighbors, neigh1)
    target      = intersection(ofcolor(I, 1), neigh2)
    O = fill(I, 3, target)
    return O
