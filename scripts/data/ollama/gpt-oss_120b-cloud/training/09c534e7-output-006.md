def dsl1(I):
    # colour‑filtering: global box of each colour >1, recolour the 1‑cells inside it
    seeds   = sfilter(palette(I), lambda c: c > 1)
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)
    patches = prapply(
                lambda c, b: recolor(c,
                                     intersection(b, ofcolor(I, 1))),
                seeds, boxes)
    O = paint(I, merge(patches))
    return O


def dsl2(I):
    # geometric: use extreme corners of each colour to build a rectangle
    seeds   = sfilter(palette(I), lambda c: c > 1)
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                box(connect(ulcorner(ofcolor(I, c)),
                                      lrcorner(ofcolor(I, c))),
                                ofcolor(I, 1))),
                seeds)
    O = paint(I, merge(patches))
    return O


def dsl3(I):
    # object‑extraction: merge all objects of the same colour,
    # take their joint box and recolour the 1‑cells inside it
    seeds   = sfilter(palette(I), lambda c: c > 1)
    allobj  = fgpartition(I)                                 # all connected components
    unions  = apply(lambda c: merge(colorfilter(allobj, c)), seeds)
    boxes   = apply(lambda o: box(o), unions)
    patches = prapply(
                lambda c, b: recolor(c,
                                     intersection(b, ofcolor(I, 1))),
                seeds, boxes)
    O = paint(I, merge(patches))
    return O


def dsl4(I):
    # relational/structural: erase the original 1‑cells, then paint the
    # part of each colour’s box that originally contained a 1‑cell
    seeds   = sfilter(palette(I), lambda c: c > 1)
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)
    masks   = apply(lambda b: intersection(b, ofcolor(I, 1)), boxes)
    patches = prapply(lambda c, m: recolor(c, m), seeds, masks)
    base    = cover(I, ofcolor(I, 1))                         # grid without any 1‑cell
    O = paint(base, merge(patches))
    return O


def dsl5(I):
    # relational with underpaint: similar to dsl4 but uses underpaint instead of paint
    seeds   = sfilter(palette(I), lambda c: c > 1)
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)
    masks   = apply(lambda b: intersection(b, ofcolor(I, 1)), boxes)
    patches = prapply(lambda c, m: recolor(c, m), seeds, masks)
    base    = cover(I, ofcolor(I, 1))
    O = underpaint(base, merge(patches))                    # only fills the cleared 1‑cells
    return O
