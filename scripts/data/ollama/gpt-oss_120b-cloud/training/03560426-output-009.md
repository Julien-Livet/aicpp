def dsl1(I):
    # GLOBAL – remove all empty outer rows/cols, delete interior empty
    # rows/cols, rotate 90° clockwise and draw the result on a fresh canvas.
    trimmed   = trim(I)
    compact  = compress(compress(trimmed))
    rotated  = rot90(compact)
    empty    = canvas(0, shape(I))
    return paint(empty, asobject(rotated))


def dsl2(I):
    # OBJECT‑LEVEL – keep only the largest coloured object (by cell count)
    # and move it to the top‑left corner (gravity up then left).
    objs        = fgpartition(I)
    largest_obj = argmax(objs, lambda o: size(o))
    pts         = toindices(largest_obj)

    # move up
    up_vec  = gravitate(pts, {(0, c) for c in interval(0, width(I), 1)})
    pts_up  = shift(pts, up_vec)

    # then move left
    left_vec = gravitate(pts_up, {(r, 0) for r in interval(0, height(I), 1)})
    final    = shift(pts_up, left_vec)

    empty = canvas(0, shape(I))
    return paint(empty, recolor(color(largest_obj), final))


def dsl3(I):
    # COLOR‑FILTER – keep the three most frequent non‑zero colours,
    # erase everything else and then compress the remaining pattern.
    nonzero   = palette(I) - {0}
    ordered   = order(nonzero, lambda c: -colorcount(I, c))
    top3      = {first(apply(lambda _: ordered, interval(0, 3, 1)))}
    mask      = ofcolor(I, top3)
    filtered  = underpaint(I, mask)
    compact   = compress(compress(trim(filtered)))
    empty     = canvas(0, shape(I))
    return paint(empty, asobject(compact))


def dsl4(I):
    # GEOMETRIC – reflect across the main diagonal (transpose) and then
    # remove any interior empty rows/cols, finally render on a blank canvas.
    transposed = dm mirror(I)          # diagonal mirror = transpose
    compact   = compress(compress(trim(transposed)))
    empty     = canvas(0, shape(I))
    return paint(empty, asobject(compact))


def dsl5(I):
    # RELATIONAL – push every coloured object upward until it touches the
    # top edge, then push the whole union leftward until it touches the
    # left edge, preserving the original vertical ordering of the objects.
    objs = fgpartition(I)

    # lift each object to the top
    lifted = apply(lambda o:
                   recolor(color(o),
                           shift(toindices(o),
                                 gravitate(toindices(o),
                                           {(0, c) for c in interval(0, width(I), 1)}))),
                   objs)

    united_up = merge(lifted)

    # shift the whole union left
    left_vec = gravitate(toindices(united_up),
                         {(r, 0) for r in interval(0, height(I), 1)})
    final    = shift(toindices(united_up), left_vec)

    empty = canvas(0, shape(I))
    return paint(empty, recolor(mostcolor(I), final))
