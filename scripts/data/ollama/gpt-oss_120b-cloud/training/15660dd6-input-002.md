def dsl1(I):
    # ── Global: compress outer uniform border, take the centre 5‑row band,
    #          then recolour the four original colours.
    interior = compress(I)                                 # strip all outer 8‑border
    # height of interior is 17 → the middle 5 rows start at row 5
    middle    = crop(interior, (5, 0), (5, 17))            # keep rows 5‑9, all columns
    step1 = replace(middle, 1, 9)                         # 1 → 9
    step2 = replace(step1, 8, 5)                           # 8 → 5
    step3 = replace(step2, 3, 8)                         # 3 → 8
    O      = replace(step3, 2, 7)                          # 2 → 7
    return O
```

def dsl2(I):
    # ── Object‑extraction: the only colour that does not touch the outer 8‑border
    #          is colour 1 (the central horizontal band).  Extract its bbox,
    #          enlarge it horizontally to the full width, and recolour.
    band1      = ofcolor(I, 1)                             # all 1‑cells
    bbox1      = backdrop(band1)                           # minimal rectangle of the band
    fullwidth  = box(bbox1)                               # same height, full width of the grid
    band_grid  = subgrid(fullwidth, I)                     # extract that rectangle
    step1 = replace(band_grid, 1, 9)
    step2 = replace(step1, 8, 5)
    step3 = replace(step2, 3, 8)
    O      = replace(step3, 2, 7)
    return O
```
