def dsl1(I):
    # 1️⃣ GLOBAL TRANSFORM – remove all‑zero border rows/columns
    O = trim(I)
    return O


def dsl2(I):
    # 2️⃣ OBJECT TEMPLATE – use the shape of colour 1 as a stencil.
    #    For every other colour c, paste that stencil coloured c
    #    at the original position of the first pixel of colour c,
    #    then trim the result.
    objs        = fgpartition(I)                               # all objects
    tmpl_objs   = colorfilter(objs, 1)                         # only the 1‑object(s)
    tmpl        = normalize(first(tmpl_objs))                  # stencil at (0,0)
    other_objs  = difference(objs, tmpl_objs)                  # everything but colour 1
    # recolour the stencil for each remaining object and move it to the
    # object's upper‑left corner
    pasted      = apply(
                    lambda o: recolor(
                                color(o),
                                shift(tmpl, ulcorner(o))
                              ),
                    other_objs
                 )
    # union of all coloured stencil cells
    all_cells   = combine(pasted, initset())
    # paint them on a blank canvas of the original size and crop
    blank       = canvas(0, shape(I))
    painted     = paint(blank, all_cells)
    O           = trim(painted)
    return O


def dsl3(I):
    # 3️⃣ COLOR FILTER – delete every colour 1, keep everything else unchanged,
    #    then compress the picture (remove empty outer rows/columns)
    no_one      = replace(I, 1, 0)            # erase 1’s
    O           = trim(no_one)                # crop to remaining content
    return O


def dsl4(I):
    # 4️⃣ GEOMETRIC – mirror the whole picture about the vertical axis
    #    and then remove any all‑zero border rows/columns
    mirrored    = vmirror(I)                  # left‑right flip
    O           = trim(mirrored)
    return O


def dsl5(I):
    # 5️⃣ RELATIONAL – move each coloured object so that its centre of mass
    #    coincides with the top‑left corner (0,0); afterwards crop.
    objs        = fgpartition(I)                               # all objects
    shifted     = apply(
                    lambda o: shift(o, negate(centerofmass(o))),
                    objs
                 )
    blank       = canvas(0, shape(I))
    painted     = apply(lambda o: paint(blank, o), shifted)      # paint each shifted object
    O           = first(painted)                               # the fully painted grid
    O           = trim(O)                                      # crop excess zeros
    return O
