def dsl1(I):
    # Global transformation: rotate 180°, replace 6→9, rotate back.
    O = rot180(I)
    O = replace(O, 6, 9)
    O = rot180(O)
    return O


def dsl2(I):
    # Geometric reasoning: double mirroring (equivalent to 180° rotation),
    # then colour change, then undo the mirrors.
    O = hmirror(I)
    O = vmirror(O)               # now the grid is rotated 180°
    O = replace(O, 6, 9)        # colour change
    O = vmirror(O)               # undo vertical mirror
    O = hmirror(O)               # undo horizontal mirror
    return O


def dsl3(I):
    # Colour‑filtering with a temporary colour.
    # 6 → 8 (temp), rotate, 8 → 9, rotate back.
    O = replace(I, 6, 8)         # 6 → temporary colour 8
    O = rot180(O)                # 180° rotation
    O = replace(O, 8, 9)         # temporary colour → final colour 9
    O = rot180(O)                # undo rotation
    return O


def dsl4(I):
    # Object‑centric: extract 6‑cells, recolour their 180° counterparts to 9,
    # and overlay onto the grid where 6‑cells were cleared.
    base = replace(I, 6, 0)               # clear original 6‑cells
    rot  = rot180(I)                       # rotate whole grid
    rot  = replace(rot, 6, 9)              # turn rotated 6‑cells into 9
    nine_patch = ofcolor(rot, 9)            # locations of the new 9’s
    nine_obj   = recolor(9, nine_patch)    # make them an object (value,coords)
    O = paint(base, nine_obj)               # paint 9‑object onto cleared grid
    return O


def dsl5(I):
    # Relational / structural reasoning: cover the original 6‑cells,
    # take the symmetric 6‑cells from the rotated grid, recolour them 9,
    # and underpaint them where the background now exists.
    patch6   = ofcolor(I, 6)                # positions of colour 6
    base     = cover(I, patch6)             # remove 6‑cells (set to 0)
    rot      = rot180(I)                    # rotate whole grid
    nine_idx = ofcolor(rot, 6)              # symmetric positions of the 6‑cells
    nine_obj = recolor(9, nine_idx)        # turn those positions into colour 9
    O = underpaint(base, nine_obj)          # paint 9’s only on background cells
    return O
