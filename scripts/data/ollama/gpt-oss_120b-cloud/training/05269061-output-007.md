def dsl1(I):
    # OBJECT‑CENTRIC → column‑mod‑3 tiling.
    # 1) find the three single‑colour objects.
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    # order them by (row+col) of their centre of mass → residue 0,1,2
    sum_of = lambda o: add(centerofmass(o)[0], centerofmass(o)[1])
    o0 = argmin(objs, sum_of)
    o1 = argmin(difference(objs, {o0}), sum_of)
    o2 = argmin(difference(objs, {o0, o1}), sum_of)

    c0 = color(o0); c1 = color(o1); c2 = color(o2)

    # build a full‑height column of a given colour
    col_obj = lambda col, colc: recolor(colc,
                    toobject(apply(lambda r: (r, col), interval(0, height(I), 1)), I))

    # three families of columns according to column‑mod‑3
    cols0 = interval(0, width(I), 3)
    cols1 = interval(1, width(I), 3)
    cols2 = interval(2, width(I), 3)

    obj0 = mapply(combine,
                  apply(lambda c: col_obj(c, c0), cols0))
    obj1 = mapply(combine,
                  apply(lambda c: col_obj(c, c1), cols1))
    obj2 = mapply(combine,
                  apply(lambda c: col_obj(c, c2), cols2))

    empty = canvas(0, shape(I))
    G0 = paint(empty, obj0)
    G1 = paint(G0,    obj1)
    O  = paint(G1,    obj2)
    return O


def dsl2(I):
    # PROTOTYPE‑3×3 → full‑grid tiling.
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)
    sum_of = lambda o: add(centerofmass(o)[0], centerofmass(o)[1])
    o0 = argmin(objs, sum_of); o1 = argmin(difference(objs,{o0}), sum_of)
    o2 = argmin(difference(objs,{o0,o1}), sum_of)
    c0 = color(o0); c1 = color(o1); c2 = color(o2)

    # colour for a given residue (r+c) % 3
    col_of = lambda r,c: branch(equality(modulo(add(r,c),3),0),c0,
                     branch(equality(modulo(add(r,c),3),1),c1,c2))

    rows = interval(0,3,1); cols = interval(0,3,1)

    # all coordinates of the 3×3 prototype
    coords = combine(apply(lambda r: apply(lambda c: (r,c), cols), rows))

    # coloured prototype object
    proto = recolor(0, toobject(coords, I))
    proto = replace(proto, 0,
            combine(apply(lambda rc:
                   initset((col_of(rc[0],rc[1]), rc)), coords))

    # tile the prototype over the whole grid
    roffs = interval(0, height(I), 3)

    empty = canvas(0, shape(I))
    tiled = apply(lambda off: paint(empty, shift(proto, off)), anchors)
    O = first(tiled)                      # any order gives the full tiling
    return O


def dsl3(I):

    mask0 = mfilter(all_coords,
            lambda rc: equality(modulo(add(rc[0],rc[1]),3),0)

    mask2 = mfilter(all_coords,

    obj1 = recolor(c1, mask1)
    obj2 = recolor(c2, mask2)

    empty = canvas(0, shape(I))
    O = paint(paint(paint(empty, obj0), obj1), obj2)
    return O
