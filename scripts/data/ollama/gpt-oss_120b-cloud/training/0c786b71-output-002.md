def dsl1(I):
    # ---------- Global symmetric tiling ----------
    # 1. Rotate the whole picture by 180°.
    # 2. Build the top half by concatenating this block with its
    #    left‑right mirror.
    # 3. Obtain the bottom half by mirroring the top half vertically.
    # 4. Stack the halves.
    base   = rot180(I)                                 # 180° rotation
    top    = hconcat(base, vmirror(base))              # left‑right reflection
    bottom = hmirror(top)                               # up‑down reflection
    O = vconcat(top, bottom)
    return O


def dsl2(I):
    # ---------- Colour‑wise construction ----------
    # Perform the same global tiling as dsl1, but build it separately
    # for every colour and finally merge the layers.
    base   = rot180(I)
    colors = palette(base)

    def tiled_layer(col):
        mask   = ofcolor(base, col)                     # cells of this colour
        solid  = canvas(col, shape(base))               # solid canvas
        layer  = paint(solid, toobject(mask, base))     # colour‑only layer
        top    = hconcat(layer, vmirror(layer))
        bottom = hmirror(top)
        return vconcat(top, bottom)

    tiled  = apply(tiled_layer, colors)                # tuple of layers
    O = merge(tiled)                                   # union of all layers
    return O


def dsl3(I):
    # ---------- Relational quadrant placement ----------
    # Create the four quadrants by mirroring the 180° rotated picture
    # and shift each quadrant to its proper location.
    base = rot180(I)                         # 180° rotation of input
    h, w = shape(base)

    tl = base                                 # top‑left
    tr = vmirror(base)                        # top‑right (LR‑mirror)
    bl = hmirror(base)                        # bottom‑left (UD‑mirror)
    br = hmirror(vmirror(base))                # bottom‑right (both mirrors)

    # shift the three non‑origin quadrants
    tr = shift(tr, (0, w))
    bl = shift(bl, (h, 0))
    br = shift(br, (h, w))

    O = merge({tl, tr, bl, br})               # overlay all quadrants
    return O


def dsl4(I):
    # ---------- Periodic replication with cropping ----------
    # 1. Produce the 2×2 periodic pattern by repeating the 180° rotated
    #    picture together with its mirrors.
    # 2. Crop the central 2×2 block (which is exactly the desired output).
    base = rot180(I)

    # build a 2×2 repetition
    row0 = hconcat(base, vmirror(base))
    row1 = hconcat(hmirror(base), hmirror(vmirror(base)))
    period = vconcat(row0, row1)

    # the whole picture already has the correct size, but we keep the
    # cropping step to emphasise the periodic‑replication idea.
    O = crop(period, (0, 0), shape(period))
    return O

    # Treat the whole picture as a single object, mirror it in the required
