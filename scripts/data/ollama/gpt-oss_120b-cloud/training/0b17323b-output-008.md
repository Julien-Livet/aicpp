def dsl1(I):
    # GLOBAL – rotate, extend as a vertical line, rotate back
    R          = rot90(I)                                             # turn main diagonal into a column
    ones_R     = ofcolor(R, 1)                                        # 1‑pixels are now on the same column
    last_one_R = astuple(lowermost(ones_R), rightmost(ones_R))       # lowest 1 in the rotated view
    start_R    = increment(last_one_R)                               # first empty cell below it
    col_R      = shoot(start_R, astuple(0, 1))                        # vertical ray downwards
    R_filled   = fill(R, 2, col_R)                                    # paint the missing cells with 2
    O          = rot270(R_filled)                                      # restore original orientation
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION – build the whole diagonal and keep only the missing part
    ones       = ofcolor(I, 1)
    first_one  = ulcorner(ones)                                       # upper‑left 1
    last_one   = astuple(lowermost(ones), rightmost(ones))
    existing   = connect(first_one, last_one)                         # diagonal segment that already exists
    full_diag  = shoot(astuple(0, 0), subtract(shape(I), astuple(1, 1))  # diagonal from (0,0) to bottom‑right
    missing    = difference(full_diag, existing)                       # cells that have to become 2
    O          = fill(I, 2, missing)
    return O


def dsl3(I):
    # COLOR‑FILTERING – recolour the missing diagonal to 2 and under‑paint it
    ones       = ofcolor(I, 1)
    last_one   = astuple(lowermost(ones), rightmost(ones))
    full_diag  = shoot(astuple(0, 0), subtract(shape(I), astuple(1, 1)))
    part_up_to = shoot(astuple(0, 0), last_one)
    mask_idx   = difference(full_diag, part_up_to)                     # indices of the missing cells
    mask_obj   = recolor(2, mask_idx)                                 # turn them into a 2‑object
    O          = underpaint(I, mask_obj)                               # paint only where background is 0
    return O


def dsl4(I):
    # RELATIONAL / STRUCTURAL – derive the unit step from the span of the 1‑pixels
    ones       = ofcolor(I, 1)
    first_one  = ulcorner(ones)                                       # first 1 on the diagonal
    last_one   = astuple(lowermost(ones), rightmost(ones))
    delta      = subtract(last_one, first_one)                         # overall displacement
    unit_step  = divide(delta, delta)                                  # normalise to (1,1)
    start      = increment(last_one)                                   # cell right after the last 1
    line       = shoot(start, unit_step)                              # continue the diagonal
    O          = fill(I, 2, line)
    return O


def dsl5(I):
    # GLOBAL – use a full‑diagonal mask built with CONNECT and keep the tail part
    ones       = ofcolor(I, 1)
