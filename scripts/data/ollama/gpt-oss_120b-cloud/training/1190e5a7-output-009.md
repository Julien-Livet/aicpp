def dsl1(I):
    # Global: periodicity of the non‑dominant colour mask,
    # fallback to the full grid size when there is no repetition.
    dom      = mostcolor(I)
    non_mask = difference(asindices(I), ofcolor(I, dom))
    obj      = toobject(non_mask, I)
    vsize    = vperiod(obj)
    hsize    = hperiod(obj)
    vsize    = branch(equality(vsize, ZERO), shape(I)[0], vsize)
    hsize    = branch(equality(hsize,   ZERO), shape(I)[1], hsize)
    O        = canvas(dom, (vsize, hsize))
    return O


def dsl2(I):
    # Object‑extraction: bounding box of the non‑dominant colour region.
    dom      = mostcolor(I)
    non_mask = difference(asindices(I), ofcolor(I, dom))
    obj      = toobject(non_mask, I)          # minimal rectangle covering the mask
    vsize    = height(obj)
    hsize    = width(obj)
    O        = canvas(dom, (vsize, hsize))
    return O


def dsl3(I):
    # Colour‑filter → clear non‑dominant → compress → use the compressed size,

    vsize    = branch(equality(comp, ()), shape(I)[0], shape(comp)[0])
    hsize    = branch(equality(comp, ()), shape(I)[1], shape(comp)[1])
    O        = canvas(dom, (vsize, hsize))
    return O
