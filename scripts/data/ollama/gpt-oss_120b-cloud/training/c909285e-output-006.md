def dsl1(I):
    # ── object‑centric: largest non‑background object ──
    bg       = mostcolor(I)                                            # background colour
    all_objs = objects(I, False, False, False)                         # every colour object

    tl       = ulcorner(bb)                                            # top‑left corner of the box

    O        = crop(I, tl, astuple(h, w))                               # extract the sub‑grid
