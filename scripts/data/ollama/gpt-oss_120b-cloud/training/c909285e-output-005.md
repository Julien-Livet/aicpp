def dsl1(I):
    # Global periodicity – crop the top‑left tile whose size is the
    # vertical and horizontal periods of the whole grid.
    hp = hperiod(asindices(I))                     # horizontal period (tile width)
    vp = vperiod(asindices(I))                     # vertical period (tile height)
    O  = crop(I, astuple(ZERO, ZERO), astuple(vp, hp))
    return O


def dsl2(I):
    # Anchor on the first cell that is not the background colour 8,
    # then crop a tile of the global periods.
    bg      = ofcolor(I, 8)                        # background cells
    nonbg   = difference(asindices(I), bg)         # cells of the pattern
    anchor  = first(nonbg)                         # top‑left occurrence of the pattern
    hp = hperiod(asindices(I))
    vp = vperiod(asindices(I))
    O  = crop(I, anchor, astuple(vp, hp))
    return O


def dsl3(I):
    # Split the grid into equal vertical and horizontal strips using the
    # periods, and take the first block (top‑left tile).
    hp      = hperiod(asindices(I))
    vp      = vperiod(asindices(I))
    n_vert  = divide(width(I), hp)                  # how many vertical slices
    n_horiz = divide(height(I), vp)                  # how many horizontal slices
    vparts  = vsplit(I, n_vert)                     # vertical slices
    firstcol = first(vparts)                        # left‑most slice
    hparts   = hsplit(firstcol, n_horiz)            # horizontal slices of that column
    O       = first(hparts)                         # top‑left tile
    return O


def dsl4(I):
    # Object‑centric: keep the largest non‑background (≠8) object,
    # take its tight bounding box and return that sub‑grid.
    objs    = objects(I, False, False, True)                     # all objects
    non8    = mfilter(objs, lambda o: not equal(color(o), 8))   # discard background
    obj     = first(non8)                                        # any non‑background object
    bbox    = backdrop(obj)                                      # its bounding box (indices)
    tl      = ulcorner(bbox)                                    # top‑left corner of the box
    h       = add(subtract(lowermost(bbox), uppermost(bbox)), ONE)
    w       = add(subtract(rightmost(bbox), leftmost(bbox)), ONE)
    O       = crop(I, tl, astuple(h, w))
    return O


def dsl5(I):
    # Colour‑filtering: turn the background colour 8 into 0,
    # then crop the minimal region that contains all non‑zero cells.
    O0      = replace(I, 8, 0)                                 # background → 0

    bbox    = backdrop(nz)                                     # bounding box of those cells
    tl      = ulcorner(bbox)

    w       = add(subtract(rightmost(bbox), leftmost(bbox)), ONE)
