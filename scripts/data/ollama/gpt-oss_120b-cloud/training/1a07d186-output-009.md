def dsl1(I):
    # GLOBAL – remove all empty rows and empty columns.
    # First remove empty rows, then rotate, remove rows (i.e. columns), rotate back.
    O = rot270(compress(rot90(compress(I))))
    return O


def dsl2(I):
    # OBJECT – keep only the single largest object (by cell count).
    OBJS     = fgpartition(I)                                 # all coloured objects
    LARGEST  = argmax(OBJS, lambda o: size(o))                # object with most cells
    O        = underpaint(canvas(0, shape(I)), LARGEST)        # draw it on a blank canvas
    return O


def dsl3(I):
    # COLOR‑FILTER – retain objects whose cells form a continuous horizontal run
    # (horizontal period equals 1).
    OBJS     = fgpartition(I)
    CONT     = sfilter(OBJS, lambda o: hperiod(o) == 1)         # continuous rows
    O        = underpaint(canvas(0, shape(I)), merge(CONT))
    return O


def dsl4(I):
    # GEOMETRIC – overlay the grid with its vertical mirror and keep only the
    # intersecting cells (common part of original and mirror).
    MIR      = vmirror(I)                                       # vertical mirror
    INTER    = intersection(asobject(I), asobject(MIR))          # common cells
    O        = paint(canvas(0, shape(I)), INTER)                # draw intersection
    return O


def dsl5(I):
    # RELATIONAL – normalise every object (move its upper‑left corner to the origin)
    # and render the collection on a blank canvas.
    OBJS     = fgpartition(I)
    NORMED   = apply(normalize, OBJS)                          # shift each to (0,0)
    O        = underpaint(canvas(0, shape(I)), merge(NORMED))
    return O
