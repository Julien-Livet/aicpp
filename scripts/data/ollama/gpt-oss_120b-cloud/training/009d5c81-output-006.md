def dsl1(I):
    # GLOBAL‑TRANSFORMATION
    #   Count all 8‑pixels.  The larger the count the higher the colour
    #   (7 = large, 3 = medium, 2 = small).  Erase the 1‑shape.
    dens = colorcount(I, 8)                                 # total 8‑pixels
    col  = branch(greater(dens, 40), 7,
           branch(greater(dens, 20), 3, 2))                # 7/3/2 by size
    O    = replace(I, 8, col)                              # recolour 8‑object
    O    = replace(O, 1, 0)                                # delete the 1‑shape
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION
    #   Use the bounding‑box area of the 8‑object to decide the colour.
    p8   = ofcolor(I, 8)                                   # 8‑object cells
    h    = height(p8)                                       # vertical size
    w    = width(p8)                                        # horizontal size
    area = multiply(h, w)                                   # bbox area
    col  = branch(greater(area, 30), 7,
           branch(greater(area, 15), 3, 2))                  # 7/3/2 by area
    O    = replace(I, 8, col)
    O    = replace(O, 1, 0)
    return O


def dsl3(I):
    # COLOR‑FILTERING (boundary 8‑pixels)
    #   Keep only those 8‑pixels that touch the interior background.
    p8          = ofcolor(I, 8)                              # all 8 cells
    interior0   = intersection(ofcolor(I, 0), backdrop(p8))   # 0‑cells inside bbox
    boundary8   = intersection(p8, box(interior0))           # 8‑cells on that interior edge
    cnt         = size(boundary8)                           # how many such cells
    col         = branch(greater(cnt, 30), 7,
                 branch(greater(cnt, 10), 3, 2))               # 7/3/2 by perimeter count
    O           = replace(I, 8, col)
    O           = replace(O, 1, 0)
    return O


def dsl4(I):

    #   bounding box in both vertical and horizontal directions.

    vert_below  = greater(com[0], cbox[0])                   # COM lower than box centre?
    horiz_right = greater(com[1], cbox[1])                   # COM right of box centre?
    col = branch(vert_below,
