def dsl1(I):
    # GLOBAL‑BBOX‑FILL‑EXCL:
    # fill the 5‑bounding box with 8,
    # but keep any whole row or whole column that contains ≥2 fives unchanged.
    five      = ofcolor(I, 5)                               # all 5‑pixels
    box       = backdrop(five)                               # bounding‑box of the 5’s
    O1        = fill(I, 8, box)                             # paint the whole box with 8

    # rows that contain at least two 5’s
    rows_multi = {r for r in asindices(I) if size(filter(lambda p: p[0]==r, five)) > 1}
    # columns that contain at least two 5’s
    cols_multi = {c for c in asindices(I) if size(filter(lambda p: p[1]==c, five)) > 1}

    # whole rows / columns as patches
    row_patch = combine(*{shoot((r, 0), RIGHT) for r in rows_multi})
    col_patch = combine(*{shoot((0, c), DOWN)  for c in cols_multi})
    # restore original colours on those rows / columns
    O2 = paint(O1, toobject(combine(row_patch, col_patch), I))
    # finally restore the original 5’s (they may have been overwritten)
    O  = paint(O2, toobject(five, O2))
    return O

    O1      = fill(I, 8, region)                    # colour the ring with 8
    O       = paint(O1, toobject(five, O1))         # restore the original 5’s
    return O
