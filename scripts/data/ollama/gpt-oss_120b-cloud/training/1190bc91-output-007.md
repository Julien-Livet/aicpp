def dsl1(I):
    # Global geometric transformation: rotate 90° clockwise then mirror horizontally.
    O = hmirror(rot90(I))
    return O


def dsl2(I):
    # Object‑level transformation: take the object whose colour is the
    # most frequent, normalise its position and paint it on a blank canvas.
    main_col   = mostcommon(I)
    objs       = fgpartition(I, False, False, True)          # all foreground objects
    main_objs  = colorfilter(objs, main_col)                # keep only those of the dominant colour
    obj        = first(main_objs)                           # there should be exactly one
    norm_obj   = normalize(obj)                              # shift to the origin
    O = paint(canvas(0, shape(I)), norm_obj)
    return O


def dsl3(I):
    # Colour‑filtering transformation: replace every background (0) cell
    # with the most common non‑zero colour in the grid.
    most = mostcommon(I)
    O = replace(I, 0, most)
    return O


def dsl4(I):
    # Geometric reasoning: mirror the whole picture along the main diagonal.
    # (dmirror = diagonal mirror)
    O = dm(I)
    return O


def dsl5(I):
    # Relational/structural reasoning: move every cell that is NOT the
    # most common colour one step towards the cluster of the most common colour.
    main_col   = mostcommon(I)
    main_patch = ofcolor(I, main_col)                        # dominant‑colour cells
    all_cells   = asindices(I)                               # every cell index
    other_patch = difference(all_cells, main_patch)          # everything else
    direction   = gravitate(other_patch, main_patch)         # direction toward the dominant cluster
    moved       = move(I, other_patch, direction)             # shift the minority cells
    # paint the dominant colour first, then the moved minority cells
    O = paint(paint(canvas(0, shape(I)), main_patch), moved)
    return O
