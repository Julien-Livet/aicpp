def dsl(I):
    objs = objects(I, False, False, True)
    merged = merge(objs)
    top = topmost(merged)
    left = leftmost(merged)
    origin = astuple(top, left)

    box = subgrid(merged, I)
    hb = height(box); wb = width(box)

    base = canvas(0, (hb, wb))

    rows_pat = merge(apply(
        lambda o: fill(base,
                       color(o),
                       {(r,c) for (r,c) in asindices(base) if r == subtract(center(o), origin)[0]}),
        objs))

    cols_pat = merge(apply(
        lambda o: fill(base,
                       color(o),
                       {(r,c) for (r,c) in asindices(base) if c == subtract(center(o), origin)[1]}),
        objs))

    pat = branch(greater(wb, hb), rows_pat, cols_pat)

    result = branch(greater(wb, hb),
                    vupscale(pat, divide(height(I), hb)),
                    hupscale(pat, divide(width(I), wb)))

    return result

                  if r == subtract(center(let_o), origin)[0]}),
        palette(I)))

    # columns pattern built colour‑by‑colour
    cols_pat  = merge(apply(
        lambda col:
            let_o = first(colorfilter(objs, col));
            fill(base,
                 col,
                 {(r, c) for (r, c) in asindices(base)
                  if c == subtract(center(let_o), origin)[1]}),
        palette(I)))

    pat       = branch(greater(wb, hb), rows_pat, cols_pat)

    result    = branch(greater(wb, hb),
                     vupscale(pat, divide(height(I), hb)),
                     hupscale(pat, divide(width(I), wb)))
    return result



def dsl5(I):
    # RELATIONAL‑REPEAT – use vertical distance between distinct rows as the repeat unit
    objs      = objects(I, False, False, True)
    merged    = merge(objs)
    top       = topmost(merged)
    left      = leftmost(merged)
    origin    = astuple(top, left)

    slice_    = subgrid(merged, I)
    hb, wb    = height(slice_), width(slice_)

    rows_set  = {center(o)[0] for o in objs}
    drow      = branch(sizefilter(rows_set, 2),
                      valmin({abs(r2 - r1) for r1 in rows_set for r2 in rows_set
                              if r1 != r2 and abs(r2 - r1) > 0},
                             identity),
                      hb)                               # fallback to slice height

    base      = canvas(0, (hb, wb))

    rows_pat  = merge(apply(
        lambda o: fill(base,
                       color(o),
                       {(r, c) for (r, c) in asindices(base)
                        if r == subtract(center(o), origin)[0]}),
        objs))

    # use the computed row‑spacing as the vertical scaling factor
    result    = vupscale(rows_pat, divide(height(I), drow))
    return result
