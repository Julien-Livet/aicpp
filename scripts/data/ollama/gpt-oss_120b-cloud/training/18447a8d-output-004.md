def dsl1(I):
    # RELATIONAL MOVE : any non‑7/8 cell that touches a 7 and an 8
    # moves one step toward a neighbouring 7 and its original location becomes 7.
    other   = palette(I) - {7, 8}
    cand    = mapply(
                lambda c: frozenset(
                            {p for p in ofcolor(I, c)
                               if dneighbors(p) & ofcolor(I, 7)
                               and dneighbors(p) & ofcolor(I, 8)}),
                other)

    dirs    = mapply(
                lambda p: gravitate(
                            frozenset({p}),
                            frozenset({first(dneighbors(p) & ofcolor(I, 7))})),
                cand)

    O = I
    O = mapply(lambda p, d: move(O, frozenset({p}), d), zip(cand, dirs))
    O = recolor(7, merge(cand))
    return O


def dsl2(I):
    # OBJECT‑LEVEL EXCHANGE : for each non‑7/8 object, swap its colour with every
    # 7 that lies inside its bounding box.
    objs    = {o for o in objects(I, univalued=False,
                                 diagonal=False, without_bg=True)
                if color(o) not in {7, 8}}

    inside7 = mapply(lambda o: intersection(ofcolor(I, 7), box(o)), objs)

    O = I
    O = mapply(lambda p, o: fill(O, color(o), toobject(p, I)),
               zip(inside7, objs))
    O = mapply(lambda o: fill(O, 7, o), objs)
    return O


def dsl3(I):
    # GLOBAL HORIZONTAL SWAP : a 7 takes the colour of its right neighbour
    # (if that neighbour is a non‑7/8 colour) and that neighbour becomes 7.
    other   = palette(I) - {7, 8}

    to_c    = mapply(
                lambda c: frozenset(
                            {p for p in ofcolor(I, 7)
                               if add(p, RIGHT) in ofcolor(I, c)}),
                other)

    orig_c  = mapply(lambda c: ofcolor(I, c), other)

    O = I
    O = mapply(lambda c:
                fill(O, c, toobject(to_c[c], I)),
                other)

    O = mapply(lambda c:
                fill(O, 7, toobject(orig_c[c], I)),
                other)
    return O


def dsl4(I):
    # MIRROR‑SHIFT‑COMBINE + RECOLOUR “sandwiched” 7.
    mirrored = vmirror(I)
    shifted  = move(mirrored, asobject(mirrored), (0, 1))
    O = combine(I, shifted)

    # least‑frequent non‑7/8 colour in the original grid
    filler = valmin(palette(I) - {7, 8},
                   lambda c: colorcount(I, c))

    sand = frozenset({p for p in ofcolor(O, 7)
                        if add(p, LEFT)  in ofcolor(I, 8)
                        and add(p, RIGHT) in ofcolor(I, 8)})
    O = recolor(filler, sand)
    return O


def dsl5(I):
    # VERTICAL PROJECTION SWAP : in each column, the leftmost non‑7/8 cell that
    # shares the column with a 7 swaps colours with that 7.
    other   = palette(I) - {7, 8}
    others  = mapply(lambda c: ofcolor(I, c), other)

    pairs   = mapply(
                lambda p: (p,

                                   if q[1] == p[1]}))

                others)

    valid   = {pair for pair in pairs if pair is not None}

                valid)
