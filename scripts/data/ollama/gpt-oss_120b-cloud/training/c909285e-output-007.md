def dsl1(I):
    # ── global: trim outer uniform rows/columns, then fill the remaining background
    bg   = mostcolor(I)                                 # most frequent colour (the “background”)
    O0   = compress(I)                                  # drop all‑zero outer borders
    newc = mostcolor(O0)                                # most common colour after trimming
    O    = replace(O0, bg, newc)                        # change the leftover background to that colour
    return O


def dsl2(I):
    # ── object‑centric: crop the tight box that encloses every non‑background cell,
    # then colour‑fill the holes with the dominant colour inside the box
    bg      = mostcolor(I)                                          # background colour
    nb      = difference(asindices(I), ofcolor(I, bg))               # all non‑background cells
    anchor  = first(nb)                                             # top‑most‑left‑most non‑bg cell
    hp      = hperiod(nb)                                           # horizontal period of the pattern
    vp      = vperiod(nb)                                           # vertical period of the pattern
    tile    = crop(I, anchor, astuple(vp, hp))                       # the repeated tile (still contains bg)
    fillcol = mostcolor(tile)                                       # colour that is most common inside the tile
    O       = replace(tile, bg, fillcol)                             # fill the background cells
    return O


def dsl3(I):
    # ── colour‑filtering: erase the global background, then recover the pattern
    # by recolouring the erased cells with the most frequent remaining colour
    bg   = mostcolor(I)                                 # background colour
    O0   = replace(I, bg, 0)                            # erase background → 0
    O1   = compress(O0)                                 # cut away outer 0‑rows/columns
    repc = mostcolor(O1)                                # dominant colour inside the cropped area
    O    = replace(O1, 0, repc)                         # recolour the erased cells
    return O


def dsl4(I):
    # ── geometric: locate the only solid rectangular block (its outline equals the block itself),
    # crop it and fill its interior with the colour that occurs most often on its border
    bg       = mostcolor(I)                                         # background colour
    objs     = objects(I, False, False, False)                       # every coloured object
    rects    = mfilter(objs, lambda o: equality(o, box(o)))         # solid rectangles only
    rect     = first(rects)                                         # the unique rectangle
    bb       = backdrop(rect)                                       # its bounding box
    tl       = ulcorner(bb)                                         # top‑left corner of the box
    h        = add(subtract(lowermost(bb), uppermost(bb)), ONE)     # height
    w        = add(subtract(rightmost(bb), leftmost(bb)), ONE)       # width
    tile     = crop(I, tl, astuple(h, w))                            # extract the rectangle
    borderc  = mostcolor(border(tile))                               # colour that dominates the border
    O        = recolor(borderc, tile)                               # fill any interior background with that colour
    return O


def dsl5(I):
    # ── relational: use the centre of mass of all non‑background cells as the anchor,
    # take a window whose size equals the discovered vertical/horizontal periods,
    # then replace the background inside that window with the colour that appears most often
    bg      = mostcolor(I)                                          # background colour
    nb      = difference(asindices(I), ofcolor(I, bg))               # all non‑background cells
    centre  = centerofmass(nb)                                      # centre of mass of the pattern
    hp      = hperiod(nb)                                           # horizontal period
    vp      = vperiod(nb)                                           # vertical period
    # top‑left corner of the window = centre – (vp‑1, hp‑1) // 2
    offset  = astuple(divide(subtract(centre[0], subtract(vp, ONE)), TWO),
                       divide(subtract(centre[1], subtract(hp, ONE)), TWO))
    tl      = add(ulcorner(nb), offset)                             # shift from an actual cell
    tile    = crop(I, tl, astuple(vp, hp))                          # extract the tile
    fillcol = mostcolor(tile)                                       # most common non‑bg colour in the tile
    O       = replace(tile, bg, fillcol)                            # recolour background cells
    return O
