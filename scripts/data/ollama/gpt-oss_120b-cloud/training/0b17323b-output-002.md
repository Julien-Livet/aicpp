def dsl1(I):
    # ---- object‑extraction ----
    # 1. locate the whole line of 1’s (they are collinear with slope 1)
    # 2. build the complete main diagonal of the grid
    # 3. cut away the part of the diagonal that ends at the last 1
    # 4. colour the remaining cells with 2
    ones        = ofcolor(I, 1)
    last_one    = astuple(lowermost(ones), rightmost(ones))
    h, w        = shape(I)
    last_cell   = subtract(astuple(h, w), astuple(1, 1))            # (h‑1,w‑1)
    full_diag   = connect(astuple(0, 0), last_cell)                # whole main diagonal
    prefix_diag = connect(astuple(0, 0), last_one)                # part up to the last 1
    to_fill     = difference(full_diag, prefix_diag)               # cells after the last 1
    O = fill(I, 2, to_fill)
    return O


def dsl2(I):
    # ---- periodicity (relational) ----
    # 1. the three 1‑pixels are equally spaced; their column‑step equals the row‑step
    # 2. use that step as a vector (Δr,Δc) = (step, step)
    # 3. start from the cell just after the last 1 and shoot along that vector
    ones        = ofcolor(I, 1)
    step        = hperiod(ones)                       # horizontal period = column gap
    vec         = astuple(step, step)                 # (Δr, Δc)
    last_one    = astuple(lowermost(ones), rightmost(ones))
    start       = increment(last_one)
    line        = shoot(start, vec)                   # cells to the border
    O = fill(I, 2, line)
    return O


def dsl3(I):
    # ---- object‑based shift ----
    # 1. the 1‑pixels form a line; the displacement between any two neighbours
    #    is the translation that needs to be repeated
    # 2. shift the last 1 by that displacement and keep shooting until the border
    ones      = ofcolor(I, 1)
    pts       = tuple(sorted(ones))
    v         = subtract(pts[1], pts[0])               # vector between successive 1’s
    start     = add(pts[-1], v)                       # first empty cell after the line
    line      = shoot(start, v)
    O = fill(I, 2, line)
    return O


def dsl4(I):
    # ---- global transformation (crop + underpaint) ----
    # 1. cut out the sub‑grid that starts right after the last 1 and reaches the bottom‑right
    # 2. fill that whole sub‑grid with 2 (it is completely background)
    # 3. paste the coloured sub‑grid back only on the background cells (underpaint)
    ones        = ofcolor(I, 1)
    last_one    = astuple(lowermost(ones), rightmost(ones))
    start       = increment(last_one)
    h, w        = shape(I)
    dims        = subtract(astuple(h, w), start)      # (remaining rows, remaining cols)
    sub_canvas  = canvas(2, dims)                     # a block of 2’s
    O           = underpaint(I, toobject(ofcolor(sub_canvas, 2), I))
    return O


def dsl5(I):
    # ---- colour‑filtering (fill only the diagonal beyond the farthest 1) ----
    # 1. the farthest 1 from the origin determines how far the diagonal is already occupied
    # 2. the main diagonal of the whole grid is built, then the prefix up to that distance
    #    is removed; the rest is coloured 2.
    ones        = ofcolor(I, 1)
    max_dist   = valmax(ones, lambda p: manhattan(initset((0, 0)), initset(p)))
    h, w       = shape(I)
    last_cell  = subtract(astuple(h, w), astuple(1, 1))
    full_diag  = connect(astuple(0, 0), last_cell)           # whole main diagonal
    # the cell that corresponds to the maximal Manhattan distance
    cut_len    = add(divide(max_dist, 2), 1)                 # because dist( (r,r) ) = 2r
    cut_point  = astuple(cut_len, cut_len)
    prefix     = connect(astuple(0, 0), cut_point)           # occupied part of the diagonal
    to_fill    = difference(full_diag, prefix)               # still‑empty diagonal cells
    O = fill(I, 2, to_fill)
    return O
