def dsl1(I):
    # GEOMETRIC – extend each object’s rectangle to the full border line.
    BG   = mostcommon(I)                                          # background colour
    CAN  = canvas(BG, shape(I))                                    # empty canvas
    CLS  = difference(palette(I), initset(BG))                     # non‑background colours
    BC   = argmax(CLS, lambda c: size(ofcolor(I, c)))              # colour of the solid border
    B    = ofcolor(I, BC)                                          # border patch (full row or column)
    OBJ  = objects(I, False, False, True)                         # all coloured objects
    BOBJ = colorfilter(OBJ, BC)                                    # border objects (usually one)
    OTH  = difference(OBJ, BOBJ)                                   # objects that are not the border
    RCT  = apply(lambda o: recolor(
                difference(backdrop(combine(o, B)), B),           # rectangle up to but not over the border
                color(o)),
                OTH)                                                # coloured rectangles
    ALL  = merge(RCT)                                              # union of rectangles
    O    = paint(CAN, ALL)                                         # paint on canvas
    return O
