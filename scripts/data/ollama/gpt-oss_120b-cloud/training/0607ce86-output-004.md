def dsl1(I):
    # ------- GLOBAL ROW‑TEMPLATE (per block) -------
    # 1. rows that contain a non‑zero colour and no 8
    good = apply(
            lambda r: any(c != 0 for c in r) and not any(c == 8 for c in r),
            I)

    # 2. turn the boolean mask into a grid of the same width (False → 0, True → 1)
    mask_grid = tuple(apply(lambda b: (1,) * width(I) if b else (0,) * width(I), good))

    # 3. split the grid at the zero‑rows of the mask – each block contains only “good’’ rows
    blocks = hsplit(mask_grid, size(mask_grid))          # a tuple of blocks (each block is a grid)

    # 4. for each block, compute the most common original row and replace the whole block
    templated = apply(
            lambda blk: let(
                rows = [I[i] for i in range(start, start + height(blk))],
                tmpl = mostcommon(rows),
                tuple(apply(lambda _: tmpl, blk) ) ),
            enumerate(blocks))

    # 5. concatenate the processed blocks back together
    O = vconcat(*templated)
    return O

def dsl3(I):

def dsl5(I):
