def dsl1(I):
    # Global transformation: flip vertically (mirror along horizontal axis)
    return hmirror(I)


def dsl2(I):
    # Object‑extraction route: treat the whole grid as an object,
    # mirror it, and paint it back onto a blank canvas.
    O = paint(canvas(0, shape(I)), hmirror(asobject(I)))
    return O


def dsl3(I):
    # Geometric composition: rotate 180° then mirror left‑right,
    # which together equals a vertical flip.
    return vmirror(rot180(I))


def dsl4(I):
    # Alternate geometric route: rotate –90° (270° clockwise), mirror left‑right,
    # then rotate back +90°. The net effect is a vertical flip.
    return rot90(vmirror(rot270(I)))


def dsl5(I):
    # Color‑filtering route: mirror each color’s patch individually,
    # merge the mirrored patches and under‑paint them onto a blank canvas.
    f = compose(hmirror, rbind(ofcolor, I))          # color → mirrored patch
    patches = apply(f, palette(I))                   # mirrored patches for all colors
    merged = combine(patches)                         # union of all mirrored patches
    O = underpaint(canvas(0, shape(I)), merged)      # render onto blank canvas
    return O
