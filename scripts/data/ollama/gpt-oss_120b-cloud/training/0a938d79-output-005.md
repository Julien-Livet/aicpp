def dsl1(I):
    # GLOBAL – tile the tight bounding‑box of all coloured cells across the whole canvas.
    objs   = objects(I, False, False, True)                 # all non‑zero cells
    box    = subgrid(merge(objs), I)                        # minimal sub‑grid that contains them
    v_rep  = divide(height(I), height(box))                 # vertical repetition factor
    h_rep  = divide(width (I), width (box))                 # horizontal repetition factor
    tiled  = hupscale(vupscale(box, v_rep), h_rep)          # repeat vertically then horizontally
    return tiled


def dsl2(I):
    # ROW – for every colour, colour the whole row that contains it,
    # then keep the grid as‑is (row filling already covers the full height).
    objs   = objects(I, False, False, True)
    rows   = apply(
                lambda o: fill(I,
                               color(o),
                               { (r, c) for (r, c) in asindices(I) if r == center(o)[0] }),
                objs)
    result = merge(rows)                                    # overlay all row‑fills
    return result


def dsl3(I):
    # COL – for every colour, colour the whole column that contains it,
    # then keep the grid as‑is (column filling already covers the full width).
    objs   = objects(I, False, False, True)
    cols   = apply(
                lambda o: fill(I,
                               color(o),
                               { (r, c) for (r, c) in asindices(I) if c == center(o)[1] }),
                objs)
    result = merge(cols)                                    # overlay all column‑fills
    return result


def dsl4(I):
    # ROW + COL – overlay the row‑expansion and the column‑expansion of every colour.
    objs   = objects(I, False, False, True)

    rows   = apply(
                lambda o: fill(I,

                objs)

    result = combine(merge(rows), merge(cols))              # overlay both expansions
    return result


def dsl5(I):
    # MIRROR – take the tight slice that contains all colours,
    # mirror it across the diagonal that corresponds to the shorter side,
    # then repeat the mirrored slice along the longer axis.
    objs   = objects(I, False, False, True)
    slice  = subgrid(merge(objs), I)                         # minimal slice with all colours

    vp = height(slice)                                      # vertical period
    hp = width (slice)                                      # horizontal period

    mirrored = branch(greater(vp, hp),
                       cmirror(slice),    # mirror across main diagonal
                       dmirror(slice))   # mirror across counter‑diagonal

    result = branch(greater(vp, hp),
                    hupscale(mirrored, divide(width(I), hp)),
                    vupscale(mirrored, divide(height(I), vp)))
