# 1️⃣  GLOBAL TRANSFORMATION – use whole‑grid symmetry & trimming
# --------------------------------------------------------------
def dsl1(I):
    # 1.  Remove the empty border that consists only of the most common colour
    bg      = mostcolor(I)                                 # background colour
    trimmed = trim(I)                                      # strip outer rows/cols that are all bg
    # 2.  Rotate the trimmed picture 180° and mirror it on the main diagonal
    rot     = rot180(trimmed)
    diag    = dmirror(rot)
    # 3.  Put the transformed patch back into a canvas of the original size,
    #     filling the previously removed border with the background colour.
    O = underfill(I, bg, asobject(diag, trimmed))
    return O

# 2️⃣  OBJECT‑EXTRACTION – expand the rare colour into full rows/cols
# -----------------------------------------------------------------
def dsl2(I):

    seed    = leastcolor(I)                               # 1 in all three trainings
    # 2.  All cells having that colour
    src     = ofcolor(I, seed)                            # set of (r,c)
    # 3.  Extract the set of rows and columns that contain a seed cell
    rows    = {r for (r, _) in src}                       # via set‑comprehension (allowed as DSL)
    cols    = {c for (_, c) in src}
    # 4.  Build a patch covering those complete rows and columns
    rows_pt = { (r, c) for r in rows for c in range(width(I)) }
    cols_pt = { (r, c) for c in cols for r in range(height(I)) }
    cross   = rows_pt | cols_pt                              # union → full cross
    # 5.  Write the cross with the seed colour, but turn the original

    O1      = fill(I, seed, cross)                           # paint cross with seed
