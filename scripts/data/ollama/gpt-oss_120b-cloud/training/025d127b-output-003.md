def dsl1(I):
    # GLOBAL bottom‑row restoration:
    # shift every coloured cell right, then repaint the cells that belong to the
    # bottom row of each object (the cells that have no neighbour below).
    all_cells   = asobject(I)                                          # all non‑zero cells
    shifted     = shift(all_cells, RIGHT)                              # whole picture shifted right
    # cells that have a neighbour below = intersection with the picture shifted up
    has_below   = intersection(all_cells, shift(all_cells, UP))
    bottom_rows = difference(all_cells, has_below)                     # true bottom rows
    empty       = canvas(0, shape(I))
    O1          = paint(empty, shifted)                               # shifted picture
    O           = paint(O1, bottom_rows)                               # restore bottom rows
    return O


def dsl2(I):
    # OBJECT‑LEVEL transformation (apply per object):
    # for each foreground object shift all cells except its bottom row.
    objs = fgpartition(I)

    def transform(obj):
        # bottom row of this object (no cell below it)
        has_below   = intersection(obj, shift(obj, UP))
        bottom_row  = difference(obj, has_below)
        non_bottom  = difference(obj, bottom_row)
        shifted     = shift(non_bottom, RIGHT)
        empty       = canvas(0, shape(I))
        return paint(paint(empty, shifted), bottom_row)   # object after its transformation

    # apply the transformation to every object and merge the results
    transformed = apply(transform, objs)                     # set of objects (grids)
    # union of all transformed objects: start from empty canvas and underpaint each one
    base = canvas(0, shape(I))
    O    = foldl(lambda acc g: underpaint(acc, g), base, transformed)  # foldl not native,
    # emulate the fold with repeated apply‑underpaint using mapply
    O    = underpaint(base, combine(*transformed))        # combine merges the object sets
    return O


def dsl3(I):

    # any coloured cell that has another coloured cell directly below it
    # is moved one step to the right; all other cells stay.

    shifted_mov = shift(movable, RIGHT)
    empty       = canvas(0, shape(I))
    O1          = paint(empty, shifted_mov)
    O           = paint(O1, static)
    return O

    # applying the same relational rule as dsl3 to each colour.
