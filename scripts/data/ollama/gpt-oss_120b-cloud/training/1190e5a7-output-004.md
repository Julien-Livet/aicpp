def dsl1(I):
    # ── Global periodicity ──
    #   Find the dominant colour, build an object of its cells,
    #   measure its vertical/horizontal periods and crop the original grid.
    dom   = mostcolor(I)
    idx   = ofcolor(I, dom)               # indices of dominant colour
    obj   = toobject(idx, I)               # object (color‑indexed cells)
    vsize = vperiod(obj)
    hsize = hperiod(obj)
    O     = crop(I, (0, 0), (vsize, hsize))
    return O


def dsl2(I):
    # ── Object‑union extraction ──
    #   Unite all connected components of the dominant colour,
    #   compute their periods and extract the elementary tile.
    dom      = mostcolor(I)
    comps    = objects(I, True, False, True)     # uni‑coloured objects
    dom_objs = colorfilter(comps, dom)           # dominant‑colour objects
    union    = merge(dom_objs)                   # union of all dominant objects
    vsize    = vperiod(union)
    hsize    = hperiod(union)
    O        = crop(I, (0, 0), (vsize, hsize))
    return O


def dsl3(I):
    # ── Colour‑filter + period cropping ──
    #   Keep only the dominant colour, then use its repetition periods
    #   to cut out the minimal repeating tile.
    dom   = mostcolor(I)
    mask  = ofcolor(I, dom)                     # indices of dominant colour
    obj   = toobject(mask, I)                   # object of those cells
    vsize = vperiod(obj)
    hsize = hperiod(obj)
    O     = crop(I, (0, 0), (vsize, hsize))
    return O


def dsl4(I):
    # ── Symmetric core extraction ──
    #   Cells unchanged by both horizontal and vertical mirrors form a core.
    #   Its periods give the tile size; crop the core itself.
    H          = hmirror(I)
    V          = vmirror(I)
    eq_h       = cellwise(I, H, 0)               # equal to horizontal mirror
    core_grid  = cellwise(eq_h, V, 0)             # also equal to vertical mirror
    dom        = mostcolor(core_grid)
    mask       = ofcolor(core_grid, dom)
    obj        = toobject(mask, I)
    vsize      = vperiod(obj)
    hsize      = hperiod(obj)
    O          = crop(core_grid, (0, 0), (vsize, hsize))
    return O


def dsl5(I):
    # ── Relational first‑occurrence tile ──
    #   Use the location of the first occurrence of the dominant colour
    #   and the overall bounding box of that colour to extract the tile.
    dom   = mostcolor(I)
    mask  = ofcolor(I, dom)                     # indices of dominant colour
    obj   = toobject(mask, I)
    occ   = first(occurrences(I, obj))        # top‑left corner of a tile
    tile  = box(mask)                          # full bounding box of dominant cells
    vsize = height(tile)
    hsize = width(tile)
    O     = crop(I, occ, (vsize, hsize))
    return O
