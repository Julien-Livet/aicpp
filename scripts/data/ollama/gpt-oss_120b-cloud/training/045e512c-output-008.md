def dsl1(I):
    # REPLICATION – upscale the whole pattern once and paste a copy at every
    # original coloured cell (any colour).  This uses the set of all non‑zero
    # locations as anchor points for the enlarged block.
    whole   = merge(objects(I, F, F, T))                     # all coloured cells
    sub     = subgrid(backdrop(whole), I)                   # tight bounding box
    big     = upscale(sub, TWO)                             # 2× enlargement
    # every coloured cell becomes an anchor for a copy of the enlarged block
    anchors = combine(apply(lambda c: ofcolor(I, c), palette(I)))   # all non‑zero cells
    copies  = apply(lambda loc: shift(asobject(big), loc), anchors)
    O       = paint(canvas(0, shape(I)), merge(copies))
    return O


def dsl2(I):
    # MIRRORING – enlarge the whole pattern once and reflect it horizontally,
    # vertically and both, then overlay the four symmetric copies.
    whole   = merge(objects(I, F, F, T))
    sub     = subgrid(backdrop(whole), I)
    big     = upscale(sub, TWO)
    horiz   = hmirror(big)          # left‑right mirror
    vert    = vmirror(big)          # top‑bottom mirror
    both    = hmirror(vert)         # both mirrors
    patches = {asobject(big), asobject(horiz), asobject(vert), asobject(both)}
    O       = paint(canvas(0, shape(I)), merge(patches))
    return O


def dsl3(I):
    # COLOR‑FILTER – treat each colour independently, upscale its exact shape,
    # and overlay the enlarged shape at the colour’s original upper‑left corner,
    # keeping the original colour cells as well.
    cols    = palette(I)

                lambda c: shift(

                cols)
    olds    = apply(lambda c: asobject(ofcolor(I, c)), cols)

    return O


def dsl4(I):
    # RELATIONAL – enlarge the whole pattern and reposition it so that its centre of
    # mass coincides with the centre of mass of the original pattern.
    whole   = merge(objects(I, F, F, T))
    sub     = subgrid(backdrop(whole), I)
    big     = upscale(sub, TWO)
    c_orig  = centerofmass(toindices(whole))          # centre of original pattern
    c_big   = centerofmass(toindices(asobject(big))) # centre of enlarged pattern
    delta   = subtract(c_orig, c_big)                 # vector needed for alignment
    shifted = shift(asobject(big), delta)
    O       = paint(canvas(0, shape(I)), shifted)
    return O
