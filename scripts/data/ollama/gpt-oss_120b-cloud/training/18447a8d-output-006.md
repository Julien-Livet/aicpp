            if color(o) not in {7, 8}}
    O = mapply(
            lambda o: fill(I, succ(color(o)), o),
            objs)
    return O


def dsl4(I):
    # MIRROR‑SHIFT‑COMBINE: mirror the whole picture vertically,
    # shift the mirror one column to the right, overlay it on the
    # original and keep the background (7) wherever both layers have 7.
    mirrored = vmirror(I)
    shifted  = move(mirrored, asobject(mirrored), (ZERO, ONE))
    O = combine(I, shifted)
    # cells that are 7 in both layers become the least‑frequent colour
    filler   = valmin(palette(I) - {7, 8},
                     lambda c: colorcount(I, c))
    both7    = intersection(ofcolor(I, 7), ofcolor(shifted, 7))
    O = recolor(filler, both7)
    return O


def dsl5(I):
    # GRAVITATE‑LEFT: repeatedly move every non‑7/8 cell one step
    # toward the nearest 8 in its row, repeating enough times to push
    # the whole colour block next to the 8‑block.
    other   = palette(I) - {7, 8}
    step    = lambda g, c: gravitate(
                               frozenset({c}),
                               frozenset({first(dneighbors(c) & ofcolor(g, 8)}))
    O = apply(
            lambda _: mapply(
                        lambda col:
                            let = ofcolor(_, col)
                            dirs = mapply(lambda p: step(_, p), let)
                            move(_, let, dirs[0]) if dirs else _,
                        other),
            range(width(I)))                # repeat for the width of the grid
    return O
