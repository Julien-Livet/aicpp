def dsl1(I):
    # cells that are 5 and have at least two orthogonal neighbours equal to 0
    zeros = ofcolor(I, 0)
    fives = ofcolor(I, 5)

    def enough_ortho(p):
        return greater(size(intersection(neighbors(p), zeros)), ONE)

    target = mfilter(fives, enough_ortho)
    O = fill(I, 2, toobject(target, I))
    return O


def dsl2(I):
    # cells that are 5 and have at least one diagonal neighbour equal to 0
    zeros = ofcolor(I, 0)
    fives = ofcolor(I, 5)

    def any_diag(p):
        return greater(size(intersection(ineighbors(p), zeros)), ZERO)

    target = mfilter(fives, any_diag)
    O = fill(I, 2, toobject(target, I))
    return O


def dsl3(I):
    # cells that are 5 and have another 5 immediately to the right (horizontal pair)
    fives = ofcolor(I, 5)
    right_pair = intersection(fives, shift(fives, LEFT))
    O = fill(I, 2, toobject(right_pair, I))

    target = intersection(fives, mirrored)
    O = fill(I, 2, toobject(target, I))
    return O
