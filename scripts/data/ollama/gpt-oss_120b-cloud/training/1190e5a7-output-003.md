def dsl1(I):
    # ── Global‑period extraction (no explicit objects) ──
    #   Periods are read directly from the index set of the dominant colour,
    #   then the original grid is cropped to that elementary tile.
    dom   = mostcolor(I)
    idx   = ofcolor(I, dom)                     # indices of dominant colour
    vsize = vperiod(idx)                        # vertical repetition length
    hsize = hperiod(idx)                        # horizontal repetition length
    O     = crop(I, (0, 0), (vsize, hsize))
    return O


def dsl2(I):
    # ── Object‑union extraction ──
    #   All connected components of the dominant colour are united,
    #   their periodicities are measured and the tile is cropped.
    dom      = mostcolor(I)
    comps    = objects(I, True, False, True)     # uni‑coloured objects
    dom_objs = colorfilter(comps, dom)           # only dominant‑colour objects
    union    = combine(dom_objs)                 # union of all dominant components
    vsize    = vperiod(union)
    hsize    = hperiod(union)
    O        = crop(I, (0, 0), (vsize, hsize))
    return O


def dsl3(I):
    # ── Colour‑filter → compression tile ──
    #   Keep only the dominant colour (others become 0),
    #   compress removes the repeating border rows/cols,
    #   the compressed grid is exactly the desired tile.
    dom   = mostcolor(I)
    F     = fill(I, 0, difference(asindices(I), ofcolor(I, dom)))  # keep only dom
    O     = compress(F)                                            # periodic tile
    return O


def dsl4(I):
    # ── Symmetric core extraction ──
    #   Cells unchanged by both horizontal and vertical mirrors form the core.
    #   Its periods give the tile size; crop the core itself.
    H     = hmirror(I)
    V     = vmirror(I)
    core  = cellwise(I, H, 0)            # equal to horizontal mirror
    core  = cellwise(core, V, 0)         # also equal to vertical mirror
    dom   = mostcolor(core)
    idx   = ofcolor(core, dom)
    vsize = vperiod(idx)
    hsize = hperiod(idx)
    O     = crop(core, (0, 0), (vsize, hsize))
    return O


def dsl5(I):
    # ── Relational tiling via first occurrence ──
    #   The bounding box of the first occurrence of the dominant colour
    #   determines the elementary tile.
    dom      = mostcolor(I)
    mask     = ofcolor(I, dom)
    obj      = toobject(mask, I)
    occ      = first(occurrences(I, obj))      # location of a representative tile
    tilebox  = box(toobject(mask, I))          # full bounding box of dominant cells
    vsize    = height(tilebox)                 # vertical size of the pattern
    hsize    = width(tilebox)                  # horizontal size of the pattern
    O        = crop(I, occ, (vsize, hsize))
    return O
