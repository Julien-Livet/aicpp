def dsl1(I):
    # Global: compress the grid to the tight bounding box of all non‑background cells,
    # mirror that compact shape horizontally, then restore the original size by filling
    # the empty area with the background colour.
    bg = 4
    C   = compress(I)                     # smallest subgrid containing the pattern
    M   = vmirror(C)                      # horizontal mirror of the compact pattern
    O   = underfill(I, bg, M)             # place the mirrored pattern back into the original canvas
    return O


def dsl2(I):
    # Object‑extraction + geometry: isolate the dominant non‑background colour,
    # erase it from the original picture, mirror the isolated object and paste it back.
    bg  = 4
    nb  = palette(I) - {bg}
    tgt = argmax(nb, lambda c: colorcount(I, c))   # most frequent colour ≠ background
    obj = ofcolor(I, tgt)                         # cells of that colour
    O   = fill(I, bg, obj)                        # remove the original occurrence
    M   = vmirror(obj)                            # mirror the object horizontally
    O   = fill(O, tgt, M)                         # paste the mirrored copy
    return O


def dsl3(I):
    # Colour‑filtering: replace every non‑background cell by the most common
    # non‑background colour, leaving background cells unchanged.
    bg  = 4
    nb  = palette(I) - {bg}
    tgt = argmax(nb, lambda c: colorcount(I, c))   # dominant colour ≠ background
    C   = canvas(tgt, shape(I))                   # full‑grid of the dominant colour
    O   = cellwise(C, I, bg)                      # keep background where it occurs
    return O


def dsl4(I):
    # Geometric: rotate the whole picture 90° clockwise and keep the rotated cells,
    # falling back to the original background colour where the rotation is empty.
    bg = 4
    R  = rot90(I)                                 # 90° clockwise rotation
    O  = cellwise(R, I, bg)                       # original background fills any gaps
    return O


def dsl5(I):
    # Relational / structural: discover the horizontal period of all coloured objects,
    # duplicate the whole coloured pattern one period to the right while preserving
    # the original pattern.
    bg   = 4
    objs = objects(I, False, False, True)         # all coloured objects (background excluded)
    patch = {c for obj in objs for c in obj}      # union of their cells
    p    = hperiod(patch)                         # basic horizontal period
    shifted = shift(patch, (0, p))                 # copy displaced one period to the right
    O = fill(I, bg, shifted)                      # paste the replicated copy over background
    return O
