    M       = dmirror(T)                                              # counter‑diagonal mirror
    objs    = objects(M, False, False, True)                           # coloured objects of the mirror
    ind    = {c[1] for obj in objs for c in obj}                       # their indices
    mir_obj = toobject(ind, M)                                         # coloured object of the mirror
    O       = underpaint(I, mir_obj)                                   # overlay on background
    return O


def dsl3(I):
    # ► Colour‑statistics – swap the background colour (4) with the most
    #   frequent non‑background colour.
    bg      = 4
    nonbg   = {c for c in palette(I) if c != bg}
    tgt     = argmax(nonbg, lambda c: colorcount(I, c))               # dominant colour
    bg_cells= ofcolor(I, bg)
    tg_cells= ofcolor(I, tgt)
    O       = recolor(tgt, bg_cells)       # paint background with dominant colour
    O       = recolor(bg,  tg_cells)       # paint former dominant cells with background
    return O


def dsl4(I):
    # ► Geometric – vertical mirror overlay, keeping original cells
    #   where they differ from the mirror (fallback colour 4).
    V       = vmirror(I)                     # vertical flip
    O       = cellwise(I, V, 4)               # keep I‑pixels, otherwise 4
    return O


def dsl5(I):
    # ► Relational / structural – replicate the pattern one vertical period downwards.
    objs    = objects(I, False, False, True)
    patch   = {c for obj in objs for c in obj}
    p       = vperiod(patch)                  # basic vertical period
    shifted = shift(patch, (p, 0))             # copy one period down
    O       = fill(I, 4, patch)               # erase original pattern
    O       = fill(O, mostcolor(I), shifted)  # paste the replicated copy
    return O
