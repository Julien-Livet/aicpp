def dsl2(I):
    # Object‑extraction – locate the pattern, crop it, then fill background with 9.
    pat   = ofcolor(I, 1) | ofcolor(I, 2) | ofcolor(I, 3) | ofcolor(I, 8)
    bbox  = backdrop(pat)                       # minimal rectangle containing the pattern
    sub   = subgrid(bbox, I)                    # crop to that rectangle
    base  = underfill(sub, 9, pat)              # everything not in pat becomes 9
    O     = replace(base, 8, 5)                 # 8 → 5
    O     = replace(O, 1, 9)                    # 1 → 9
    O     = replace(O, 3, 8)                    # 3 → 8
    O     = replace(O, 2, 7)                    # 2 → 7
    return O
