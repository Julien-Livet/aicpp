def dsl1(I):
    # ----------- object‑extraction (non‑5 components) -----------------
    # make the separator colour 5 become background (0)
    G = switch(I, 5, 0)

    # each connected component of non‑background cells is a rectangular block
    blocks = objects(G, False, False, True)

    # recolour every block with the most frequent colour that appears inside it
    recoloured = apply(
        lambda o: recolor(
            mostcolor(subgrid(o, I)),
            o),
        blocks)

    O = paint(I, merge(recoloured))
    return O


def dsl2(I):
    # ----------- global cut at full rows/columns of colour 5 -------
    rows = mfilter(
        interval(0, shape(I)[0], 1),
        lambda i: size(
            intersection(
                ofcolor(I, 5),
                apply(lambda j: (i, j), interval(0, shape(I)[1], 1))
            )) == shape(I)[1])

    cols = mfilter(
        interval(0, shape(I)[1], 1),
        lambda j: size(
            intersection(
                ofcolor(I, 5),
                apply(lambda i: (i, j), interval(0, shape(I)[0], 1))
            )) == shape(I)[0])

    rcuts = order(combine({0}, combine(set(rows), {shape(I)[0]})), lambda a, b: a < b)
    ccuts = order(combine({0}, combine(set(cols), {shape(I)[1]})), lambda a, b: a < b)

    blocks = apply(
        lambda r: apply(
            lambda c: recolor(
                mostcolor(
                    crop(
                        I,
                        (list(rcuts)[r], list(ccuts)[c]),
                        (list(rcuts)[r + 1] - list(rcuts)[r],
                         list(ccuts)[c + 1] - list(ccuts)[c]))),
                shoot(
                    (list(rcuts)[r], list(ccuts)[c]),
                    (list(rcuts)[r + 1] - list(rcuts)[r],
                     list(ccuts)[c + 1] - list(ccuts)[c]))
            ),
            interval(0, size(ccuts) - 1, 1)),
        interval(0, size(rcuts) - 1, 1))

    O = paint(I, merge(flatten(blocks)))
    return O

def dsl4(I):
    # ----------- geometric – rotate, cut, rotate back ---------------
    R = rot90(I)                                   # separators become vertical lines

    # line objects in the rotated view
    hlines = apply(ulcorner,
                   mfilter(objects(R, True, False, False), hline))
    vlines = apply(ulcorner,
                   mfilter(objects(R, True, False, False), vline))

    rcuts = order(combine({0}, combine(set(hlines), {shape(R)[0]})), lambda a, b: a < b)
    ccuts = order(combine({0}, combine(set(vlines), {shape(R)[1]})), lambda a, b: a < b)

    blocks = apply(
        lambda r: apply(
            lambda c: recolor(
                mostcolor(
                    crop(
                        R,
                        (list(rcuts)[r], list(ccuts)[c]),
                        (list(rcuts)[r + 1] - list(rcuts)[r],
                         list(ccuts)[c + 1] - list(ccuts)[c]))),
                shoot(
                    (list(rcuts)[r], list(ccuts)[c]),
                    (list(rcuts)[r + 1] - list(rcuts)[r],
                     list(ccuts)[c + 1] - list(ccuts)[c]))
            ),
            interval(0, size(ccuts) - 1, 1)),
        interval(0, size(rcuts) - 1, 1))

    Orot = paint(R, merge(flatten(blocks)))
    O = rot270(Orot)                               # back to original orientation
    return O


def dsl5(I):
    # ----------- relational – nearest separator rectangle -----------
    # positions of full‑row and full‑column separators (colour 5)
    sep_rows = mfilter(
        interval(0, shape(I)[0], 1),
        lambda i: size(
            intersection(
                ofcolor(I, 5),
                apply(lambda j: (i, j), interval(0, shape(I)[1], 1))
            )) == shape(I)[1])

    sep_cols = mfilter(
        interval(0, shape(I)[1], 1),
        lambda j: size(
            intersection(
                ofcolor(I, 5),
                apply(lambda i: (i, j), interval(0, shape(I)[0], 1))
            )) == shape(I)[0])

    # for every non‑separator cell, find the rectangle delimited by the
    # nearest separator rows above/below and columns left/right
    non5 = mfilter(asindices(I), lambda p: index(I, p) != 5)

    rects = apply(
        lambda p: recolor(
            mostcolor(
                subgrid(
                    mfilter(
                        asindices(I),
                        lambda q:
                            q[0] > argmax(sep_rows, lambda r: r if r < p[0] else -ONE) and
                            q[0] < argmin(sep_rows, lambda r: r if r > p[0] else ONE) and
                            q[1] > argmax(sep_cols, lambda c: c if c < p[1] else -ONE) and
                            q[1] < argmin(sep_cols, lambda c: c if c > p[1] else ONE)
                    ),
                    I)),
            {p}),
        non5)

    O = paint(I, merge(rects))
    return O
