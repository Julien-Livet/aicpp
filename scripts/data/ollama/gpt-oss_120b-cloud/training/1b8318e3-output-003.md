def dsl1(I):
    # GLOBAL –  compress →  rotate 180°  →  hmirror  (single pipeline)
    #   1.  Remove empty outer rows/cols.

    #   3.  Mirror the result left‑right.

    R   = rot180(C)                  # 180° rotation
    O   = hmirror(R)                 # horizontal mirror
    return O


def dsl2(I):

    #           with the centre of the largest colour‑5 object.

    ctr_five    = center(big_five)                        # its centre

    def move_to_five(o):

        d = subtract(ctr_five, center(o))
        return move(I, o, d)

    O = cover(O, big_five)                                # keep the 5‑object unchanged
