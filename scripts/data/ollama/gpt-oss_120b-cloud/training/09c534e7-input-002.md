```python
def dsl1(I):
    # ---- colour‑filtering (seed expansion) -----------------
    # seeds are all colours > 1
    seeds   = mfilter(palette(I), lambda c: c > 1)

    # for each seed colour build the patch of 1‑cells that lie inside its
    # orthogonal out‑box and recolour that patch with the seed colour
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                outbox(ofcolor(I, c)),
                                ofcolor(I, 1))
                         ),
                seeds)

    # paint all recoloured patches on the original grid
    O = paint(I, combine(patches))
    return O
```
