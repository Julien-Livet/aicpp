def dsl1(I):
    # GLOBAL‑TRANSFORMATION : trim outer background, binarise and count 8‑connected components
    bg      = mostcolor(I)
    mask    = difference(asindices(I), ofcolor(I, bg))          # foreground cells
    empty   = canvas(0, shape(I))
    binG    = paint(empty, recolor(1, mask))                  # binary picture (0/1)
    trimmed = compress(trim(binG))                            # remove empty borders, then compress
    comps   = objects(trimmed, False, True, True)            # diagonal (8‑neighbour) components
    O = canvas(size(comps), (1, 1))
    return O


def dsl2(I):

    bg      = mostcolor(I)
    parts   = fgpartition(I)                                 # each colour → its objects
    fgs     = sfilter(parts,
                     lambda o: flip(equality(color(o), bg)))   # keep only non‑background objects
    O = canvas(size(fgs), (1, 1))
    return O


def dsl3(I):
    # COLOR‑FILTERING : build a set of binary masks, one per colour (except background),
    # recolour each mask to 1 and count the total number of connected components.
    bg      = mostcolor(I)
    # objects of each distinct colour (univalued=True gives one object per colour region)
    col_objs = objects(I, True, False, False)
    # drop background objects
    non_bg   = sfilter(col_objs, lambda o: flip(equality(color(o), bg)))
    # turn every coloured object into a binary object (colour 1)
    bin_objs = mapply(lambda o: recolor(1, o), non_bg)
    # union of all binary objects gives a binary picture
    empty    = canvas(0, shape(I))
    binG     = paint(empty, combine(bin_objs, {}))
    comps    = objects(binG, False, True, True)              # diagonal connectivity
    O = canvas(size(comps), (1, 1))
    return O


def dsl4(I):
