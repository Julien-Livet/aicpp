def dsl1(I):
    # replace a 5 that has at least two orthogonal neighbours equal to 0
    zeros  = ofcolor(I, 0)
    fives  = ofcolor(I, 5)

    def enough_zero_neighbours(p):
        return greater(size(intersection(neighbors(p), zeros)), ONE)   # size > 1

    target = mfilter(fives, enough_zero_neighbours)
    O = fill(I, 2, target)
    return O


def dsl2(I):
    # replace a 5 that has another 5 immediately to its right (horizontal run)
    fives = ofcolor(I, 5)

    # shift the whole set one step left; the intersection are the cells
    # that have a 5 on their right side
    right_adjacent = intersection(fives, shift(fives, LEFT))

    O = fill(I, 2, right_adjacent)
    return O


def dsl3(I):
    # replace every 5 that lies on the outer border of the grid
    fives = ofcolor(I, 5)

    def on_border(p):
        # a single‑cell patch is created with the index and tested
        return bordering({p}, I)

    border_fives = mfilter(fives, on_border)
    O = fill(I, 2, border_fives)
    return O


def dsl4(I):
    # replace a 5 that is also a 5 in the diagonal mirror of the grid
    fives        = ofcolor(I, 5)
    mirrored_five = ofcolor(dmirror(I), 5)

    target = intersection(fives, mirrored_five)
    O = fill(I, 2, target)
    return O


def dsl5(I):

    zeros = ofcolor(I, 0)
    fives = ofcolor(I, 5)

    def diagonal_zero(p):
        return greater(size(intersection(ineighbors(p), zeros)), ZERO)   # any diagonal zero

    target = mfilter(fives, diagonal_zero)
    O = fill(I, 2, target)
    return O
