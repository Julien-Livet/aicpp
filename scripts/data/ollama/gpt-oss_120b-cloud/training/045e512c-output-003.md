def dsl1(I):
    # GLOBAL – keep only the non‑zero rectangle and enlarge it uniformly 2×
    O = compress(I)          # smallest rectangle containing every non‑zero cell
    O = upscale(O, 2)        # uniform 2‑fold enlargement
    return O


def dsl2(I):
    # GLOBAL – discard the outer zero border and then double the width and height
    O = trim(I)              # remove rows / columns that are completely background
    O = hupscale(O, 2)       # double the horizontal dimension
    O = vupscale(O, 2)       # double the vertical dimension
    return O


def dsl3(I):
    # OBJECT‑LEVEL – extract the overall bounding box of all coloured cells,
    # then enlarge that sub‑grid.
    objs   = objects(I, True, False, True)            # every coloured object
    all_ix = merge(apply(toindices, objs))            # union of all its indices
    bb     = backdrop(all_ix)                         # bounding‑box indices
    O      = subgrid(bb, I)                           # smallest sub‑grid that contains everything
    O      = upscale(O, 2)                             # 2‑fold enlargement
    return O


def dsl4(I):
    # OBJECT‑LEVEL – compute exact corner coordinates of the occupied region,
    # crop with those coordinates and enlarge.
    objs   = objects(I, True, False, True)
    all_ix = merge(apply(toindices, objs))            # union of all indices
    ul     = ulcorner(all_ix)                         # upper‑left corner of the box
    lr     = lrcorner(all_ix)                         # lower‑right corner of the box
    dims   = add(subtract(lr, ul), (ONE, ONE))         # width × height of the box
    O      = crop(I, ul, dims)                        # exact rectangle
    O      = upscale(O, 2)                            # double its size
    return O


def dsl5(I):
    # OBJECT‑LEVEL – scale each colour separately and paint the enlarged
    # objects onto a fresh canvas that has the size of the globally enlarged
    # picture (compress → upscale gives the required canvas size).
    colours      = palette(I)                                 # all colours that appear
    canvas_sz    = shape(upscale(compress(I), 2))              # size of the final picture
    empty_canvas = canvas(0, canvas_sz)                        # blank canvas

    def scale_and_paint(col):
        pts   = ofcolor(I, col)                                # cells of this colour
        sub   = subgrid(backdrop(pts), I)                      # its minimal sub‑grid
        big   = upscale(sub, 2)                                 # enlarge 2×
        ul    = ulcorner(pts)                                   # original upper‑left corner
        new_ul = multiply(ul, (TWO, TWO))                      # where the enlarged block must be placed
        shifted = shift(toobject(big, empty_canvas), new_ul)    # move it to the right spot
        return paint(empty_canvas, shifted)                    # paint over the blank canvas

    # apply the above to every colour and keep the last (fully painted) canvas
    O = apply(scale_and_paint, colours)
    # `apply` returns a container of canvases; the last element is the fully painted one
    O = last(O)
    return O
