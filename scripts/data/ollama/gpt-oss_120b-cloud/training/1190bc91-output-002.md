def dsl1(I):
    # -----------------------------------------------------------------
    # 1. Object‑wise north‑west propagation.
    #    Every non‑zero cell paints all cells that are strictly north‑west
    #    of it.  The nearest (in Chebyshev distance) source wins because
    #    the propagation is performed from the farthest offsets to the
    #    smallest offset.
    # -----------------------------------------------------------------
    objs   = fgpartition(I, False, False, True)                     # all coloured cells
    # list of offsets (dx,dy) = (‑k,‑k) for k = 0 … max dimension
    offs   = interval(0, maximum({height(I), width(I)}))
    # shift the whole set of objects by every offset
    shifts = apply(lambda k: move(I, objs, (-k, -k)), offs)          # tuple of grids
    # combine them starting from the largest offset → smallest (nearest)
    O      = fold(combine, shifts)                                   # fold is a helper combinator
    return O


def dsl2(I):
    # -----------------------------------------------------------------
    # 2. Fill the north‑west rectangle of each source, ordered by distance.
    #    For every coloured cell a rectangle from the origin to that cell
    #    (inclusive) is filled with its colour.  Rectangles are applied from
    #    the farthest cells to the nearest, so the nearest colour overwrites.
    # -----------------------------------------------------------------
    objs   = fgpartition(I, False, False, True)                     # each non‑zero cell
    # distance of a cell from the origin (Chebyshev)
    dist   = lambda o: max(position((0,0), o)[0], position((0,0), o)[1])
    # order objects by decreasing distance
    ordered = order(objs, lambda o: -dist(o))
    # for each object create its bounding‑box patch and recolour it
    patches = apply(lambda o: recolor(color(o), backdrop(o)), ordered)
    # overlay all patches
    O      = fold(combine, patches)
    return O


def dsl3(I):
    # -----------------------------------------------------------------
    # 3. Paint the whole background with the most frequent non‑zero colour.
    #    This is a pure colour‑filtering hypothesis.

    nonzero   = difference(palette(I), initset(0))          # all colours except 0
    fillcol   = mostcommon(nonzero)                        # most common colour
    bg        = ofcolor(I, 0)                              # background cells
    O         = underfill(I, fillcol, bg)                  # colour the background
    return O


def dsl4(I):
    # -----------------------------------------------------------------
    # 4. Gravitate every coloured cell one step towards the top‑left corner.
    #    The direction is computed once for the whole set of dominant‑colour

    # -----------------------------------------------------------------
    dominant = mostcommon(palette(I))                     # dominant colour
    srcs     = ofcolor(I, dominant)                       # its cells
    dir      = gravitate(srcs, initset((0, 0)))            # direction to (0,0)
    O        = move(I, srcs, dir)                         # shift them one step
    return O


def dsl5(I):
    # -----------------------------------------------------------------
    # 5. Replicate the pattern formed by the most common colour using its
    #    vertical period.  The pattern is tiled vertically, then merged.
    # -----------------------------------------------------------------
    objs        = fgpartition(I, False, False, True)                     # all objects
    commoncol   = mostcommon(palette(I))                                 # most common colour
    commonobjs  = colorfilter(objs, commoncol)                            # objects of that colour
    period      = vperiod(commonobjs)                                     # vertical period
    reps        = interval(0, divide(height(I), period))                  # how many repeats
    tiles       = apply(lambda k: move(I, commonobjs, (k*period, 0)), reps)
    tiled       = fold(combine, tiles)                                   # whole tiling
    O           = combine(I, tiled)                                      # overlay on original
    return O
