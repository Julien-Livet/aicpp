def dsl1(I):
    # GLOBAL‑PIXELATION – down‑scale the whole picture and then restore its size.
    G = downscale(I, 2)                 # make a coarse version
    O = upscale(G, 2)                   # bring it back to the original resolution
    return O


def dsl2(I):
    # OBJECT‑SELECTION – keep only the largest coloured component.
    OBJ    = objects(I, univalued=False, diagonal=False, without_bg=True)
    BIG    = argmax(OBJ, size)          # component with maximal size
    O      = canvas(0, shape(I))
    O      = paint(O, BIG)             # render that component, everything else stays 0
    return O


def dsl3(I):
    # COLOUR‑FILTERING – delete the “marker” colours 3, 5, 8 and any isolated cell.
    MARKERS   = combine(combine(ofcolor(I, 3), ofcolor(I, 5)), ofcolor(I, 8))
    OBJ       = objects(I, univalued=False, diagonal=False, without_bg=True)
    SINGLETON = mfilter(OBJ, lambda o: equality(size(o), 1))
    BAD       = combine(MARKERS, SINGLETON)   # cells that must become background
    O         = fill(I, 0, BAD)                # replace them by 0
    return O


def dsl4(I):
    # GEOMETRIC‑OVERLAY – superimpose horizontal and vertical mirrors, original cells win.
    H = hmirror(I)                         # flip vertically
    V = vmirror(I)                         # flip horizontally
    M = cellwise(H, V, fallback=0)         # union of the two mirrors
    O = cellwise(I, M, fallback=0)          # original colour dominates where it exists
    return O


def dsl5(I):
    # RELATIONAL‑OUTLINE – draw the bounding‑box outline of every coloured region,
    # using the region’s own colour.
    OBJ = objects(I, univalued=False, diagonal=False, without_bg=True)
    O   = canvas(0, shape(I))
    O   = apply(lambda o: paint(O, recolor(color(o), box(o))), OBJ)
    return O
