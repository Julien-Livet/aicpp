def dsl1(I):
    # -------------------------------------------------
    # Strategy 1 – Row‑wise replication (object extraction)
    # -------------------------------------------------
    # 1.  Find all cells whose colour is 5 (the “anchor” colour).
    anchor   = ofcolor(I, 5)

    # 2.  Extend each anchor cell horizontally to the full width of the grid
    #     (this yields the whole rows that contain a 5 in column 0).
    rows    = mapply(lambda d: shift(anchor, (0, d)),
                    interval(ZERO, width(I), ONE))

    # 3.  Turn the colour 5 in those rows into 0.
    cleared = recolor(0, anchor)
    I0      = paint(I, cleared)

    # 4.  Extract the rows (with the leftmost colour cleared) as a sub‑grid.
    pattern = subgrid(rows, I0)

    # 5.  Count how many completely empty rows lie below the lowest
    #     non‑zero row, and how many rows belong to the pattern.
    empty_rows   = subtract(height(I),
                           add(valmax({ r for r,_ in ofcolor(I,0) },
                                     lambda x: x), ONE))
    pattern_rows = size({ r for r,_ in anchor })

    # 6.  How many whole repetitions of the pattern fit into the empty space?
    repeats = divide(empty_rows, pattern_rows)

    # 7.  Build the repeated block by concatenating the pattern ‘repeats’ times.
    repeat_one   = lambda g: vconcat(g, pattern)
    repeat_block  = power(repeat_one, subtract(repeats, ONE))(pattern)

    # 8.  Keep the original content up to the first empty row
    #     (i.e. all rows that contain a non‑zero cell) and append the block.
    nonzero   = difference(asindices(I), ofcolor(I, 0))
    nz_rows   = mapply(lambda d: shift(nonzero, (0, d)),
                        interval(ZERO, width(I), ONE))
    base      = subgrid(nz_rows, I)

    O = vconcat(base, repeat_block)
    # 9.  Remove any trailing rows that are still completely zero.
    O = trim(O)
    return O
