def dsl1(I):
    # GLOBAL EXTREME COLUMNS – colour the leftmost and rightmost
    # non‑zero columns of the whole picture.
    colours   = {c for c in palette(I) if c > 0}
    masks     = apply(lambda c: ofcolor(I, c), colours)          # set of patches
    union     = merge(masks)                                    # all coloured cells
    leftcol   = leftmost(union)
    rightcol  = rightmost(union)
    lefts  = recolor(4, sfilter(union, lambda rc: rc[1] == leftcol))
    rights = recolor(2, sfilter(union, lambda rc: rc[1] == rightcol))
    O = paint(I, lefts)
    O = paint(O, rights)
    return O


def dsl2(I):
    # OBJECT‑LEVEL EXTREME COLUMNS – for every connected component
    # recolour its own leftmost column with 4 and its rightmost column with 2.
    objs      = fgpartition(I)                                 # all objects (incl. bg)
    coloured  = mfilter(objs, lambda o: color(o) > 0)           # drop background
    lefts  = merge(apply(
                lambda o: recolor(
                    4,
                    sfilter(toindices(o),
                            lambda rc: rc[1] == leftmost(toindices(o)))
                ), coloured))
    rights = merge(apply(
                lambda o: recolor(
                    2,
                    sfilter(toindices(o),
                            lambda rc: rc[1] == rightmost(toindices(o)))
                ), coloured))
    O = paint(I, lefts)
    O = paint(O, rights)
    return O


def dsl3(I):
    # RECTANGLE‑SIDES PER OBJECT – draw only those cells of the
    # bounding‑box outline that belong to the object itself, then colour.
    objs      = fgpartition(I)
    coloured  = mfilter(objs, lambda o: color(o) > 0)
    lefts  = merge(apply(
                lambda o: recolor(
                    4,
                    intersection(
                        sfilter(box(toindices(o)),
                                 lambda rc: rc[1] == leftmost(toindices(o))),
                        toindices(o))
                ), coloured))
    rights = merge(apply(
                lambda o: recolor(
                    2,
                    intersection(
                        sfilter(box(toindices(o)),
                                 lambda rc: rc[1] == rightmost(toindices(o))),
                        toindices(o))
                ), coloured))
    O = paint(I, lefts)
    O = paint(O, rights)
    return O


def dsl4(I):
    # COLOUR‑BASED EXTREME COLUMNS – treat each colour as a group,
    # colour the leftmost column of that colour with 4 and the rightmost
    # column with 2 (inside each colour’s cells only).
    colours   = {c for c in palette(I) if c > 0}
    lefts  = merge(apply(
                lambda c: recolor(
                    4,
                    sfilter(ofcolor(I, c),

    rights = merge(apply(

                             lambda rc: rc[1] == rightmost(ofcolor(I, c)))
                ), colours))
    O = paint(I, lefts)
    O = paint(O, rights)
    return O


def dsl5(I):

    # line at its leftmost and rightmost columns and keep only the cells that
    # already belong to the object.
    objs      = fgpartition(I)
    coloured  = mfilter(objs, lambda o: color(o) > 0)
