def dsl1(I):
    # GLOBAL‑REPLACE: find the most frequent colour that is not 0,1,4
    # and replace every 1 with it in one step.
    other   = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main    = argmax(other, lambda c: colorcount(I, c))
    O       = replace(I, 1, main)               # colour‑swap 1 → main
    return O


def dsl2(I):
    # OBJECT‑PAINT: treat all 1‑cells as a single patch,
    # recolour that patch to the dominant non‑{0,1,4} colour,
    # then paint it back on the original grid.
    other   = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main    = argmax(other, lambda c: colorcount(I, c))
    ones    = ofcolor(I, 1)                     # patch of all 1‑cells
    O       = paint(I, recolor(main, ones))
    return O


def dsl3(I):
    # PER‑STRIP: rows that consist entirely of 4 act as separators.
    # 1) turn those separator rows into 0 (so the whole picture is one strip);
    # 2) recolour all 1‑cells inside each connected component with the same
    #    dominant colour;
    # 3) restore the original 4‑rows.
    other   = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main    = argmax(other, lambda c: colorcount(I, c))

    # 1) erase the 4‑rows
    no4     = paint(I, recolor(0, ofcolor(I, 4)))

    # 2) recolour each component separately
    comps   = objects(no4, False, False, True)                     # connected components (0‑background)
    recol   = lambda comp: \
        branch(contains(comp, lambda p: index(I, p) == 1),
              paint(comp, recolor(main, ofcolor(I, 1))),
              comp)
    newc    = apply(recol, comps)
    merged  = merge(newc)

    # 3) put the 4‑rows back
    O       = paint(merged, recolor(4, ofcolor(I, 4)))
    return O


def dsl4(I):
    # PALETTE‑MAP: build a recolour‑patch for every colour in the palette,
    # keep all colours unchanged except colour 1,
    # which is remapped to the dominant non‑{0,1,4} colour.
    other   = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main    = argmax(other, lambda c: colorcount(I, c))

    # patches for all colours
    patches = apply(lambda c: branch(c == 1,
                                   recolor(main, ofcolor(I, 1)),
                                   recolor(c, ofcolor(I, c))),
                    palette(I))

    O = paint(I, merge(patches))
    return O


def dsl5(I):
    # RELATIVE‑PATCH: locate the dominant colour (as in dsl1) and
    # construct a patch that maps every 1‑cell to that colour.
    # The patch is then painted over the original grid.
    other   = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main    = argmax(other, lambda c: colorcount(I, c))

    # map each coordinate that currently holds a 1 to the dominant colour
    map_one = recolor(main, ofcolor(I, 1))

    O = paint(I, map_one)
    return O
