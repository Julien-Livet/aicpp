def dsl1(I):
    # ----------------------------------------------------------------------
    # 1️⃣  Global – crop the “anchor” rows, clear the leading 5 and tile them
    # ----------------------------------------------------------------------
    anchor      = ofcolor(I, 5)                                            # {(r,0)}
    rows_set    = { r for r,_ in anchor }                                 # set of anchor rows
    rows_patch  = mapply(lambda d: shift(anchor, (0, d)),
                        interval(ZERO, width(I), ONE)                     # whole anchor rows
    cleared     = recolor(0, anchor)                                       # 5 → 0 inside the block
    I0          = paint(I, cleared)                                        # grid with leading 5 cleared
    pattern_g   = subgrid(rows_patch, I0)                                   # pattern (cleared)

    # ----- how many copies fit in the empty area below the data ? ----------
    nz          = difference(asindices(I), ofcolor(I, 0))                   # all non‑zero cells
    max_nz_row  = valmax(nz, lambda p: p[0])                               # lowest occupied row
    empty_rows  = subtract(height(I), add(max_nz_row, ONE))                 # rows of pure 0 under the data
    repeat_cnt  = divide(empty_rows, size(rows_set))                       # how many pattern blocks fit

    # ----- build the repeated block by shifting the pattern down ----------------
    h           = size(rows_set)                                           # height of one block
    pat_obj     = asobject(pattern_g)                                      # pattern as an object
    shifts      = mapply(lambda i: shift(pat_obj,
                                         (multiply(add(i, ONE), h), ZERO)),
                        interval(ZERO, repeat_cnt, ONE))                # i = 0 … repeat_cnt‑1,
                                                                       # first copy placed just below the original
    repeated    = merge(shifts)                                            # all shifted copies together

    # ----- keep the original picture (including the leading 5) ----------------
    base        = subgrid(backdrop(nz), I)                                 # original non‑zero part
    O           = trim(paint(base, repeated))                             # overlay copies and trim trailing zeros
    return O

    max_nz_row  = valmax(nz, lambda p: p[0])

    repeat_one  = lambda g: vconcat(g, doubled_g)
    repeated_g  = branch(greater(repeat_cnt, ONE),

    O           = trim(vconcat(base, repeated_g))
    return O
