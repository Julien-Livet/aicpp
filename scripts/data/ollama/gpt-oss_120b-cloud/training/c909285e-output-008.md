def dsl1(I):
    # period‑based crop, then replace the global background with the tile's dominant colour
    bg      = mostcolor(I)
    nb      = difference(asindices(I), ofcolor(I, bg))
    anchor  = first(nb)                                 # top‑left non‑bg cell
    hp      = hperiod(nb)                               # horizontal period
    vp      = vperiod(nb)                               # vertical period
    tile    = crop(I, anchor, astuple(vp, hp))          # the fundamental tile
    fillcol = mostcolor(tile)                           # dominant colour inside the tile
    O       = replace(tile, bg, fillcol)                # recolour background cells
    return O


def dsl2(I):
    # split the grid by the detected periods and take the first block
    bg     = mostcolor(I)
    nb     = difference(asindices(I), ofcolor(I, bg))
    hp     = hperiod(nb)
    vp     = vperiod(nb)
    rows   = hsplit(I, vp)                # horizontal strips of height = vertical period
    first_row = first(rows)                # the top strip
    cols   = vsplit(first_row, hp)         # vertical blocks of width = horizontal period
    tile   = first(cols)                   # the top‑left tile
    fillcol = mostcolor(tile)
    O      = replace(tile, bg, fillcol)
    return O


def dsl3(I):
    # erase the global background, trim outer uniform margin, then extract the tile via periods
    bg      = mostcolor(I)
    O0      = replace(I, bg, 0)                         # background -> 0
    O1      = compress(O0)                            # remove outer 0‑rows/columns
    nb      = difference(asindices(O1), ofcolor(O1, 0))
    anchor  = first(nb)
    hp      = hperiod(nb)
    vp      = vperiod(nb)
    tile    = crop(O1, anchor, astuple(vp, hp))
    fillcol = mostcolor(tile)
    O       = replace(tile, 0, fillcol)                # refill former background
    return O


def dsl4(I):
    # extract tile as in dsl1, then fill background with the colour that dominates its border
    bg      = mostcolor(I)
    nb      = difference(asindices(I), ofcolor(I, bg))
    anchor  = first(nb)
    hp      = hperiod(nb)
    vp      = vperiod(nb)
    tile    = crop(I, anchor, astuple(vp, hp))
    borderc = mostcolor(border(tile))                 # most common colour on the tile border
    O       = replace(tile, bg, borderc)               # recolour background cells
    return O


def dsl5(I):
    # use the centre of mass of all non‑bg cells as anchor for the tile
    bg      = mostcolor(I)
    nb      = difference(asindices(I), ofcolor(I, bg))
    centre  = centerofmass(nb)                         # (row, col) centre of mass
    hp      = hperiod(nb)
    vp      = vperiod(nb)
    # top‑left corner = centre – floor(period/2)
    tl      = subtract(centre,
                astuple(divide(vp, TWO), divide(hp, TWO)))
    tile    = crop(I, tl, astuple(vp, hp))
    fillcol = mostcolor(tile)
    O       = replace(tile, bg, fillcol)
    return O
