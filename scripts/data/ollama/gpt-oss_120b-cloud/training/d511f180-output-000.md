def dsl2(I):
    # Object‑level manipulation: extract colour‑5 and colour‑8 objects,
    # erase them, then paint them back with swapped colours.
    p5 = ofcolor(I, 5)                     # all cells of colour 5
    p8 = ofcolor(I, 8)                     # all cells of colour 8
    O  = cover(I, p5)                      # remove 5‑cells (set to background)
    O  = cover(O, p8)                      # remove 8‑cells
    O  = paint(O, recolor(p5, 8))          # paint former 5‑cells with 8
    O  = paint(O, recolor(p8, 5))          # paint former 8‑cells with 5
    return O


def dsl3(I):
    # Colour‑filter + combine: build a single patch for each colour,
    # recolour and overlay.
    objs   = objects(I, True, False, True)          # all coloured objects
    f5     = colorfilter(objs, 5)                     # set of colour‑5 objects
    f8     = colorfilter(objs, 8)                     # set of colour‑8 objects
    p5     = combine(*tuple(f5)) if f5 else initset()  # unified 5‑patch
    p8     = combine(*tuple(f8)) if f8 else initset()  # unified 8‑patch
    O      = cover(I, p5)                              # erase 5‑cells
    O      = cover(O, p8)                              # erase 8‑cells
    O      = paint(O, recolor(p5, 8))                  # paint with 8
    O      = paint(O, recolor(p8, 5))                  # paint with 5
    return O
