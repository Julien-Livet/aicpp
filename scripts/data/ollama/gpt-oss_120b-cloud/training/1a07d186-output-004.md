def dsl1(I):
    # GLOBAL – compress the non‑empty region and paste it at the top‑left corner.
    C    = compress(I)                     # compact pattern (remove empty rows/cols)
    OBJ  = asobject(C)                     # turn the compact grid into an object
    O    = underpaint(canvas(0, shape(I)), OBJ)  # render on a blank canvas of original size
    return O

    COMBINED  = merge(VLINES)                                   # union of the selected objects
    O         = underpaint(canvas(0, shape(I)), COMBINED)       # render on a blank canvas
    return O

    # RELATIONAL / STRUCTURAL – shift each object downward by its leftmost column index.
    OBJS      = fgpartition(I)                                                   # all objects
    SHIFTED   = apply(lambda o: shift(toindices(o), (leftmost(toindices(o)), 0)), OBJS)  # vertical shift
    COMBINED  = merge(SHIFTED)                                                    # reunite shifted objects
    O         = underpaint(canvas(0, shape(I)), COMBINED)                        # paint on blank grid
    return O
