def dsl1(I):
    # GLOBAL – remove the uniform border (background colour) and
    #          uniformly down‑scale the remaining rectangle to 7×7.
    G = trim(I)                                          # cut away outer rows/cols of bg colour
    sf = divide(height(G), 7)                           # scaling factor (height == width after trim)
    O  = downscale(G, sf)                               # shrink to 7×7
    return O


def dsl2(I):
    # OBJECT – paint the largest foreground object on a 7×7 canvas
    objs   = fgpartition(I)                             # all solid foreground objects
    motif  = argmax(objs, lambda o: size(o))            # the biggest one
    bgcol  = argmax(palette(I), lambda c: colorcount(I, c))
    O      = canvas(bgcol, (7, 7))                       # canvas coloured with background
    O      = paint(O, motif)                             # draw the chosen object (keeps its colours)
    return O


def dsl3(I):
    # COLOR‑FILTER – keep the colour that forms a full horizontal
    #                and a full vertical line, then crop to its tight box.
    horiz_ok = apply(lambda c: hline(ofcolor(I, c)), palette(I))
    vert_ok  = apply(lambda c: vline(ofcolor(I, c)), palette(I))
    frame_c  = first(intersection(horiz_ok, vert_ok))    # colour present in both a H‑line and V‑line
    isolated = ofcolor(I, frame_c)                       # keep only that colour
    O        = compress(isolated)                        # strip empty rows/cols → 7×7
    return O


def dsl4(I):
    # GEOMETRIC – centre a 7×7 window on the centre‑of‑mass of the
    #            most frequent colour region.
    C        = argmax(palette(I), lambda c: colorcount(I, c))
    region   = ofcolor(I, C)
    centre   = centerofmass(region)                      # (row, col) of mass centre
    start    = (subtract(centre[0], THREE), subtract(centre[1], THREE))
    O        = crop(I, start, (7, 7))                   # 7×7 window centred there
    return O


def dsl5(I):
    # RELATIONAL – locate every pair of equal‑coloured horizontal neighbours,
    #              recolour the pair with the background colour,
    #              and render the result on a 7×7 canvas.
    cols       = palette(I)
    pairsets   = apply(lambda c:
                      intersection(ofcolor(I, c),
                                   shift(ofcolor(I, c), LEFT)),
                      cols)                               # all left‑right equal‑colour pairs
    pair_union = merge(pairsets)                         # union of all such pairs
    bgcol      = argmax(palette(I), lambda c: colorcount(I, c))
    pat_obj    = recolor(bgcol, pair_union)              # colour the whole pattern with bg colour
    O          = canvas(bgcol, (7, 7))
    O          = paint(O, pat_obj)
    return O
