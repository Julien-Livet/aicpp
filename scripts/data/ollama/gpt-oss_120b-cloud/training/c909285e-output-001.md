def dsl1(I):
    # ── GLOBAL – biggest solid rectangle, down‑scaled to 7×7 ──
    # colour that occupies the largest rectangular block
    C   = argmax(palette(I), lambda c: colorcount(I, c))
    P   = ofcolor(I, C)                     # all cells with that colour
    B   = backdrop(P)                       # its bounding box (set of indices)

    # rectangle borders
    top    = valmin(B, lambda p: p[0])
    left   = valmin(B, lambda p: p[1])
    bottom = valmax(B, lambda p: p[0])
    right  = valmax(B, lambda p: p[1])

    # crop exactly the rectangle
    G = crop(I, (top, left),
                (add(subtract(bottom, top), ONE),
                 add(subtract(right, left), ONE)))

    # uniform down‑scale to 7×7 (both dimensions are divisible by 7)
    sf = divide(height(G), 7)
    O  = downscale(G, sf)
    return O


def dsl2(I):
    # ── OBJECT – smallest non‑background object placed on 7×7 canvas ──
    objs      = fgpartition(I)                         # all solid objects
    motif     = argmin(objs, lambda o: size(o))         # the smallest one
    bgcol     = argmax(palette(I), lambda c: colorcount(I, c))
    O         = canvas(bgcol, (7, 7))                   # 7×7 canvas with border colour
    O         = paint(O, motif)                         # draw the motif (keeps its colours)
    return O


def dsl3(I):
    # ── COLOR‑FILTER – keep the colour that forms both a full H‑line and V‑line,
    #                 then crop to its tight bounding box (result is 7×7) ──
    horiz_ok = apply(lambda c: hline(ofcolor(I, c)), palette(I))
    vert_ok  = apply(lambda c: vline(ofcolor(I, c)), palette(I))
    frame_col = first(intersection(horiz_ok, vert_ok))        # colour that is both lines

    # isolate that colour, everything else becomes background (0)
    isolated = replace(I, frame_col, frame_col)               # no‑op, just to keep type
    isolated = replace(isolated,
                       frame_col,                         # keep frame colour
                       frame_col)                        # other colours are set to 0 by default
    # crop away empty rows/cols => 7×7
    O = compress(isolated)
    return O


def dsl4(I):
    # ── GEOMETRIC – intersection of dominant H‑line and V‑line gives centre,
    #                 crop a 7×7 window centred there ──
    hcol = first(apply(lambda c: hline(ofcolor(I, c)), palette(I)))
    vcol = first(apply(lambda c: vline(ofcolor(I, c)), palette(I)))

    H    = ofcolor(I, hcol)
    V    = ofcolor(I, vcol)

    centre_pt = center(intersection(H, V))

    start = (subtract(centre_pt[0], THREE), subtract(centre_pt[1], THREE))
    O = crop(I, start, (7, 7))
    return O


def dsl5(I):
    # ── RELATIONAL – find a “00” horizontal pattern, recolour it with border colour,
    #                 then render on 7×7 canvas ──
    zeros      = ofcolor(I, ZERO)                # all 0‑cells
    shifted    = shift(zeros, RIGHT)               # those cells moved one step right
    pattern    = intersection(zeros, shifted)      # cells that have a 0 on their left

    border_col = argmax(palette(I), lambda c: colorcount(I, c))
    # turn the pattern into an object coloured with the border colour
    pat_obj = recolor(border_col, pattern)

    O = canvas(border_col, (7, 7))
    O = paint(O, pat_obj)
    return O
