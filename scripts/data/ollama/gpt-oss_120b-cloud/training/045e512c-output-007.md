def dsl1(I):
    # GLOBAL – take the union of all non‑zero cells, crop to its tight box,
    # enlarge 2×, then paste the enlarged block back at the original top‑left
    # corner of the pattern.
    whole   = merge(objects(I, F, F, T))                     # all coloured cells
