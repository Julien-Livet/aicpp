def dsl1(I):
    # 📦  Object‑level translation:
    #   • collect every cell that does **not** have the background colour 4
    #   • shift the whole set so that its leftmost column touches the rightmost column
    #   • keep the original colours while moving the cells
    objs   = objects(I, False, False, True)                # all non‑4 objects
    patch  = {c for obj in objs for c in obj}             # union of their cells
    delta  = (0, subtract(rightmost(patch), leftmost(patch)))  # horizontal offset
    O = move(I, patch, delta)                            # translate the patch
    return O


def dsl2(I):
    # 🎨  Colour‑statistics swap:
    #   • find the most frequent colour that is **not** the background (4)
    #   • exchange this colour with the background colour everywhere
    bg     = 4
    cand   = palette(I)                                   # all colours present
    target = mostcommon({c for c in cand if c != bg})     # dominant non‑4 colour
    O = switch(I, bg, target)                           # swap background ↔ dominant
    return O


def dsl3(I):

    #   • reflect the complete grid across the main diagonal (counter‑diagonal)
    #   • keep the original grid unchanged
    O = dmirror(I)                                        # diagonal mirror
    return O


def dsl4(I):
    # ↔  Horizontal periodic replication:
    #   • compute the horizontal period of the pattern formed by all non‑4 cells
    #   • copy the pattern once to the right by exactly that period
    objs   = objects(I, False, False, True)
    patch  = {c for obj in objs for c in obj}
    p      = hperiod(patch)                               # fundamental horizontal repeat
    shifted = shift(patch, (0, p))                        # same shape moved one period right
    O = fill(I, 4, patch)                                 # clear original pattern
    O = fill(O, mostcolor(I), shifted)                    # paste the copy (retain original colours)
    return O


def dsl5(I):

    #   • flip the whole picture vertically
    #   • overlay the flipped picture on top of the original,
    #     keeping the original colour wherever the two grids differ
    V = vmirror(I)                                        # vertical mirror
    # cellwise: if a cell differs from the original keep the original colour,
    # otherwise keep the flipped colour (fallback never used here)
    O = cellwise(I, V, 0)                                 # combine the two views
