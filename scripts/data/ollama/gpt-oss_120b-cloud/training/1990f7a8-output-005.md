def dsl1(I):
    # OBJECT‑EXTRACTION + SQUARE‑PADDING
    # 1️⃣  extract the 2‑cells, 2️⃣  build the minimal square that contains them,
    #    padding with 0 if the bounding box is rectangular,
    #    finally down‑scale by 2.
    patch   = ofcolor(I, 2)                                 # set of 2‑cells
    top     = uppermost(patch)                             # row of the topmost 2
    left    = leftmost(patch)                              # column of the leftmost 2
    h       = height(patch)                               # height of the bounding box
    w       = width(patch)                                # width  of the bounding box
    size    = maximum({h, w})                              # side length of the enclosing square
    obj     = toobject(patch, I)                           # coloured object (colour, position)
    shifted = shift(obj, (-top, -left))                    # move it to the origin
    square  = canvas(0, (size, size))                      # empty square canvas
    filled  = paint(square, shifted)                       # paste the object on the canvas
    O       = downscale(filled, 2)                          # final 7×7 output
    return O


def dsl2(I):
    # GLOBAL‑PERIOD ↓‑SCALE + TRIM
    # The pattern of 2’s repeats every hperiod cells (2 for all examples).
    # Down‑scale the whole picture by that period and remove empty borders.
    period = hperiod(ofcolor(I, 2))
    O = trim(downscale(I, period))
    return O


def dsl3(I):
    # COLOR‑FILTERING + COMPRESS + PERIODic ↓‑SCALE
    # Turn the background into a non‑zero colour so that compress keeps the pattern,
    # then down‑scale by the horizontal period and trim.
    I1     = replace(I, 0, 1)                              # background → 1
    period = hperiod(ofcolor(I, 2))
    O = trim(downscale(I1, period))
    return O


def dsl4(I):
    # SUBGRID‑EXTRACTION + SQUARE‑PAD + ↓‑SCALE
    # Take the exact sub‑grid that contains all 2‑cells, pad it to a square,
    # then down‑scale by 2.
    patch   = ofcolor(I, 2)                                 # 2‑cells
    sub     = subgrid(patch, I)                            # minimal rectangle (with background)
    h       = height(sub)
    w       = width(sub)
    size    = maximum({h, w})                              # side of the square we need
    # pad vertically if needed
    pad_v   = canvas(0, (size - h, w))
    vpad    = branch(size > h, vconcat(sub, pad_v), sub)
    # pad horizontally if needed
    pad_h   = canvas(0, (size, size - w))
    square  = branch(size > w, hconcat(vpad, pad_h), vpad)
    O = downscale(square, 2)
    return O


def dsl5(I):

    # then down‑scale by 2 and trim the empty border.

    left  = leftmost(patch)
