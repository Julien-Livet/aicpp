def dsl1(I):
    # colour‑filtering: recolour 1‑cells that lie inside the box of any colour>1
    seeds   = sfilter(palette(I), lambda c: c > 1)                     # colours to use
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)              # their boxes
    patches = prapply(lambda c, b: recolor(c,                               # recolour the 1‑mask
                                          intersection(b,
                                                       ofcolor(I, 1))),
                     seeds, boxes)
    O = paint(I, merge(patches))                                       # paint on top
    return O


def dsl2(I):
    # object‑extraction: treat each colour as a single merged object,
    # then recolour 1‑cells inside its bounding box
    seeds = sfilter(palette(I), lambda c: c > 1)
    objs  = objects(I, False, False, True)                             # every object
    unions = apply(lambda c: merge(colorfilter(objs, c)), seeds)        # one object per colour
    boxes = apply(lambda o: box(o), unions)                            # its box
    patches = prapply(lambda c, b: recolor(c,
                                          intersection(b,
                                                       ofcolor(I, 1))),
                     seeds, boxes)
    O = paint(I, merge(patches))
    return O


def dsl3(I):
    # geometric (corner‑rectangle): build a rectangle from the UL and LR corners
