def dsl1(I):
    # GLOBAL – fill the whole axis‑aligned bounding box of the 5‑pixels with 8
    five      = ofcolor(I, 5)                       # { (r,c) … }

    O         = paint(O1, toobject(five, O1))        # restore the original 5’s
    return O

    O1         = fill(I, 8, adjacents)

def dsl3(I):
