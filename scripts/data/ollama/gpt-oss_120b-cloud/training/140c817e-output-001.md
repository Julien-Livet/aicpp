def dsl1(I):
    """
    GLOBAL‑TRANSFORMATION:
    Remove the uniform background border, rotate the remaining picture 180°,
    mirror it on the main diagonal and paste it back onto a canvas of the
    original size filled with the original background colour.
    """
    bg      = mostcolor(I)                 # background colour (most frequent)
    trimmed = trim(I)                      # cut away the empty border of bg‑cells
    rot     = rot180(trimmed)              # 180° rotation
    diag    = dmirror(rot)                 # mirror on the main diagonal
    O       = underfill(I, bg, diag)       # place the transformed patch on a canvas
    return O


def dsl2(I):
    """
    OBJECT‑EXTRACTION + ADJACENCY:
    •  Find the rare colour (seed).
    •  Build the full cross formed by the rows and columns of all seed cells.
    •  Paint the cross with the seed colour, turn the original seed cells into
       seed+1 (value 2) and colour every orthogonal neighbour of the cross with 3.
    """
    seed   = leastcolor(I)                     # the rare colour (1)
    src    = ofcolor(I, seed)                  # set of seed cells

    # ----- build the cross (full rows + full columns of every seed cell)
    def lines(p):
        h = combine(shoot(p, RIGHT), shoot(p, LEFT))   # whole row of p
        v = combine(shoot(p, DOWN),  shoot(p, UP))     # whole column of p
        return combine(h, v)
    cross  = merge(apply(lines, src))          # union of all rows/columns

    # ----- paint the cross with the seed colour
    O0 = fill(I, seed, cross)

    # ----- orthogonal neighbours of the cross (excluding the cross itself)
    nbrs   = merge(apply(neighbors, cross))
    adj    = difference(nbrs, cross)

    # ----- paint neighbours with 3 and original seeds with 2
    O1 = paint(O0, recolor(3, adj))
    O  = paint(O1, recolor(seed + 1, src))
    return O

    •  Replace every orthogonal neighbour of a seed cell by colour 3.

    GEOMETRIC‑REASONING:
