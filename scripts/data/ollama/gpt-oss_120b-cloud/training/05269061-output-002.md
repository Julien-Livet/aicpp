def dsl1(I):
    # Frequency‑based periodic tiling.
    # 1) Find the three non‑zero colours.
    # 2) Order them by ascending occurrence count.
    # 3) Paint the whole grid column‑wise with that order (period 3).
    cols   = palette(I) - {0}
    c0     = valmin(cols, lambda c: colorcount(I, c))          # least frequent
    c1     = valmin(cols - {c0}, lambda c: colorcount(I, c))   # middle
    c2     = valmax(cols - {c0, c1}, lambda c: colorcount(I, c))# most frequent
    h0     = canvas(c0, (height(I), 1))
    h1     = canvas(c1, (height(I), 1))
    h2     = canvas(c2, (height(I), 1))
    trio   = hconcat(h0, hconcat(h1, h2))                      # 3‑column pattern
    # repeat the 3‑column pattern enough times and crop to the original width
    repeats = (width(I) // 3) + 2
    wide   = hconcat(*([trio] * repeats))
    O      = crop(wide, (0, 0), shape(I))
    return O


def dsl2(I):
    # Object‑centric tiling.
    # Extract the smallest non‑background object and paint it at every
    # possible position that keeps it inside the canvas.
    objs   = objects(I, univalued=True, diagonal=False, without_bg=True)
    pat    = argmin(objs, lambda o: shape(subgrid(o, I))[0] * shape(subgrid(o, I))[1])
    h, w   = shape(subgrid(pat, I))
    H, W   = shape(I)
    offsets = asindices(canvas(0, ((H - h + 1, W - w + 1)))          # all top‑left anchors
    base    = canvas(0, shape(I))
    O = apply(lambda off: paint(base, shift(pat, off)), offsets)
    return O


def dsl3(I):
    # Swap the most frequent and the least frequent non‑zero colours.
    cols   = palette(I) - {0}
    most   = valmax(cols, lambda c: colorcount(I, c))
    least  = valmin(cols, lambda c: colorcount(I, c))
    O = switch(I, most, least)
    return O


def dsl4(I):
