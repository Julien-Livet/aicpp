def dsl1(I):
    # GLOBAL – crop the rows that contain a 5, clear the 5s and tile the cropped block
    anchor      = ofcolor(I, 5)                                   # cells with colour 5
    rows_set    = { r for r,_ in anchor }                         # rows that hold a 5
    rows_patch  = mapply(lambda d: shift(anchor, (0, d)),
                        interval(ZERO, width(I), ONE))           # whole rows of the block
    cleared     = recolor(0, anchor)                             # erase the 5s
    I0          = paint(I, cleared)                               # grid with the block cleared
    pattern_g   = subgrid(rows_patch, I0)                        # cleared rows as a grid

    pat_obj     = asobject(pattern_g)                            # {(value,(r,c))}
    nz          = difference(asindices(I), ofcolor(I, 0))          # all non‑zero cells
    max_nz_row  = valmax(nz, lambda p: p[0])                     # lowest occupied row
    empty_rows  = subtract(height(I), add(max_nz_row, ONE))       # rows that are still empty
    repeat_cnt  = divide(empty_rows, size(rows_set))             # how many full copies fit

    # build the tiled copies
    shifted     = mapply(lambda i: shift(pat_obj,
                                         (multiply(add(i, ONE), size(rows_set)), ZERO)),
                        interval(ZERO, repeat_cnt, ONE))
    tiled_obj   = merge(shifted)

    base_grid   = subgrid(backdrop(nz), I)                        # keep original picture
    O           = trim(paint(base_grid, tiled_obj))               # overlay the copies
    return O

def dsl4(I):
    # GEOMETRIC – build the repeating period by mirroring the cleared rows and tile it
    anchor      = ofcolor(I, 5)
    rows_set    = { r for r,_ in anchor }

    rows_patch  = mapply(lambda d: shift(anchor, (0, d)),
                         interval(ZERO, width(I), ONE))
    cleared     = recolor(0, anchor)
    I0          = paint(I, cleared)
    pattern_g   = subgrid(rows_patch, I0)                     # cleared rows

    # create one full period (original rows + their vertical mirror)
    mirrored    = vmirror(pattern_g)
    period_g    = vconcat(pattern_g, mirrored)

    nz          = difference(asindices(I), ofcolor(I, 0))
    max_nz_row  = valmax(nz, lambda p: p[0])
    empty_rows  = subtract(height(I), add(max_nz_row, ONE))
    period_h    = height(period_g)
    repeat_cnt  = divide(empty_rows, period_h)

    period_obj  = asobject(period_g)
    shifted     = mapply(lambda i: shift(period_obj,
                                         (multiply(add(i, ONE), period_h), ZERO)),
                         interval(ZERO, repeat_cnt, ONE))
    tiled_obj   = merge(shifted)

    base        = subgrid(backdrop(nz), I)
    O           = trim(paint(base, tiled_obj))
    return O
