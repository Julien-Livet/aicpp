def dsl1(I):
    # GLOBAL – trim, recolour, compute vertical period of the pattern,
    # then keep a single period (top block).
    c   = trim(I)
    c1  = replace(c, 8, 5)
    c2  = replace(c1, 1, 9)
    c3  = replace(c2, 3, 8)
    G   = replace(c3, 2, 7)                     # fully recoloured interior
    pat = ofcolor(G, 9) | ofcolor(G, 8) | ofcolor(G, 7)
    per = vperiod(pat)                         # height of one repeating block
    w   = width(G)
    O   = crop(G, (0, 0), (per, w))
    return O
```
