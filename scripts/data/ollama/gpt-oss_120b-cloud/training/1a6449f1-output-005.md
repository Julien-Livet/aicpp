def dsl1(I):
    # Global – crop to the smallest repeating tile of the whole picture
    vp = vperiod(asindices(I))                 # vertical period of the full index set
    hp = hperiod(asindices(I))                 # horizontal period of the full index set
    O  = crop(I, ORIGIN, (vp, hp))            # first tile of that period
    return O


def dsl2(I):
    # Object extraction – tight box around the part of the picture that stays unchanged
    # after a horizontal mirror (the invariant cells)
    mirrored   = hmirror(I)                                    # horizontally mirrored picture
    invariant  = intersection(asobject(I), asobject(mirrored))   # cells that are identical in both
    O          = subgrid(toindices(invariant), I)              # minimal rectangle enclosing them
    return O


def dsl3(I):
    # Colour filtering – delete the background colour and then keep the minimal
    # repeating tile of what remains (the non‑background pattern)
    bg        = mostcolor(I)                         # background colour (most frequent)
    no_bg     = replace(I, bg, -1)                    # mark background with a temporary colour
    non_bg    = difference(asindices(no_bg), ofcolor(no_bg, -1))
    vp        = vperiod(non_bg)                      # vertical period of the non‑background pattern
    hp        = hperiod(non_bg)                      # horizontal period of the non‑background pattern
    O         = crop(no_bg, ORIGIN, (vp, hp))        # first tile of that pattern
    return O


def dsl4(I):
    # Geometric – keep the part of the grid that stays unchanged
    # after a vertical mirror (vertical symmetry axis)
    mirrored   = vmirror(I)                                   # vertically mirrored picture
    invariant  = intersection(asobject(I), asobject(mirrored))# cells unchanged by the mirror
    O          = subgrid(toindices(invariant), I)             # minimal rectangle around them
    return O


def dsl5(I):
    # Relational – region formed by the rarest colour together with its
    # orthogonal neighbours; if that region is empty, return the whole picture
    rare      = leastcolor(I)                                 # colour occurring the fewest times
    rc        = ofcolor(I, rare)                              # cells of that colour
    neigh     = combine(rc,
                        combine(shift(rc, DOWN),
                                combine(shift(rc, UP),
                                        combine(shift(rc, LEFT),
                                                shift(rc, RIGHT)))))
    O         = subgrid(neigh, I)                             # tight box around the neighbourhood
    return O
