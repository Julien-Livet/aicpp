def dsl1(I):
    # Global colour‑cycle using a temporary placeholder (0)
    O0 = replace(I, 9, 0)
    O1 = replace(O0, 1, 9)
    O2 = replace(O1, 3, 1)
    O  = replace(O2, 0, 3)
    return O


def dsl2(I):
    # Object‑level colour‑cycle: next colour is the smallest greater colour,
    # wrapping to the smallest when none is greater.
    colors   = palette(I) - {7, 8}
    sortedc  = tuple(order(colors, lambda c: c))
    succ = lambda c: branch(
                equality(c, last(sortedc)),
                first(sortedc),
                argmin({col for col in sortedc if greater(col, c)},
                       lambda x: x))
    objs = {o for o in objects(I,
                               univalued=False,
                               diagonal=False,
                               without_bg=True)
            if color(o) not in {7, 8}}
    O = mapply(lambda o: fill(I,
                               succ(color(o)),
                               toindices(o)),
               objs)
    return O


def dsl3(I):
    # Sequential fills that rotate the three non‑7/8 colours.
    O1 = fill(I, 1, ofcolor(I, 9))
    O2 = fill(O1, 3, ofcolor(O1, 1))
    O  = fill(O2, 9, ofcolor(O2, 3))
    return O


def dsl4(I):
    # Row‑wise processing: treat each row as a tiny grid
    # and apply the same three‑step colour‑cycle to every row.
    rows   = vsplit(I, height(I))
    proc   = lambda R: fill(
                fill(
                    fill(R, 1, ofcolor(R, 9)),
                3, ofcolor(R, 1)),
            9, ofcolor(R, 3))
    O = vconcat(*apply(proc, rows))
    return O


def dsl5(I):
    # Colour‑cycle built from cover + fill (removing then repainting).
    O0 = cover(I, ofcolor(I, 9))
    O1 = fill(O0, 1, ofcolor(I, 9))
    O2 = cover(O1, ofcolor(I, 1))
    O3 = fill(O2, 3, ofcolor(I, 1))
