def dsl1(I):
    # ---- OBJECT‑LEVEL: recolor the leftmost column of each connected component with 4
    #                  and the rightmost column with 2 (2 painted first, then 4)
    objs    = fgpartition(I)                                 # all colour‑connected objects
    non_bg  = mfilter(objs, lambda o: color(o) > 0)          # drop the background object
    # left‑most cells of every object
    lefts   = apply(lambda p: recolor(4,
                         {(r, c) for (r, c) in p if c == leftmost(p)}),
                    non_bg)
    # right‑most cells of every object
    rights  = apply(lambda p: recolor(2,
                         {(r, c) for (r, c) in p if c == rightmost(p)}),
                    non_bg)
    O = I
    for p in rights: O = paint(O, p)      # 2 first
    for p in lefts: O = paint(O, p)      # 4 overwrites when width‑1
    return O


def dsl2(I):

    objs    = fgpartition(I)
    non_bg  = mfilter(objs, lambda o: color(o) > 0)
    lefts   = apply(lambda p: recolor(4,
                         {(r, leftmost(box(p))) for (r, _) in p}),
                    non_bg)
    rights  = apply(lambda p: recolor(2,
                         {(r, rightmost(box(p))) for (r, _) in p}),
                    non_bg)
    O = I
    for p in rights: O = paint(O, p)

def dsl5(I):
