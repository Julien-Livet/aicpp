def dsl1(I):
    # --- Global propagation via successive north‑west shifts (far → near) ---
    maxd   = maximum({height(I), width(I)})
    offs   = interval(0, maxd + 1, 1)                     # 0 … maxd
    shifted = apply(lambda k: move(I, fgpartition(I), (-k, -k)), offs)
    O      = fold(combine, shifted)                      # overlay, nearest wins
    return O


def dsl2(I):
    # --- Same propagation but generated in reverse order (near → far) ---
    maxd   = maximum({height(I), width(I)})
    offs   = interval(maxd, -1, -1)                      # maxd … 0
    shifted = apply(lambda k: move(I, fgpartition(I), (-k, -k)), offs)
    O      = fold(combine, shifted)                      # farthest drawn first → nearest overwrites
    return O


def dsl3(I):
    # --- Build a rectangle from the origin to each coloured cell independently ---
    nonzero = difference(palette(I), initset(0))          # all colours except 0
    # for every colour c produce a rectangle that covers (0,0) … every cell of colour c
    rects   = apply(
                lambda c: recolor(
                            c,
                            backdrop(
                                combine(
                                    ofcolor(I, c),          # patch of colour c
                                    initset((0, 0))         # the origin cell
                                )
                            )
                         ),
                nonzero)
    O = fold(combine, rects)                           # farthest rectangles first → nearest colour wins
    return O


def dsl4(I):
    # --- Propagation using the “gravitate” primitive (direction to the origin) ---
    srcs   = fgpartition(I)                              # all coloured objects
    maxd   = maximum({height(I), width(I)})
    offs   = interval(0, maxd + 1, 1)
    # move every source object by (‑k,‑k) for each offset k and overlay
    shifted = apply(lambda k: move(I, srcs, (-k, -k)), offs)
    O = fold(combine, shifted)
    return O


def dsl5(I):
    # --- Propagation via repeated “cover” of the blank canvas ---
    blank   = canvas(0, shape(I))                         # empty grid of same size
    maxd    = maximum({height(I), width(I)})
    offs    = interval(0, maxd + 1, 1)
    # start from a blank canvas and repeatedly cover it with a shifted copy
    O = fold(lambda acc, k: cover(acc, move(I, fgpartition(I), (-k, -k))), 
             apply(lambda k: k, offs),                     # the list of offsets is the iterable for fold
             init=blank)
    return O
