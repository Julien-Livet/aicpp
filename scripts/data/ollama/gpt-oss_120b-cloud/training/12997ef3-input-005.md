```python
    O = trim(hupscale(compress(I), 2))
    return O
```

```python
def dsl2(I):
    # OBJECT‑EXTRACTION – keep colours 1 and 2 together, discard everything else,
    # paint the merged object on a blank canvas and crop.
    c1 = ofcolor(I, 1)
    c2 = ofcolor(I, 2)
    merged = combine(c1, c2)                                 # union of the two index sets
    O = trim(paint(canvas(0, shape(I)), toobject(merged, I)))
    return O
```
