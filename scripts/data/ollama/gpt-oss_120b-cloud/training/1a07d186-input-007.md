```python
def dsl1(I):
    # GLOBAL – trim empty outer border, then restore original size
    C   = compress(I)                                 # remove outer empty rows / cols
    OBJ = asobject(C)                                 # treat the compact grid as an object
    O   = underpaint(canvas(0, shape(I)), OBJ)        # paint it back on a blank canvas
    return O
```

## DSL
```python
def dsl2(I):
    # OBJECT EXTRACTION – keep only objects that occupy more than one cell
    OBJS      = fgpartition(I)                        # all coloured objects
    BIG       = sizefilter(OBJS, lambda o: size(o) > 1) # discard single‑pixel objects
    COMBINED  = merge(BIG)                            # union of the remaining objects
    O         = underpaint(canvas(0, shape(I)), COMBINED)
    return O
```
