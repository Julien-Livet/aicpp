# ----------------------------------------------------------------------
# 1. OBJECT‑LEVEL HORIZONTAL COMPRESS
#    Move every non‑7/8 object left until it touches the block of 8’s,
#    filling the vacated cells with 7 (the most frequent colour).
# ----------------------------------------------------------------------
def dsl1(I):
    # all coloured objects that are not 7 or 8
    objs = {o for o in objects(I, univalued=False,
                              diagonal=False, without_bg=True)
            if color(o) not in {7, 8}}

    # column of the right‑most 8 (global, because the 8‑block is left‑aligned)
    right8 = valmax(ofcolor(I, 8), lambda p: p[1])          # (row, col)
    target_col = add(right8[1], ONE)                       # first free column

    O = I
    # shift each object horizontally and repaint the original cells with 7
    O = mapply(
            lambda o: fill(
                    recolor(7, O),                     # set original cells to 7
                    shift(o, astuple(ZERO, subtract(target_col, leftmost(o)))),
                    color(o)),
            objs)
    return O


# ----------------------------------------------------------------------
# 2. BOUNDING‑BOX SWAP
#    For every non‑7/8 object, exchange its colour with the 7‑cells that lie
#    inside its bounding box.  The 7‑cells become the new location of the
#    object; the old location is recoloured 7.
# ----------------------------------------------------------------------
def dsl2(I):
    objs = {o for o in objects(I, univalued=False,
                              diagonal=False, without_bg=True)
            if color(o) not in {7, 8}}

    # colours of the objects replace the 7‑cells that are inside their boxes
    O = mapply(lambda o: fill(I, color(o),
                intersection(ofcolor(I, 7), box(o))), objs)

    # original positions become 7 again
    O = mapply(lambda o: fill(O, 7, o), objs)
    return O


# ----------------------------------------------------------------------
# 3. ROW‑PROJECTION REPLACE
#    For each row, collect the right‑most non‑7/8 colour block,
#    slide it left to sit immediately after the trailing 8’s of that row,
#    and turn the emptied cells to 7.
# ----------------------------------------------------------------------
def dsl3(I):
    other = palette(I) - {7, 8}
    # global right‑most 8 column (the same for every row)
    right8 = valmax(ofcolor(I, 8), lambda p: p[1])
    new_start = add(right8[1], ONE)

    O = I
    # treat each colour independently
    O = mapply(
            lambda c:
                let = ofcolor(I, c)                       # all cells of colour c
                leftmost_c = valmin(let, lambda p: p[1])   # first column of that colour
                shiftvec = astuple(ZERO, subtract(new_start, leftmost_c[1]))
                fill(recolor(7, O), shift(let, shiftvec), c),
            other)
    return O


# ----------------------------------------------------------------------
# 4. GRAVITATE‑TO‑8
#    Every non‑7/8 cell moves in one step toward the nearest 8 in the same
#    row; repeating this move once for every column of the row pushes the
#    whole colour block against the 8‑block.
# ----------------------------------------------------------------------
def dsl4(I):
    other = palette(I) - {7, 8}
    O = I
    # perform a number of single‑step gravitations equal to the width of the grid
    O = apply(
            lambda _: mapply(
                lambda c:
                    let = ofcolor(O, c)
                    dirs = mapply(
                              lambda p: gravitate(
                                            frozenset({p}),
                                            frozenset({first(dneighbors(p) & ofcolor(O, 8)})),
                              let)
                    move(O, let, dirs[0]) if dirs else O,
                other),
            range(width(I)))          # repeat enough times to bring blocks all the way left
    return O


# ----------------------------------------------------------------------
# 5. COMBINE‑AFTER‑MIRROR‑SHIFT
#    Mirror the whole picture vertically, shift the mirror one column right,
#    overlay it on the original and finally recolour the cells that are 7 in
#    both layers with the least‑frequent non‑7/8 colour (which turns out to be
#    the colour that should occupy the vacated space).
# ----------------------------------------------------------------------
def dsl5(I):
    mirrored = vmirror(I)
    shifted  = move(mirrored, asobject(mirrored), (0, ONE))
    O = combine(I, shifted)

    filler = valmin(palette(I) - {7, 8},
                   lambda c: colorcount(I, c))

    sand = frozenset({p for p in ofcolor(O, 7)
                     if add(p, LEFT)  in ofcolor(I, 8)
                     and add(p, RIGHT) in ofcolor(I, 8)})
    O = recolor(filler, sand)
    return O
