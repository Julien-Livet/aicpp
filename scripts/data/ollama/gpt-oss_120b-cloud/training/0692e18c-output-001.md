def dsl1(I):
    # ---------- Global up‑scale + binary mask ----------
    fg   = argmax(difference(palette(I), initset(0)), lbind(colorcount, I))
    inv  = replace(replace(I, fg, 0), 0, fg)                 # swap fg ↔ 0
    N    = height(I)                                        # = width(I)
    # each pixel of inv becomes an N×N block
    big  = vupscale(hupscale(inv, N), N)
    # binary mask: 1 exactly where fg occurs, also expanded to N×N blocks
    mask = upscale(replace(I, fg, 1), N)
    # keep blocks only where mask == 1
    O    = fill(big, 0, ofcolor(mask, 0))
    return O


def dsl2(I):
    # ---------- Object extraction + shift + fold ----------
    fg   = argmax(difference(palette(I), initset(0)), lbind(colorcount, I))
    inv  = replace(replace(I, fg, 0), 0, fg)                 # inverse pattern
    N    = height(I)
    Pobj = asobject(inv)                                    # set of (color,idx)
    O    = canvas(0, (N*N, N*N))
    # for every fg cell place a shifted copy of Pobj
    O = fold(
            ofcolor(I, fg),
            lambda acc idx: paint(acc,
                                 shift(Pobj, (idx[0]*N, idx[1]*N))),
            O)
    return O


def dsl3(I):
    # ---------- Tiling the inverse pattern then masking ----------
    fg   = argmax(difference(palette(I), initset(0)), lbind(colorcount, I))
    inv  = replace(replace(I, fg, 0), 0, fg)                 # inverse pattern
    N    = height(I)

    # repeat the inverse pattern N times horizontally and vertically
    row  = hconcat(*apply(lambda _: inv, interval(0, N, 1)))
    tiled = vconcat(*apply(lambda _: row, interval(0, N, 1)))

    # binary mask of fg cells expanded to N×N blocks
    mask = upscale(replace(I, fg, 1), N)

    # keep only the tiled blocks that correspond to fg cells
    O = fill(tiled, 0, ofcolor(mask, 0))
    return O


def dsl4(I):
    # ---------- Geometric mirroring of the layout ----------
    fg   = argmax(difference(palette(I), initset(0)), lbind(colorcount, I))
    inv  = replace(replace(I, fg, 0), 0, fg)                 # inverse pattern
    N    = height(I)

    # mirror the input along the main diagonal – gives the block layout
    layout = rot90(hmirror(I))                               # equivalent to diagonal mirror
    fg_cells = ofcolor(layout, fg)

    O = canvas(0, (N*N, N*N))
    O = fold(
            fg_cells,
            lambda acc idx: paint(acc,
                                 shift(asobject(inv), (idx[0]*N, idx[1]*N))),
            O)
    return O


def dsl5(I):
    # ---------- Relational – adjacency gives the same mask ----------
    fg   = argmax(difference(palette(I), initset(0)), lbind(colorcount, I))
    inv  = replace(replace(I, fg, 0), 0, fg)                 # inverse pattern
    N    = height(I)

    # cells that are fg or adjacent (8‑neighbour) to fg – in these inputs this set == fg
    fg_cells = ofcolor(I, fg)
    adj      = apply(neighbors, fg_cells)
    target   = union(fg_cells, flatten(adj))

    O = canvas(0, (N*N, N*N))
    O = fold(
            target,
            lambda acc idx: paint(acc,
                                 shift(asobject(inv), (idx[0]*N, idx[1]*N))),
            O)
