def dsl1(I):
    # RECTANGLE‑FILL (backdrop + underpaint)
    BG   = mostcommon(I)
    CAN  = canvas(BG, shape(I))
    CLS  = difference(palette(I), initset(BG))
    BC   = argmax(CLS, lambda c: size(ofcolor(I, c)))          # colour of the solid border
    B    = ofcolor(I, BC)                                     # set of border cells
    OBJ  = objects(I, False, False, True)                     # all coloured objects
    BOBJ = colorfilter(OBJ, BC)                               # the border object(s)
    OTH  = difference(OBJ, BOBJ)                              # interior objects
    RCT  = apply(lambda o:
                recolor(
                    color(o),
                    difference(
                        backdrop(combine(toindices(o), B)),
                        B)),
                OTH)                                          # rectangles of interior objects
    O    = underpaint(CAN, merge(RCT))                        # paint them on a background canvas
    return O
