    col_bg = { (r, c) for (r, c) in asindices(col_slice)
               if index(col_slice, (r, c)) == ZERO }
    col_result = hupscale(combine(col_filled, col_bg),
                          divide(width(I), hp))

    O = branch(greater(vp, hp), col_result, row_result)
    return O


def dsl3(I):
    # SUBGRID‑ORIENTED : extract the tight bounding box, fill it,
    # then upscale the box along the shorter side.
    objs   = objects(I, False, False, True)
    all_idx = merge(objs)                                 # coloured cells (with colors)

    # tight bounding box that contains every coloured cell
    box   = subgrid(all_idx, I)

    rows = apply(lambda o: center(o)[0], objs)
    cols = apply(lambda o: center(o)[1], objs)

    # decide which orientation to repeat
    O = branch(greater(size(rows), size(cols)),
               vupscale(box,
                         divide(height(I), height(box))),
               hupscale(box,
                         divide(width(I), width(box))))
    return O


def dsl4(I):
    # SYMMETRY‑DRIVEN : mirror the minimal slice across the axis that has the
    # smaller period and then repeat the mirrored slice.
    objs = objects(I, False, False, True)

    rows = apply(lambda o: center(o)[0], objs)
    cols = apply(lambda o: center(o)[1], objs)

    vp = add(subtract(maximum(rows), minimum(rows)), ONE)
    hp = add(subtract(maximum(cols), minimum(cols)), ONE)

    # row‑slice (vertical repetition)
    row_slice = crop(I, (minimum(rows), ZERO), (vp, width(I)))
    row_mirror = dmirror(row_slice)          # reflect across counter‑diagonal
    row_pat = combine(row_slice, row_mirror) # overlay original and its mirror
    row_result = vupscale(row_pat,
                          divide(height(I), vp))

    # column‑slice (horizontal repetition)
    col_slice = crop(I, (ZERO, minimum(cols)), (height(I), hp))
    col_mirror = cmirror(col_slice)          # reflect across main diagonal
    col_pat = combine(col_slice, col_mirror)
    col_result = hupscale(col_pat,
                          divide(width(I), hp))

    O = branch(greater(vp, hp), col_result, row_result)
    return O


def dsl5(I):
    # ROW‑&‑COLUMN EXPANSION : each coloured cell expands to both its full
    # row and its full column; the union of all such expansions yields the
    # final pattern (no explicit orientation test needed because the two
    # operations together already generate the required tiling).
    objs = objects(I, False, False, True)

    rows_exp = apply(
        lambda o: fill(I,
                       color(o),
                       { (r, c) for (r, c) in asindices(I)
                         if r == center(o)[0] }),
        objs)
    cols_exp = apply(
        lambda o: fill(I,
                       color(o),
                       { (r, c) for (r, c) in asindices(I)
                         if c == center(o)[1] }),
        objs)

    O = combine(merge(rows_exp), merge(cols_exp))
    return O
