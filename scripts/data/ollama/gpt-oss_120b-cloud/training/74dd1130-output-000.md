def dsl1(I):
    # Global geometric: reflect across counter‑diagonal then rotate 180°
    O = rot180(cmirror(I))
    return O


def dsl2(I):
    # Explicit construction via indexing (object‑extraction style)
    h, w = shape(I)
    # build each row of the transpose by collecting column values
    O = apply(
            lambda j: apply(lambda i: index(I, (i, j)),
                           interval(0, h, 1)),
            interval(0, w, 1))
    return O


def dsl3(I):
    # Geometric: rotate 90° clockwise then mirror vertically
    O = vmirror(rot90(I))
    return O


def dsl4(I):
    # Geometric: rotate 270° (counter‑clockwise 90°) then mirror horizontally
    O = hmirror(rot270(I))
    return O


def dsl5(I):
    # Direct diagonal mirror (main diagonal) – equivalent to transpose
    O = dmirror(I)
    return O
