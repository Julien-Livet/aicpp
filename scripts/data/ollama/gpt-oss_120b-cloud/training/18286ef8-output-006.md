def dsl1(I):
    # GLOBAL – rotate 180°, change 6→9, rotate back.
    R = rot180(I)                 # view the whole picture upside‑down
    R = replace(R, 6, 9)         # colour change in the rotated view
    O = rot180(R)                # restore original orientation
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION – copy the (possibly changed) colour from the
    # 180°‑symmetric cell onto each original 6‑cell.
    six = ofcolor(I, 6)                           # locations of the 6‑cells
    sh  = add(shape(I), (-1, -1))                # (h‑1 , w‑1)
    sym = apply(lambda p: (sh[0] - p[0], sh[1] - p[1]), six)   # symmetry

    # colours that appear at the symmetric locations after the colour swap
    R      = replace(rot180(I), 6, 9)            # rotate & change 6→9
    symcol = toobject(sym, R)                    # colour‑object on the symmetric sites

    # erase the original 6‑cells and paint the gathered colours there
    O = replace(I, 6, 0)
    O = paint(O, toobject(six, symcol))
    return O


def dsl3(I):
    # GEOMETRIC – obtain a 180° rotation by two mirrors,
    # perform the colour change, then undo the mirrors.
    O = hmirror(vmirror(I))          # rot180 using mirrors
    O = replace(O, 6, 9)              # colour change while rotated
    O = vmirror(hmirror(O))           # undo the two mirrors
    return O


def dsl4(I):
    # RELATIONAL / STRUCTURAL – reflect each 6‑cell through the centre
    # of the grid, take the colour that sits there (after 6→9),
    # and write it back onto the original location.
    six = ofcolor(I, 6)
    centre = divide(add(shape(I), (1, 1)), 2)          # centre of the grid (rounded down)
    # reflection through the centre
    refl = apply(lambda p: (add(multiply((2, 2), centre), (-1, -1))[0] - p[0],
                           add(multiply((2, 2), centre), (-1, -1))[1] - p[1]), six)

    R = replace(I, 6, 9)               # colour change on the whole picture
    reflcol = toobject(refl, R)        # colours at the reflected positions

    O = replace(I, 6, 0)
    O = paint(O, toobject(six, reflcol))
    return O


def dsl5(I):
    # COLOR‑FILTERING – protect original 6‑cells with a temporary colour,
    # rotate, change the temporary colour, then merge the result back.
    T = replace(I, 6, 8)               # 6 → temporary colour 8
    R = rot180(T)                      # rotate the temporary picture
    R = replace(R, 8, 9)               # 8 (original 6) becomes 9 in the rotated view

    # take the colour of every cell that was 8 in the original picture,
    # but read that colour from the rotated picture.
    tmpobj = toobject(ofcolor(T, 8), R)   # object of 9’s positioned at the original 6‑cells
    O = paint(I, tmpobj)                # overlay the new colours
    O = replace(O, 6, 0)               # clear any remaining original 6‑cells
    return O
