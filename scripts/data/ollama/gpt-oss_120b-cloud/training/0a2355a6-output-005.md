def dsl1(I):
    # OBJECT‑LEVEL: per component interior →2, edge →1, singletons →3
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    def recol(o):
        interior = intersection(
            o,
            shift(o, UP),
            shift(o, DOWN),
            shift(o, LEFT),
            shift(o, RIGHT)
        )
        border   = difference(o, interior)
        return merge({
            recolor(2, interior),
            recolor(1, border),
            branch(size(o) == 1, recolor(3, o), frozenset())
        })

    return paint(canvas(0, shape(I)), merge(apply(recol, objs)))


def dsl2(I):
    # GLOBAL MORPHOLOGY: compute all interior cells of colour 8 at once,
    # then colour borders 1, singletons 3
    mask      = ofcolor(I, 8)                         # indices of 8
    mask_obj  = toobject(mask, I)                      # turn into object

    interior  = intersection(
        mask_obj,
        shift(mask_obj, UP),
        shift(mask_obj, DOWN),
        shift(mask_obj, LEFT),
        shift(mask_obj, RIGHT)
    )
    border    = difference(mask_obj, interior)

    # singletons are objects of size 1 in the original grid
    singletons = mfilter(
        objects(I, univalued=True, diagonal=False, without_bg=True),
        lambda o: size(o) == 1)

    return paint(
        canvas(0, shape(I)),
        merge({
            recolor(2, interior),
            recolor(1, border),
            recolor(3, union(singletons))          # union makes a single patch
        })
    )


def dsl3(I):
    # COLOR‑FILTERING: start from all 8→2, then overlay border objects (touch frame) →1,
    # finally overlay size‑1 objects →3
    base = replace(I, 8, 2)

    objs          = objects(I, univalued=True, diagonal=False, without_bg=True)
    border_objs   = mfilter(objs, lambda o: bordering(o, I))
    singleton_objs = mfilter(objs, lambda o: size(o) == 1)

    O1 = paint(base, merge(apply(lambda o: recolor(1, o), border_objs)))
    O2 = paint(O1,   merge(apply(lambda o: recolor(3, o), singleton_objs)))

def dsl4(I):
    # BOUNDING‑BOX: components that fully fill their bounding box are “solid”
    # → colour 2, the rest (touching edge of component) →1, singletons →3
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    solid = mfilter(
        objs,
        lambda o: size(o) == size(backdrop(o))   # full rectangle
    )
    rest  = difference(objs, solid)
    singles = mfilter(rest, lambda o: size(o) == 1)

    def recol(o):
        return branch(contained(solid, o), recolor(2, o),
               branch(contained(singles, o), recolor(3, o),
               recolor(1, o))

    return paint(canvas(0, shape(I)), merge(apply(recol, objs)))


def dsl5(I):
    # RELATIONAL‑PIXEL: classify every 8‑cell by its orthogonal neighbours
    # interior (all 4 neighbours also 8) →2,
    # edge (at least one background neighbour) →1,
    # isolated (size‑1 component) →3
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    def cell_type(o):
        # o is a single‑cell object (size 1)
        interior = intersection(
            shift(o, UP),
            shift(o, DOWN),
            shift(o, LEFT),
            shift(o, RIGHT)
        )
        return branch(size(intersection(o, interior)) > 0, recolor(2, o),
               recolor(1, o))

    # recolor all non‑singleton cells
    non_single = mfilter(objs, lambda o: size(o) > 1)
    recolored_non = merge(apply(lambda o: merge(apply(cell_type, {c}) for c in o), non_single))

    # recolor singletons to 3
    singles = mfilter(objs, lambda o: size(o) == 1)
    recolored_single = recolor(3, union(singles))

    return paint(canvas(0, shape(I)), merge({recolored_non, recolored_single}))
