# 1️⃣  OBJECT‑BASED REASONING
#    – Detect a uniform row (height==1) → horizontal tiling.
#    – Otherwise detect a uniform column (width==1) → vertical tiling.
# ----------------------------------------------------------------------
def dsl1(I):
    # all same‑color objects in the input
    OBJ = objects(I, univalued=True, diagonal=False, without_bg=False)

    # ── horizontal case ──
    # keep only objects that are exactly one row high (uniform rows)
    ROWS = sfilter(OBJ, lambda o: height(o) == ONE)
    # if any such row exists we will use it
    HCOND = positive(size(ROWS))

    # the (first) uniform row tells us where to place the horizontal block
    R0    = fst(ROWS)                     # (row, col) of its upper‑left cell
    ROWIX = fst(R0)                       # its row index
    START = multiply(ROWIX, THREE)        # 0, 3 or 6

    # build the horizontal tile (repeat each input row three times)
    ROW0  = crop(I, (ZERO, ZERO), (ONE, THREE))
    ROW1  = crop(I, (ONE,  ZERO), (ONE, THREE))
    ROW2  = crop(I, (TWO,  ZERO), (ONE, THREE))
    HROW0 = hconcat(ROW0, ROW0, ROW0)
    HROW1 = hconcat(ROW1, ROW1, ROW1)
    HROW2 = hconcat(ROW2, ROW2, ROW2)
    HTILE = vconcat(HROW0, vconcat(HROW1, HROW2))

    # insert the tile at the proper vertical offset inside a 9×9 zero canvas
    CANV  = canvas(ZERO, (NINE, NINE))
    HOUT  = fill(CANV, ONE,  # dummy value – the real values will be overwritten
                 asindices(HTILE))
    HOUT  = move(HOUT, asobject(HTILE, I), (START, ZERO))

    # ── vertical case ──
    # keep only objects that are exactly one column wide (uniform columns)
    COLS  = sfilter(OBJ, lambda o: width(o) == ONE)
    VCOND = positive(size(COLS))

    # vertical tile = repeat the whole 3×3 input three times downwards
    VBLOCK = vconcat(I, vconcat(I, I))

    # place it at the left side of a zero canvas
    CANV2 = canvas(ZERO, (NINE, NINE))
    VOUT  = fill(CANV2, ONE,
                 asindices(VBLOCK))
    VOUT  = move(VOUT, asobject(VBLOCK, I), (ZERO, ZERO))

    # choose the appropriate output
    O = branch(HCOND, HOUT, VOUT)
    return O
