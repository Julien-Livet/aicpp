def dsl1(I):
    # GLOBAL‑TRANSFORMATION (crop → diagonal → shift → fill)
    ones      = ofcolor(I, 1)
    last_one  = astuple(lowermost(ones), rightmost(ones))
    start     = increment(last_one)                                   # first empty cell after the 1‑line
    h, w      = shape(I)
    end       = subtract(astuple(h, w), astuple(1, 1))                # bottom‑right corner
    dims      = add(subtract(end, start), astuple(1, 1))              # size of the lower‑right sub‑grid
    subgrid   = crop(I, start, dims)                                  # isolate that region
    diag      = connect(astuple(0, 0), subtract(dims, astuple(1, 1)))  # its main diagonal
    O         = fill(I, 2, shift(diag, start))                        # paint the diagonal back
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION (full diagonal ∧ background → fill)
    ones       = ofcolor(I, 1)
    h, w       = shape(I)
    end        = subtract(astuple(h, w), astuple(1, 1))
    full_diag  = connect(astuple(0, 0), end)                           # complete main diagonal
    background = ofcolor(I, 0)                                         # cells that are still empty
    mask       = intersection(full_diag, background)                  # empty cells on the diagonal
    O          = fill(I, 2, mask)
    return O


def dsl3(I):
    # COLOR‑FILTERING (remaining diagonal cells → underfill)
    ones       = ofcolor(I, 1)
    h, w       = shape(I)
    end        = subtract(astuple(h, w), astuple(1, 1))
    full_diag  = connect(astuple(0, 0), end)                           # whole diagonal
    remaining  = difference(full_diag, ones)                           # cells of the diagonal not already coloured 1
    O          = underfill(I, 2, remaining)                          # colour only background cells among them
    return O


def dsl4(I):
    # GEOMETRIC REASONING (shoot from the cell after the last 1 with step (1,1))
    ones      = ofcolor(I, 1)
    last_one  = astuple(lowermost(ones), rightmost(ones))
    start     = increment(last_one)                                   # first empty cell after the line of 1's
    line      = shoot(start, astuple(1, 1))                            # continue the diagonal
    O         = fill(I, 2, line)
    return O


def dsl5(I):
    # RELATIONAL / STRUCTURAL (derive unit step from the period of the 1‑pixels)
    ones      = ofcolor(I, 1)
    step_h    = hperiod(ones)                                          # horizontal spacing of the 1’s
    step_v    = vperiod(ones)                                          # vertical spacing of the 1’s
    unit_step = astuple(divide(step_h, step_h), divide(step_v, step_v))# reduces to (1,1)
    last_one  = astuple(lowermost(ones), rightmost(ones))
    start     = increment(last_one)
    line      = shoot(start, unit_step)                                # continuation of the diagonal
    O         = fill(I, 2, line)
    return O
