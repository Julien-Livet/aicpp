def dsl1(I):
    # Global‑majority row: replace every non‑empty row with the most common row,
    # otherwise make the row all zeros.
    tmpl   = mostcommon(I, lambda x: x)                     # most frequent row pattern
    zero_r = tuple([0] * width(I))                         # blank row
    O = tuple(apply(lambda r: tmpl if any(c != 0 for c in r) else zero_r,
                I))
    return O


def dsl2(I):
    # Object‑repetition: keep only those objects that appear more than once.
    all_objs      = objects(I, True, False, True)                      # every solid object
    repeat_objs   = sfilter(all_objs,
                           lambda o: size(occurrences(I, o)) > 1)      # occurring ≥2 times
    keep_union    = merge(repeat_objs)                                 # union of repeating objects
    O = paint(I, keep_union)                                          # paint them on the original grid
    return O


def dsl3(I):
    # Color‑frequency filter: keep colors that occupy at least 10 % of the grid.
    total_cells = size(asindices(I))
    common_cols = sfilter(palette(I),
                         lambda c: colorcount(I, c) * 10 > total_cells)
    mask        = initset()
    mask        = apply(lambda col: combine(mask, ofcolor(I, col)), common_cols)
    O = paint(canvas(0, shape(I)), mask)                               # background 0, keep selected colors
    return O


def dsl4(I):
    # Geometric‑periodicity: keep objects whose vertical period is small
    # (i.e. they repeat regularly along the column direction).
    objs        = objects(I, True, False, True)
    short_period = sfilter(objs,
                          lambda o: vperiod(o) != 0 and vperiod(o) < width(I)//2)
    O = paint(I, merge(short_period))
    return O


def dsl5(I):
    # Relational adjacency: keep objects that have at least one neighbouring
    # object of the same colour (i.e. they belong to a repeated structure).
    objs    = objects(I, True, False, True)
    adj_objs = sfilter(objs,
                       lambda o: any(adjacent(o, p) for p in objs if p != o))
    O = paint(I, merge(adj_objs))
    return O
