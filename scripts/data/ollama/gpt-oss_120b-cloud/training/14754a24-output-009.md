def dsl1(I):
    # ----- GLOBAL – expanded Manhattan‑2 neighbourhood of every 4 -----------------
    # 1. all cells that are ≤2 steps from any 4 (full diamond)
    step = lambda p: combine(p,
                combine(
                    combine(shift(p, UP),    shift(p, DOWN)),
                    combine(shift(p, LEFT),  shift(p, RIGHT))
                ))
    diamond = power(step, 2)(ofcolor(I, 4))

    # 2. keep only those cells that are actually colour 5
    target = intersection(ofcolor(I, 5), diamond)

    # 3. recolour the selected 5‑cells with 2
    return fill(I, 2, toobject(target, I))


def dsl2(I):
    # ----- OBJECT‑EXTRACTION – per‑4 expansion, exact two‑step straight lines -----
    fours = ofcolor(I, 4)

    # helper: two‑step cells in a given direction, requiring a clear (0) middle cell
    def two_step(dir_vec):
        one   = intersection(shift(fours, dir_vec), ofcolor(I, 0))
        two   = shift(one, dir_vec)                         # the far cell
        return two

    # combine the four cardinal directions
    reach = combine(
                combine(fours, two_step(UP)),
                combine(two_step(DOWN),
                    combine(two_step(LEFT), two_step(RIGHT)))
    target = intersection(ofcolor(I, 5), reach)
    return fill(I, 2, toobject(target, I))


def dsl3(I):
    # ----- COLOR‑FILTERING – 5 becomes 2 iff a 4 is orthogonally adjacent -------
    fours   = ofcolor(I, 4)

    # all orthogonal neighbours of any 4
    neigh4  = combine(
                combine(shift(fours, UP),    shift(fours, DOWN)),
                combine(shift(fours, LEFT),  shift(fours, RIGHT)))

    # keep just the 5‑cells that sit on those neighbour positions
    target = intersection(ofcolor(I, 5), neigh4)
    return fill(I, 2, toobject(target, I))


def dsl4(I):
    # ----- GEOMETRIC – use vertical symmetry to locate the “mirrored” 5’s -----
    # 1. mirror the whole grid vertically (around the horizontal centre)
    mir = vmirror(I)

    # 2. cells that are 5 both in the original and in the mirrored image
    common5 = intersection(ofcolor(I, 5), ofcolor(mir, 5))

    # 3. from those, select the ones that are also adjacent (Manhattan‑1) to a 4
    fours   = ofcolor(I, 4)
    neigh4  = combine(

                combine(shift(fours, LEFT),  shift(fours, RIGHT)))
    target = intersection(common5, neigh4)
    return fill(I, 2, toobject(target, I))


def dsl5(I):
    # ----- RELATIONAL – a 5 turns to 2 when it has a 4 within Manhattan‑2
    #    *and* every cell on a shortest orthogonal path to that 4 is background (0)
    fours = ofcolor(I, 4)

    # distance‑1 cells that are 0 (potential clean stepping stones)
    clear1 = intersection(
                combine(
                    combine(shift(fours, UP),    shift(fours, DOWN)),
                    combine(shift(fours, LEFT),  shift(fours, RIGHT))),
                ofcolor(I, 0))

    # distance‑2 clean cells: go through a clear1 cell then one more step
    clear2 = combine(
                combine(shift(clear1, UP),    shift(clear1, DOWN)),
                combine(shift(clear1, LEFT),  shift(clear1, RIGHT)))

    # all reachable cells (1‑step or 2‑step via clear background)
    reach = combine(
                combine(fours,                     # the 4‑cells themselves
                    combine(shift(fours, UP),    shift(fours, DOWN),
                    shift(fours, LEFT),          shift(fours, RIGHT)),

    # keep only the original 5‑cells that lie in the reachable set

    return fill(I, 2, toobject(target, I))
