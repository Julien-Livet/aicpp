def dsl1(I):
    # GLOBAL‑INTERIOR‑FILL
    #   – for every coloured object compute its bounding box,
    #   – fill the empty cells inside that box with the most frequent colour of the object,
    #   – keep the original coloured cells.
    OBJ   = objects(I, univalued=False, diagonal=False, without_bg=True)
    O     = canvas(0, shape(I))
    # paint the original objects
    O = apply(lambda o: paint(O, o), OBJ)
    # fill each interior with the object's dominant colour

def dsl2(I):
    # OBJECT‑SIZE FILTER (keep only components larger than one cell)
    OBJ   = objects(I, univalued=False, diagonal=False, without_bg=True)
    KEEP  = mfilter(OBJ, lambda o: greater(size(o), 1))
    O     = canvas(0, shape(I))
    # render each kept component on the empty canvas
    O = apply(lambda o: paint(O, o), KEEP)
    return O


def dsl3(I):
    # COLOUR FILTERING – remove the “marker” colours 3, 5, 8 and any singleton cell
    MARKERS   = combine(combine(ofcolor(I, 3), ofcolor(I, 5)), ofcolor(I, 8))
    OBJ       = objects(I, univalued=False, diagonal=False, without_bg=True)
    SINGLETON = mfilter(OBJ, lambda o: equality(size(o), 1))
    SINGLE    = mapply(lambda o: o, SINGLETON)               # union of the singleton cells
    BAD       = combine(MARKERS, SINGLE)                     # all cells to be cleared
    O         = fill(I, 0, BAD)                              # replace them by background
    return O


def dsl4(I):
    # GEOMETRIC REASONING – overlay horizontal and vertical mirrors,
    #                     original cells dominate
    H = hmirror(I)                         # horizontal flip
    V = vmirror(I)                         # vertical flip
    M = cellwise(H, V, fallback=0)       # union of the two mirrors
    O = cellwise(I, M, fallback=0)        # keep original cells where they exist
    return O


def dsl5(I):
