def dsl1(I):
    # GLOBAL – compress to the minimal non‑zero rectangle then enlarge 2×
    O = compress(I)          # cut away outer empty rows/cols
    O = upscale(O, 2)       # uniform 2‑fold scaling
    return O


def dsl2(I):
    # GLOBAL – trim outer zero border then upscale with separate vertical /
    # horizontal operations (different primitives from dsl1)
    O = trim(I)              # remove outer rows/cols that are all background
    O = hupscale(O, 2)       # double the width
    O = vupscale(O, 2)      # double the height
    return O


def dsl3(I):
    # OBJECT‑LEVEL – extract the bounding box of *all* coloured cells,
    # then upscale the extracted subgrid.
    objs   = objects(I, True, False, True)               # all coloured objects
    all_ix = combine(apply(toindices, objs))            # union of their indices
    bb     = backdrop(all_ix)                           # bounding‑box indices
    O      = subgrid(bb, I)                             # smallest subgrid containing everything
    O      = upscale(O, 2)                              # 2‑fold enlargement
    return O


def dsl4(I):
    # OBJECT‑LEVEL – compute the exact corner coordinates of the overall
    # occupied region, crop with those coordinates, then upscale.
    objs   = objects(I, True, False, True)
    all_ix = combine(apply(toindices, objs))
    start  = ulcorner(backdrop(all_ix))                 # upper‑left corner of the box
    end    = lrcorner(backdrop(all_ix))                 # lower‑right corner of the box
    dims   = add(subtract(end, start), (ONE, ONE))      # width × height of the box
    O      = crop(I, start, dims)                      # crop to the exact rectangle
    O      = upscale(O, 2)                             # double its size
    return O


def dsl5(I):
    # OBJECT‑LEVEL – scale each individual object 2×, then repaint them
    # onto a freshly created canvas of the correct final size.
    objs          = objects(I, True, False, True)
    # size of the final canvas = size of (compress → upscale) of the whole picture
    canvas_size   = shape(upscale(compress(I), 2))
    O0            = canvas(0, canvas_size)

    # function that turns one object into a 2× enlarged patch and paints it
    def paint_scaled(obj):
        # extract the object's minimal subgrid
        sub       = subgrid(backdrop(toindices(obj)), I)
        # enlarge it
        big       = upscale(sub, 2)
        # locate where the original object's ul‑corner would map after scaling
        orig_ul   = ulcorner(toindices(obj))
        new_ul    = multiply(orig_ul, (TWO, TWO))
        # shift the enlarged patch to the correct place and paint it
        shifted   = shift(toobject(big, O0), new_ul)   # convert to object then shift
        return paint(O0, shifted)

    O = apply(paint_scaled, objs)                     # paint all scaled objects
    return O
