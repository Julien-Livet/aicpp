def dsl1(I):

def dsl2(I):

    cells_6 = ofcolor(I, 6)                     # patch of the target colour
    recoloured = recolor(9, cells_6)            # give that patch colour 9
    O = paint(I, recoloured)                    # paint it back onto the grid
    return O


def dsl3(I):
    # COLOUR FILTERING – select objects of colour 6 and fill them with 9.
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)
    six_objs = colorfilter(objs, 6)              # only the objects whose colour is 6
    O = fill(I, 9, combine(*six_objs))        # fill all their cells with 9
    return O


def dsl4(I):
    # GEOMETRIC – perform a 180° turn via mirrors, replace, then undo mirrors.
    O = hmirror(vmirror(I))                    # 180° rotation by mirroring
    O = replace(O, 6, 9)                       # colour change in mirrored view
    O = vmirror(hmirror(O))                    # restore original orientation
    return O


def dsl5(I):

    recoloured = recolor(9, unchanged)         # recolour the (unchanged) patch
    O = paint(I, recoloured)                   # paint back onto the grid
