def dsl1(I):
    # colour‑filtering: recolour the 1‑cells that lie inside the
    # orthogonal bounding box of every colour > 1.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                box(ofcolor(I, c)),   # full box of colour c
                                ofcolor(I, 1)         # all 1‑cells
                            )
                         ),
                seeds)
    O = paint(I, merge(patches))
    return O


def dsl2(I):
    # object‑extraction: treat each non‑background object of colour >1 as a
    # whole, take its orthogonal out‑box and recolour the 1‑cells that fall
    # inside that out‑box.
    objs    = sfilter(fgpartition(I), lambda o: color(o) > 1)
    patches = apply(
                lambda o: recolor(
                            color(o),
                            intersection(
                                outbox(o),           # orthogonal box of the object
                                ofcolor(I, 1)       # all 1‑cells
                            )
                         ),
                objs)
    O = paint(I, merge(patches))
    return O


def dsl3(I):
    # geometric reasoning via corners: for each colour >1 connect the
    # upper‑left and lower‑right corners, turn the segment into a rectangle
    # with `box`, intersect with the 1‑cells and recolour.
    seeds = sfilter(palette(I), lambda c: c > 1)
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                box(
                                    connect(
                                        ulcorner(ofcolor(I, c)),   # UL corner of the colour‑c cells
                                        lrcorner(ofcolor(I, c))    # LR corner of the colour‑c cells
                                    )
                                ),
                                ofcolor(I, 1)               # all 1‑cells
                            )
                         ),
                seeds)
    O = paint(I, merge(patches))
    return O


def dsl4(I):
    # global transformation with `cover`: first erase all 1‑cells,
    # then paint the full bounding‑boxes of the seed colours.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    boxes   = apply(lambda c: box(ofcolor(I, c)), seeds)      # pure boxes, no masking
    base    = cover(I, ofcolor(I, 1))                        # remove every 1‑cell
    O = paint(base, merge(apply(lambda b,c: recolor(c, b), boxes, seeds)))
    return O


def dsl5(I):
    # relational rule using `gravitate`: for every colour >1 build the set of
    # 1‑cells that are reachable by moving orthogonally from a seed cell
    # without leaving its orthogonal bounding box; this is exactly the
    # intersection of the box and the 1‑cells.
    seeds   = sfilter(palette(I), lambda c: c > 1)
    patches = apply(
                lambda c: recolor(
                            c,
                            intersection(
                                box(ofcolor(I, c)),          # bounding box of colour c
                                ofcolor(I, 1)                # all 1‑cells
                            )
                         ),
                seeds)
    O = paint(I, merge(patches))
