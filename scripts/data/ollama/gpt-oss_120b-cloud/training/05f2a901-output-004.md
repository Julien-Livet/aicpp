    rot = rot90(I)                     # turn the picture 90° clockwise

    obj2 = ofcolor(rot, TWO)
    obj8 = ofcolor(rot, EIGHT)

    up2   = uppermost(obj2);   low2  = lowermost(obj2)
    left2 = leftmost(obj2);    right2 = rightmost(obj2)
    up8   = uppermost(obj8);   low8  = lowermost(obj8)
    left8 = leftmost(obj8);    right8 = rightmost(obj8)

    # after rotation, a horizontal shift suffices when the original rows overlap
    rows_overlap = both(flip(greater(up8, low2)),
                         flip(greater(up2, low8)))

    h_gap = branch(greater(left8, right2),
                   subtract(left8, add(right2, ONE)),
                   subtract(left2, add(right8, ONE)))

    dc_sign = branch(greater(left2, right8), NEG_ONE,
                branch(greater(right2, left8), ONE, ZERO))

    dc = branch(rows_overlap, multiply(dc_sign, h_gap), ZERO)

    moved   = recolor(TWO, shift(obj2, astuple(ZERO, dc)))
    cleared = cover(rot, obj2)
    new_rot = paint(cleared, moved)

    O = rot270(new_rot)               # rotate back to original orientation
    return O


def dsl4(I):
    # RELATIONAL REASONING: use the “gravitate” primitive
    obj2 = ofcolor(I, TWO)
    obj8 = ofcolor(I, EIGHT)

    # gives the minimal (dr,dc) that makes obj2 adjacent to obj8
    move = gravitate(obj2, obj8)

    moved   = recolor(TWO, shift(obj2, move))
    cleared = cover(I, obj2)
    O = paint(cleared, moved)
    return O


def dsl5(I):
    # COLOR‑FILTERING + APPLY: shift the whole object in one higher‑order step
    obj2 = ofcolor(I, TWO)
    obj8 = ofcolor(I, EIGHT)

    move = gravitate(obj2, obj8)               # direction vector

    # apply the shift to the whole patch via a singleton container
    shifted_set = apply(lambda p: recolor(TWO, shift(p, move)),
                       initset(obj2))
    shifted_obj = first(shifted_set)

    cleared = cover(I, obj2)
    O = paint(cleared, shifted_obj)
    return O
