def dsl1(I):
    # ---------- GLOBAL TRANSFORMATION ----------
    # Remove all‑zero frontiers (empty rows/columns) from the whole grid.
    # The result is the trimmed version of the input.
    O = trim(I)
    return O


def dsl2(I):
    # ---------- OBJECT EXTRACTION ----------
    # For each non‑zero colour, extract the object, then move it to the
    # top‑left corner of an empty canvas of the original size.
    # Objects are processed in descending colour order so that lower
    # colours overwrite higher ones when they overlap.
    colors = palette(I) - {0}
    empty = canvas(0, shape(I))
    def place(col):
        obj = ofcolor(I, col)                         # patch of this colour
        # shift the object until its upper‑most and left‑most cells touch the border
        up   = gravitate(obj, { (0, c) for c in range(width(I)) })
        left = gravitate(move(I, obj, up), { (r, 0) for r in range(height(I)) })
        shifted = move(I, obj, add(up, left))
        return paint(empty, shifted)
    O = apply(place, colors)          # apply to every colour, combine by painting
    return O


def dsl3(I):
    # ---------- COLOR FILTERING ----------
    # Keep only the three most frequent non‑zero colours, erase everything else,
    # then recompress the remaining pattern by removing empty rows and columns.
    freq = mostcommon(palette(I) - {0})
    top3 = { freq, other(freq, {col for col in palette(I) - {0}}) }
    filtered = underpaint(I, colorfilter(ofcolor(I, top3), 0))  # keep top‑3 colours
    O = trim(filtered)
    return O


def dsl4(I):
    # ---------- GEOMETRIC REASONING ----------
    # Mirror the whole picture horizontally (left‑right) and then shift the
    # mirrored picture upward until it touches the top border.
    mirrored = hmirror(I)
    shift_up = gravitate(ofcolor(mirrored, 0), { (0, c) for c in range(width(I)) })
    O = move(mirrored, asobject(mirrored, I), shift_up)
    return O


def dsl5(I):
    # ---------- RELATIONAL / STRUCTURAL REASONING ----------
    # Treat each coloured component as a patch.  For every pair of patches,
    # compute the vector that would make the first patch adjacent to the second
    # (using gravitate).  Then move the first patch by that vector, processing
    # patches in order of increasing colour so that later patches overwrite earlier ones.
    cols = sorted(palette(I) - {0})
    empty = canvas(0, shape(I))
    def pack(col):
        src = ofcolor(I, col)
        # find the nearest already‑placed patch (if any)
        placed = ofcolor(empty, col)           # empty at first iteration
        dir_vec = gravitate(src, placed) if size(placed) > 0 else (0,0)
        shifted = move(I, src, dir_vec)
        return paint(empty, shifted)
    O = apply(pack, cols)
    return O
