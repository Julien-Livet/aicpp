def dsl1(I):
    # object‑extraction + hole‑count (correct mapping)
    p8   = ofcolor(I, 8)                                          # 8‑object
    a    = multiply(height(p8), width(p8))                         # bounding‑box area
    cnt  = colorcount(p8, 8)                                      # number of 8‑pixels
    holes = subtract(a, cnt)                                      # empty cells inside the box
    col  = branch(greater(holes, 30), 7,                           # many holes → 7
            branch(greater(holes, 10), 3, 2))                      # medium → 3, few → 2
    O    = replace(I, 8, col)                                     # recolour 8‑object
    O    = replace(O, 1, 0)                                      # erase the 1‑shape
    return O


def dsl2(I):
    # global transformation – down‑scaled 8‑density (reversed thresholds)
    small = colorcount(downscale(I, 2), 8)                         # 8‑pixels after 2× downscale
    col   = branch(greater(small, 35), 7,
            branch(greater(small, 15), 3, 2))                     # dense → 7, medium → 3, sparse → 2
    O     = replace(I, 8, col)
    O     = replace(O, 1, 0)
    return O


def dsl3(I):
    # object‑extraction + hole‑ratio (holes / area)
    p8    = ofcolor(I, 8)
    a     = multiply(height(p8), width(p8))                        # area
    cnt   = colorcount(p8, 8)
    holes = subtract(a, cnt)
    ratio = divide(holes, a)                                      # floor‑ratio
    col   = branch(greater(ratio, 0),                               # any hole present
            branch(greater(holes, 30), 7,
                branch(greater(holes, 10), 3, 2)), 2)                # default fallback
    O     = replace(I, 8, col)
    O     = replace(O, 1, 0)
    return O


def dsl4(I):
    # geometric reasoning – sum of dimensions (height + width)
    p8   = ofcolor(I, 8)
    h    = height(p8)
    w    = width(p8)
    dim  = add(h, w)                                              # combined size
    col  = branch(greater(dim, 20), 7,                             # very large → 7
            branch(greater(dim, 12), 3, 2))                        # medium → 3, else 2
    O    = replace(I, 8, col)
    O    = replace(O, 1, 0)
    return O


def dsl5(I):
    # relational – proportion of empty cells inside bounding box
    p8    = ofcolor(I, 8)
    a     = multiply(height(p8), width(p8))                        # total cells in box
    cnt   = colorcount(p8, 8)                                      # occupied cells
    empty = subtract(a, cnt)                                      # empty cells
    prop  = divide(empty, cnt)                                     # empty‑to‑filled ratio
    col   = branch(greater(prop, 2), 7,                            # many empties → 7
            branch(greater(prop, 1), 3, 2))                       # medium → 3, few → 2
    O     = replace(I, 8, col)
    O     = replace(O, 1, 0)
    return O
