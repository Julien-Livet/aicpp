def dsl1(I):
    # GLOBAL‑PERIOD ↓‑SCALE + TRIM
    # Down‑scale the whole picture by the horizontal repetition period of the 2‑cells
    # (the pattern repeats every 2 cells) and then cut away the empty border.
    period = hperiod(ofcolor(I, 2))
    O = trim(downscale(I, period))
    return O


def dsl2(I):
    # SUBGRID → SQUARE → ↓‑SCALE
    # Take the exact rectangle that contains all 2‑cells, pad it on the right/bottom
    # to make it square, then down‑scale by a factor of 2.
    patch   = ofcolor(I, 2)                     # all 2‑cells
    sub     = subgrid(patch, I)                # minimal rectangle (with background)
    h       = height(sub)
    w       = width(sub)
    size    = maximum({h, w})                   # side length of the square we need
    pad_v   = canvas(0, (size - h, w))          # rows to add at the bottom
    pad_h   = canvas(0, (size,   size - w))    # columns to add on the right
    square  = hconcat(vconcat(sub, pad_v), pad_h)
    O = downscale(square, 2)
    return O


def dsl3(I):
    # BACKGROUND‑FLIP + PERIODIC ↓‑SCALE + TRIM
    # Turn the background into a non‑zero colour so that it is kept when we
    # compress, then down‑scale by the horizontal period of the 2‑cells and trim.
    I1      = replace(I, 0, 1)                  # change background 0 → 1
    period  = hperiod(ofcolor(I, 2))
    O = trim(downscale(I1, period))
    return O


def dsl4(I):
    # ROTATE‑180 + ↓‑SCALE + TRIM
    # Extract the sub‑grid that holds the 2‑cells, rotate it 180°, then
    # down‑scale by 2 and remove empty borders.
    patch = ofcolor(I, 2)
    sub   = subgrid(patch, I)
    rot   = rot180(sub)
    O = trim(downscale(rot, 2))
    return O


def dsl5(I):
    # LEFT‑TOP ALIGN + ↓‑SCALE + TRIM
    # Slide the whole 2‑cell pattern until its leftmost cell touches the left
    # border and its topmost cell touches the top border, then down‑scale and trim.
    patch = ofcolor(I, 2)
    left  = leftmost(patch)
    top   = uppermost(patch)
