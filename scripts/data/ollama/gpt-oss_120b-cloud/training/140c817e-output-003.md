        h = combine(shoot(p, RIGHT), shoot(p, LEFT))   # whole row of p
        v = combine(shoot(p, DOWN),  shoot(p, UP))    # whole column of p
        return combine(h, v)
    cross      = merge(apply(lines, src))            # union of all rows/columns
    O0         = fill(I, seed, cross)               # paint the cross with seed colour
    nbrs       = merge(apply(neighbors, cross))      # orthogonal neighbours of the cross
    adj        = difference(nbrs, cross)             # keep only the outside neighbours
    cols       = {c for (_, c) in src}               # columns that contain a seed
    colmask    = { (r, c) for r in range(height(I)) for c in cols }
    adj2       = difference(adj, colmask)            # keep only horizontal neighbours
    O1         = paint(O0, recolor(3, adj2))         # colour those neighbours with 3
    O          = paint(O1, recolor(seed + 1, src))   # recolour original seeds with 2
    return O


def dsl2(I):
    # object‑extraction – obtain seed cells as objects, then build the cross
    seed   = leastcolor(I)
    objs   = fgpartition(I)                    # each colour region as an object
    srcobj = colorfilter(objs, seed)           # objects of the seed colour
    src    = merge(srcobj)                     # set of seed cells
    rows   = { r for (r, _) in src }
    cols   = { c for (_, c) in src }
    cross  = combine(
                { (r, c) for r in rows for c in range(width(I)) },

    adj2   = difference(adj, colmask)
    O1     = paint(O0, recolor(3, adj2))

    return O


def dsl3(I):
    # geometric – shift a unit‑size cross to every seed location
    seed   = leastcolor(I)
    src    = ofcolor(I, seed)

def dsl4(I):
    # colour‑filtering – recolour seeds and their neighbours first,
    # then paint the cross using the original seed colour
    seed   = leastcolor(I)
    src    = ofcolor(I, seed)
    # recolour seeds → 2
    Oa     = paint(I, recolor(seed + 1, src))
    # neighbours of the original seeds → 3
    neigh  = difference(merge(apply(neighbors, src)), src)
    Ob     = paint(Oa, recolor(3, neigh))
    # rows/cols that contain a seed
    rows   = { r for (r, _) in src }
    cols   = { c for (_, c) in src }
    cross  = combine(
                { (r, c) for r in rows for c in range(width(I)) },
                { (r, c) for c in cols for r in range(height(I)) })
    # fill the cross with the seed colour, but keep the recoloured seeds unchanged
    cross_without_src = difference(cross, src)
    Oc = fill(Ob, seed, cross_without_src)
    return Oc


def dsl5(I):
    # global – use the horizontal frontier of the cross and filter out the seed columns
    seed   = leastcolor(I)
