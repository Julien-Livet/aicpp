def dsl1(I):
    # GLOBAL – compress, double‑size, then crop back to the original shape.
    C   = compress(I)                                 # cut away empty outer border
    H   = hupscale(C, 2)                               # double width
    S   = vupscale(H, 2)                               # double height  →  overall 2× scaling
    O   = crop(S, (0, 0), (height(I), width(I)))       # keep the original dimensions
    return O


def dsl2(I):
    # OBJECT – colour every cell that lies inside the bounding rectangle of a
    # non‑background object (including the rectangle’s border).
    BG   = mostcommon(I)                               # background colour
    OBJ  = objects(I, False, False, True)              # all coloured objects
    def paint_bbox(o):
        col  = color(o)                                 # object's colour
        rect = box(o)                                   # its bounding rectangle (outline)
        return fill(I, col, rect)                       # colour the whole rectangle
    O = apply(paint_bbox, OBJ)                         # apply to every object
    return O


def dsl3(I):
    # COLOUR – shift every non‑background colour one step forward
    # (0→1, 1→2, …) while leaving the background unchanged.
    BG   = mostcommon(I)                               # background colour
    COLS = palette(I)                                 # all colours present
    def bump(c):
        return ifelse(equal(c, BG), BG, increment(c))
    O = mapply(lambda c: replace(I, c, bump(c)), COLS) # recolour each colour
    return O


def dsl4(I):
    # GEOMETRIC – reflect the picture horizontally (left↔right) and
    # keep any non‑zero cell that appears in either the original or the mirror.
    M = hmirror(I)                                    # mirror left ↔ right
    O = combine(I, M)                                 # union of occupied cells
    return O


def dsl5(I):
    # RELATIONAL – for every pair of distinct coloured objects, draw a line
    # joining the centres of their bounding boxes.
    OBJ   = objects(I, False, False, True)            # all coloured objects
    CENT  = apply(lambda o: center(o), OBJ)          # centre of each object
    PAIRS = apply(lambda p: shoot(p[0], add(p[1], NEG_ONE)),      # line between centres
                 product(CENT, CENT))
    # colour each line with the colour of the first object in the pair
    def paint_pair(p):
        col = color(p[0])
        line = shoot(center(p[0]), add(center(p[1]), NEG_ONE))
        return fill(I, col, line)
    O = apply(paint_pair, product(OBJ, OBJ))
    return O
