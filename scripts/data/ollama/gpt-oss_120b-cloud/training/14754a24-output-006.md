def dsl1(I):
    # ── two‑step Manhattan dilation using explicit nested combines ──
    fours = ofcolor(I, 4)

    d1 = combine(
            combine(combine(shift(fours, UP), shift(fours, DOWN)),
                    combine(shift(fours, LEFT), shift(fours, RIGHT))))

    d2 = combine(
            combine(combine(shift(d1, UP), shift(d1, DOWN)),
                    combine(shift(d1, LEFT), shift(d1, RIGHT))))

    reach   = combine(combine(fours, d1), d2)                 # cells ≤2 from any 4
    target  = intersection(ofcolor(I, 5), reach)              # 5‑cells that are close
    O = fill(I, 2, toobject(target, I))
    return O


def dsl2(I):
    # ── Manhattan dilation by enumerating all offsets ≤2 (shift‑list) ──
    fours = ofcolor(I, 4)

    offs = {
        UP, DOWN, LEFT, RIGHT,
        astuple(2, 0), astuple(-2, 0),
        astuple(0, 2), astuple(0, -2),
        astuple(1, 1), astuple(1, -1),
        astuple(-1, 1), astuple(-1, -1)
    }

    shifted = apply(rbind(shift, fours), offs)                # each offset applied
    reach   = combine(fours, combine(*shifted))               # union of all patches
    target  = intersection(ofcolor(I, 5), reach)
    O = fill(I, 2, toobject(target, I))
    return O


def dsl3(I):

    hull   = box(fours)                                        # bounding rectangle of all 4s
