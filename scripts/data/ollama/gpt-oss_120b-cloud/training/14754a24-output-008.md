def dsl1(I):
    # relational / path‑based rule:
    # a 5 becomes 2 iff it is 1 step from a 4,
    # or 2 steps from a 4 **and** the intermediate cell is background (0).
    fours   = ofcolor(I, 4)

    # distance‑1 cells (cardinal)
    d1 = combine(
            combine(shift(fours, UP),    shift(fours, DOWN)),
            combine(shift(fours, LEFT),  shift(fours, RIGHT))
        )

    # distance‑2 cells, but only through a clear (0) intermediate cell
    inter_up    = intersection(shift(fours, UP),    ofcolor(I, 0))
    inter_down  = intersection(shift(fours, DOWN),  ofcolor(I, 0))
    inter_left  = intersection(shift(fours, LEFT),  ofcolor(I, 0))
    inter_right = intersection(shift(fours, RIGHT), ofcolor(I, 0))

    d2 = combine(
            combine(shift(inter_up,    UP),    shift(inter_down,    DOWN)),
            combine(shift(inter_left,  LEFT),  shift(inter_right,  RIGHT))
        )

    reach = combine(combine(fours, d1), d2)
    target = intersection(ofcolor(I, 5), reach)
    return fill(I, 2, toobject(target, I))


def dsl2(I):
    # global‑style rule: start from the full Manhattan‑2 neighbourhood
    # (obtained by two applications of the neighbourhood primitive) and
    # subtract every diagonal cell (those that need a turn).
    fours = ofcolor(I, 4)

    step  = lambda p: combine(p,
                combine(
                    combine(shift(p, UP),    shift(p, DOWN)),
                    combine(shift(p, LEFT),  shift(p, RIGHT))
                ))
    reach_all = power(step, 2)(fours)               # distance ≤2 (includes diagonals)

    diag = combine(
            combine(shift(fours, astuple(1, 1)),  shift(fours, astuple(1, -1))),
            combine(shift(fours, astuple(-1, 1)), shift(fours, astuple(-1, -1))),

            combine(shift(fours, astuple(-2, 2)), shift(fours, astuple(-2, -2)))
        )
    reach = difference(reach_all, diag)              # keep only pure cardinal cells
    target = intersection(ofcolor(I, 5), reach)

def dsl3(I):
    # object‑extraction rule: treat each 4 as an object, expand it
    # independently in the four cardinal directions (up to two steps) while
    # requiring the intermediate cell to be background.
    fours = ofcolor(I, 4)

    # helper that for a given direction returns the two‑step cells that have a clear intermediate
    def expand_dir(dir_vec):
        one   = intersection(shift(fours, dir_vec), ofcolor(I, 0))
        two   = shift(one, dir_vec)
        return combine(shift(fours, dir_vec), two)

    left  = expand_dir(LEFT)

    reach = combine(combine(fours, up), combine(down, combine(left, right)))

    return fill(I, 2, toobject(target, I))
