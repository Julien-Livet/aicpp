def dsl1(I):
    # 1️⃣ GLOBAL – simply remove all‑zero border rows/columns.
    O = trim(I)
    return O


def dsl2(I):
    # 2️⃣ OBJECT‑EXTRACTION – keep only the cells of colour 1,
    #   discard everything else and crop to the remaining region.
    ones   = ofcolor(I, 1)                     # indices of colour 1
    blank  = canvas(0, shape(I))               # empty canvas of original size
    only_1 = paint(blank, toobject(ones, I))   # paint only those cells
    O      = trim(only_1)
    return O


def dsl3(I):
    # 3️⃣ COLOR‑FILTER – erase colour 1, keep all other colours unchanged,
    #   then rotate the result 90° clockwise and finally crop.
    no_one = replace(I, 1, 0)          # turn 1 → 0
    rot    = rot90(no_one)             # rotate
    O      = trim(rot)
    return O


def dsl4(I):
    # 4️⃣ GEOMETRIC – mirror the whole picture about the horizontal axis
    #   (top↔bottom) and then remove any all‑zero border rows/columns.
    mirrored = hmirror(I)
    O        = trim(mirrored)
    return O


def dsl5(I):
    # 5️⃣ RELATIONAL – for every non‑zero object move its centre of mass to
    #   the top‑left corner (0,0); finally overlay all shifted objects
    #   on a blank canvas and crop the result.
    objs      = fgpartition(I)                               # all objects
    shifted   = apply(
                 lambda o: shift(
                               o,
                               subtract((0, 0), centerofmass(o))
                           ),
                 objs)                                        # move each to origin
    blank     = canvas(0, shape(I))
    painted   = apply(lambda o: paint(blank, o), shifted)    # paint each shifted object
    O         = trim(first(painted))                           # combine and trim
    return O
