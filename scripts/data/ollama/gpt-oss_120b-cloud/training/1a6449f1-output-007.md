def dsl1(I):
    # 1️⃣  Periodic‑shift intersection
    #    Find the vertical and horizontal periods, shift the whole coloured
    #    set by one whole period and keep only the cells that stay on top of
    #    themselves.  The tight bounding box of that invariant part is the tile.
    obj   = asobject(I)
    vp    = vperiod(obj)                                   # vertical period
    hp    = hperiod(obj)                                   # horizontal period
    shifted = shift(obj, (vp, hp))                         # move one period down‑right
    invariant = intersection(obj, shifted)                 # cells unchanged by the shift
    O = subgrid(toindices(invariant), I)                   # bounding box around them
    return O


def dsl2(I):
    # 2️⃣  Largest monochrome component (object extraction)
    #    Remove the background colour, split the remaining pixels into
    #    connected components and keep the biggest one.
    bg      = mostcolor(I)                                 # background colour
    no_bg   = replace(I, bg, -1)                           # mark background with a sentinel
    comps   = objects(no_bg, univalued=True,
                     diagonal=False, without_bg=True)   # all monochrome objects
    largest = argmax(comps, lambda o: size(o))             # component with most cells
    O = subgrid(toindices(largest), I)                     # its minimal rectangle
    return O


def dsl3(I):
    # 3️⃣  180° rotational symmetry
    #    Cells that look the same after a half‑turn are kept; the bounding box of
    #    that region is the output.
    rot      = rot180(I)                                   # rotate whole picture
    invariant = intersection(asobject(I), asobject(rot))
    O = subgrid(toindices(invariant), I)
    return O


def dsl4(I):
    # 4️⃣  Double‑mirror (horizontal & vertical) invariance
    #    Keep only cells that survive both a horizontal and a vertical mirror.
    hm       = hmirror(I)
    vm       = vmirror(I)
    invariant = intersection(
                    asobject(I),
                    intersection(asobject(hm), asobject(vm))
                )
    O = subgrid(toindices(invariant), I)
    return O


def dsl5(I):
    # 5️⃣  Trim + periodic tiling
    #    First discard any uniform border (trim), then crop the first tile
    #    using the vertical and horizontal periods of the remaining pattern.
    trimmed = trim(I)                                      # remove outer uniform fringe
    vp      = vperiod(asobject(trimmed))
    hp      = hperiod(asobject(trimmed))
    O = crop(trimmed, ORIGIN, (vp, hp))                    # first repeating tile
    return O
