    # keep cells whose colour is unchanged by BOTH a vertical and a horizontal mirror;
    # recolour those cells to 2 (leaves background unchanged)

    hmir   = ofcolor(hmirror(I), None)          # all cells after horizontal mirror

    sym_hv = intersection(sym_v, hmir)            # additionally invariant horizontally
    O = fill(I, 2, toobject(sym_hv, I))
