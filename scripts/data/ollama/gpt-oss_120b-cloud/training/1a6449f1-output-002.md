# 1️⃣  GLOBAL‑TRANSFORMATION  (remove the dominant non‑zero colour)

def dsl1(I):
    # treat 0 as a “dummy” colour so that mostcolor ignores it
    I_nozero   = replace(I, 0, -1)                 # temporary –1 will never be the answer
    dom        = mostcolor(I_nozero)                # most frequent colour except 0
    # erase that colour, turn –1 back into 0 and delete empty rows/cols
    O          = compress(replace(replace(I, dom, 0), -1, 0))
    return O


# -------------------------------------------------
# 2️⃣  OBJECT‑EXTRACTION  (tight box around everything that is NOT the background colour)
# -------------------------------------------------
def dsl2(I):
    bg         = mostcolor(I)                       # background (the most common colour)
    non_bg     = difference(asindices(I), ofcolor(I, bg))
    O          = subgrid(non_bg, I)                 # minimal rectangle that contains all non‑background cells
    return O

def dsl4(I):
    bg          = mostcolor(I)                      # background colour
    pat_cells   = difference(asindices(I), ofcolor(I, bg))
    vp          = vperiod(pat_cells)                # vertical period of the pattern
    hp          = hperiod(pat_cells)                # horizontal period of the pattern
    O           = crop(I, ORIGIN, (vp, hp))         # first tile of the repeating layout
    return O

# 5️⃣  RELATIONAL / STRUCTURAL  (region of the least‑frequent colour together with its orthogonal neighbours)

def dsl5(I):
