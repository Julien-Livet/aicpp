def dsl1(I):
    # 1. Object‑extraction + axis‑aligned translation (direct computation)
    obj2 = ofcolor(I, TWO)
    obj8 = ofcolor(I, EIGHT)

    up2   = uppermost(obj2);   low2  = lowermost(obj2)
    left2 = leftmost(obj2);   right2 = rightmost(obj2)
    up8   = uppermost(obj8);   low8  = lowermost(obj8)
    left8 = leftmost(obj8);   right8 = rightmost(obj8)

    # do the row intervals intersect ?
    rows_overlap = both(
        flip(greater(up8, low2)),
        flip(greater(up2, low8))
    )

    # horizontal shift (positive → right)
    h_shift = branch(
        greater(left8, right2),                           # 8 is to the right of 2
        subtract(left8, add(right2, ONE)),               # move right until adjacent
        subtract(left2, add(right8, ONE))                # otherwise move left
    )

    # vertical shift (positive → down)
    v_shift = branch(
        greater(up8, low2),                               # 8 is below 2
        subtract(up8, add(low2, ONE)),                  # move down until adjacent
        subtract(up2, add(low8, ONE))                    # otherwise move up
    )

    dr = branch(rows_overlap, ZERO, v_shift)
    dc = branch(rows_overlap, h_shift, ZERO)

    moved      = recolor(TWO, shift(obj2, astuple(dr, dc)))
    cleared    = cover(I, obj2)
    O = paint(cleared, moved)
    return O
