def dsl1(I):
    # GLOBAL TRANSFORMATION
    # Rotate the whole picture 180° and then upscale it by a factor of 2
    O = vupscale(hupscale(rot180(I)), 2)

def dsl2(I):
    # OBJECT‑LEVEL EXTRACTION
    # Keep the object of the most frequent colour, normalise its position,
    # and paint it on a blank canvas of the original size.
    main_col   = mostcommon(I)
    objs       = fgpartition(I)                     # all foreground objects
    main_objs  = colorfilter(objs, main_col)        # objects of the dominant colour
    obj        = first(main_objs)                   # the (single) dominant object
    norm_obj   = normalize(obj)                     # shift to the origin
    O = paint(canvas(0, shape(I)), norm_obj)
    return O


def dsl3(I):
    # COLOUR‑FILTERING
    # Replace every background cell (0) with the most common colour present
    # in the grid (including 0 itself – the rule works for the given tasks).
    most = mostcommon(I)
    O = replace(I, 0, most)
    return O


def dsl4(I):
    # GEOMETRIC REASONING
    # Mirror the picture along the main diagonal (counter‑diagonal).
    O = dmirror(I)
    return O


def dsl5(I):
    # RELATIONAL / STRUCTURAL REASONING
    # Move all cells that are NOT of the most common colour one step towards the
    # cluster of the most common colour, then repaint the whole picture.
    main_col   = mostcommon(I)
    main_patch = ofcolor(I, main_col)                # dominant‑colour cells
    all_cells  = asindices(I)                       # every cell index
    other_patch = difference(all_cells, main_patch)  # cells of other colours
    direction  = gravitate(other_patch, main_patch) # direction toward the dominant cluster
    moved      = move(I, other_patch, direction)   # shift the minority cells
    # repaint dominant colour first, then the moved minority cells
    O = paint(paint(canvas(0, shape(I)), main_patch), moved)
    return O
