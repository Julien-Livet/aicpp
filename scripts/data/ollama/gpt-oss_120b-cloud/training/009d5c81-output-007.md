def dsl1(I):
    # GLOBAL – total number of 8‑pixels decides the colour
    dens = colorcount(I, 8)                                 # total 8‑cells
    col  = branch(greater(dens, 30), 7,                     # >30 → large
           branch(greater(dens, 15), 3, 2))                 # 16‑30 → medium, else small
    O = replace(I, 8, col)                                 # recolour the 8‑object
    O = replace(O, 1, 0)                                    # erase the 1‑shape
    return O


def dsl2(I):
    # OBJECT‑EXTRACTION – bounding‑box area (inclusive thresholds)
    p8   = ofcolor(I, 8)                                   # 8‑object
    h    = height(p8)                                      # bbox height
    w    = width(p8)                                       # bbox width
    area = multiply(h, w)                                   # bbox area
    col  = branch(greater(area, 29), 7,                     # ≥30 → large
           branch(greater(area, 14), 3, 2))                # 15‑29 → medium, else small
    O = replace(I, 8, col)
    O = replace(O, 1, 0)
    return O


def dsl3(I):
    # COLOR‑FILTERING – length of the interior boundary of the 8‑object
    p8          = ofcolor(I, 8)
    interior0   = intersection(ofcolor(I, 0), backdrop(p8))   # 0‑cells inside the bbox
    boundary8   = intersection(p8, box(interior0))          # 8‑cells on that interior edge
    cnt         = size(boundary8)                           # boundary length
    col         = branch(greater(cnt, 30), 7,
                branch(greater(cnt, 15), 3, 2))
    O = replace(I, 8, col)
    O = replace(O, 1, 0)
    return O


def dsl4(I):
    # OBJECT‑SIZE – direct pixel count of the 8‑object
    p8  = ofcolor(I, 8)
    sz  = size(p8)                                          # number of 8‑cells
    col = branch(greater(sz, 30), 7,
           branch(greater(sz, 15), 3, 2))
    O = replace(I, 8, col)
    O = replace(O, 1, 0)
    return O


def dsl5(I):
    # GEOMETRIC‑DENSITY – empty cells inside the bounding box
    p8   = ofcolor(I, 8)
    h    = height(p8)
    w    = width(p8)
    area = multiply(h, w)                                   # bbox area
    sz   = size(p8)                                          # filled cells
    diff = subtract(area, sz)                                # empty cells inside bbox
    col  = branch(greater(diff, 15), 2,                       # many empty → small
           branch(greater(diff, 5), 3, 7))                  # few empty → medium / large
    O = replace(I, 8, col)
    O = replace(O, 1, 0)
    return O
