def dsl1(I):
    # Horizontal‑strip processing: each block delimited by full rows of colour 4
    # – if a block contains a colour ≠ 0,1,4, swap that colour with 1 inside the block
    h, w = shape(I)
    sep_rows = {r for r in range(h) if all(index(I, (r, c)) == 4 for c in range(w))}
    strips   = hsplit(I, len(sep_rows) + 1)

    def swap_in_strip(seg):
        other   = sfilter(palette(seg), lambda c: c not in {0, 1, 4})
        main    = branch(size(other) == 0,
                         None,
                         argmax(other, lambda c: colorcount(seg, c)))
        # no other colour → nothing to do
        O = seg
        O = branch(main is None,
                  O,
                  paint(O,
                        merge(apply(lambda c: recolor(c, ofcolor(seg, 1)), {main})),
                        merge(apply(lambda c: recolor(1, ofcolor(seg, c)), {main})))
        return O

    # re‑assemble the strips
    processed = apply(swap_in_strip, strips)          # tuple of processed strips
    # fold the tuple with vconcat (pairwise reduction)
    def fold_vconcat(t):
        return branch(size(t) == 1,
                     first(t),
                     vconcat(first(t), fold_vconcat(tail(t))))
    return fold_vconcat(processed)


def dsl2(I):
    # Global colour‑swap: exchange every 1 with the most frequent colour
    # that is not 0, 1 or 4.
    other_colors = sfilter(palette(I), lambda c: c not in {0, 1, 4})
    main         = argmax(other_colors, lambda c: colorcount(I, c))
    O = paint(I, recolor(main, ofcolor(I, 1)))
    O = paint(O, recolor(1,    ofcolor(I, main)))
    return O

def dsl5(I):
    # Column‑wise visibility using “gravitate” → swap 1 with the colour it sees
    ones   = ofcolor(I, 1)
    fours  = ofcolor(I, 4)
    others = sfilter(palette(I), lambda c: c not in {0, 1, 4})

    sees_up = lambda c: {
        cell for cell in ones
        if intersection(

                ofcolor(I, c))
           and not intersection(
                {(r, cell[1]) for r in range(cell[0] + 1, len(I))},
                fours)
    }

    sees_down = lambda c: {
        cell for cell in ofcolor(I, c)
        if intersection(
                {(r, cell[1]) for r in range(0, cell[0])},
                ones)
           and not intersection(
                {(r, cell[1]) for r in range(0, cell[0])},
                fours)
    }

    recol_up   = apply(lambda c: recolor(c, sees_up(c)),   others)
    recol_down = apply(lambda c: recolor(1, sees_down(c)), others)

    O = paint(I, merge(recol_up))
    O = paint(O, merge(recol_down))
