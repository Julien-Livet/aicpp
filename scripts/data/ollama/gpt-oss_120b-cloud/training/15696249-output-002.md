def dsl1(I):
    # OBJECT‑EXTRACTION: detect a uniform row vs column and tile accordingly
    OBJ   = objects(I, univalued=True, diagonal=False, without_bg=False)
    ROWS  = sfilter(OBJ, lambda o: height(o) == ONE)     # full‑row objects
    COLS  = sfilter(OBJ, lambda o: width(o)  == ONE)     # full‑column objects
    HCOND = positive(size(ROWS))

    YOFF = branch(positive(size(ROWS)), multiply(uppermost(first(ROWS)), THREE), ZERO)
    XOFF = branch(positive(size(COLS)), multiply(leftmost(first(COLS)), THREE), ZERO)

    HROW = hconcat(hconcat(I, I), I)      # 3 × 9 tile
    VCOL = vconcat(vconcat(I, I), I)      # 9 × 3 tile
    CANV = canvas(ZERO, (9, 9))

    HOUT = move(CANV, asobject(HROW), (YOFF, ZERO))
    VOUT = move(CANV, asobject(VCOL), (ZERO, XOFF))

    O = branch(HCOND, HOUT, VOUT)
    return O
