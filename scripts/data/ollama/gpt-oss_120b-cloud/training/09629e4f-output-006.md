def dsl1(I):
    # ----------- global – periodic separators (full rows/cols of 5) -------------
    # objects that are whole separator lines
    sep_objs      = objects(I, True, False, False)
    full_rows    = mfilter(sep_objs, hline)
    full_cols    = mfilter(sep_objs, vline)

    # row / column indices of the separator lines
    rows = apply(lambda o: ulcorner(o)[0], full_rows)
    cols = apply(lambda o: ulcorner(o)[1], full_cols)

    # cut positions (including the outer borders)
    rcuts = order(combine({0}, combine(rows, {shape(I)[0]})), lambda a, b: a < b)
    ccuts = order(combine({0}, combine(cols, {shape(I)[1]})), lambda a, b: a < b)

    # recolour every rectangular block delimited by two consecutive cuts
    blocks = apply(
        lambda r: apply(
            lambda c: recolor(
                mostcolor(
                    crop(
                        I,
                        (list(rcuts)[r], list(ccuts)[c]),
                        (list(rcuts)[r + 1] - list(rcuts)[r],
                         list(ccuts)[c + 1] - list(ccuts)[c])
                    )
                ),
                shoot(
                    (list(rcuts)[r], list(ccuts)[c]),
                    (list(rcuts)[r + 1] - list(rcuts)[r],
                     list(ccuts)[c + 1] - list(ccuts)[c])
                )
            ),
            interval(0, size(ccuts) - 1, 1)
        ),
        interval(0, size(rcuts) - 1, 1)
    )
    O = paint(I, merge(flatten(blocks)))
    return O


def dsl2(I):
    # ------------------- object‑extraction (original idea) ----------------------
    # turn the separator colour (5) into background so that each coloured block
    # becomes a separate connected component.
    S = switch(I, 5, 0)

    # all connected components in the modified grid
    all_objs = objects(S, False, False, False)

    # keep only those components that do NOT contain the separator colour 5
    blocks = colorfilter(all_objs, 5)

    # recolour each block with its dominant non‑5 colour taken from the ORIGINAL grid
    recoloured = apply(
        lambda o: recolor(
            mostcolor(subgrid(o, I)),
            o
        ),
        blocks
    )
    O = paint(I, merge(recoloured))
    return O


def dsl3(I):
    # ------------------- geometric (rotate → global → rotate back) -------------
    # after a 90° rotation the horizontal separator lines become vertical,
    # letting us reuse the global‑cut logic.
    R = rot90(I)

    # separator objects in the rotated grid
    sep_objs   = objects(R, True, False, False)
    full_rows = mfilter(sep_objs, hline)      # were vertical lines before rotation
    full_cols = mfilter(sep_objs, vline)      # were horizontal lines before rotation

    rows = apply(lambda o: ulcorner(o)[0], full_rows)
    cols = apply(lambda o: ulcorner(o)[1], full_cols)

    rcuts = order(combine({0}, combine(rows, {shape(R)[0]})), lambda a, b: a < b)
    ccuts = order(combine({0}, combine(cols, {shape(R)[1]})), lambda a, b: a < b)

    blocks = apply(
        lambda r: apply(
            lambda c: recolor(
                mostcolor(
                    crop(
                        R,
                        (list(rcuts)[r], list(ccuts)[c]),
                        (list(rcuts)[r + 1] - list(rcuts)[r],
                         list(ccuts)[c + 1] - list(ccuts)[c])
                    )
                ),
                shoot(
                    (list(rcuts)[r], list(ccuts)[c]),
                    (list(rcuts)[r + 1] - list(rcuts)[r],
                     list(ccuts)[c + 1] - list(ccuts)[c])
                )
            ),
            interval(0, size(ccuts) - 1, 1)
        ),
        interval(0, size(rcuts) - 1, 1)
    )
    Orot = paint(R, merge(flatten(blocks)))
    O = rot270(Orot)          # back to original orientation
    return O


def dsl4(I):
    # ------------------- local colour majority (3×3 neighbourhood) -----------
    # work only on cells that are not separators (color 5)
    non5 = mfilter(asindices(I), lambda p: index(I, p) != 5)

    # for each such cell take its 3×3 neighbourhood (clipped at the border)
    # and recolour the centre with the most frequent colour inside the patch
    patches = apply(
        lambda p: recolor(
            mostcolor(
                crop(
                    I,
                    (max(p[0] - 1, 0), max(p[1] - 1, 0)),
                    (min(3, shape(I)[0] - max(p[0] - 1, 0)),
                     min(3, shape(I)[1] - max(p[1] - 1, 0)))
            ),
            {p}
        ),
        non5
    )
    O = paint(I, merge(patches))
    return O


def dsl5(I):
    # ------------------- relational / structural (nearest separators) ----------
    # set of separator cells (colour 5)
    sep = ofcolor(I, 5)

    # for every non‑separator cell locate its nearest separator in the
    # four cardinal directions; the rectangle bounded by those four separators
    # is the block that the cell belongs to.
    rects = apply(
        lambda p: recolor(
            mostcolor(
                subgrid(
                    mfilter(asindices(I),
                        lambda q:
                            q[0] > gravitate({p}, sep)[0] + p[0] and
                            q[0] < -gravitate({p}, sep)[0] + p[0] and
                            q[1] > gravitate({p}, sep)[1] + p[1] and
                            q[1] < -gravitate({p}, sep)[1] + p[1]
                    ),
                    I
                )
            ),
            {p}
        ),
        mfilter(asindices(I), lambda p: index(I, p) != 5)
    )
    O = paint(I, merge(rects))
    return O
