def dsl1(I):
    # OBJECT‑CENTRIC → fill each column according to its (col % 3) residue.
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    # order the three single‑colour objects by centre‑of‑mass (gives residues 0,1,2)
    sum_of = lambda o: add(centerofmass(o)[0], centerofmass(o)[1])
    o0 = argmin(objs, sum_of)
    o1 = argmin(difference(objs, {o0}), sum_of)
    o2 = argmin(difference(objs, {o0, o1}), sum_of)

    c0 = color(o0); c1 = color(o1); c2 = color(o2)

    # a full‑height column of a given colour
    col_obj = lambda col, colc: recolor(colc,
                     toobject(apply(lambda r: (r, col), interval(0, height(I), 1)), I))

    obj0 = merge(apply(lambda c: col_obj(c, c0), interval(0, width(I), 3)))
    obj1 = merge(apply(lambda c: col_obj(c, c1), interval(1, width(I), 3)))
    obj2 = merge(apply(lambda c: col_obj(c, c2), interval(2, width(I), 3)))

    empty = canvas(0, shape(I))
    G0 = paint(empty, obj0)
    G1 = paint(G0,    obj1)
    O  = paint(G1,    obj2)
    return O
```

def dsl2(I):
    # GEOMETRIC → rotate the diagonal line to horizontal,
    # copy it across the whole rotated canvas, then undo the rotation.
    rot = rot90(I)                                 # diagonal → horizontal line
    line = argmax(objects(rot, univalued=True, diagonal=False,
                        without_bg=True), lambda o: size(o))

    h, w = shape(subgrid(line, rot))

    row_offs = interval(0, height(I) - h + 1, 1)
    col_offs = interval(0, width(I) - w + 1, 1)
    anchors = merge(apply(lambda r: apply(lambda c: (r, c), col_offs), row_offs))

    empty = canvas(0, shape(rot))
    tiled = apply(lambda off: paint(empty, shift(line, off)), anchors)
    Orot = merge(tiled)          # full tiling in rotated space
    O = rot270(Orot)             # undo the rotation
    return O
```

def dsl3(I):
    # COLOR‑FILTER → create three masks (row+col) % 3 = 0,1,2
    # and recolour them with the three input colours.
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    sum_of = lambda o: add(centerofmass(o)[0], centerofmass(o)[1])
    o0 = argmin(objs, sum_of)
    o1 = argmin(difference(objs, {o0}), sum_of)
    o2 = argmin(difference(objs, {o0, o1}), sum_of)

    c0 = color(o0); c1 = color(o1); c2 = color(o2)

    # all grid coordinates
    all_coords = combine(apply(lambda r:
                       apply(lambda c: (r, c), interval(0, width(I), 1)),
                       interval(0, height(I), 1)))

    # mask for a given residue k (k = 0,1,2)
    def mask(k):
        return apply(lambda rc:
               branch(equality(add(rc[0], rc[1]), add(k, mul(3, div(subtract(add(rc[0], rc[1]), 3)))), initset(rc), set()),
               all_coords)

    # build three masks (the above trick builds a set containing the coordinate
    # iff (row+col) == k (mod 3))
    mk0 = merge(mask(0))
    mk1 = merge(mask(1))
    mk2 = merge(mask(2))

    obj0 = recolor(c0, mk0)
    obj1 = recolor(c1, mk1)
    obj2 = recolor(c2, mk2)

    empty = canvas(0, shape(I))
    O = paint(paint(paint(empty, obj0), obj1), obj2)
    return O

def dsl4(I):
    # NORMALISE‑PATCH → extract the three diagonal strips,
    # normalise each to the origin and tile them everywhere they fit.
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)
    sum_of = lambda o: add(centerofmass(o)[0], centerofmass(o)[1])
    o0 = argmin(objs, sum_of); o1 = argmin(difference(objs,{o0}), sum_of)
    o2 = argmin(difference(objs,{o0,o1}), sum_of)

    cols = palette(I) - {0}
    empty = canvas(0, shape(I))

    # normalise a coloured patch to (0,0) and keep its colour
    norm = lambda c: let(
        patch   = ofcolor(I, c),
        obj     = toobject(patch, I),
        shifted = shift(obj, (-uppermost(patch), -leftmost(patch))),
        recolor(c, shifted))

    normed = apply(norm, cols)          # three normalised diagonal strips
    allobj = merge(normed)              # union of the three strips

    # size of a single strip (they are identical)
    h, w = shape(subgrid(first(normed), I))

    row_offs = interval(0, height(I) - h + 1, 1)
    col_offs = interval(0, width(I) - w + 1, 1)
    anchors = merge(apply(lambda r: apply(lambda c: (r, c), col_offs), row_offs))

    tiled = apply(lambda off: paint(empty, shift(allobj, off)), anchors)
    O = merge(tiled)                    # overlay all copies
    return O

def dsl5(I):
    # PROTOTYPE‑TILING → build a 3×3 tile from the three colours
    # (ordered by centre‑of‑mass) and repeat it over the whole grid.
    objs = objects(I, univalued=True, diagonal=False, without_bg=True)

    sum_of = lambda o: add(centerofmass(o)[0], centerofmass(o)[1])
    o0 = argmin(objs, sum_of); o1 = argmin(difference(objs,{o0}), sum_of)
    o2 = argmin(difference(objs,{o0,o1}), sum_of)

    c0 = color(o0); c1 = color(o1); c2 = color(o2)

    # 3×3 prototype where cell (r,c) gets colour list[(r+c) % 3]
    row0 = combine(apply(lambda c: initset((c0, (0, c))) if equality(c,0) else
                     initset((c1, (0, c))) if equality(c,1) else
                     initset((c2, (0, c))), interval(0,3,1)))
    row1 = combine(apply(lambda c: initset((c2, (1, c))) if equality(c,0) else
                     initset((c0, (1, c))) if equality(c,1) else
                     initset((c1, (1, c))), interval(0,3,1)))
    row2 = combine(apply(lambda c: initset((c1, (2, c))) if equality(c,0) else
                     initset((c2, (2, c))) if equality(c,1) else
                     initset((c0, (2, c))), interval(0,3,1)))

    proto_coords = combine(combine(row0, row1), row2)   # set of (colour,coord)

    proto = replace(canvas(0, (3,3)),
                    0,
                    combine(apply(lambda rc: initset((rc[0], rc[1])), proto_coords)))

    col_offs = interval(0, width(I), 3)

    empty = canvas(0, shape(I))
    tiled = apply(lambda off: paint(empty, shift(proto, off)), anchors)
    O = merge(tiled)                     # any order yields the full tiling
