def dsl1(I):
    # GLOBAL TRANSFORMATION – compress the non‑empty region and render it
    # at the top‑left corner of a blank canvas of the original size.
    C      = compress(I)                                   # compact pattern
    OBJ    = asobject(C, I)                                # turn it into an object
    O      = underpaint(canvas(0, shape(I)), OBJ)          # paint on blank canvas
    return O


def dsl2(I):
    # OBJECT EXTRACTION – keep only the objects that form a full vertical line.
    OBJS      = fgpartition(I)                                      # all coloured objects
    VLINES    = mfilter(OBJS, lambda o: vline(toindices(o)))        # select vertical lines
    COMBINED  = merge(VLINES)                                       # union of the selected patches
    O         = underpaint(canvas(0, shape(I)), COMBINED)            # render on a blank grid
    return O


def dsl3(I):
    # COLOR FILTERING – erase every cell of colour 2 (replace with 0).
    O = replace(I, 2, 0)
    return O


def dsl4(I):
    # GEOMETRIC REASONING – rotate the whole grid 180° and overlay it
    # with the original (the overlay keeps the original colour where it is non‑zero).
    R           = rot180(I)                                          # 180° rotation
    ORIGINAL    = asobject(I, I)                                     # original as an object
    ROTATED_OBJ = asobject(R, I)                                     # rotated as an object
    COMBINED    = merge({ORIGINAL, ROTATED_OBJ})                     # union of both objects
    O           = underpaint(canvas(0, shape(I)), COMBINED)          # render on blank canvas
    return O


def dsl5(I):
    # RELATIONAL / STRUCTURAL – shift each object vertically downward by an
    # amount equal to its leftmost column index.
    OBJS      = fgpartition(I)                                          # all objects
    SHIFTED   = apply(lambda o: shift(o, (leftmost(o), 0)), OBJS)      # vertical shift
    COMBINED  = merge(SHIFTED)                                       # reunite the shifted objects
    O         = underpaint(canvas(0, shape(I)), COMBINED)            # paint on a blank grid
    return O
