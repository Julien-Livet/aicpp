def dsl1(I):
    # ── Global periodicity extraction ──
    # Find the smallest repeating block using the horizontal and vertical periods,
    # then return the first block (top‑left tile).
    idx      = asindices(I)                                 # all cell indices
    obj      = asobject(idx, I)                              # whole grid as an object
    hp       = hperiod(obj)                                 # horizontal period (width of a tile)
    vp       = vperiod(obj)                                 # vertical period   (height of a tile)
    w        = width(I)
    h        = height(I)
    n_vert   = divide(w, hp)                                 # how many tiles fit horizontally
    n_horiz  = divide(h, vp)                                 # how many tiles fit vertically
    vparts   = vsplit(I, n_vert)                             # vertical slices (each width = hp)
    firstcol = list(vparts)[0]                               # left‑most tile column
    hparts   = hsplit(firstcol, n_horiz)                     # horizontal slices (each height = vp)
    O        = list(hparts)[0]                               # top‑left tile
    return O


def dsl2(I):
    # ── Object‑centric extraction ──
    # Partition the grid into colour‑connected objects (ignoring background 8),
    # keep the largest non‑8 object and return its tight bounding box.
    objs        = objects(I, False, False, True)             # all objects, background ignored
    non8_objs   = mfilter(objs, lambda o: not equal(color(o), 8))
    biggest     = argmax(non8_objs, lambda o: size(o))
    bbox        = backdrop(biggest)
    tl          = ulcorner(bbox)
    h           = add(subtract(lowermost(bbox), uppermost(bbox)), ONE)
    w           = add(subtract(rightmost(bbox), leftmost(bbox)), ONE)
    O           = crop(I, tl, (h, w))
    return O
