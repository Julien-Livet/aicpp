def dsl1(I):
    # relational – build rows+cols by shooting from each seed cell
    seed   = leastcolor(I)                     # the rare colour (1)
    src    = ofcolor(I, seed)                  # set of seed cells
    def lines(p):
        h = combine(shoot(p, RIGHT), shoot(p, LEFT))   # whole row of p
        v = combine(shoot(p, DOWN),  shoot(p, UP))    # whole column of p
        return combine(h, v)
    cross  = merge(apply(lines, src))          # union of all rows/columns
    O0     = fill(I, seed, cross)              # paint the cross with the seed colour
    nbrs   = merge(apply(neighbors, cross))    # orthogonal neighbours of the cross
    adj    = difference(nbrs, cross)           # exclude the cross itself
    O1     = paint(O0, recolor(3, adj))        # colour neighbours with 3
    O      = paint(O1, recolor(seed + 1, src))# recolour original seeds with 2
    return O


def dsl2(I):
    # object‑extraction – use row/column index sets to fill the cross
    seed   = leastcolor(I)                     # 1
    src    = ofcolor(I, seed)                  # seed cells
    rows   = { r for (r, _) in src }           # rows containing a seed
    cols   = { c for (_, c) in src }           # columns containing a seed
    cross  = combine(
                { (r, c) for r in rows for c in range(width(I)) },
                { (r, c) for c in cols for r in range(height(I)) })
    O0     = fill(I, seed, cross)              # paint the cross with seed colour
    nbrs   = merge(apply(neighbors, cross))    # neighbours of the cross
    adj    = difference(nbrs, cross)           # keep only the outside neighbours
    O1     = paint(O0, recolor(3, adj))        # colour neighbours with 3
    O      = paint(O1, recolor(seed + 1, src)) # recolour original seeds with 2
    return O


def dsl3(I):
    # geometric – shift a full‑grid cross built at the origin to each seed
    seed   = leastcolor(I)                     # 1
    src    = ofcolor(I, seed)                  # seed cells
    # a full‑grid cross centred at (0,0)
    base   = combine(
                combine(shoot(ORIGIN, RIGHT), shoot(ORIGIN, LEFT)),
                combine(shoot(ORIGIN, DOWN),  shoot(ORIGIN, UP)))
    # move that cross to every seed position and unite
    cross  = merge(apply(lambda p: shift(base, p), src))
    O0     = fill(I, seed, cross)               # paint cross with seed colour

    O1     = paint(O0, recolor(3, adj))         # colour neighbours with 3
    O      = paint(O1, recolor(seed + 1, src)) # recolour original seeds with 2
