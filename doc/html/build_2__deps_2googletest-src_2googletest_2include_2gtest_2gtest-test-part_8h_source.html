51// A copyable object representing the result of a test part (i.e. an
52// assertion or an explicit FAIL(), ADD_FAILURE(), or SUCCESS()).
53//
54// Don't inherit from TestPartResult as its destructor is not virtual.
55class GTEST_API_ [[nodiscard]] TestPartResult {
56 public:
57 // The possible outcomes of a test part (i.e. an assertion or an
58 // explicit SUCCEED(), FAIL(), or ADD_FAILURE()).
59 enum Type {
60 kSuccess, // Succeeded.
61 kNonFatalFailure, // Failed but the test can continue.
62 kFatalFailure, // Failed and the test should be terminated.
63 kSkip // Skipped.
64 };
65
66 // C'tor. TestPartResult does NOT have a default constructor.
67 // Always use this constructor (with parameters) to create a
68 // TestPartResult object.
69 TestPartResult(Type a_type, std::string_view a_file_name, int a_line_number,
70 std::string_view a_message)
71 : type_(a_type),
72 file_name_(a_file_name),
73 line_number_(a_line_number),
74 summary_(ExtractSummary(a_message)),
75 message_(a_message) {}
76
77 // Gets the outcome of the test part.
78 Type type() const { return type_; }
79
80 // Gets the name of the source file where the test part took place, or
81 // NULL if it's unknown.
82 const char* file_name() const {
83 return file_name_.empty() ? nullptr : file_name_.c_str();
84 }
85
86 // Gets the line in the source file where the test part took place,
87 // or -1 if it's unknown.
88 int line_number() const { return line_number_; }
89
90 // Gets the summary of the failure message.
91 const char* summary() const { return summary_.c_str(); }
92
93 // Gets the message associated with the test part.
94 const char* message() const { return message_.c_str(); }
95
96 // Returns true if and only if the test part was skipped.
97 bool skipped() const { return type_ == kSkip; }
98
99 // Returns true if and only if the test part passed.
100 bool passed() const { return type_ == kSuccess; }
101
102 // Returns true if and only if the test part non-fatally failed.
103 bool nonfatally_failed() const { return type_ == kNonFatalFailure; }
104
105 // Returns true if and only if the test part fatally failed.
106 bool fatally_failed() const { return type_ == kFatalFailure; }
107
108 // Returns true if and only if the test part failed.
109 bool failed() const { return fatally_failed() || nonfatally_failed(); }
110
111 private:
112 Type type_;
113
114 // Gets the summary of the failure message by omitting the stack
115 // trace in it.
116 static std::string ExtractSummary(std::string_view message);
117
118 // The name of the source file where the test part took place, or
119 // "" if the source file is unknown.
120 std::string file_name_;
121 // The line in the source file where the test part took place, or -1
122 // if the line number is unknown.
123 int line_number_;
124 std::string summary_; // The test failure summary.
125 std::string message_; // The test failure message.
126};
