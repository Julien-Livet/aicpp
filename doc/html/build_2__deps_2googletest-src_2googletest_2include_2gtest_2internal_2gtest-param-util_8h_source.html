97// Interface for iterating over elements provided by an implementation
98// of ParamGeneratorInterface<T>.
99template <typename T>
100class [[nodiscard]] ParamIteratorInterface {
101 public:
102 virtual ~ParamIteratorInterface() = default;
103 // A pointer to the base generator instance.
104 // Used only for the purposes of iterator comparison
105 // to make sure that two iterators belong to the same generator.
106 virtual const ParamGeneratorInterface<T>* BaseGenerator() const = 0;
107 // Advances iterator to point to the next element
108 // provided by the generator. The caller is responsible
109 // for not calling Advance() on an iterator equal to
110 // BaseGenerator()->End().
111 virtual void Advance() = 0;
112 // Clones the iterator object. Used for implementing copy semantics
113 // of ParamIterator<T>.
114 virtual ParamIteratorInterface* Clone() const = 0;
115 // Dereferences the current iterator and provides (read-only) access
116 // to the pointed value. It is the caller's responsibility not to call
117 // Current() on an iterator equal to BaseGenerator()->End().
118 // Used for implementing ParamGenerator<T>::operator*().
119 virtual const T* Current() const = 0;
120 // Determines whether the given iterator and other point to the same
121 // element in the sequence generated by the generator.
122 // Used for implementing ParamGenerator<T>::operator==().
123 virtual bool Equals(const ParamIteratorInterface& other) const = 0;
124};

126// Class iterating over elements provided by an implementation of
127// ParamGeneratorInterface<T>. It wraps ParamIteratorInterface<T>
128// and implements the const forward iterator concept.
129template <typename T>
130class [[nodiscard]] ParamIterator {
131 public:
132 typedef T value_type;
133 typedef const T& reference;
134 typedef ptrdiff_t difference_type;
135
136 // ParamIterator assumes ownership of the impl_ pointer.
137 ParamIterator(const ParamIterator& other) : impl_(other.impl_->Clone()) {}
138 ParamIterator& operator=(const ParamIterator& other) {
139 if (this != &other) impl_.reset(other.impl_->Clone());
140 return *this;
141 }
142
143 const T& operator*() const { return *impl_->Current(); }
144 const T* operator->() const { return impl_->Current(); }
145 // Prefix version of operator++.
146 ParamIterator& operator++() {
147 impl_->Advance();
148 return *this;
149 }
150 // Postfix version of operator++.
151 ParamIterator operator++(int /*unused*/) {
152 ParamIteratorInterface<T>* clone = impl_->Clone();
153 impl_->Advance();
154 return ParamIterator(clone);
155 }
156 bool operator==(const ParamIterator& other) const {
157 return impl_.get() == other.impl_.get() || impl_->Equals(*other.impl_);
158 }
159 bool operator!=(const ParamIterator& other) const {
160 return !(*this == other);
161 }
162
163 private:
164 friend class ParamGenerator<T>;
165 explicit ParamIterator(ParamIteratorInterface<T>* impl) : impl_(impl) {}
166 std::unique_ptr<ParamIteratorInterface<T>> impl_;
167};

183// Wraps ParamGeneratorInterface<T> and provides general generator syntax
184// compatible with the STL Container concept.
185// This class implements copy initialization semantics and the contained
186// ParamGeneratorInterface<T> instance is shared among all copies
187// of the original object. This is possible because that instance is immutable.
188template <typename T>
189class [[nodiscard]] ParamGenerator {
190 public:
191 typedef ParamIterator<T> iterator;
192
193 explicit ParamGenerator(ParamGeneratorInterface<T>* impl) : impl_(impl) {}
194 ParamGenerator(const ParamGenerator& other) : impl_(other.impl_) {}
195
196 ParamGenerator& operator=(const ParamGenerator& other) {
197 impl_ = other.impl_;
198 return *this;
199 }
200
201 iterator begin() const { return iterator(impl_->Begin()); }
202 iterator end() const { return iterator(impl_->End()); }
203
204 private:
205 std::shared_ptr<const ParamGeneratorInterface<T>> impl_;

208// Generates values from a range of two comparable values. Can be used to
209// generate sequences of user-defined types that implement operator+() and
210// operator<().
211// This class is used in the Range() function.
212template <typename T, typename IncrementT>
213class [[nodiscard]] RangeGenerator : public ParamGeneratorInterface<T> {
214 public:
215 RangeGenerator(T begin, T end, IncrementT step)
216 : begin_(begin),
217 end_(end),
218 step_(step),
219 end_index_(CalculateEndIndex(begin, end, step)) {}
220 ~RangeGenerator() override = default;
221
222 ParamIteratorInterface<T>* Begin() const override {
223 return new Iterator(this, begin_, 0, step_);
224 }
225 ParamIteratorInterface<T>* End() const override {
226 return new Iterator(this, end_, end_index_, step_);
227 }
228
229 private:
230 class Iterator : public ParamIteratorInterface<T> {
231 public:
232 Iterator(const ParamGeneratorInterface<T>* base, T value, int index,
233 IncrementT step)
234 : base_(base), value_(value), index_(index), step_(step) {}
235 ~Iterator() override = default;
236
237 const ParamGeneratorInterface<T>* BaseGenerator() const override {
238 return base_;
239 }
240 void Advance() override {
241 value_ = static_cast<T>(value_ + step_);
242 index_++;
243 }
244 ParamIteratorInterface<T>* Clone() const override {
245 return new Iterator(*this);
246 }
247 const T* Current() const override { return &value_; }
248 bool Equals(const ParamIteratorInterface<T>& other) const override {
249 // Having the same base generator guarantees that the other
250 // iterator is of the same type and we can downcast.
251 GTEST_CHECK_(BaseGenerator() == other.BaseGenerator())
252 << "The program attempted to compare iterators "
253 << "from different generators." << std::endl;
254 const int other_index =
255 CheckedDowncastToActualType<const Iterator>(&other)->index_;
256 return index_ == other_index;
257 }
258
259 private:
260 Iterator(const Iterator& other)
261 : ParamIteratorInterface<T>(),
262 base_(other.base_),
263 value_(other.value_),
264 index_(other.index_),
265 step_(other.step_) {}
266
267 // No implementation - assignment is unsupported.
268 void operator=(const Iterator& other);
269
270 const ParamGeneratorInterface<T>* const base_;
271 T value_;
272 int index_;
273 const IncrementT step_;
274 }; // class RangeGenerator::Iterator
275
276 static int CalculateEndIndex(const T& begin, const T& end,
277 const IncrementT& step) {
278 int end_index = 0;
279 for (T i = begin; i < end; i = static_cast<T>(i + step)) end_index++;
280 return end_index;
281 }
282
283 // No implementation - assignment is unsupported.
284 void operator=(const RangeGenerator& other);
285
286 const T begin_;
287 const T end_;
288 const IncrementT step_;
289 // The index for the end() iterator. All the elements in the generated
290 // sequence are indexed (0-based) to aid iterator comparison.
291 const int end_index_;
292}; // class RangeGenerator

294// Generates values from a pair of STL-style iterators. Used in the
295// ValuesIn() function. The elements are copied from the source range
296// since the source can be located on the stack, and the generator
297// is likely to persist beyond that stack frame.
298template <typename T>
299class [[nodiscard]] ValuesInIteratorRangeGenerator
300 : public ParamGeneratorInterface<T> {
301 public:
302 template <typename ForwardIterator>
303 ValuesInIteratorRangeGenerator(ForwardIterator begin, ForwardIterator end)
304 : container_(begin, end) {}
305 ~ValuesInIteratorRangeGenerator() override = default;
306
307 ParamIteratorInterface<T>* Begin() const override {
308 return new Iterator(this, container_.begin());
309 }
310 ParamIteratorInterface<T>* End() const override {
311 return new Iterator(this, container_.end());
312 }
313
314 private:
315 typedef typename ::std::vector<T> ContainerType;
316
317 class Iterator : public ParamIteratorInterface<T> {
318 public:
319 Iterator(const ParamGeneratorInterface<T>* base,
320 typename ContainerType::const_iterator iterator)
321 : base_(base), iterator_(iterator) {}
322 ~Iterator() override = default;
323
324 const ParamGeneratorInterface<T>* BaseGenerator() const override {
325 return base_;
326 }
327 void Advance() override {
328 ++iterator_;
329 value_.reset();
330 }
331 ParamIteratorInterface<T>* Clone() const override {
332 return new Iterator(*this);
333 }
334 // We need to use cached value referenced by iterator_ because *iterator_
335 // can return a temporary object (and of type other then T), so just
336 // having "return &*iterator_;" doesn't work.
337 // value_ is updated here and not in Advance() because Advance()
338 // can advance iterator_ beyond the end of the range, and we cannot
339 // detect that fact. The client code, on the other hand, is
340 // responsible for not calling Current() on an out-of-range iterator.
341 const T* Current() const override {
342 if (value_.get() == nullptr) value_.reset(new T(*iterator_));
343 return value_.get();
344 }
345 bool Equals(const ParamIteratorInterface<T>& other) const override {
346 // Having the same base generator guarantees that the other
347 // iterator is of the same type and we can downcast.
348 GTEST_CHECK_(BaseGenerator() == other.BaseGenerator())
349 << "The program attempted to compare iterators "
350 << "from different generators." << std::endl;
351 return iterator_ ==
352 CheckedDowncastToActualType<const Iterator>(&other)->iterator_;
353 }
354
355 private:
356 Iterator(const Iterator& other)
357 // The explicit constructor call suppresses a false warning
358 // emitted by gcc when supplied with the -Wextra option.
359 : ParamIteratorInterface<T>(),
360 base_(other.base_),
361 iterator_(other.iterator_) {}
362
363 const ParamGeneratorInterface<T>* const base_;
364 typename ContainerType::const_iterator iterator_;
365 // A cached value of *iterator_. We keep it here to allow access by
366 // pointer in the wrapping iterator's operator->().
367 // value_ needs to be mutable to be accessed in Current().
368 // Use of std::unique_ptr helps manage cached value's lifetime,
369 // which is bound by the lifespan of the iterator itself.
370 mutable std::unique_ptr<const T> value_;
371 }; // class ValuesInIteratorRangeGenerator::Iterator
372
373 // No implementation - assignment is unsupported.
374 void operator=(const ValuesInIteratorRangeGenerator& other);
375
376 const ContainerType container_;
377}; // class ValuesInIteratorRangeGenerator

499// INTERNAL IMPLEMENTATION - DO NOT USE IN USER CODE.
500//
501// ParameterizedTestSuiteInfo accumulates tests obtained from TEST_P
502// macro invocations for a particular test suite and generators
503// obtained from INSTANTIATE_TEST_SUITE_P macro invocations for that
504// test suite. It registers tests with all values generated by all
505// generators when asked.
506template <class TestSuite>
507class [[nodiscard]] ParameterizedTestSuiteInfo
508 : public ParameterizedTestSuiteInfoBase {
509 public:
510 // ParamType and GeneratorCreationFunc are private types but are required
511 // for declarations of public methods AddTestPattern() and
512 // AddTestSuiteInstantiation().
513 using ParamType = typename TestSuite::ParamType;
514 // A function that returns an instance of appropriate generator type.
515 typedef ParamGenerator<ParamType>(GeneratorCreationFunc)();
516 using ParamNameGeneratorFunc = std::string(const TestParamInfo<ParamType>&);
517
518 explicit ParameterizedTestSuiteInfo(std::string name,
519 CodeLocation code_location)
520 : test_suite_name_(std::move(name)),
521 code_location_(std::move(code_location)) {}
522
523 // Test suite base name for display purposes.
524 const std::string& GetTestSuiteName() const override {
525 return test_suite_name_;
526 }
527 // Test suite id to verify identity.
528 TypeId GetTestSuiteTypeId() const override { return GetTypeId<TestSuite>(); }
529 // TEST_P macro uses AddTestPattern() to record information
530 // about a single test in a LocalTestInfo structure.
531 // test_suite_name is the base name of the test suite (without invocation
532 // prefix). test_base_name is the name of an individual test without
533 // parameter index. For the test SequenceA/FooTest.DoBar/1 FooTest is
534 // test suite base name and DoBar is test base name.
535 void AddTestPattern(const char*, const char* test_base_name,
536 TestMetaFactoryBase<ParamType>* meta_factory,
537 CodeLocation code_location) {
538 tests_.emplace_back(
539 new TestInfo(test_base_name, meta_factory, std::move(code_location)));
540 }
541 // INSTANTIATE_TEST_SUITE_P macro uses AddGenerator() to record information
542 // about a generator.
543 int AddTestSuiteInstantiation(std::string instantiation_name,
544 GeneratorCreationFunc* func,
545 ParamNameGeneratorFunc* name_func,
546 const char* file, int line) {
547 instantiations_.emplace_back(std::move(instantiation_name), func, name_func,
548 file, line);
549 return 0; // Return value used only to run this method in namespace scope.
550 }
551 // UnitTest class invokes this method to register tests in this test suite
552 // right before running tests in RUN_ALL_TESTS macro.
553 // This method should not be called more than once on any single
554 // instance of a ParameterizedTestSuiteInfoBase derived class.
555 // UnitTest has a guard to prevent from calling this method more than once.
556 void RegisterTests() override {
557 bool generated_instantiations = false;
558
559 std::string test_suite_name;
560 std::string test_name;
561 for (const std::shared_ptr<TestInfo>& test_info : tests_) {
562 for (const InstantiationInfo& instantiation : instantiations_) {
563 const std::string& instantiation_name = instantiation.name;
564 ParamGenerator<ParamType> generator((*instantiation.generator)());
565 ParamNameGeneratorFunc* name_func = instantiation.name_func;
566 const char* file = instantiation.file;
567 int line = instantiation.line;
568
569 if (!instantiation_name.empty())
570 test_suite_name = instantiation_name + "/";
571 else
572 test_suite_name.clear();
573 test_suite_name += test_suite_name_;
574
575 size_t i = 0;
576 std::set<std::string> test_param_names;
577 for (const auto& param : generator) {
578 generated_instantiations = true;
579
580 test_name.clear();
581
582 std::string param_name =
583 name_func(TestParamInfo<ParamType>(param, i));
584
585 GTEST_CHECK_(IsValidParamName(param_name))
586 << "Parameterized test name '" << param_name
587 << "' is invalid (contains spaces, dashes, or any "
588 "non-alphanumeric characters other than underscores), in "
589 << file << " line " << line << "" << std::endl;
590
591 GTEST_CHECK_(test_param_names.count(param_name) == 0)
592 << "Duplicate parameterized test name '" << param_name << "', in "
593 << file << " line " << line << std::endl;
594
595 if (!test_info->test_base_name.empty()) {
596 test_name.append(test_info->test_base_name).append("/");
597 }
598 test_name += param_name;
599
600 test_param_names.insert(std::move(param_name));
601
602 MakeAndRegisterTestInfo(
603 test_suite_name, test_name.c_str(),
604 nullptr, // No type parameter.
605 PrintToString(param).c_str(), test_info->code_location,
606 GetTestSuiteTypeId(),
607 SuiteApiResolver<TestSuite>::GetSetUpCaseOrSuite(file, line),
608 SuiteApiResolver<TestSuite>::GetTearDownCaseOrSuite(file, line),
609 test_info->test_meta_factory->CreateTestFactory(param));
610 ++i;
611 } // for param
612 } // for instantiation
613 } // for test_info
614
615 if (!generated_instantiations) {
616 // There are no generaotrs, or they all generate nothing ...
617 InsertSyntheticTestCase(GetTestSuiteName(), code_location_,
618 !tests_.empty());
619 }
620 } // RegisterTests

660 // Check for empty string

702 // Looks up or creates and returns a structure containing information about
703 // tests and instantiations of a particular test suite.
704 template <class TestSuite>
705 ParameterizedTestSuiteInfo<TestSuite>* GetTestSuitePatternHolder(
706 std::string test_suite_name, CodeLocation code_location) {
707 ParameterizedTestSuiteInfo<TestSuite>* typed_test_info = nullptr;
708
709 auto item_it = suite_name_to_info_index_.find(test_suite_name);
710 if (item_it != suite_name_to_info_index_.end()) {
711 auto* test_suite_info = test_suite_infos_[item_it->second];
712 if (test_suite_info->GetTestSuiteTypeId() != GetTypeId<TestSuite>()) {
713 // Complain about incorrect usage of Google Test facilities
714 // and terminate the program since we cannot guaranty correct
715 // test suite setup and tear-down in this case.
716 ReportInvalidTestSuiteType(test_suite_name.c_str(), code_location);
717 posix::Abort();
718 } else {
719 // At this point we are sure that the object we found is of the same
720 // type we are looking for, so we downcast it to that type
721 // without further checks.
722 typed_test_info =
723 CheckedDowncastToActualType<ParameterizedTestSuiteInfo<TestSuite>>(
724 test_suite_info);
725 }
726 }
727 if (typed_test_info == nullptr) {
728 typed_test_info = new ParameterizedTestSuiteInfo<TestSuite>(
729 test_suite_name, std::move(code_location));
730 suite_name_to_info_index_.emplace(std::move(test_suite_name),
731 test_suite_infos_.size());
732 test_suite_infos_.push_back(typed_test_info);
733 }
734 return typed_test_info;
735 }
736 void RegisterTests() {
737 for (auto& test_suite_info : test_suite_infos_) {
738 test_suite_info->RegisterTests();
739 }

826template <typename... T>
827class [[nodiscard]] CartesianProductGenerator
828 : public ParamGeneratorInterface<::std::tuple<T...>> {
829 public:
830 typedef ::std::tuple<T...> ParamType;
831
832 CartesianProductGenerator(const std::tuple<ParamGenerator<T>...>& g)
833 : generators_(g) {}
834 ~CartesianProductGenerator() override = default;
835
836 ParamIteratorInterface<ParamType>* Begin() const override {
837 return new Iterator(this, generators_, false);
838 }
839 ParamIteratorInterface<ParamType>* End() const override {
840 return new Iterator(this, generators_, true);
841 }
842
843 private:
844 template <class I>
845 class IteratorImpl;
846 template <size_t... I>
847 class IteratorImpl<std::index_sequence<I...>>
848 : public ParamIteratorInterface<ParamType> {
849 public:
850 IteratorImpl(const ParamGeneratorInterface<ParamType>* base,
851 const std::tuple<ParamGenerator<T>...>& generators,
852 bool is_end)
853 : base_(base),
854 begin_(std::get<I>(generators).begin()...),
855 end_(std::get<I>(generators).end()...),
856 current_(is_end ? end_ : begin_) {
857 ComputeCurrentValue();
858 }
859 ~IteratorImpl() override = default;
860
861 const ParamGeneratorInterface<ParamType>* BaseGenerator() const override {
862 return base_;
863 }
864 // Advance should not be called on beyond-of-range iterators
865 // so no component iterators must be beyond end of range, either.
866 void Advance() override {
867 assert(!AtEnd());
868 // Advance the last iterator.
869 ++std::get<sizeof...(T) - 1>(current_);
870 // if that reaches end, propagate that up.
871 AdvanceIfEnd<sizeof...(T) - 1>();
872 ComputeCurrentValue();
873 }
874 ParamIteratorInterface<ParamType>* Clone() const override {
875 return new IteratorImpl(*this);
876 }
877
878 const ParamType* Current() const override { return current_value_.get(); }
879
880 bool Equals(const ParamIteratorInterface<ParamType>& other) const override {
881 // Having the same base generator guarantees that the other
882 // iterator is of the same type and we can downcast.
883 GTEST_CHECK_(BaseGenerator() == other.BaseGenerator())
884 << "The program attempted to compare iterators "
885 << "from different generators." << std::endl;
886 const IteratorImpl* typed_other =
887 CheckedDowncastToActualType<const IteratorImpl>(&other);
888
889 // We must report iterators equal if they both point beyond their
890 // respective ranges. That can happen in a variety of fashions,
891 // so we have to consult AtEnd().
892 if (AtEnd() && typed_other->AtEnd()) return true;
893
894 bool same = true;
895 bool dummy[] = {
896 (same = same && std::get<I>(current_) ==
897 std::get<I>(typed_other->current_))...};
898 (void)dummy;
899 return same;
900 }
901
902 private:
903 template <size_t ThisI>
904 void AdvanceIfEnd() {
905 if (std::get<ThisI>(current_) != std::get<ThisI>(end_)) return;
906
907 bool last = ThisI == 0;
908 if (last) {
909 // We are done. Nothing else to propagate.
910 return;
911 }
912
913 constexpr size_t NextI = ThisI - (ThisI != 0);
914 std::get<ThisI>(current_) = std::get<ThisI>(begin_);
915 ++std::get<NextI>(current_);
916 AdvanceIfEnd<NextI>();
917 }
918
919 void ComputeCurrentValue() {
920 if (!AtEnd())
921 current_value_ = std::make_shared<ParamType>(*std::get<I>(current_)...);
922 }
923 bool AtEnd() const {
924 bool at_end = false;
925 bool dummy[] = {
926 (at_end = at_end || std::get<I>(current_) == std::get<I>(end_))...};
927 (void)dummy;
928 return at_end;
929 }
930
931 const ParamGeneratorInterface<ParamType>* const base_;
932 std::tuple<typename ParamGenerator<T>::iterator...> begin_;
933 std::tuple<typename ParamGenerator<T>::iterator...> end_;
934 std::tuple<typename ParamGenerator<T>::iterator...> current_;
935 std::shared_ptr<ParamType> current_value_;
936 };
937
938 using Iterator = IteratorImpl<std::make_index_sequence<sizeof...(T)>>;
939
940 std::tuple<ParamGenerator<T>...> generators_;
941};
